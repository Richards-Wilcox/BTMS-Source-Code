Private Sub cmdSave_Click()

Private Sub LoadPartInfo(ByVal strPartNum As String)
    On Error GoTo Error_Handler
    Const cstrErrorSource As String = cstrApplicationName & "." & cstrModuleName & ".LoadPartInfo()"
    Dim llngErrNo As Long
    Dim lstrErrDescr As String
    Dim lstrErrSource As String
    llngErrNo = 0
    lstrErrDescr = ""
    lstrErrSource = ""

    If Trim(strPartNum) = "" Then Exit Sub

    Me.txtPtQty = Format("1", "Fixed")

    Dim llngAffected As Long
    Dim lstrSQLQuery As String
    Dim lobjMainRS As ADODB.Recordset
'    Dim lobjInvRS As ADODB.Recordset
    Dim llngPkgQty As Integer
    Dim lintMult As Integer
    Dim ldblMult As Double
    Dim ldblSprWgt As Double
    If strPartNum = "4" Then
        txtPtQty.Text = DrWidth - 2
        If MeasU = 0 Then
            ClearConObj
            aCon.Inch = DrWidth - 2
            aCon.mmcalc
            txtPtQty = Round(aCon.mm, 0)
        End If
    End If
    If strPartNum = "4-0069" Then
        txtPtQty.Text = RoundUp(DrWidth / 144)
    End If
    
     '-- Mod Hanobver  MultId changed to strMultId
'    lstrSQLQuery = "select DescImp, DescMet, DescOther, MultID, SpecialMult, UM,"
    lstrSQLQuery = "select DescImp, DescMet, DescOther, " & strMultId & ", SpecialMult, UM,"
    lstrSQLQuery = lstrSQLQuery & " PkgQty, CallPrice, PhantomPN, G, I, N, T,J,M,S,V, ShipWeight, SpringingWeight"
    lstrSQLQuery = lstrSQLQuery & " from main where partnum=" & SQLString(Trim(strPartNum))
    Set lobjMainRS = CnnBTMS.Execute(lstrSQLQuery, llngAffected, adCmdText)
    If Not lobjMainRS Is Nothing Then
        lobjMainRS.ActiveConnection = Nothing
        If Not (lobjMainRS.BOF And lobjMainRS.EOF) Then
            lobjMainRS.MoveFirst
            Dim lstrFieldName As String

            lstrFieldName = "DescImp"
            If MeasU = 0 Then lstrFieldName = "DescMet"
            txtPtDesc.Text = IIf(IsNull(lobjMainRS.Fields(lstrFieldName).Value), "", Trim(lobjMainRS.Fields(lstrFieldName).Value))
            Dim UM As String
            UM = GetUMfromInventory(strPartNum)
            If UM = "INCHES" And MeasU = 0 Then UM = "MM"
            cboUM.Clear
            cboUM.AddItem UM
            cboUM.ListIndex = 0
            cboUM.Enabled = False
            
'            If cboUM.List(cboUM.ListIndex) = "LFT" Then
'                cboUM.Clear
'                cboUM.AddItem IIf(MeasU = 1, "INCHES", "MM")
'                cboUM.ListIndex = 0
'            End If
            
            llngPkgQty = IIf(lobjMainRS.Fields("PkgQty").Value > 0, lobjMainRS.Fields("PkgQty").Value, 0)
            lblPkgQty.Caption = "Package Qty: " & IIf(llngPkgQty > 0, Format(llngPkgQty, cstrQtyMask), "N/A")
            
            ldblMult = Format(IIf(IsNull(lobjMainRS.Fields("SpecialMult").Value), "", Trim(lobjMainRS.Fields("SpecialMult").Value)), cstrQtyMask)
            
            If chkCall.Value = 0 Then
                chkCall.Value = IIf(lobjMainRS.Fields("CallPrice").Value = True, 1, 0)
            End If
            
            txtShipWgt.Text = IIf(IsNull(lobjMainRS.Fields("ShipWeight").Value), "", Format(lobjMainRS.Fields("ShipWeight").Value, cstrQtyMask))
            
            txtSprWgt.Text = IIf(IsNull(lobjMainRS.Fields("SpringingWeight").Value), "", Format(lobjMainRS.Fields("SpringingWeight").Value, cstrQtyMask))
            
            If mblnTruss = True Then
                ldblSprWgt = IIf(IsNull(lobjMainRS.Fields("SpringingWeight").Value), 0, lobjMainRS.Fields("SpringingWeight").Value)
                ldblSprWgt = (ldblSprWgt / GetLengthFromMain(strPartNum))
                ldblSprWgt = ldblSprWgt * (DrWidth - 2)
                txtSprWgt.Text = ldblSprWgt
                ClearConObj
                aCon.Inch = DrWidth - 2
                aCon.mmcalc
                Me.txtPtNotes.Text = "Cut to " & IIf(MeasU = 1, GetImpString(DrWidth - 2), Round(aCon.mm, 0) & " mm")
            End If
            
            If cboUM.List(cboUM.ListIndex) <> "EACH" Then
                txtShipWgt.Text = DBLValue(txtShipWgt.Text) / 12
                txtSprWgt.Text = DBLValue(txtSprWgt.Text) / 12
                If cboUM.List(cboUM.ListIndex) = "MM" Then
                    txtShipWgt.Text = DBLValue(txtShipWgt.Text) * 0.03937008
                    txtSprWgt.Text = DBLValue(txtSprWgt.Text) * 0.03937008
                End If
            End If
            
            If MeasU = 0 Then
                ClearConObj
                aCon.LB = DBLValue(txtSprWgt.Text)
                aCon.KGcalc
                txtSprWgt.Text = aCon.KG
                ClearConObj
                aCon.LB = DBLValue(txtShipWgt.Text)
                aCon.KGcalc
                txtShipWgt.Text = aCon.KG
            End If
            If DBLValue(txtSprWgt.Text) > 0.01 Then
                txtSprWgt.Text = Format(DBLValue(txtSprWgt.Text), "Fixed")
            Else
                txtSprWgt.Text = Format(DBLValue(txtSprWgt.Text), "0.0000")
            End If
            If DBLValue(txtShipWgt.Text) > 0.01 Then
                txtShipWgt.Text = Format(DBLValue(txtShipWgt.Text), "Fixed")
            Else
                txtShipWgt.Text = Format(DBLValue(txtShipWgt.Text), "0.0000")
            End If
            ' Calculate Cost of part
            Dim ldblCost As Double
            Select Case intCheck
                Case 1: ldblCost = DBLValue(lobjMainRS.Fields("G").Value)
                Case 2: ldblCost = DBLValue(lobjMainRS.Fields("I").Value)
                Case 3: ldblCost = DBLValue(lobjMainRS.Fields("N").Value)
                Case 4: ldblCost = DBLValue(lobjMainRS.Fields("T").Value)
                
                Case 5: ldblCost = DBLValue(lobjMainRS.Fields("J").Value)
                Case 6: ldblCost = DBLValue(lobjMainRS.Fields("M").Value)
                Case 7: ldblCost = DBLValue(lobjMainRS.Fields("S").Value)
                Case 8: ldblCost = DBLValue(lobjMainRS.Fields("V").Value)
                
                
            End Select
            
            ldblCost = Decrypt(GetSym(strPartNum), ldblCost)

 '-- Mod Hanobver  MultId changed to strMultId
            lintMult = IIf(IsNull(lobjMainRS.Fields(strMultId).Value), "", Trim(lobjMainRS.Fields(strMultId).Value))
            If lobjMainRS.Fields("SpecialMult").Value > 0 Then
                ldblCost = ldblCost * DBLValue(lobjMainRS.Fields("SpecialMult").Value)
            Else
                ldblCost = ldblCost * (MultipliersCase(DBLValue(Trim(lintMult))))
            End If

            If cboUM.List(cboUM.ListIndex) <> "EACH" Then
                ldblCost = ldblCost / 12
                If cboUM.List(cboUM.ListIndex) = "MM" Then
                    ldblCost = ldblCost * 0.03937008
                End If
            End If
            
            lblBookCost.Caption = IIf(ldblCost < 0.01, Format(ldblCost, "0.0000"), Format(ldblCost, "Fixed"))
            
            If chkCall.Value = 0 Then
                
                txtPtCost.Text = IIf(DBLValue(lblBookCost.Caption) < 1, Format(lblBookCost.Caption, "0.0000"), Format(lblBookCost.Caption, "Fixed"))
            Else
                txtPtCost.Text = Format("0", "0.0000")
            End If
            
            lblTotalCost.Caption = Format(DBLValue(txtPtQty.Text) * DBLValue(txtPtCost.Text), cstrDollarMask)
            
            lobjMainRS.Close
        End If
    End If
    
    txtPtDesc.Enabled = False
    txtPtCost.Enabled = False
    txtShipWgt.Enabled = False
'    txtSprWgt.Enabled = False
    chkCall.Enabled = False
    chkBtmWgt.Value = IIf(strPartNum = "4", 1, 0)
    If strPartNum = "4" Then Me.txtPN.Text = "COMM-NSTK" & "-" & Left("000", 3 - Len(CStr(CurrSpItem))) & CStr(CurrSpItem)
    Set lobjMainRS = Nothing

    Exit Sub

Error_Handler:
    llngErrNo = IIf(llngErrNo = 0, Err.Number, llngErrNo)
    lstrErrDescr = IIf(Len(lstrErrDescr) = 0, Err.Description, Err.Description)
    lstrErrSource = IIf(Len(Trim(Err.Source)) = 0, "", Err.Source & vbCrLf) & cstrErrorSource
    Call Err.Raise(llngErrNo, lstrErrSource, lstrErrDescr)
End Sub

Private Sub SaveData()

    On Error GoTo Error_Handler
    Const cstrErrorSource As String = cstrApplicationName & "." & cstrModuleName & ".SaveData()"
    Dim llngErrNo As Long
    Dim lstrErrDescr As String
    Dim lstrErrSource As String
    llngErrNo = 0
    lstrErrDescr = ""
    lstrErrSource = ""

    Dim lintCount As Integer
    If isEmptyArraySpItems Then
        lintCount = 0
    Else
        lintCount = UBound(varSpItem()) + 1
    End If
    
    ReDim Preserve varSpItem(lintCount)
    
    If mblnTruss = True Then
    
        varSpItem(lintCount).intType = 2  'truss
    End If
    
    varSpItem(lintCount).strPN = Trim(txtPN.Text)
    varSpItem(lintCount).strDescription = Trim(txtPtDesc.Text)
    varSpItem(lintCount).sngQty = DBLValue(txtPtQty.Text)
    varSpItem(lintCount).strUM = IIf(cboUM.List(cboUM.ListIndex) = "MM", "INCHES", cboUM.List(cboUM.ListIndex))
    
    If cboUM.ListIndex >= 0 Then
        If cboUM.List(cboUM.ListIndex) = "MM" Or cboUM.List(cboUM.ListIndex) = "INCHES" Then
            If MeasU = 0 Then
                ClearConObj
                aCon.mm = DBLValue(txtPtQty.Text)
                aCon.Inchcalc
                varSpItem(lintCount).sngLength = aCon.Inch
                varSpItem(lintCount).sngQty = aCon.Inch
            Else
                varSpItem(lintCount).sngLength = DBLValue(txtPtQty.Text)
            End If
        End If
    End If
    
    varSpItem(lintCount).sngPrice = DBLValue(txtPtCost.Text)
    varSpItem(lintCount).sngShippingWgt = DBLValue(txtShipWgt.Text)
    varSpItem(lintCount).sngSpringingWgt = DBLValue(txtSprWgt.Text)
    
    If MeasU = 0 Then
        ClearConObj
        aCon.KG = varSpItem(lintCount).sngShippingWgt
        aCon.LBcalc
        varSpItem(lintCount).sngShippingWgt = DBLValue(aCon.LB) * 25.5
        ClearConObj
        aCon.KG = varSpItem(lintCount).sngSpringingWgt
        aCon.LBcalc
        varSpItem(lintCount).sngSpringingWgt = DBLValue(aCon.LB) * 25.4
    End If
    varSpItem(lintCount).blnBtmWgt = IIf(chkBtmWgt.Value = 1, True, False)
    varSpItem(lintCount).strNotes = Trim(txtPtNotes.Text)
    

''''''''''Change in .NET''''''''''''''''
    If Left(txtPN.Text, 9) = "COMM-NSTK" Then CurrSpItem = CurrSpItem + 1
''''''''''END Change in .NET
    
    'add notes to springing, that weight will affect the springing.
    Dim W As Single
    W = varSpItem(lintCount).sngSpringingWgt * varSpItem(lintCount).sngQty
    If MeasU = 0 Then
        ClearConObj
        aCon.LB = W
        aCon.KGcalc
        W = aCon.KG
    End If
    frmGrid.txtSprNotes.Text = frmGrid.txtSprNotes.Text & IIf(frmGrid.txtSprNotes.Text <> "", vbCrLf, "") & "Part Number: " & varSpItem(lintCount).strPN & " added " & Format(W, "Fixed") & " " & IIf(MeasU = 0, "kg", "lb") & " to " & IIf(varSpItem(lintCount).blnBtmWgt = True, "extra bottom ", "sections ") & "weight."
    Exit Sub

Error_Handler:
    llngErrNo = IIf(llngErrNo = 0, Err.Number, llngErrNo)
    lstrErrDescr = IIf(Len(lstrErrDescr) = 0, Err.Description, lstrErrDescr & vbCrLf & Err.Description)
    lstrErrSource = IIf(Len(Trim(Err.Source)) = 0, "", Err.Source & vbCrLf) & cstrErrorSource
    Call Err.Raise(llngErrNo, lstrErrSource, lstrErrDescr)

End Sub
