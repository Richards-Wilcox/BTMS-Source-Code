Public Function GetSpecialPart(ByVal lstrType As String) As String
    On Error GoTo Error_Handler
    Const cstrErrorSource As String = cstrApplicationName & "." & cstrModuleName & ".GetSpecialPart()"
    Dim llngErrNo As Long
    Dim lstrErrDescr As String
    Dim lstrErrSource As String
    llngErrNo = 0
    lstrErrDescr = ""
    lstrErrSource = ""

    Dim lstrCurrVal As String
    Dim llngNextVal As Long
    
    GetSpecialPart = ""
    aReg.OpenKey LocalMachine, "\Software\Richards-Wilcox Canada\BTMS", True 'opens or creates key
    
    Select Case lstrType
    Case "NSTK"
        lstrCurrVal = aReg.ReadValue("LastNumber")
        llngNextVal = DBLValue(lstrCurrVal)
        If llngNextVal = 0 Or llngNextVal > 999 Then llngNextVal = 1
'        aReg.WriteValue "LastNumber", CStr(llngNextVal + 1)
    Case "ASECT"
        lstrCurrVal = aReg.ReadValue("LastNumberASECT")
        llngNextVal = DBLValue(lstrCurrVal)
        If llngNextVal = 0 Or llngNextVal > 999 Then llngNextVal = 1
'        aReg.WriteValue "LastNumberASECT", CStr(llngNextVal + 1)
    Case "RSECT"
        lstrCurrVal = aReg.ReadValue("LastNumberRSECT")
        llngNextVal = DBLValue(lstrCurrVal)
        If llngNextVal = 0 Or llngNextVal > 999 Then llngNextVal = 1
'        aReg.WriteValue "LastNumberRSECT", CStr(llngNextVal + 1)
    End Select
    aReg.CloseKey 'close key
    
    GetSpecialPart = CStr(llngNextVal + 1)
    GetSpecialPart = "COMM-" & lstrType & "-" & Left("000", 3 - Len(GetSpecialPart)) & GetSpecialPart
    
    Exit Function

Error_Handler:
    llngErrNo = IIf(llngErrNo = 0, Err.Number, llngErrNo)
    lstrErrDescr = IIf(Len(lstrErrDescr) = 0, Err.Description, lstrErrDescr & vbCrLf & Err.Description)
    lstrErrSource = IIf(Len(Trim(Err.Source)) = 0, "", Err.Source & vbCrLf) & cstrErrorSource
    Call Err.Raise(llngErrNo, lstrErrSource, lstrErrDescr)
End Function

Public Function AllocateQuote(ByVal lngQuoteID As Long) As Boolean

    On Error GoTo Error_Handler
    Const cstrErrorSource As String = cstrApplicationName & "." & cstrModuleName & ".AllocateQuote()"
    Dim llngErrNo As Long
    Dim lstrErrDescr As String
    Dim lstrErrSource As String
    llngErrNo = 0
    lstrErrDescr = ""
    lstrErrSource = ""

    Dim lstrSQL As String
    Dim llngAffected As Long
    Dim lobjQuoteDetRS As ADODB.Recordset
    Dim lblnInTrans As Boolean
    Dim dblQtyReq As Double
    Dim ldblStock   As Double
    Dim ldblResult As Double
    Dim lobjINVRS As ADODB.Recordset
    lblnInTrans = False
    

    'Get BoM Details for part to Allocate
    lstrSQL = "select QuotePtNum, QuotePtQty,QuoteTotAmountReq from QuoteDetails"
    lstrSQL = lstrSQL & " where QuoteID=" & lngQuoteID
    Set lobjQuoteDetRS = CnnBTMS.Execute(lstrSQL, llngAffected, adCmdText)
    If Not lobjQuoteDetRS Is Nothing Then
        If Not (lobjQuoteDetRS.BOF And lobjQuoteDetRS.EOF) Then
            lobjQuoteDetRS.MoveFirst
            CnnBTMS.BeginTrans
            lblnInTrans = True
            
            Do While lobjQuoteDetRS.EOF = False
                If IsNull(lobjQuoteDetRS.Fields("QuoteTotAmountReq").Value) Then
                    dblQtyReq = 0
                Else
                    dblQtyReq = DBLValue(lobjQuoteDetRS.Fields("QuoteTotAmountReq").Value)
                End If
                If dblQtyReq > 0 Then
                    lstrSQL = "select PartNum, Qtyinstock from Inventory "
                    lstrSQL = lstrSQL & " Where PartNum=" & SQLString(lobjQuoteDetRS.Fields("QuotePtNum").Value)
                    
                    Set lobjINVRS = CnnBTMS.Execute(lstrSQL, llngAffected, adCmdText And adExecuteNoRecords)
                    If Not lobjINVRS Is Nothing And Not (lobjINVRS.BOF And lobjINVRS.EOF) Then
                        
                        lobjINVRS.MoveFirst
                        lobjINVRS.Find "PartNum=" & SQLString(lobjQuoteDetRS.Fields("QuotePtNum").Value)
                        If lobjINVRS.EOF = False And lobjINVRS.BOF = False Then
                            ldblStock = lobjINVRS.Fields("QtyInStock").Value
                        End If
                    End If
                    
                    ldblResult = ldblStock - dblQtyReq
                    If ldblResult < 0.01 Then ldblResult = 0
                    lstrSQL = "Update Inventory Set "
                    lstrSQL = lstrSQL & "QtyInStock=" & Replace(ldblResult, ",", ".")
                    lstrSQL = lstrSQL & " Where PartNum=" & SQLString(lobjQuoteDetRS.Fields("QuotePtNum").Value)
                    Call CnnBTMS.Execute(lstrSQL, llngAffected, adCmdText And adExecuteNoRecords)
                    Set lobjINVRS = Nothing
                    
                End If
                lobjQuoteDetRS.MoveNext
            Loop
            lobjQuoteDetRS.Close
        End If
    End If
    
    lstrSQL = "Update BoM Set "
    lstrSQL = lstrSQL & " BOMAlloc=True, BOMDoneDate=" & SQLString(Now())
    lstrSQL = lstrSQL & " where QuoteID=" & lngQuoteID
    Call CnnBTMS.Execute(lstrSQL, llngAffected, adCmdText And adExecuteNoRecords)
    
    lstrSQL = "Update Quote Set "
    lstrSQL = lstrSQL & " QuoteAlloc=True, QuoteDoneDate=" & SQLString(Now())
    lstrSQL = lstrSQL & " where QuoteID=" & lngQuoteID
    Call CnnBTMS.Execute(lstrSQL, llngAffected, adCmdText And adExecuteNoRecords)
    
    CnnBTMS.CommitTrans
    AllocateQuote = True
    lblnInTrans = False
    Set lobjQuoteDetRS = Nothing
    Exit Function

Error_Handler:
    llngErrNo = IIf(llngErrNo = 0, Err.Number, llngErrNo)
    lstrErrDescr = IIf(Len(lstrErrDescr) = 0, Err.Description, lstrErrDescr & vbCrLf & Err.Description)
    lstrErrSource = IIf(Len(Trim(Err.Source)) = 0, "", Err.Source & vbCrLf) & cstrErrorSource
    If lblnInTrans = True Then CnnBTMS.RollbackTrans
    Call Err.Raise(llngErrNo, lstrErrSource, lstrErrDescr)
End Function

Public Function CheckBOMforAllocation(ByVal lngBOMID As Long, ByVal mstrBOMUM As String, ByVal pblnNow As Boolean) As Boolean
    
    On Error GoTo Error_Handler
    Const cstrErrorSource As String = cstrApplicationName & "." & cstrModuleName & ".CheckBOMforAllocation()"
    Dim llngErrNo As Long
    Dim lstrErrDescr As String
    Dim lstrErrSource As String
    llngErrNo = 0
    lstrErrDescr = ""
    lstrErrSource = ""

    Dim lstrSQL As String
    Dim llngAffected As Long
    Dim lobjINVRS As ADODB.Recordset
    Dim lobjMainRS As ADODB.Recordset
    Dim lobjBOMDetRS As ADODB.Recordset
    Dim lblnInTrans As Boolean
    Dim ldblQtyReq As Double
    Dim ldblQtyStk As Double
    Dim lintCounter As Integer
    Dim ldblQtyResult As Double
    Dim ldblRecmd As Double
    Dim ldblLength As Double
    Dim ldblTempLen As Double
    lblnInTrans = False

    ' Get BoM Details for part to Allocate
    lstrSQL = "select BOMPtNum,BOMPtDesc, BOMPtUM,BOMPtTotAmountReq,BOMPtQtyReq from BOMDetails"
    lstrSQL = lstrSQL & " where BOMID=" & lngBOMID
    Set lobjBOMDetRS = CnnBTMS.Execute(lstrSQL, llngAffected, adCmdText)
    If Not lobjBOMDetRS Is Nothing Then
        If Not (lobjBOMDetRS.BOF And lobjBOMDetRS.EOF) Then
            lobjBOMDetRS.MoveFirst
            CnnBTMS.BeginTrans
            lblnInTrans = True
            Do While lobjBOMDetRS.EOF = False
                ' Find out if Inventory record Exists
                lstrSQL = "Select I.PartNum, M.DescImp, M.DescMet, M.DescOther, I.QtyInStock, I.QtyReserved, I.QtyOnOrder, I.QtyBkOrder, I.RestockingLevel, I.QtyMin ,I.QtyOnPos From Inventory I"
                lstrSQL = lstrSQL & " Left join Main M on I.PartNum = M.PartNum"
                lstrSQL = lstrSQL & " where I.PartNum=" & SQLString(lobjBOMDetRS.Fields("BOMPtNum").Value)
                Set lobjINVRS = CnnBTMS.Execute(lstrSQL, llngAffected, adCmdText)
                If Not lobjINVRS Is Nothing Then
                    If Not (lobjINVRS.BOF And lobjINVRS.EOF) Then
                        
                        lobjINVRS.MoveFirst
                        If IsNull(lobjINVRS.Fields("QtyInStock").Value) Then
                            ldblQtyStk = 0
                        Else
                            ldblQtyStk = DBLValue(lobjINVRS.Fields("QtyInStock").Value)
                        End If
                        
                        If IsNull(lobjBOMDetRS.Fields("BOMPtTotAmountReq").Value) Then
                            ldblQtyReq = 0
                        Else
                            ldblQtyReq = DBLValue(lobjBOMDetRS.Fields("BOMPtTotAmountReq").Value)
                        End If
                        
                        ldblQtyResult = ldblQtyStk - IIf(pblnNow = True, 0, DBLValue(lobjINVRS.Fields("QtyReserved").Value)) - ldblQtyReq
                        If Format(ldblQtyResult, "Fixed") < 0 Then 'if insufficient stock
                            lintCounter = lintCounter + 1
                            If lintCounter >= 2 Then
                                frmBOMAlloc.grdData1.Rows = frmBOMAlloc.grdData1.Rows + 1
                            End If
                            frmBOMAlloc.grdData1.TextMatrix(lintCounter, 0) = lobjINVRS.Fields("PartNum").Value
                            frmBOMAlloc.grdData1.TextMatrix(lintCounter, 1) = IIf(UCase(mstrBOMUM) = "METRIC", lobjINVRS.Fields("DescMet").Value, lobjINVRS.Fields("DescImp").Value)
                            frmBOMAlloc.grdData1.TextMatrix(lintCounter, 2) = IIf(IsNull(lobjINVRS.Fields("DescOther").Value), "", lobjINVRS.Fields("DescOther").Value)
                            frmBOMAlloc.grdData1.TextMatrix(lintCounter, 3) = lobjBOMDetRS.Fields("BOMPTUM").Value
                            
                            If mstrBOMUM = "Metric" And lobjBOMDetRS.Fields("BOMPTUM").Value = "INCHES" Then
                                If Format(lobjINVRS.Fields("QtyInStock").Value, "Fixed") > 0 Then
                                    ClearConObj
                                    aCon.Inch = DBLValue(lobjINVRS.Fields("QtyInStock").Value)
                                    aCon.mmcalc
                                    frmBOMAlloc.grdData1.TextMatrix(lintCounter, 4) = Round(aCon.mm, 0)
                                Else
                                    frmBOMAlloc.grdData1.TextMatrix(lintCounter, 4) = 0
                                End If
                            Else
                                frmBOMAlloc.grdData1.TextMatrix(lintCounter, 4) = IIf(lobjBOMDetRS.Fields("BOMPtUM").Value = "INCHES", Format(lobjINVRS.Fields("QtyInStock").Value, "Fixed"), Round(lobjINVRS.Fields("QtyInStock").Value, 0))
                            End If
                            
                            
                            frmBOMAlloc.grdData1.TextMatrix(lintCounter, 5) = IIf(lobjBOMDetRS.Fields("BOMPtUM").Value = "INCHES" And mstrBOMUM <> "Metric", Format(lobjBOMDetRS.Fields("BOMPtQtyReq").Value, "Fixed"), Round(lobjBOMDetRS.Fields("BOMPtQtyReq").Value, 0))
                            
                            If mstrBOMUM = "Metric" And lobjBOMDetRS.Fields("BOMPTUM").Value = "INCHES" Then
                                If Format(lobjINVRS.Fields("QtyReserved").Value, "Fixed") > 0 Then
                                    ClearConObj
                                    aCon.Inch = DBLValue(lobjINVRS.Fields("QtyReserved").Value)
                                    aCon.mmcalc
                                    frmBOMAlloc.grdData1.TextMatrix(lintCounter, 6) = Round(aCon.mm, 0)
                                Else

                                End If

Public Sub UnAllocateBOM(ByVal lngBOMID As Long)

    'Get BoM Details for part to UnAllocate
    lstrSQL = "select BOMPtNum, BOMPtTotAmountReq from BOMDetails where BOMID=" & lngBOMID
    Set lobjBOMDetRS = CnnBTMS.Execute(lstrSQL, llngAffected, adCmdText)
    If Not lobjBOMDetRS Is Nothing Then
        If Not (lobjBOMDetRS.BOF And lobjBOMDetRS.EOF) Then
            lobjBOMDetRS.MoveFirst
            Do While lobjBOMDetRS.EOF = False
                ' Find out if Inventory record Exists
                lstrSQL = "Select PartNum, QtyInStock, QtyReserved From Inventory"
                lstrSQL = lstrSQL & " where PartNum=" & SQLString(lobjBOMDetRS.Fields("BOMPtNum").Value)
                Set lobjINVRS = CnnBTMS.Execute(lstrSQL, llngAffected, adCmdText)
                If Not lobjINVRS Is Nothing Then
                    If Not (lobjINVRS.BOF And lobjINVRS.EOF) Then
                        lobjINVRS.MoveFirst
                        If IsNull(lobjINVRS.Fields("QtyInStock").Value) Then
                            dblQtyStk = 0
                        Else
                            dblQtyStk = DBLValue(lobjINVRS.Fields("QtyInStock").Value)
                        End If
                        If IsNull(lobjBOMDetRS.Fields("BOMPtTotAmountReq").Value) Then
                            dblQtyRes = 0
                        Else
                             dblQtyRes = DBLValue(lobjINVRS.Fields("QtyReserved").Value) - DBLValue(lobjBOMDetRS.Fields("BOMPtTotAmountReq").Value)
                        End If
                        
                        lobjINVRS.Close
                                                
                        If lobjBOMRS.Fields("BOMReserveDate").Value > 0 Then 'unreserve
                            lstrSQL = "Update Inventory Set "
                            lstrSQL = lstrSQL & "QtyReserved=" & Replace(dblQtyRes, ",", ".")
                            lstrSQL = lstrSQL & " Where PartNum=" & SQLString(lobjBOMDetRS.Fields("BOMPtNum").Value)
                            Call CnnBTMS.Execute(lstrSQL, llngAffected, adCmdText And adExecuteNoRecords)
                        Else 'unallocate
                            lstrSQL = "Update Inventory Set "
                            lstrSQL = lstrSQL & "QtyInStock=QtyInStock+" & Replace(lobjBOMDetRS.Fields("BOMPtTotAmountReq").Value, ",", ".")
                            lstrSQL = lstrSQL & " Where PartNum=" & SQLString(lobjBOMDetRS.Fields("BOMPtNum").Value)
                            Call CnnBTMS.Execute(lstrSQL, llngAffected, adCmdText And adExecuteNoRecords)
                        End If
                    End If
                Else
                    ' No Inventory record exits - DO NOTHING!!!!
                End If
            
                Set lobjINVRS = Nothing
                lobjBOMDetRS.MoveNext
                
            Loop
            If lobjBOMRS.Fields("BOMReserveDate").Value > 0 Then
                lstrSQL = "Update BoM Set BOMReserveDate=NULL" '& SQLString(Format(Now(), cstrDateMask))
                lstrSQL = lstrSQL & ", BOMAlloc=0"
                lstrSQL = lstrSQL & " where BOMID=" & lngBOMID
                Call CnnBTMS.Execute(lstrSQL, llngAffected, adCmdText And adExecuteNoRecords)
            Else
                lstrSQL = "Update BoM Set BOMDoneDate=NULL" '& SQLString(Format(Now(), cstrDateMask))
                lstrSQL = lstrSQL & ", BOMAlloc=0"
                lstrSQL = lstrSQL & " where BOMID=" & lngBOMID
                Call CnnBTMS.Execute(lstrSQL, llngAffected, adCmdText And adExecuteNoRecords)
            End If
        End If
    End If
    CnnBTMS.CommitTrans
    lblnInTrans = False
    Set lobjBOMDetRS = Nothing
    
    Exit Sub

Public Sub SetRestockingLevels(ByVal lngPOID As Long)
    On Error GoTo Error_Handler
    Const cstrErrorSource As String = cstrApplicationName & "." & cstrModuleName & ".SetRestockingLevels()"
    Dim llngErrNo As Long
    Dim lstrErrDescr As String
    Dim lstrErrSource As String
    llngErrNo = 0
    lstrErrDescr = ""
    lstrErrSource = ""

    Dim lstrSQL As String
    Dim lobjINVRS As ADODB.Recordset
    Dim llngAffected As Long
    Dim lobjPODetRS As ADODB.Recordset
    Dim lobjMainRS As ADODB.Recordset
    Dim lobjCategoryRS As ADODB.Recordset
    Dim ldblOrd As Double
    Dim lstrCat As String
    
    lstrSQL = "select PartNum,QtyMin,QtyInStock,RestockingLevel, InvUM from Inventory"
    Set lobjINVRS = New ADODB.Recordset
    lobjINVRS.CursorType = adOpenForwardOnly
    lobjINVRS.CursorLocation = adUseClient
    lobjINVRS.Open lstrSQL, CnnBTMS, adOpenForwardOnly, adLockOptimistic
    lstrSQL = "select POPtNum, POPtQtyOrd, POPtUM, POID from PODetails where POID=" & lngPOID
    Set lobjPODetRS = CnnBTMS.Execute(lstrSQL, llngAffected, adCmdText)
    
    lstrSQL = "select PartNum,SubCategory from Main"
    Set lobjMainRS = CnnBTMS.Execute(lstrSQL, llngAffected, adCmdText)
    lstrSQL = "select SubCategory, MinQty from CatSub"
    Set lobjCategoryRS = CnnBTMS.Execute(lstrSQL, llngAffected, adCmdText)
    
    If Not lobjPODetRS Is Nothing Then
        If Not (lobjPODetRS.EOF = True And lobjPODetRS.BOF = True) Then
            lobjPODetRS.MoveFirst
            Do While lobjPODetRS.EOF = False
                    
                lobjINVRS.MoveFirst
                lobjINVRS.Find "PartNum=" & SQLString(lobjPODetRS.Fields("POPtNum").Value)
                If lobjINVRS.EOF = False And lobjINVRS.BOF = False Then
                    If lobjPODetRS.Fields("POPtUM").Value = "EACH" And lobjINVRS.Fields("InvUM").Value = "INCHES" Then
                        ldblOrd = DBLValue(lobjPODetRS.Fields("POPtQtyOrd").Value) * GetLengthFromMain(lobjPODetRS.Fields("POPtNum").Value)
                    ElseIf lobjPODetRS.Fields("POPtUM").Value = "LFT" Then
                        ldblOrd = DBLValue(lobjPODetRS.Fields("POPtQtyOrd").Value) * 12
                    Else
                        ldblOrd = DBLValue(lobjPODetRS.Fields("POPtQtyOrd").Value)
                    End If

                    lobjMainRS.MoveFirst
                    lobjMainRS.Find "PartNum=" & SQLString(lobjPODetRS.Fields("POPtNum").Value)
                    If lobjMainRS.EOF = False And lobjMainRS.BOF = False Then
                        lobjCategoryRS.MoveFirst
                        lobjCategoryRS.Find "SubCategory=" & SQLString(lobjMainRS.Fields("SubCategory").Value)
                        If lobjCategoryRS.EOF = False And lobjCategoryRS.BOF = False Then
                            
                            lobjINVRS.Fields("QtyMin").Value = RoundUp(ldblOrd * IIf(IsNull(lobjCategoryRS.Fields("MinQty").Value), 0, lobjCategoryRS.Fields("MinQty").Value))
                            lobjINVRS.Fields("RestockingLevel").Value = RoundUp(ldblOrd - lobjINVRS.Fields("QtyMin").Value)
                            If lobjINVRS.Fields("RestockingLevel").Value = 0 Then
                                lobjINVRS.Fields("RestockingLevel").Value = lobjINVRS.Fields("QtyMin").Value
                            End If
                            
                            lobjINVRS.Update
                        End If
                    End If
                End If
                lobjPODetRS.MoveNext
            Loop
        End If
    End If
    
    Exit Sub

Error_Handler:
    llngErrNo = IIf(llngErrNo = 0, Err.Number, llngErrNo)
    lstrErrDescr = IIf(Len(lstrErrDescr) = 0, Err.Description, lstrErrDescr & vbCrLf & Err.Description)
    lstrErrSource = IIf(Len(Trim(Err.Source)) = 0, "", Err.Source & vbCrLf) & cstrErrorSource
    Call Err.Raise(llngErrNo, lstrErrSource, lstrErrDescr)

End Sub

                            If WasPOPreviouslyReceived(plngPOID) = True Then lobjINVRS.Fields("QtyBkOrder").Value = DBLValue(lobjINVRS.Fields("QtyBkOrder").Value) - (DBLValue(lobjPODetRS.Fields("POPtQtyBO").Value) * GetLengthFromMain(lobjPODetRS.Fields("POPtNum").Value))
                        ElseIf lobjPODetRS.Fields("POPtUM").Value = "LFT" Then
                            lobjINVRS.Fields("QtyOnOrder").Value = DBLValue(lobjINVRS.Fields("QtyOnOrder").Value) - (DBLValue(lobjPODetRS.Fields("POPtQtyBO").Value) * 12)
                            If lobjINVRS.Fields("QtyOnOrder").Value < 0 Then lobjINVRS.Fields("QtyOnOrder").Value = 0 'again the rounding issue with VB. (see note in frmInvPOReceipts, BuildRS)
                            If WasPOPreviouslyReceived(plngPOID) = True Then
                                lobjINVRS.Fields("QtyBkOrder").Value = DBLValue(lobjINVRS.Fields("QtyBkOrder").Value) - (DBLValue(lobjPODetRS.Fields("POPtQtyBO").Value) * 12)
                                If lobjINVRS.Fields("QtyBkOrder").Value < 0 Then lobjINVRS.Fields("QtyBkOrder").Value = 0
                            End If
                        Else
                            lobjINVRS.Fields("QtyOnOrder").Value = DBLValue(lobjINVRS.Fields("QtyOnOrder").Value) - DBLValue(lobjPODetRS.Fields("POPtQtyBO").Value)
                            If WasPOPreviouslyReceived(plngPOID) = True Then lobjINVRS.Fields("QtyBkOrder").Value = DBLValue(lobjINVRS.Fields("QtyBkOrder").Value) - DBLValue(lobjPODetRS.Fields("POPtQtyBO").Value)
                        End If
                        lobjINVRS.Update
                    End If
                End If
                lobjINVRS.Close
                lobjPODetRS.MoveNext
            Loop
            Set lobjINVRS = Nothing
            Set lobjPODetRS = Nothing
        End If
    End If
        
    Exit Sub

Error_Handler:
    llngErrNo = IIf(llngErrNo = 0, Err.Number, llngErrNo)

Public Sub ResetInventory(ByVal plngPOID As Long)
    On Error GoTo Error_Handler
    Const cstrErrorSource As String = cstrApplicationName & "." & cstrModuleName & ".ResetInventory()"
    Dim llngErrNo As Long
    Dim lstrErrDescr As String
    Dim lstrErrSource As String
    llngErrNo = 0
    lstrErrDescr = ""
    lstrErrSource = ""

    Dim lstrSQL As String
    Dim llngAffected As Long
    Dim lobjINVRS As ADODB.Recordset
    Dim lobjPODetRS As ADODB.Recordset
    Set lobjINVRS = New ADODB.Recordset
    
    lstrSQL = "select * from podetails where POID=" & plngPOID
    Set lobjPODetRS = CnnBTMS.Execute(lstrSQL, llngAffected, adCmdText)
    If Not lobjPODetRS Is Nothing Then
        If Not (lobjPODetRS.EOF And lobjPODetRS.BOF) Then
            lobjPODetRS.MoveFirst
            Do While lobjPODetRS.EOF = False
                lstrSQL = "select * from Inventory where PartNum=" & SQLString(lobjPODetRS.Fields("POPtNum").Value)
                lobjINVRS.Open lstrSQL, CnnBTMS, adOpenForwardOnly, adLockOptimistic
                If Not lobjINVRS Is Nothing Then
                    If Not (lobjINVRS.EOF And lobjINVRS.BOF) Then
                        If lobjINVRS.Fields("InvUM").Value = "INCHES" And lobjPODetRS.Fields("POPtUM").Value = "EACH" Then
                            lobjINVRS.Fields("QtyOnOrder").Value = DBLValue(lobjINVRS.Fields("QtyOnOrder").Value) + (DBLValue(lobjPODetRS.Fields("POPtQtyOrd").Value) * GetLengthFromMain(lobjPODetRS.Fields("POPtNum").Value))
                            lobjINVRS.Fields("QtyInStock").Value = DBLValue(lobjINVRS.Fields("QtyInStock").Value) - (DBLValue(lobjPODetRS.Fields("POPtQtyRcvd").Value) * GetLengthFromMain(lobjPODetRS.Fields("POPtNum").Value))
                        ElseIf lobjPODetRS.Fields("POPtUM").Value = "LFT" Then
                            lobjINVRS.Fields("QtyOnOrder").Value = DBLValue(lobjINVRS.Fields("QtyOnOrder").Value) + (DBLValue(lobjPODetRS.Fields("POPtQtyOrd").Value) * 12)
                            lobjINVRS.Fields("QtyInStock").Value = DBLValue(lobjINVRS.Fields("QtyInStock").Value) - (DBLValue(lobjPODetRS.Fields("POPtQtyRcvd").Value) * 12)
                            If DBLValue(lobjINVRS.Fields("QtyInStock").Value) < 0 Then lobjINVRS.Fields("QtyInStock").Value = 0 'see note in frmInvPOReceipts, BuildRs regarding rounding in VB.
                        Else
                            lobjINVRS.Fields("QtyOnOrder").Value = DBLValue(lobjINVRS.Fields("QtyOnOrder").Value) + DBLValue(lobjPODetRS.Fields("POPtQtyOrd").Value)
                            lobjINVRS.Fields("QtyInStock").Value = DBLValue(lobjINVRS.Fields("QtyInStock").Value) - DBLValue(lobjPODetRS.Fields("POPtQtyRcvd").Value)
                        End If
                        lobjINVRS.Update
                    End If
                End If
                lobjINVRS.Close
                lobjPODetRS.MoveNext
            Loop
            Set lobjINVRS = Nothing
            Set lobjPODetRS = Nothing
        End If
    End If
        
    Exit Sub

Error_Handler:
    llngErrNo = IIf(llngErrNo = 0, Err.Number, llngErrNo)

Public Function GetQtyOnActivePOs(ByVal plstrPartNum As String) As Double

    On Error GoTo Error_Handler
    Const cstrErrorSource As String = cstrApplicationName & "." & cstrModuleName & ".GetQtyOnActivePOs()"
    Dim llngErrNo As Long
    Dim lstrErrDescr As String
    Dim lstrErrSource As String
    llngErrNo = 0
    lstrErrDescr = ""
    lstrErrSource = ""
    ' will return the total qty sitting on active purchase orders for the passed part number.
    'the qty returned will be transfered to the UM in the inventory table: PVC bushing is each on the PO but the total number of inches will be returned.
    
    Dim lstrSQL As String
    Dim llngAffected As Long
    Dim lobjPOActiveRS As ADODB.Recordset
    Dim lobjPODetailsRS As ADODB.Recordset
    Dim ldblQty As Double
    Dim MainUM As String
    Dim InvUM As String
    
    lstrSQL = lstrSQL & "(select sum(POPtQtyOrd) from PODetails where poptnum=" & SQLString(plstrPartNum)
    lstrSQL = lstrSQL & " and poid IN ( select POID from PO where POStatus='Active' and PORunThroughInventory=true))"
    Set lobjPOActiveRS = CnnBTMS.Execute(lstrSQL, llngAffected, adCmdText)
    If Not lobjPOActiveRS Is Nothing Then
        If Not (lobjPOActiveRS.EOF And lobjPOActiveRS.BOF) Then
            lobjPOActiveRS.ActiveConnection = Nothing
            If IsNull(lobjPOActiveRS.Fields(0).Value) Then
                ldblQty = 0
            Else
                ldblQty = DBLValue(lobjPOActiveRS.Fields(0).Value)
            End If
        End If
    End If
            
    lstrSQL = "select POPtUM from PODetails where POPtNum=" & SQLString(plstrPartNum)
    Set lobjPODetailsRS = CnnBTMS.Execute(lstrSQL, llngAffected, adCmdText)
    If Not lobjPODetailsRS Is Nothing Then
        If Not (lobjPODetailsRS.EOF And lobjPODetailsRS.BOF) Then
            lobjPODetailsRS.ActiveConnection = Nothing
            MainUM = lobjPODetailsRS.Fields("POptUM").Value
        End If
    End If
    
    Set lobjPODetailsRS = Nothing
    Set lobjPOActiveRS = Nothing
    
    lstrSQL = "select InvUM from Inventory where PartNum=" & SQLString(plstrPartNum)
    Set lobjPODetailsRS = CnnBTMS.Execute(lstrSQL, llngAffected, adCmdText)
    If Not lobjPODetailsRS Is Nothing Then
        If Not (lobjPODetailsRS.EOF And lobjPODetailsRS.BOF) Then
            lobjPODetailsRS.ActiveConnection = Nothing
            InvUM = lobjPODetailsRS.Fields("InvUM").Value
        End If
    End If
    Set lobjPODetailsRS = Nothing
    
    If (InvUM = "INCHES" Or InvUM = "") And MainUM = "EACH" Then
        ldblQty = ldblQty * GetLengthFromMain(plstrPartNum)
    ElseIf MainUM = "LFT" Then
        ldblQty = ldblQty * 12
    End If
    
    GetQtyOnActivePOs = ldblQty
    Exit Function

Error_Handler:
    llngErrNo = IIf(llngErrNo = 0, Err.Number, llngErrNo)
    lstrErrDescr = IIf(Len(lstrErrDescr) = 0, Err.Description, lstrErrDescr & vbCrLf & Err.Description)
    lstrErrSource = IIf(Len(Trim(Err.Source)) = 0, "", Err.Source & vbCrLf) & cstrErrorSource
    Call Err.Raise(llngErrNo, lstrErrSource, lstrErrDescr)

End Function

    Dim lobjINVUM As ADODB.Recordset

Public Sub ExportToText(ByVal lstrFilepath As String, lstrExcluedDesc As String, lstrExcludedPN As String)
    On Error GoTo Error_Handler
    Const cstrErrorSource As String = cstrApplicationName & "." & cstrModuleName & ".ExportToText()"
    Dim llngErrNo As Long
    Dim lstrErrDescr As String
    Dim lstrErrSource As String
    llngErrNo = 0
    lstrErrDescr = ""
    lstrErrSource = ""

    Dim lstrSQL As String
    Dim lobjRS As ADODB.Recordset
    Dim llngAffected As Long
    Dim i As Integer
    Dim lstrLine As String
    Dim lstrPriceField As String
    Dim ldblPrice As Double
    
    Select Case intCheck
    Case 1: lstrPriceField = "G"
    Case 2: lstrPriceField = "I"
    Case 3: lstrPriceField = "N"
    Case 4: lstrPriceField = "T"
    
        
    Case 5: lstrPriceField = "J"
    Case 6: lstrPriceField = "M"
    Case 7: lstrPriceField = "S"
    Case 8: lstrPriceField = "V"

    
    End Select
    
    lstrSQL = "Select INV.PartNum as PN ,MAIN.PartNLocal as PNL,"
    lstrSQL = lstrSQL & IIf(MeasU = 0, "MAIN.DescMet as D1", "MAIN.DescImp as D1")
    lstrSQL = lstrSQL & ",MAIN.DescOther as DescO, INV.InvUM as UM, INV.QtyInStock as InStock ,INV.QtyOnPOs as OnPos"
    lstrSQL = lstrSQL & ",INV.QtyOnOrder  as OnOrder, Inv.QtyBkOrder as BO, INV.QtyReserved as ReservedQ, Main.UM as UMM, Main.Length as Length, Inv.QtyMin as MinQty, Inv.RestockingLevel as RestckLev,Main." & lstrPriceField & " as Cost From Inventory INV"
    lstrSQL = lstrSQL & " LEFT JOIN Main Main ON MAIN.PartNum = INV.PartNum"
    lstrSQL = lstrSQL & " order by Inv.PartNum"
    Set lobjRS = New ADODB.Recordset
    
    lobjRS.CursorLocation = adUseClient 'open disconnected recordset
    lobjRS.Open lstrSQL, CnnBTMS, , adLockBatchOptimistic, adCmdText
    If Not lobjRS Is Nothing Then
        If Not (lobjRS.EOF And lobjRS.BOF) Then
            Set lobjRS.ActiveConnection = Nothing
            Open lstrFilepath For Output As #1   ' Open file for output.
            lstrLine = "Part Number"
            lstrLine = lstrLine & vbTab & "Description"
            lstrLine = lstrLine & vbTab & "UM"
            lstrLine = lstrLine & vbTab & "In Stock"
            lstrLine = lstrLine & vbTab & "On POs"
            lstrLine = lstrLine & vbTab & "On Order"
            lstrLine = lstrLine & vbTab & "B.O."
            lstrLine = lstrLine & vbTab & "Reserved"
            lstrLine = lstrLine & vbTab & "Minimum"
            lstrLine = lstrLine & vbTab & "Restocking Level"
            lstrLine = lstrLine & vbTab & "Inventory Value"
            Print #1, lstrLine
            lobjRS.MoveFirst
            Do While lobjRS.EOF = False
                lstrLine = ""
                
                ldblPrice = Decrypt(GetSym(lobjRS.Fields("PN").Value), lobjRS.Fields("Cost").Value)

                If lobjRS.Fields("UMM").Value = "LFT" Then
                    ldblPrice = (ldblPrice / 12) * DBLValue(lobjRS.Fields("InStock").Value)
                ElseIf lobjRS.Fields("UM").Value = "INCHES" And lobjRS.Fields("UMM").Value = "EACH" Then
                    ldblPrice = (ldblPrice / DBLValue(lobjRS.Fields("Length").Value)) * DBLValue(lobjRS.Fields("InStock").Value)
                Else
                    ldblPrice = ldblPrice * DBLValue(lobjRS.Fields("InStock").Value)
                End If
                lobjRS.Fields("Cost").Value = ldblPrice
                lobjRS.Update
            lobjRS.MoveNext
            Loop
            
            lobjRS.MoveFirst
            Do While lobjRS.EOF = False
                lstrLine = ""
                For i = 0 To lobjRS.Fields.Count - 1
                    
                    If lobjRS.Fields.Item(i).Name <> lstrExcludedPN And _
                    lobjRS.Fields.Item(i).Name <> lstrExcluedDesc And _
                    lobjRS.Fields.Item(i).Name <> "UMM" And _
                    lobjRS.Fields.Item(i).Name <> "Length" Then
                        If lstrLine <> "" Then lstrLine = lstrLine & vbTab
                        lstrLine = lstrLine & lobjRS.Fields(i)
                    End If
                Next
                Print #1, lstrLine
                lobjRS.MoveNext
            Loop
            Close #1
        End If
    End If

    Exit Sub

Error_Handler:
    llngErrNo = IIf(llngErrNo = 0, Err.Number, llngErrNo)
    lstrErrDescr = IIf(Len(lstrErrDescr) = 0, Err.Description, lstrErrDescr & vbCrLf & Err.Description)
    lstrErrSource = IIf(Len(Trim(Err.Source)) = 0, "", Err.Source & vbCrLf) & cstrErrorSource
    Call Err.Raise(llngErrNo, lstrErrSource, lstrErrDescr)

End Sub

                            lobjRS.Fields.Item(i).Name <> "S" Then
                            
                            If lstrLine <> "" Then lstrLine = lstrLine & vbTab
                            lstrLine = lstrLine & lobjRS.Fields(i)
                        End If
                    Next
                    
                End If
                Print #1, lstrLine
                lobjRS.MoveNext
            Loop
            Close #1
        End If
    End If

    Exit Sub

Error_Handler:
    llngErrNo = IIf(llngErrNo = 0, Err.Number, llngErrNo)
    lstrErrDescr = IIf(Len(lstrErrDescr) = 0, Err.Description, lstrErrDescr & vbCrLf & Err.Description)
    lstrErrSource = IIf(Len(Trim(Err.Source)) = 0, "", Err.Source & vbCrLf) & cstrErrorSource
    Call Err.Raise(llngErrNo, lstrErrSource, lstrErrDescr)

End Sub
