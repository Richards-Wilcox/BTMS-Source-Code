Private Sub Form_Load()

    On Error GoTo Error_Handler
    Const cstrErrorSource As String = cstrApplicationName & "." & cstrModuleName & ".Form_Load()"
    Dim llngErrNo As Long
    Dim lstrErrDescr As String
    Dim lstrErrSource As String
    llngErrNo = 0
    lstrErrDescr = ""
    lstrErrSource = ""

    Dim currBPP As Long
    currBPP = GetDeviceCaps(hdc, BITSPIXEL)
    If currBPP <= 8 Then Call GoGrey(Me)
    If WgtLock = True Then
        Me.chkLock.Value = 1
    Else
        chkLock.Value = 0
    End If
    
    Me.Caption = IIf(MeasU = 1, "Weight Summary in Pounds", "Weight Summary in Kilograms")

    Grid1.Clear
'    Grid1.Left = (Me.Width - Grid1.Width) / 2
    Grid1.ColWidth(0) = Grid1.Width * 0.25
    Grid1.ColWidth(1) = Grid1.Width * 0.13
    Grid1.ColWidth(2) = Grid1.Width * 0.13
    Grid1.ColWidth(3) = Grid1.Width - Grid1.ColWidth(0) - Grid1.ColWidth(1) - Grid1.ColWidth(2) - 100
    
    Grid1.TextMatrix(0, 1) = "Calculated"
    Grid1.TextMatrix(0, 2) = "New"
    Grid1.TextMatrix(0, 3) = "Reason for change"
    
    Grid1.TextMatrix(1, 0) = "Sections Weight"
    Grid1.TextMatrix(2, 0) = "Lite Weight"
    Grid1.TextMatrix(3, 0) = "Pass Door Weight"
    Grid1.TextMatrix(4, 0) = "Hardware Weight"
    Grid1.TextMatrix(5, 0) = "Extra Wgt on Bottom"
    Grid1.TextMatrix(6, 0) = "Truss Weight"
    Grid1.TextMatrix(7, 0) = "Total Weight"
    
    If MeasU = 1 Then
        Grid1.TextMatrix(1, 1) = Format(SecWgtOld, "Fixed") 'Format(IIf(SecWgtOld = 0, SecWgt, SecWgtOld), "Fixed")
        Grid1.TextMatrix(2, 1) = Format(LiteWgtOld, "Fixed") ' Format(IIf(LiteWgtOld = 0, LiteWgt, LiteWgtOld), "Fixed")
        Grid1.TextMatrix(3, 1) = Format(varPassDoor.sngWgtOld, "Fixed") 'Format(IIf(varPassDoor.sngWgtOld = 0, varPassDoor.sngWgt, varPassDoor.sngWgtOld), "Fixed")
        Grid1.TextMatrix(4, 1) = Format(HWWgtOld, "Fixed") 'Format(IIf(HWWgtOld = 0, HWWgt, HWWgtOld), "Fixed")
        Grid1.TextMatrix(5, 1) = Format(MRSBWgtOld, "Fixed") 'Format(IIf(MRSBWgtOld = 0, MRSBWgt, MRSBWgtOld), "Fixed")
        Grid1.TextMatrix(6, 1) = Format(TrussWgtOld, "Fixed") 'Format(IIf(TrussWgtOld = 0, TrussWgt, TrussWgtOld), "Fixed")
        Grid1.TextMatrix(1, 2) = Format(SecWgt, "Fixed")
        Grid1.TextMatrix(2, 2) = Format(LiteWgt, "Fixed")
        Grid1.TextMatrix(3, 2) = Format(varPassDoor.sngWgt, "Fixed")
        Grid1.TextMatrix(4, 2) = Format(HWWgt, "Fixed")
        Grid1.TextMatrix(5, 2) = Format(MRSBWgt, "Fixed")
        Grid1.TextMatrix(6, 2) = Format(TrussWgt, "Fixed")
    Else
    
        aCon.LB = SecWgtOld
        aCon.KGcalc
        Grid1.TextMatrix(1, 1) = Format(aCon.KG, "Fixed")
        aCon.LB = LiteWgtOld
        aCon.KGcalc
        Grid1.TextMatrix(2, 1) = Format(aCon.KG, "Fixed")
        aCon.LB = varPassDoor.sngWgtOld
        aCon.KGcalc
        Grid1.TextMatrix(3, 1) = Format(aCon.KG, "Fixed")
        aCon.LB = HWWgtOld
        aCon.KGcalc
        Grid1.TextMatrix(4, 1) = Format(aCon.KG, "Fixed")
        aCon.LB = MRSBWgtOld
        aCon.KGcalc
        Grid1.TextMatrix(5, 1) = Format(aCon.KG, "Fixed")
        aCon.LB = TrussWgtOld
        aCon.KGcalc
        Grid1.TextMatrix(6, 1) = Format(aCon.KG, "Fixed")
    
    
    
    
    
    
        aCon.LB = SecWgt
        aCon.KGcalc
        Grid1.TextMatrix(1, 2) = Format(aCon.KG, "Fixed")
        aCon.LB = LiteWgt
        aCon.KGcalc
        Grid1.TextMatrix(2, 2) = Format(aCon.KG, "Fixed")
        aCon.LB = varPassDoor.sngWgt
        aCon.KGcalc
        Grid1.TextMatrix(3, 2) = Format(aCon.KG, "Fixed")
        aCon.LB = HWWgt
        aCon.KGcalc
        Grid1.TextMatrix(4, 2) = Format(aCon.KG, "Fixed")
        aCon.LB = MRSBWgt
        aCon.KGcalc
        Grid1.TextMatrix(5, 2) = Format(aCon.KG, "Fixed")
        aCon.LB = TrussWgt
        aCon.KGcalc
        Grid1.TextMatrix(6, 2) = Format(aCon.KG, "Fixed")

Private Sub Grid1_KeyPressEdit(ByVal Row As Long, ByVal Col As Long, KeyAscii As Integer)
    
    On Error GoTo Error_Handler
    Const cstrErrorSource As String = cstrApplicationName & "." & cstrModuleName & ".Grid1_KeyPressEdit()"
    Dim llngErrNo As Long
    Dim lstrErrDescr As String
    Dim lstrErrSource As String
    llngErrNo = 0
    lstrErrDescr = ""
    lstrErrSource = ""
    
    If Col = 2 Then
        If Not (Len(Grid1.Text) = Grid1.EditSelLength) And Len(Grid1.EditText) >= 6 Then
            Beep
            KeyAscii = 0
            Exit Sub
        End If
        If Not IsNumeric(Chr$(KeyAscii)) Xor KeyAscii = 8 Then
            If (DecimU = 0 And KeyAscii = 46) Or (DecimU = 1 And KeyAscii = 44) Then
                Beep
                KeyAscii = 0
            End If
        End If
    End If
    If Col = 3 Then
        If Not (Len(Grid1.Text) = Grid1.EditSelLength) And Len(Grid1.EditText) >= 30 Then
            Beep
            KeyAscii = 0
            Exit Sub
        End If
    End If
    Exit Sub

Error_Handler:
    llngErrNo = IIf(llngErrNo = 0, Err.Number, llngErrNo)
    lstrErrDescr = IIf(Len(lstrErrDescr) = 0, Err.Description, Err.Description)
    Call DisplayError(llngErrNo, IIf(Len(Trim(Err.Source)) = 0, "", Err.Source & vbCrLf) & cstrErrorSource, lstrErrDescr)

End Sub
