Private Sub Form_Unload(Cancel As Integer)

    If Not gobjAgent Is Nothing Then
        If gobjAgent.ActorIsVisible = True Then
            gobjAgent.ActorStopAll
            gobjAgent.HideAgent
        End If
    End If

End Sub
Private Sub MousePos1_MouseOver(ControlName As String, Index As Long, CtrlData As String, PosX As Long, PosY As Long)
    On Error GoTo Error_Handler
    Const cstrErrorSource As String = cstrApplicationName & "." & cstrModuleName & ".MousePos1_MouseOver()"
    Dim llngErrNo As Long
    Dim lstrErrDescr As String
    Dim lstrErrSource As String
    llngErrNo = 0
    lstrErrDescr = ""
    lstrErrSource = ""
    
    If Me.Visible = False Then Exit Sub
    If gobjAgent.ActorIsLoaded = False Then Exit Sub
    
    If gobjAgent.ActorIsVisible = True Then
        If Not CtrlData = "" Then Call AgentFieldHelp(PosX, PosY, CtrlData)
    End If
    Exit Sub

Error_Handler:
    llngErrNo = IIf(llngErrNo = 0, Err.Number, llngErrNo)
    lstrErrDescr = IIf(Len(lstrErrDescr) = 0, Err.Description, lstrErrDescr & vbCrLf & Err.Description)
    lstrErrSource = IIf(Len(Trim(Err.Source)) = 0, "", Err.Source & vbCrLf) & cstrErrorSource
    Call Err.Raise(llngErrNo, lstrErrSource, lstrErrDescr)
End Sub

Private Sub cmdAccept_Click()

    On Error GoTo Error_Handler
    Const cstrErrorSource As String = cstrApplicationName & "." & cstrModuleName & ".cmdAccept_Click()"
    Dim llngErrNo As Long
    Dim lstrErrDescr As String
    Dim lstrErrSource As String
    llngErrNo = 0
    lstrErrDescr = ""
    lstrErrSource = ""
    Screen.MousePointer = vbHourglass
    frmSprSolutions.Grid1.ColSort(cboSort1.ItemData(cboSort1.ListIndex)) = IIf(optAscending1.Value = True, flexSortGenericAscending, flexSortGenericDescending)
    If cboSort2.ItemData(cboSort2.ListIndex) > cboSort1.ItemData(cboSort1.ListIndex) Then
        frmSprSolutions.Grid1.ColSort(cboSort2.ItemData(cboSort2.ListIndex)) = IIf(optAscending2.Value = True, flexSortGenericAscending, flexSortGenericDescending)
        frmSprSolutions.Grid1.Select 1, cboSort1.ItemData(cboSort1.ListIndex), 1, cboSort2.ItemData(cboSort2.ListIndex)
    Else
        frmSprSolutions.Grid1.Select 1, cboSort1.ItemData(cboSort1.ListIndex) ', 1, cboSort2.ItemData(cboSort2.ListIndex)
    End If
    
    frmSprSolutions.Grid1.Sort = flexSortUseColSort
    Unload Me
    frmSprSolutions.Show
    Screen.MousePointer = vbDefault
    Exit Sub

Error_Handler:
    Screen.MousePointer = vbDefault
    llngErrNo = IIf(llngErrNo = 0, Err.Number, llngErrNo)
    lstrErrDescr = IIf(Len(lstrErrDescr) = 0, Err.Description, lstrErrDescr & vbCrLf & Err.Description)
    Call DisplayError(llngErrNo, IIf(Len(Trim(Err.Source)) = 0, "", Err.Source & vbCrLf) & cstrErrorSource, lstrErrDescr)

End Sub
