Option Explicit
Const cstrModuleName As String = "frmSecRemoveSpItems"
Private mblnInFocus As Boolean

Private Sub cmdCancel_Click()
    On Error GoTo Error_Handler
    Const cstrErrorSource As String = cstrApplicationName & "." & cstrModuleName & ".cmdCancel_Click()"
    Dim llngErrNo As Long
    Dim lstrErrDescr As String
    Dim lstrErrSource As String
    llngErrNo = 0
    lstrErrDescr = ""
    lstrErrSource = ""
    
    Screen.MousePointer = vbHourglass
    Unload Me
    Screen.MousePointer = vbDefault
    Exit Sub

Error_Handler:
    Screen.MousePointer = vbDefault
    llngErrNo = IIf(llngErrNo = 0, Err.Number, llngErrNo)
    lstrErrDescr = IIf(Len(lstrErrDescr) = 0, Err.Description, lstrErrDescr & vbCrLf & Err.Description)
    Call DisplayError(llngErrNo, IIf(Len(Trim(Err.Source)) = 0, "", Err.Source & vbCrLf) & cstrErrorSource, lstrErrDescr)

End Sub

Private Sub cmdRemove_Click()
    On Error GoTo Error_Handler
    Const cstrErrorSource As String = cstrApplicationName & "." & cstrModuleName & ".cmdRemove_Click()"
    Dim llngErrNo As Long
    Dim lstrErrDescr As String
    Dim lstrErrSource As String
    llngErrNo = 0
    lstrErrDescr = ""
    lstrErrSource = ""
    Screen.MousePointer = vbHourglass
    Dim i As Integer, lstrPN As String, lintInd As Integer
    Dim larrTemp() As SpecialItem
    Dim UpperBound As Integer, lintCntr As Integer
    
    lintInd = -1
    For i = 0 To lstSpItems.ListCount - 1
        If lstSpItems.Selected(i) = True Then
            lstrPN = Mid(lstSpItems.Text, 1, InStr(1, lstSpItems.Text, vbTab) - 1)
            lintInd = i
            Exit For
        End If
    Next
    If Not isEmptyArraySpItems Then
        For i = 0 To UBound(varSpItem())
            If varSpItem(i).strPN <> lstrPN Then
                ReDim Preserve larrTemp(lintCntr)
                larrTemp(lintCntr).sngLength = varSpItem(i).sngLength
                larrTemp(lintCntr).sngPrice = varSpItem(i).sngPrice
                larrTemp(lintCntr).sngPrice = varSpItem(i).sngPrice
                larrTemp(lintCntr).sngQty = varSpItem(i).sngQty
                larrTemp(lintCntr).sngShippingWgt = varSpItem(i).sngShippingWgt
                larrTemp(lintCntr).sngSpringingWgt = varSpItem(i).sngSpringingWgt
                larrTemp(lintCntr).strDescription = varSpItem(i).strDescription
                larrTemp(lintCntr).strNotes = varSpItem(i).strNotes
                larrTemp(lintCntr).strPN = varSpItem(i).strPN
                larrTemp(lintCntr).strUM = varSpItem(i).strUM
                lintCntr = lintCntr + 1
            End If
        Next
        ReDim varSpItem(0)
        Erase varSpItem
        If lintCntr > 0 Then
            ReDim varSpItem(UBound(larrTemp))
            For i = 0 To UBound(larrTemp())
                varSpItem(i).sngLength = larrTemp(i).sngLength
                varSpItem(i).sngPrice = larrTemp(i).sngPrice
                varSpItem(i).sngPrice = larrTemp(i).sngPrice
                varSpItem(i).sngQty = larrTemp(i).sngQty
                varSpItem(i).sngShippingWgt = larrTemp(i).sngShippingWgt
                varSpItem(i).sngSpringingWgt = larrTemp(i).sngSpringingWgt
                varSpItem(i).strDescription = larrTemp(i).strDescription
                varSpItem(i).strNotes = larrTemp(i).strNotes
                varSpItem(i).strPN = larrTemp(i).strPN
                varSpItem(i).strUM = larrTemp(i).strUM
            Next
        End If
    End If
    ReDim larrTemp(0)
    Erase larrTemp
    If lintInd > -1 Then lstSpItems.RemoveItem (lintInd)
    Screen.MousePointer = vbDefault
    Exit Sub

Error_Handler:
    Screen.MousePointer = vbDefault
    llngErrNo = IIf(llngErrNo = 0, Err.Number, llngErrNo)
    lstrErrDescr = IIf(Len(lstrErrDescr) = 0, Err.Description, lstrErrDescr & vbCrLf & Err.Description)
    Call DisplayError(llngErrNo, IIf(Len(Trim(Err.Source)) = 0, "", Err.Source & vbCrLf) & cstrErrorSource, lstrErrDescr)

End Sub

Private Sub Form_Load()
    On Error GoTo Error_Handler
    Const cstrErrorSource As String = cstrApplicationName & "." & cstrModuleName & ".Form_Load()"
    Dim llngErrNo As Long
    Dim lstrErrDescr As String
    Dim lstrErrSource As String
    llngErrNo = 0
    lstrErrDescr = ""
    lstrErrSource = ""
    Dim currBPP As Long
    currBPP = GetDeviceCaps(hdc, BITSPIXEL)
    If currBPP <= 8 Then Call GoGrey(Me)
    Me.Move Screen.Width / 2 - Me.Width / 2, Screen.Height / 2 - Me.Height / 2
    Dim i As Integer
    If Not isEmptyArraySpItems Then
        For i = 0 To UBound(varSpItem)
           lstSpItems.AddItem varSpItem(i).strPN & vbTab & varSpItem(i).strDescription
        Next
    End If
    LoadHelpText Me
    MousePos1.Init Me
    mblnInFocus = True
    Exit Sub

Error_Handler:
    llngErrNo = IIf(llngErrNo = 0, Err.Number, llngErrNo)
    lstrErrDescr = IIf(Len(lstrErrDescr) = 0, Err.Description, lstrErrDescr & vbCrLf & Err.Description)
    Call DisplayError(llngErrNo, IIf(Len(Trim(Err.Source)) = 0, "", Err.Source & vbCrLf) & cstrErrorSource, lstrErrDescr)

End Sub

End Sub

Private Sub MousePos1_MouseOver(ControlName As String, Index As Long, CtrlData As String, PosX As Long, PosY As Long)
    On Error GoTo Error_Handler
    Const cstrErrorSource As String = cstrApplicationName & "." & cstrModuleName & ".MousePos1_MouseOver()"
    Dim llngErrNo As Long
    Dim lstrErrDescr As String
    Dim lstrErrSource As String
    llngErrNo = 0
    lstrErrDescr = ""
    lstrErrSource = ""
    
    If Me.Visible = False Or mblnInFocus = False Then Exit Sub
    If gobjAgent.ActorIsLoaded = False Then Exit Sub
    
    If gobjAgent.ActorIsVisible = True Then
        If Not CtrlData = "" Then Call AgentFieldHelp(PosX, PosY, CtrlData)
    End If
    Exit Sub

Error_Handler:
    llngErrNo = IIf(llngErrNo = 0, Err.Number, llngErrNo)
    lstrErrDescr = IIf(Len(lstrErrDescr) = 0, Err.Description, lstrErrDescr & vbCrLf & Err.Description)
    lstrErrSource = IIf(Len(Trim(Err.Source)) = 0, "", Err.Source & vbCrLf) & cstrErrorSource
    Call Err.Raise(llngErrNo, lstrErrSource, lstrErrDescr)
End Sub
