Option Explicit
Const cstrModuleName As String = "frmInvPrint"
Private mblnInFocus As Boolean

Private Sub cmdCancel_Click()
    On Error GoTo Error_Handler
    Const cstrErrorSource As String = cstrApplicationName & "." & cstrModuleName & ".cmdCancel_Click()"
    Dim llngErrNo As Long
    Dim lstrErrDescr As String
    Dim lstrErrSource As String
    llngErrNo = 0
    lstrErrDescr = ""
    lstrErrSource = ""
    
    Unload Me
    
    Exit Sub

Error_Handler:
    llngErrNo = IIf(llngErrNo = 0, Err.Number, llngErrNo)
    lstrErrDescr = IIf(Len(lstrErrDescr) = 0, Err.Description, lstrErrDescr & vbCrLf & Err.Description)
    Call DisplayError(llngErrNo, IIf(Len(Trim(Err.Source)) = 0, "", Err.Source & vbCrLf) & cstrErrorSource, lstrErrDescr)

End Sub

Private Sub cmdPrint_Click()
    On Error GoTo Error_Handler
    Const cstrErrorSource As String = cstrApplicationName & "." & cstrModuleName & ".cmdPrint_Click()"
    Dim llngErrNo As Long
    Dim lstrErrDescr As String
    Dim lstrErrSource As String
    llngErrNo = 0
    lstrErrDescr = ""
    lstrErrSource = ""

    If Me.optPNDefault = True Then
        frmPrint.mblnPNValue = True 'default PN
    Else
        frmPrint.mblnPNValue = False
    End If
    
    If Me.optDescDefault = True Then
        frmPrint.mblnDefaultDescr = True 'default Desc
    Else
        frmPrint.mblnDefaultDescr = False
    End If
    
    If Me.optBelowMin = True Then
        frmPrint.mblnInStock = True 'In Stock Report (items where stock >0)
    Else
        frmPrint.mblnInStock = False
    End If
    
    If Me.optInStock = True Then
        frmPrint.mblnBelowMin = True 'Below Min Report (items where stock < = QtyMin)
    Else
        frmPrint.mblnBelowMin = False
    End If
    
    frmPrint.mstrReportName = "Inventory Listing.rpt"
    frmPrint.mstrTitle = "Inventory Listing"
    
    Unload Me
    frmPrint.Show vbModal
    
    Exit Sub

Error_Handler:
    llngErrNo = IIf(llngErrNo = 0, Err.Number, llngErrNo)
    lstrErrDescr = IIf(Len(lstrErrDescr) = 0, Err.Description, lstrErrDescr & vbCrLf & Err.Description)
    Call DisplayError(llngErrNo, IIf(Len(Trim(Err.Source)) = 0, "", Err.Source & vbCrLf) & cstrErrorSource, lstrErrDescr)
    
End Sub
Private Sub Form_KeyDown(KeyCode As Integer, Shift As Integer)
    On Error GoTo Error_Handler
    Const cstrErrorSource As String = cstrApplicationName & "." & cstrModuleName & ".Form_KeyDown()"
    Dim llngErrNo As Long
    Dim lstrErrDescr As String
    Dim lstrErrSource As String
    llngErrNo = 0
    lstrErrDescr = ""
    lstrErrSource = ""
    
    If KeyCode = 112 Then
        ShowHideAgent Me, "Hello!"
    End If

    Exit Sub

Error_Handler:
    llngErrNo = IIf(llngErrNo = 0, Err.Number, llngErrNo)
    lstrErrDescr = IIf(Len(lstrErrDescr) = 0, Err.Description, lstrErrDescr & vbCrLf & Err.Description)
    Call DisplayError(llngErrNo, IIf(Len(Trim(Err.Source)) = 0, "", Err.Source & vbCrLf) & cstrErrorSource, lstrErrDescr)
End Sub

Private Sub MousePos1_MouseOver(ControlName As String, Index As Long, CtrlData As String, PosX As Long, PosY As Long)
    On Error GoTo Error_Handler
    Const cstrErrorSource As String = cstrApplicationName & "." & cstrModuleName & ".MousePos1_MouseOver()"
    Dim llngErrNo As Long
    Dim lstrErrDescr As String
    Dim lstrErrSource As String
    llngErrNo = 0
    lstrErrDescr = ""
    lstrErrSource = ""
    
    If Me.Visible = False Or mblnInFocus = False Then Exit Sub
    If gobjAgent.ActorIsLoaded = False Then Exit Sub
    
    If gobjAgent.ActorIsVisible = True Then
        If Not CtrlData = "" Then Call AgentFieldHelp(PosX, PosY, CtrlData)
    End If
    Exit Sub

Error_Handler:
    llngErrNo = IIf(llngErrNo = 0, Err.Number, llngErrNo)
    lstrErrDescr = IIf(Len(lstrErrDescr) = 0, Err.Description, lstrErrDescr & vbCrLf & Err.Description)
    lstrErrSource = IIf(Len(Trim(Err.Source)) = 0, "", Err.Source & vbCrLf) & cstrErrorSource
    Call Err.Raise(llngErrNo, lstrErrSource, lstrErrDescr)
End Sub


Private Sub Form_Load()
    On Error GoTo Error_Handler
    Const cstrErrorSource As String = cstrApplicationName & "." & cstrModuleName & ".Form_Load()"
    Dim llngErrNo As Long
    Dim lstrErrDescr As String
    Dim lstrErrSource As String
    llngErrNo = 0
    lstrErrDescr = ""
    lstrErrSource = ""
    Dim currBPP As Long
    
    Me.Move Screen.Width / 2 - Me.Width / 2, Screen.Height / 2 - Me.Height / 2
    
    currBPP = GetDeviceCaps(hdc, BITSPIXEL)
    If currBPP <= 8 Then Call GoGrey(Me)
    LoadHelpText Me
    MousePos1.Init Me
    mblnInFocus = True
    
    Exit Sub

Error_Handler:
    llngErrNo = IIf(llngErrNo = 0, Err.Number, llngErrNo)
    lstrErrDescr = IIf(Len(lstrErrDescr) = 0, Err.Description, lstrErrDescr & vbCrLf & Err.Description)
    Call DisplayError(llngErrNo, IIf(Len(Trim(Err.Source)) = 0, "", Err.Source & vbCrLf) & cstrErrorSource, lstrErrDescr)

End Sub
