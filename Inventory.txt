    lstrErrSource = ""

    Dim lstrSQLQuery As String
    Dim lobjInventoryRS As ADODB.Recordset
    
    mintMode = MODE_VIEW
    Call SetControls(mintMode)

    lstrSQLQuery = "Select MAIN.PartNum,MAIN.PartNLocal,"
    lstrSQLQuery = lstrSQLQuery & IIf(MeasU = 0, "MAIN.DescMet", "MAIN.DescImp")

    Set lobjInventoryRS = CnnBTMS.Execute(lstrSQLQuery, llngAffected, adCmdText)

        lobjInventoryRS.ActiveConnection = Nothing
        If Not (lobjInventoryRS.BOF And lobjInventoryRS.EOF) Then
            lobjInventoryRS.MoveFirst

            lstrFieldName = IIf(MeasU = 0, "DescMet", "DescImp")

            lstrFieldName = "PhantomPN"
            chkPhantom.Value = IIf(lobjInventoryRS.Fields(lstrFieldName).Value = True, 1, 0)
            lstrFieldName = "CallPrice"
            chkCall.Value = IIf(lobjInventoryRS.Fields(lstrFieldName).Value = True, 1, 0)
            lstrFieldName = "INVUM"

            If MeasU = 0 And txtUM.Text = "INCHES" Then txtUM.Text = "MM"
            
            lstrFieldName = "SpringingWeight"
            If Not IsNull(lobjInventoryRS.Fields(lstrFieldName).Value) Then
                If MeasU = 0 Then
                    ClearConObj
                    aCon.LB = lobjInventoryRS.Fields(lstrFieldName).Value
                    aCon.KGcalc
                    lblWeight.Caption = Format(aCon.KG, cstrQtyMask)
                Else
                    lblWeight.Caption = Format(lobjInventoryRS.Fields(lstrFieldName).Value, cstrQtyMask)

            Else

            End If
            
            lstrFieldName = "Height"
            If Not IsNull(lobjInventoryRS.Fields(lstrFieldName).Value) Then
                If MeasU = 0 Then
                    ClearConObj
                    aCon.Inch = lobjInventoryRS.Fields(lstrFieldName).Value
                    aCon.mmcalc
                    lblHeight.Caption = Round(aCon.mm, 0)
                Else
                    lblHeight.Caption = Format(lobjInventoryRS.Fields(lstrFieldName).Value, cstrQtyMask)
                End If
            Else

                Else
                    lblLength.Caption = Format(lobjInventoryRS.Fields(lstrFieldName).Value, cstrQtyMask)
                End If
            Else
                lblLength.Caption = Format(0, "Fixed")
            End If
            
            lstrFieldName = "ShipWeight"
            If Not IsNull(lobjInventoryRS.Fields(lstrFieldName).Value) Then
                If MeasU = 0 Then
                    ClearConObj
                    aCon.LB = lobjInventoryRS.Fields(lstrFieldName).Value
                    aCon.KGcalc

                    txtBackOrder.Text = Format(lobjInventoryRS.Fields(lstrFieldName).Value, cstrQtyMask)

            Else
                txtBackOrder.Text = Format(0, "Fixed")

            lstrFieldName = "QtyOnOrder"

                If MeasU = 0 And lobjInventoryRS.Fields("INVUM").Value = "INCHES" Then

Private Sub SaveData()
    On Error GoTo Error_Handler
    Const cstrErrorSource As String = cstrApplicationName & "." & cstrModuleName & ".SaveData()"
    Dim llngErrNo As Long
    Dim lstrErrDescr As String
    Dim lstrErrSource As String
    llngErrNo = 0
    lstrErrDescr = ""
    lstrErrSource = ""

    Dim lstrSQL As String
    Dim lstrWhereClause As String
    Dim lstrExcludeFieldList As String
    Dim llngAffected As Long
    Dim lobjADORS As ADODB.Recordset
    
    ' Copy the fields from the screen to an ADO Recordset
    Set lobjADORS = BuildRS()
    
    ' Process through the recordset and insert or update rows
    If Not lobjADORS Is Nothing Then
        If Not (lobjADORS.BOF And lobjADORS.EOF) Then
            lobjADORS.MoveFirst
            lstrExcludeFieldList = "InventoryID"
            Do While lobjADORS.EOF = False
                ' Check to see if the record exists in the
                ' Inventory table already
                Dim lobjParentRS As ADODB.Recordset
                mintMode = MODE_ADD
                lstrSQL = "select PartNum from Inventory where PartNum=" & SQLString(mstrPartNum)
                Set lobjParentRS = CnnBTMS.Execute(lstrSQL, llngAffected, adCmdText)
                If Not lobjParentRS Is Nothing Then
                    If Not (lobjParentRS.BOF And lobjParentRS.EOF) Then
                        lobjParentRS.MoveFirst
                        ' Reset the Mode so that we do an update of the existing part
                        mintMode = MODE_EDIT
                    End If
                    lobjParentRS.Close
                End If
                Set lobjParentRS = Nothing
            
                If mintMode = MODE_ADD Then
                    lstrErrDescr = "Error building insert statement for " & cstrTableName
                    lstrSQL = Build_Insert(lobjADORS, cstrTableName, lstrExcludeFieldList)
                    lstrErrDescr = "Error inserting data into " & cstrTableName
                Else
                    lstrErrDescr = "Error building update statement for " & cstrTableName
                    lstrWhereClause = "PartNum=" & SQLString(mstrPartNum)
                    lstrSQL = Build_Update(lobjADORS, cstrTableName, lstrWhereClause, lstrExcludeFieldList)
                    lstrErrDescr = "Error updating " & cstrTableName
                End If
                
                Call CnnBTMS.Execute(lstrSQL, 0, adCmdText And adExecuteNoRecords)
                lstrErrDescr = ""
                lobjADORS.MoveNext
            Loop
            
            lobjADORS.Close
        End If
    End If
    
    Set lobjADORS = Nothing
    Exit Sub

Error_Handler:
    llngErrNo = IIf(llngErrNo = 0, Err.Number, llngErrNo)
    lstrErrDescr = IIf(Len(lstrErrDescr) = 0, Err.Description, Err.Description)
    lstrErrSource = IIf(Len(Trim(Err.Source)) = 0, "", Err.Source & vbCrLf) & cstrErrorSource
    Call Err.Raise(llngErrNo, lstrErrSource, lstrErrDescr)
End Sub

Private Function BuildRS() As ADODB.Recordset
    On Error GoTo Error_Handler
    Const cstrErrorSource As String = cstrApplicationName & "." & cstrModuleName & ".BuildRS()"
    Dim llngErrNo As Long
    Dim lstrErrDescr As String
    Dim lstrErrSource As String
    llngErrNo = 0
    lstrErrDescr = ""
    lstrErrSource = ""

    Dim lstrSQL As String
    Dim lstrSchema As String
    Dim llngAffected As Long
    Dim lobjADORS As ADODB.Recordset
    
    ' Get empty recordset
    lstrErrDescr = "Error getting empty ADO recordset."
    lstrSQL = "Select * from " & cstrTableName & " where 1=0"
    Set lobjADORS = New ADODB.Recordset
    lobjADORS.CursorType = adOpenForwardOnly
    lobjADORS.CursorLocation = adUseClient
    lobjADORS.Open lstrSQL, CnnBTMS, adOpenForwardOnly, adLockOptimistic
    lobjADORS.AddNew
    Set lobjADORS.ActiveConnection = Nothing
    lstrErrDescr = ""
    
    lobjADORS.Fields("PartNum").Value = Trim(lblPartNum.Caption)
    'lobjADORS.Fields("PartNLocal").Value = Trim(lbllocal.caption)
    
'    lobjADORS.Fields("QtyBkOrder").Value = DBLValue(txtBackOrder.Text)
'    lobjADORS.Fields("QtyOnOrder").Value = DBLValue(txtOnOrder.Text)
    lobjADORS.Fields("QtyInStock").Value = DBLValue(txtInStock.Text)
'    lobjADORS.Fields("QtyMin").Value = DBLValue(txtMinQty.Text)
'    lobjADORS.Fields("QtyReserved").Value = DBLValue(txtReserved.Text)
    If Me.txtUM = "MM" Then
        ClearConObj
        If DBLValue(txtInStock.Text) > 0 Then
            aCon.mm = DBLValue(txtInStock.Text)
            aCon.Inchcalc
            lobjADORS.Fields("QtyInStock").Value = aCon.Inch
        End If
    End If
    Set BuildRS = lobjADORS
    Set lobjADORS = Nothing
    Exit Function

Error_Handler:
    llngErrNo = IIf(llngErrNo = 0, Err.Number, llngErrNo)
    lstrErrDescr = IIf(Len(lstrErrDescr) = 0, Err.Description, Err.Description)
    lstrErrSource = IIf(Len(Trim(Err.Source)) = 0, "", Err.Source & vbCrLf) & cstrErrorSource
    Call Err.Raise(llngErrNo, lstrErrSource, lstrErrDescr)
End Function
