Private Sub LoadData()
    On Error GoTo Error_Handler
    Const cstrErrorSource As String = cstrApplicationName & "." & cstrModuleName & ".LoadData()"
    Dim llngErrNo As Long
    Dim lstrErrDescr As String
    Dim lstrErrSource As String
    llngErrNo = 0
    lstrErrDescr = ""
    lstrErrSource = ""

    Dim llngAffected As Long
    Dim llngRecords As Long
    Dim lstrSQL As String
    Dim lobjPOActiveRS As ADODB.Recordset
    Dim lobjPODetailsRS As ADODB.Recordset
    Dim lobjINVRS As ADODB.Recordset
    Dim lsngQty As Single
    Dim MainUM As String
    Dim InvUM As String
    Dim i As Integer
    Screen.MousePointer = vbHourglass
    
    grdData.Clear
    llngRecords = 0
    'get set of data from inventory table, pull out description from main.
    lstrSQL = "Select INV.PartNum as PN,MAIN.PartNLocal as PNL,"
    lstrSQL = lstrSQL & IIf(MeasU = 0, "MAIN.DescMet as D1", "MAIN.DescImp as D1")
    lstrSQL = lstrSQL & ",MAIN.DescOther as DescO, INV.InvUM as UM, INV.QtyInStock as InStock ,INV.QtyOnPOs as OnPos"
    lstrSQL = lstrSQL & ",INV.QtyOnOrder as OnOrder,INV.QtyReserved as ReservedQ, Main.UM as UMM, Main.Length as Length, Inv.QtyMin as MinQty, Inv.RestockingLevel as RestckLev From Inventory INV"
    lstrSQL = lstrSQL & " LEFT JOIN Main Main ON MAIN.PartNum = INV.PartNum"
    lstrSQL = lstrSQL & " order by Inv.PartNum"

    Set mobjADORS = New ADODB.Recordset
    
    mobjADORS.CursorLocation = adUseClient 'open disconnected recordset
    mobjADORS.Open lstrSQL, CnnBTMS, , adLockBatchOptimistic, adCmdText

    If Not mobjADORS Is Nothing Then
        mobjADORS.ActiveConnection = Nothing
        If Not (mobjADORS.BOF And mobjADORS.EOF) Then
            
            mobjADORS.MoveFirst
            Do While mobjADORS.EOF = False 'loop though the data in the recordset
                lsngQty = 0
                'perform standard to metric conversion
                If MeasU = 0 Then
                    ClearConObj
                    If mobjADORS.Fields("UM").Value = "INCHES" Then
                        mobjADORS.Fields("UM").Value = "MM"

                        If mobjADORS.Fields("InStock").Value > 0 Then
                            aCon.Inch = DBLValue(mobjADORS.Fields("InStock").Value)
                            aCon.mmcalc
                            mobjADORS.Fields("InStock").Value = Round(aCon.mm, 0)
                        End If
                        If mobjADORS.Fields("OnPos").Value > 0 Then
                            aCon.Inch = DBLValue(mobjADORS.Fields("OnPos").Value)
                            aCon.mmcalc
                            mobjADORS.Fields("OnPos").Value = Round(aCon.mm, 0)
                        End If
                        If mobjADORS.Fields("OnOrder").Value > 0 Then
                            aCon.Inch = DBLValue(mobjADORS.Fields("OnOrder").Value)
                            aCon.mmcalc
                            mobjADORS.Fields("OnOrder").Value = Round(aCon.mm, 0)
                        End If
                        If mobjADORS.Fields("ReservedQ").Value > 0 Then
                            aCon.Inch = DBLValue(mobjADORS.Fields("ReservedQ").Value)
                            aCon.mmcalc
                            mobjADORS.Fields("ReservedQ").Value = Round(aCon.mm, 0)
                        End If
                        If mobjADORS.Fields("MinQty").Value > 0 Then
                            aCon.Inch = DBLValue(mobjADORS.Fields("MinQty").Value)
                            aCon.mmcalc
                            mobjADORS.Fields("MinQty").Value = Round(aCon.mm, 0)
                        End If
                        If mobjADORS.Fields("RestckLev").Value > 0 Then
                            aCon.Inch = DBLValue(mobjADORS.Fields("RestckLev").Value)
                            aCon.mmcalc
                            mobjADORS.Fields("RestckLev").Value = Round(aCon.mm, 0)
                        End If

                    End If
                End If

                mobjADORS.Update
                mobjADORS.MoveNext
            Loop
            
            'set the grid on the screen to the recordset.

Public Function ParseDocument(wField As Object, WordDoc As Object) As String
    On Error GoTo Error_Handler
    Const cstrErrorSource As String = cstrApplicationName & "." & cstrModuleName & ".GetNewResult()"
    Dim llngErrNo As Long
    Dim lstrErrDescr As String
    Dim lstrErrSource As String
    llngErrNo = 0
    lstrErrDescr = ""
    lstrErrSource = ""
    
    Dim lstrStopPos As Long
    Dim lstrWord As String
    Dim lstrText As String
    Dim lstrWordValue As String
    Dim wRange As Object
    
    ' These three lines strip down the field code to find
    ' out it's name
    lstrStopPos = InStrRev(wField.Code, "\*")
    lstrWord = Left(wField.Code, lstrStopPos - 3)
    lstrWord = Right(lstrWord, Len(lstrWord) - 14)
    Set wRange = WordDoc.Range
    
    ' Check this field hasn't already appeared in this
    ' document.
    Select Case UCase(lstrWord)
        Case "RPTTITLE":    lstrWordValue = "Inventory Listing"
            
        ' I don't simply want to insert a string -
        ' I wish to insert a table at the Product Field.
        Case "RPTLIST", "RPTHEADER"
                            
                lstrText = "Part No" & "~" & "Local Part No" & "~" & "Description" & "~" & "In Stock" & "~" & "Min Qty" & "~" & "On Order" & "~" & "Resved" & "~" & "BkOrder" & vbCrLf
                If UCase(lstrWord) = "RPTLIST" Then
                    Dim lstrSQLQuery As String
                    Dim llngAffected As String
                    lstrText = ""
                    lstrSQLQuery = "select MAIN.PartNum, MAIN.PartNLocal, MAIN.DescImp, INVENTORY.QtyInStock, "
                    lstrSQLQuery = lstrSQLQuery & "INVENTORY.QtyMin, INVENTORY.QtyOnOrder, INVENTORY.QtyReserved, INVENTORY.QtyBkOrder"
                    lstrSQLQuery = lstrSQLQuery & " from MAIN LEFT JOIN INVENTORY ON MAIN.PartNum = INVENTORY.PartNum"
                    lstrSQLQuery = lstrSQLQuery & " order by MAIN.PartNum"
                    Dim lobjINVRS As ADODB.Recordset
                    Set lobjINVRS = CnnBTMS.Execute(lstrSQLQuery, llngAffected, adCmdText)
                    
                    lstrText = " ~ ~ ~ ~ ~ ~ ~ " & vbCrLf
                    If Not lobjINVRS Is Nothing Then
                        lobjINVRS.ActiveConnection = Nothing
                        If Not (lobjINVRS.BOF And lobjINVRS.EOF) Then
                            lobjINVRS.MoveFirst
                            lstrText = ""
                            lstrText = lobjINVRS.GetString(adClipString, -1, "~", vbCrLf)
                            lobjINVRS.Close
                        End If
                    End If
                    Set lobjINVRS = Nothing
                End If
                
                ' Get the range (location) of the product field
                Set wRange = wField.Code
                ' Delete the field, as any text will be inserted into the
                ' {} of the existing field.
                wField.Delete
                
                ' Enter our table information including headers.
                ' Ideally, I would get this data from an ADO recordset
                ' using GetString().
                With wRange
            
                    .Text = lstrText
                    .FormattedText.Font.Name = "Arial"
                    .FormattedText.Font.Size = "8"
                
                    ' Once the data is there, we can convert it to a table
                    ' structure and format it to look pretty!
                    If UCase(lstrWord) = "RPTLIST" Then
                        .ConvertToTable "~", , , 50, wdTableFormatColorful2, , , , , 0
                    Else
                        .ConvertToTable "~", , , 50, wdTableFormatColorful2
                    End If
                                                   
                    .Columns(1).Width = 65
                    .Columns(2).Width = 80
                    .Columns(3).Width = 180
                End With
                
                ' Send back blank string as field does not exist anymore
                lstrWordValue = ""
    
        Case Else
            
            ' Get the value of the field from the user
            lstrWordValue = lstrWord
    
    End Select
    
    ParseDocument = lstrWordValue
    Exit Function

Error_Handler:
    llngErrNo = IIf(llngErrNo = 0, Err.Number, llngErrNo)
    lstrErrDescr = IIf(Len(lstrErrDescr) = 0, Err.Description, Err.Description)
    lstrErrSource = IIf(Len(Trim(Err.Source)) = 0, "", Err.Source & vbCrLf) & cstrErrorSource
    Call Err.Raise(llngErrNo, lstrErrSource, lstrErrDescr)
End Function
