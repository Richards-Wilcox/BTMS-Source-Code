Private Sub cmdExport_Click()
On Error GoTo Error_Handler
    Const cstrErrorSource As String = cstrApplicationName & "." & cstrModuleName & ".cmdExport_Click()"
    Dim llngErrNo As Long
    Dim lstrErrDescr As String
    Dim lstrErrSource As String
    llngErrNo = 0
    lstrErrDescr = ""
    lstrErrSource = ""
    
    CommonDialog1.DefaultExt = ".txt"
    CommonDialog1.FileName = GetTempDir & "RWInventoryListing.txt"
    CommonDialog1.Filter = "Text (*.txt)|*.txt"
    CommonDialog1.CancelError = True
    CommonDialog1.FLAGS = cdlOFNOverwritePrompt Or cdlOFNHideReadOnly
    CommonDialog1.ShowSave
    Set frmInvExportOptions.mobjParent = Me
    frmInvExportOptions.Show vbModal

        Call ExportToText_PriceLookup(CommonDialog1.FileName, lstrExclDesc, lstrExclPN)
    End If
    Exit Sub

    lstrErrDescr = IIf(Len(lstrErrDescr) = 0, Err.Description, lstrErrDescr & vbCrLf & Err.Description)
