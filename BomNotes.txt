Private Sub Form_KeyDown(KeyCode As Integer, Shift As Integer)

Private Sub Form_Load()
    On Error GoTo Error_Handler
    Const cstrErrorSource As String = cstrApplicationName & "." & cstrModuleName & ".Form_Load()"
    Dim llngErrNo As Long
    Dim lstrErrDescr As String
    Dim lstrErrSource As String
    llngErrNo = 0
    lstrErrDescr = ""
    lstrErrSource = ""

    currBPP = GetDeviceCaps(hdc, BITSPIXEL)

    Me.Move Screen.Width / 2 - Me.Width / 2, (Screen.Height / 2 - Me.Height / 2) - 500

Private Sub LoadData()
    On Error GoTo Error_Handler
    Const cstrErrorSource As String = cstrApplicationName & "." & cstrModuleName & ".LoadData()"
    Dim llngErrNo As Long
    Dim lstrErrDescr As String
    Dim lstrErrSource As String
    llngErrNo = 0
    lstrErrDescr = ""
    lstrErrSource = ""

    mintMode = MODE_VIEW
    
    If mlngBOMID <= 0 Then Exit Sub
    
    Dim llngAffected As Long
    Dim lstrSQLQuery As String
    Dim lobjBOMRS As ADODB.Recordset
    
    lstrSQLQuery = "select * from BoM where BOMID=" & mlngBOMID
    Set lobjBOMRS = CnnBTMS.Execute(lstrSQLQuery, llngAffected, adCmdText)
    If Not lobjBOMRS Is Nothing Then
        lobjBOMRS.ActiveConnection = Nothing
        If Not (lobjBOMRS.BOF And lobjBOMRS.EOF) Then
            lobjBOMRS.MoveFirst
            Dim lstrFieldName As String
            lstrFieldName = "BOMNum"
            lblBOMNum.Caption = IIf(IsNull(lobjBOMRS.Fields(lstrFieldName).Value), mstrDocNum, Trim(lobjBOMRS.Fields(lstrFieldName).Value))
            lstrFieldName = "BOMDate"
            lblDate.Caption = IIf(IsNull(lobjBOMRS.Fields(lstrFieldName).Value), "", Format(lobjBOMRS.Fields(lstrFieldName).Value, "Short Date"))
            lstrFieldName = "CustName"
            lblName.Caption = IIf(IsNull(lobjBOMRS.Fields(lstrFieldName).Value), "", Trim(lobjBOMRS.Fields(lstrFieldName).Value))
            
            lstrFieldName = "BOMAdditionalWrapping"
            txtAddWrap.Text = IIf(IsNull(lobjBOMRS.Fields(lstrFieldName).Value), "", Trim(lobjBOMRS.Fields(lstrFieldName).Value))
            lstrFieldName = "BOMCartonBox"
            chkRWBox.Value = IIf(lobjBOMRS.Fields(lstrFieldName).Value = True, 1, 0)
            lstrFieldName = "BOMSectionNotes"
            txtSectionNotes.Text = IIf(IsNull(lobjBOMRS.Fields(lstrFieldName).Value), "", Trim(lobjBOMRS.Fields(lstrFieldName).Value))
            lstrFieldName = "BOMTrackNotes"
            txtTrackNotes.Text = IIf(IsNull(lobjBOMRS.Fields(lstrFieldName).Value), "", Trim(lobjBOMRS.Fields(lstrFieldName).Value))
            lstrFieldName = "BOMTorsionNotes"
            txtTorsionNotes.Text = IIf(IsNull(lobjBOMRS.Fields(lstrFieldName).Value), "", Trim(lobjBOMRS.Fields(lstrFieldName).Value))
            lstrFieldName = "BOMShipNotes"
            txtShipNotes.Text = IIf(IsNull(lobjBOMRS.Fields(lstrFieldName).Value), "", Trim(lobjBOMRS.Fields(lstrFieldName).Value))
            lstrFieldName = "BOMStdWrapping"
            txtStdWrap.Text = IIf(IsNull(lobjBOMRS.Fields(lstrFieldName).Value), "", Trim(lobjBOMRS.Fields(lstrFieldName).Value))
            lstrFieldName = "POID"
            mlngPOID = IIf(IsNull(lobjBOMRS.Fields(lstrFieldName).Value), 0, lobjBOMRS.Fields(lstrFieldName).Value)
            lstrFieldName = "QuoteID"
            mlngQuoteID = IIf(IsNull(lobjBOMRS.Fields(lstrFieldName).Value), 0, lobjBOMRS.Fields(lstrFieldName).Value)
            
            lobjBOMRS.Close
        End If
    End If
    Set lobjBOMRS = Nothing
    Call SetControls(mintMode)
    Exit Sub

Error_Handler:
    llngErrNo = IIf(llngErrNo = 0, Err.Number, llngErrNo)
    lstrErrDescr = IIf(Len(lstrErrDescr) = 0, Err.Description, Err.Description)
    lstrErrSource = IIf(Len(Trim(Err.Source)) = 0, "", Err.Source & vbCrLf) & cstrErrorSource
    Call Err.Raise(llngErrNo, lstrErrSource, lstrErrDescr)
End Sub

Private Sub SetControls(ByVal intMode)
    On Error GoTo Error_Handler
    Const cstrErrorSource As String = cstrApplicationName & "." & cstrModuleName & ".SetControls()"
    Dim llngErrNo As Long
    Dim lstrErrDescr As String
    Dim lstrErrSource As String
    llngErrNo = 0
    lstrErrDescr = ""
    lstrErrSource = ""

    Select Case intMode
        Case MODE_EDIT
                        cmdEdit.Caption = "Cancel &Edit"
                        cmdEdit.Enabled = True
                        cmdSave.Enabled = True
                        cmdBack.Enabled = False
                        If gblnWordFlag = True Then cmdSpell1.Enabled = True
                        If gblnWordFlag = True Then cmdSpell2.Enabled = True
                        If gblnWordFlag = True Then cmdSpell3.Enabled = True
                        If gblnWordFlag = True Then cmdSpell4.Enabled = True
                        If gblnWordFlag = True Then cmdSpell5.Enabled = True
                        If gblnWordFlag = True Then cmdSpell7.Enabled = True
                        
                        txtSectionNotes.Enabled = True
                        txtTrackNotes.Enabled = True
                        txtTorsionNotes.Enabled = True
                        txtStdWrap.Enabled = True
                        txtAddWrap.Enabled = True
                        txtShipNotes.Enabled = True
                        txtBOMNotes.Enabled = True
                        chkRWBox.Enabled = True
                        
                        txtSectionNotes.BackColor = &HFFFFFF
                        txtTrackNotes.BackColor = &HFFFFFF
                        txtTorsionNotes.BackColor = &HFFFFFF
                        txtStdWrap.BackColor = &HFFFFFF
                        txtAddWrap.BackColor = &HFFFFFF
                        txtShipNotes.BackColor = &HFFFFFF
                        txtBOMNotes.BackColor = &HFFFFFF
        Case Else
                        cmdEdit.Caption = "&Edit Notes"
                        cmdEdit.Enabled = True
                        If mlngPOID + mlngQuoteID > 0 Then cmdEdit.Enabled = False
                        cmdSave.Enabled = False
                        cmdBack.Enabled = True
                        If gblnWordFlag = True Then cmdSpell1.Enabled = False
                        If gblnWordFlag = True Then cmdSpell2.Enabled = False
                        If gblnWordFlag = True Then cmdSpell3.Enabled = False
                        If gblnWordFlag = True Then cmdSpell4.Enabled = False
                        If gblnWordFlag = True Then cmdSpell5.Enabled = False
                        If gblnWordFlag = True Then cmdSpell7.Enabled = False
                        
                        txtSectionNotes.Enabled = False
                        txtTrackNotes.Enabled = False
                        txtTorsionNotes.Enabled = False
                        txtStdWrap.Enabled = False
                        txtAddWrap.Enabled = False
                        txtShipNotes.Enabled = False
                        txtBOMNotes.Enabled = False
                        chkRWBox.Enabled = False
                        
                        txtSectionNotes.BackColor = &HE0E0E0
                        txtTrackNotes.BackColor = &HE0E0E0
                        txtTorsionNotes.BackColor = &HE0E0E0
                        txtStdWrap.BackColor = &HE0E0E0
                        txtAddWrap.BackColor = &HE0E0E0
                        txtShipNotes.BackColor = &HE0E0E0
                        txtBOMNotes.BackColor = &HE0E0E0
                        
    End Select
    
    Exit Sub

Error_Handler:
    llngErrNo = IIf(llngErrNo = 0, Err.Number, llngErrNo)
    lstrErrDescr = IIf(Len(lstrErrDescr) = 0, Err.Description, Err.Description)
    lstrErrSource = IIf(Len(Trim(Err.Source)) = 0, "", Err.Source & vbCrLf) & cstrErrorSource
    Call Err.Raise(llngErrNo, lstrErrSource, lstrErrDescr)
End Sub

Private Function BuildRS() As ADODB.Recordset
    On Error GoTo Error_Handler
    Const cstrErrorSource As String = cstrApplicationName & "." & cstrModuleName & ".BuildRS()"
    Dim llngErrNo As Long
    Dim lstrErrDescr As String
    Dim lstrErrSource As String
    llngErrNo = 0
    lstrErrDescr = ""
    lstrErrSource = ""

    Dim lstrSQL As String
    Dim lstrSchema As String
    Dim llngAffected As Long
    Dim lobjADORS As ADODB.Recordset
    
    ' Get empty recordset
    lstrErrDescr = "Error getting empty ADO recordset."
    lstrSQL = "Select * from " & cstrTableName & " where 1=0"
    Set lobjADORS = New ADODB.Recordset
    lobjADORS.CursorType = adOpenForwardOnly
    lobjADORS.CursorLocation = adUseClient
    lobjADORS.Open lstrSQL, CnnBTMS, adOpenForwardOnly, adLockOptimistic
    lobjADORS.AddNew
    Set lobjADORS.ActiveConnection = Nothing
    lstrErrDescr = ""
    
    lobjADORS.Fields("BOMSectionNotes").Value = Trim(txtSectionNotes.Text)
    lobjADORS.Fields("BOMTrackNotes").Value = Trim(txtTrackNotes.Text)
    lobjADORS.Fields("BOMTorsionNotes").Value = Trim(txtTorsionNotes.Text)
    lobjADORS.Fields("BOMShipNotes").Value = Trim(txtShipNotes.Text)
    lobjADORS.Fields("BOMAdditionalWrapping").Value = Trim(txtAddWrap.Text)
    lobjADORS.Fields("BOMStdWrapping").Value = Left(Trim(txtStdWrap.Text), 50)
    lobjADORS.Fields("BOMCartonBox").Value = IIf(chkRWBox.Value = 1, True, False)
    
    Set BuildRS = lobjADORS
    Set lobjADORS = Nothing
    Exit Function

Error_Handler:
    llngErrNo = IIf(llngErrNo = 0, Err.Number, llngErrNo)
    lstrErrDescr = IIf(Len(lstrErrDescr) = 0, Err.Description, Err.Description)
    lstrErrSource = IIf(Len(Trim(Err.Source)) = 0, "", Err.Source & vbCrLf) & cstrErrorSource
    Call Err.Raise(llngErrNo, lstrErrSource, lstrErrDescr)
End Function
