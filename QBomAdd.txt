Private Function BuildQuoteDetailRS(ByVal lngBOMID As Long) As ADODB.Recordset
    On Error GoTo Error_Handler
    Const cstrErrorSource As String = cstrApplicationName & "." & cstrModuleName & ".BuildQuoteDetailRS()"
    Dim llngErrNo As Long
    Dim lstrErrDescr As String
    Dim lstrErrSource As String
    llngErrNo = 0
    lstrErrDescr = ""
    lstrErrSource = ""

    Dim lstrSQL As String
    Dim lstrSchema As String
    Dim llngAffected As Long
    Dim lobjADORS As ADODB.Recordset
    Dim lobjBOMDetRS As ADODB.Recordset
    Dim lobjQuoteRS As ADODB.Recordset

''''''''''Change in .NET''''''''''''''''
    Dim lobjMainRS As ADODB.Recordset
    Dim lobjBOMRS As ADODB.Recordset
    Dim ldblCost As Double
    Dim lstrPriceField As String
    Dim MeasSys As Integer
    Dim lstrPONumArray() As String
    Dim lintCount As Integer
    ReDim lstrPONumArray(0)
    Dim lstrDocumentName As String
    
    Select Case intCheck
        Case 1:  lstrPriceField = "G"
        Case 2:  lstrPriceField = "I"
        Case 3:  lstrPriceField = "N"
        Case 4:  lstrPriceField = "T"
        Case 5: lstrPriceField = "J"
        Case 6: lstrPriceField = "M"
        Case 7: lstrPriceField = "S"
        Case 8: lstrPriceField = "V"

        
    End Select
    
    lstrSQL = "Select bommeassys,bomdrqty from BoM where BOMID=" & lngBOMID
    Set lobjBOMRS = CnnBTMS.Execute(lstrSQL, llngAffected, adCmdText)
    If Not lobjBOMRS Is Nothing Then
        If lobjBOMRS.EOF = False And lobjBOMRS.BOF = False Then
            MeasSys = lobjBOMRS.Fields("BomMeasSys").Value
        End If
        Set lobjBOMRS = Nothing
    End If
    
''''''''''END Change in .NET
    
    ' Get empty recordset
    lstrErrDescr = "Error getting empty ADO recordset."
    lstrSQL = "Select * from QuoteDetails where 1=0"
    Set lobjADORS = New ADODB.Recordset
    lobjADORS.CursorType = adOpenForwardOnly
    lobjADORS.CursorLocation = adUseClient
    lobjADORS.Open lstrSQL, CnnBTMS, adOpenForwardOnly, adLockOptimistic
    Set lobjADORS.ActiveConnection = Nothing
    lstrErrDescr = ""
       
    lstrSQL = "Select * from BOMDetails where BOMID=" & lngBOMID
    Set lobjBOMDetRS = CnnBTMS.Execute(lstrSQL, llngAffected, adCmdText)
    If Not lobjBOMDetRS Is Nothing Then
        lobjBOMDetRS.ActiveConnection = Nothing
        If Not (lobjBOMDetRS.BOF And lobjBOMDetRS.EOF) Then
            lobjBOMDetRS.MoveFirst
            Do While lobjBOMDetRS.EOF = False
                lobjADORS.AddNew
                lobjADORS.Fields("QuoteID").Value = mlngQuoteID
                lobjADORS.Fields("QuoteNum").Value = mstrQuoteName
                lobjADORS.Fields("QuotePtNum").Value = IIf(IsNull(lobjBOMDetRS.Fields("BOMPtNum").Value), "", Trim(lobjBOMDetRS.Fields("BOMPtNum").Value))
                lobjADORS.Fields("QuotePtDesc").Value = IIf(IsNull(lobjBOMDetRS.Fields("BOMPtDesc").Value), "N/A", Trim(lobjBOMDetRS.Fields("BOMPtDesc").Value))

''''''''''Change in .NET''''''''''''''''
                ' blast off the call price fields, since QuotePtCost is a List Price
'                lobjADORS.Fields("QuotePtCallPrice").Value = IIf(IsNull(lobjBOMDetRS.Fields("BOMCallPrice").Value), 0, lobjBOMDetRS.Fields("BOMCallPrice").Value)
'                If lobjADORS.Fields("QuotePtCallPrice").Value = True Then lobjADORS.Fields("QuoteCallPriceExpDate").Value = IIf(IsNull(lobjBOMDetRS.Fields("BOMCallPriceExpDate").Value), Now() + 30, lobjBOMDetRS.Fields("BOMCallPriceExpDate").Value)
                lobjADORS.Fields("QuotePtCallPrice").Value = IIf(IsNull(lobjBOMDetRS.Fields("BOMCallPrice").Value), 0, lobjBOMDetRS.Fields("BOMCallPrice").Value)
                If lobjADORS.Fields("QuotePtCallPrice").Value = True And Mid(lobjBOMDetRS.Fields("BOMPtNum").Value, 1, 4) <> "COMM" Then
                   
                    lstrDocumentName = lobjBOMDetRS.Fields("BOMNum")
                    lstrPONumArray(lintCount) = lobjBOMDetRS.Fields("BOMPtNum")
                    ReDim Preserve lstrPONumArray(UBound(lstrPONumArray) + 1)
                    lintCount = lintCount + 1

                    lobjADORS.Fields("QuoteCallPriceExpDate").Value = Null
                    lobjADORS.Fields("QuotePtCallPrice").Value = False
                
                'non-stock item transfer call price date
                ElseIf lobjADORS.Fields("QuotePtCallPrice").Value = True And Mid(lobjBOMDetRS.Fields("BOMPtNum").Value, 1, 4) = "COMM" Then
                    lobjADORS.Fields("QuoteCallPriceExpDate").Value = IIf(IsNull(lobjBOMDetRS.Fields("BOMCallPriceExpDate").Value), Now() + 30, lobjBOMDetRS.Fields("BOMCallPriceExpDate").Value)
                
                End If
                    
''''''''''END Change in .NET
                
                lobjADORS.Fields("QuotePtCost").Value = IIf(IsNull(lobjBOMDetRS.Fields("BOMPtMtCost").Value), 0, lobjBOMDetRS.Fields("BOMPtMtCost").Value)
                lobjADORS.Fields("QuotePtNotes").Value = IIf(IsNull(lobjBOMDetRS.Fields("BOMPtNotes").Value), "", Trim(lobjBOMDetRS.Fields("BOMPtNotes").Value))
                lobjADORS.Fields("QuotePtPhantom").Value = IIf(IsNull(lobjBOMDetRS.Fields("BOMPtPhantomPN").Value), 0, lobjBOMDetRS.Fields("BOMPtPhantomPN").Value)
                lobjADORS.Fields("QuotePtQty").Value = IIf(IsNull(lobjBOMDetRS.Fields("BOMPtQtyReq").Value), 0, lobjBOMDetRS.Fields("BOMPtQtyReq").Value)
                lobjADORS.Fields("QuoteTotAmountReq").Value = IIf(IsNull(lobjBOMDetRS.Fields("BOMPtTotAmountReq").Value), 0, lobjBOMDetRS.Fields("BOMPtTotAmountReq").Value)
                lobjADORS.Fields("QuoteDiscount").Value = IIf(IsNull(lobjBOMDetRS.Fields("BOMDiscount").Value), 0, lobjBOMDetRS.Fields("BOMDiscount").Value)
                lobjADORS.Fields("QuotePtMultID").Value = IIf(IsNull(lobjBOMDetRS.Fields("BOMPtMultID").Value), 0, lobjBOMDetRS.Fields("BOMPtMultID").Value)
                lobjADORS.Fields("QuoteUM").Value = IIf(IsNull(lobjBOMDetRS.Fields("BOMPtUM").Value), "EACH", Trim(lobjBOMDetRS.Fields("BOMPtUM").Value))
                lobjADORS.Fields("QuoteTotalCost").Value = IIf(IsNull(lobjBOMDetRS.Fields("BOMPtMtCost").Value), 0, lobjBOMDetRS.Fields("BOMPtMtCost").Value) * IIf(IsNull(lobjBOMDetRS.Fields("BOMPtTotAmountReq").Value), 0, lobjBOMDetRS.Fields("BOMPtTotAmountReq").Value)
                lobjADORS.Fields("QuotePtWgt").Value = IIf(IsNull(lobjBOMDetRS.Fields("BOMPtShipWgt").Value), 0, lobjBOMDetRS.Fields("BOMPtShipWgt").Value)
                
                ''''''''''Change in .NET''''''''''''''''
                 '-- Mod Hanobver  MultId changed to strMultId
'                lstrSQL = "Select PartNum, descimp, Length, shipweight, UM, SpecialMult, MultID, " & lstrPriceField & " from Main where PartNum=" & SQLString(lobjADORS.Fields("QuotePtNum").Value)
                lstrSQL = "Select PartNum, descimp, Length, shipweight, UM, SpecialMult," & strMultId & ", " & lstrPriceField & " from Main where PartNum=" & SQLString(lobjADORS.Fields("QuotePtNum").Value)
        
                
                Set lobjMainRS = CnnBTMS.Execute(lstrSQL, llngAffected, adCmdText)
                If Not lobjMainRS Is Nothing Then
                    lobjMainRS.ActiveConnection = Nothing
                    If Not (lobjMainRS.BOF And lobjMainRS.EOF) Then
                    
                        ldblCost = Decrypt(GetSym(lobjBOMDetRS.Fields("BOMPtNum").Value), lobjMainRS.Fields(lstrPriceField).Value)
                        
                        If DBLValue(lobjMainRS.Fields("SpecialMult").Value) > 0 Then
                            ldblCost = DBLValue(lobjMainRS.Fields("SpecialMult").Value) * ldblCost
                        Else
                        
                         '-- Mod Hanobver  MultId changed to strMultId
                            If Not Trim(lobjMainRS.Fields(strMultId).Value) = "" Then
                                ldblCost = ldblCost * (MultipliersCase(DBLValue(Trim(lobjMainRS.Fields(strMultId).Value))))
                            End If
                        End If
                        
                        lobjADORS.Fields("QuotePtCost").Value = ldblCost
                        lobjADORS.Fields("QuoteDiscount").Value = IIf(IsNull(lobjMainRS.Fields("SpecialMult").Value), 0, lobjMainRS.Fields("SpecialMult").Value)
                         '-- Mod Hanobver  MultId changed to strMultId
                         lobjADORS.Fields("QuotePtMultID").Value = IIf(IsNull(lobjMainRS.Fields(strMultId).Value), 0, lobjMainRS.Fields(strMultId).Value)
 
                        
                        lobjADORS.Fields("QuoteUM").Value = lobjMainRS.Fields("UM").Value
                        lobjADORS.Fields("QuotePtWgt").Value = lobjMainRS.Fields("shipweight").Value
                        If MeasSys = 0 Then lobjADORS.Fields("QuotePtDesc").Value = IIf(IsNull(lobjMainRS.Fields("DescImp").Value), "N/A", Trim(lobjMainRS.Fields("DescImp").Value))
                        
                        ''''''''''Change in .NET''''''''''''''''
                        If lobjMainRS.Fields("UM").Value = "LFT" Then
                            lobjADORS.Fields("QuotePtWgt").Value = lobjMainRS.Fields("ShipWeight").Value / 12
                            lobjADORS.Fields("QuotePtCost").Value = ldblCost / 12
                            lobjADORS.Fields("QuoteUM").Value = "INCHES"
                        End If
                        
                        If lobjMainRS.Fields("UM").Value = "EACH" And lobjBOMDetRS.Fields("BOMPtUM").Value = "INCHES" And Left(lobjADORS.Fields("QuotePtNum").Value, 4) <> "433-" Then
                            lobjADORS.Fields("QuotePtWgt").Value = Round((DBLValue(lobjMainRS.Fields("ShipWeight").Value) / DBLValue(lobjMainRS.Fields("Length").Value)), 2)
                            lobjADORS.Fields("QuotePtCost").Value = (ldblCost / lobjMainRS.Fields("Length").Value)
                            lobjADORS.Fields("QuoteUM").Value = "INCHES"
                        End If
                        
                        If Left(lobjADORS.Fields("QuotePtNum").Value, 4) = "433-" Then
                        
                            lstrSQL = "select BOMWireType from BoM where BOMID=" & lngBOMID
                            Set lobjBOMRS = CnnBTMS.Execute(lstrSQL, llngAffected, adCmdText)
                            
                            If Not lobjBOMRS Is Nothing Then
                                If lobjBOMRS.EOF = False And lobjBOMRS.BOF = False Then
                                    If lobjBOMRS.Fields("BOMWireType").Value = "Round" Then
                                        lobjADORS.Fields("QuoteUM").Value = "INCHES"
                                    Else
                                        lobjADORS.Fields("QuoteUM").Value = "COILS"
                                    End If
                                End If
                            End If
                        End If
                        ''''''''''END Change in .NET
                    End If
                End If
'               '''''''''END Change in .NET
                lobjBOMDetRS.MoveNext
            Loop
            lobjBOMDetRS.Close
        End If
    End If
    
''''''''''Change in .NET''''''''''''''''
    'part no. found on active POs
    Dim lstrData As String
    Dim i As Integer
    
    If lintCount >= 1 And mblnFlagRemoveTag = False Then
        lstrData = lstrData & "The following part number(s) price have been converted back to List Pricing on the Quote. The price will not be changed on the Bill-Of-Material. " & vbNewLine '& mstrQuoteName
        For i = 0 To UBound(lstrPONumArray) - 1
            lstrData = lstrData & lstrPONumArray(i) & vbNewLine
        Next
        MsgBox lstrData, vbApplicationModal + vbInformation, "Values Converted"
        mblnflagMessageshown = True
     End If
''''''''''END Change in .NET

    Set lobjBOMDetRS = Nothing
    Set BuildQuoteDetailRS = lobjADORS
    Set lobjADORS = Nothing
    Exit Function

Error_Handler:
    llngErrNo = IIf(llngErrNo = 0, Err.Number, llngErrNo)
    lstrErrDescr = IIf(Len(lstrErrDescr) = 0, Err.Description, Err.Description)
    lstrErrSource = IIf(Len(Trim(Err.Source)) = 0, "", Err.Source & vbCrLf) & cstrErrorSource
    Call Err.Raise(llngErrNo, lstrErrSource, lstrErrDescr)
End Function

Private Sub AddToQuote(ByVal lngBOMID As Long)
     On Error GoTo Error_Handler
    Const cstrErrorSource As String = cstrApplicationName & "." & cstrModuleName & ".AddToQuote()"
    Dim llngErrNo As Long
    Dim lstrErrDescr As String
    Dim lstrErrSource As String
    llngErrNo = 0
    lstrErrDescr = ""
    lstrErrSource = ""

    Dim lstrSQL As String
    Dim lintMode As Integer
    Dim lstrWhereClause As String
    Dim lstrExcludeFieldList As String
    Dim llngAffected As Long
    Dim lobjADORS As ADODB.Recordset
    Dim lobjParentRS As ADODB.Recordset
    
'''''''''Change in .NET''''''''''''''''
    Dim lstrNumArray() As String
    Dim lintCount As Integer
    ReDim lstrNumArray(0)
    
    mblnflagMessageshown = False 'initial value
     
''''''''''END Change in .NET
    
    ' Copy the fields from the screen to an ADO Recordset
    Set lobjADORS = BuildQuoteDetailRS(lngBOMID)
    CnnBTMS.BeginTrans
    ' Process through the recordset and insert or update rows
    If Not lobjADORS Is Nothing Then
        If Not (lobjADORS.BOF And lobjADORS.EOF) Then
            lobjADORS.MoveFirst
            lstrExcludeFieldList = ""
            
            Do While lobjADORS.EOF = False
                ' Check to see if the record exists in the
                ' BOMDetails table already
                lstrSQL = "select QuotePtNum, QuotePtQty,QuoteTotAmountReq, QuotePtCallPrice, QuotePtCost,QuotePtMultID,QuoteDiscount from QuoteDetails"
                lstrSQL = lstrSQL & " where QuoteID=" & mlngQuoteID
                lstrSQL = lstrSQL & " and QuotePtNum=" & SQLString(lobjADORS.Fields("QuotePtNum").Value)
                Set lobjParentRS = CnnBTMS.Execute(lstrSQL, llngAffected, adCmdText)
                If Not lobjParentRS Is Nothing Then
                    If Not (lobjParentRS.BOF And lobjParentRS.EOF) Then
                        lobjParentRS.MoveFirst

Private Sub CopyLenDetails(ByVal BOMID As Long, ByVal BOMName As String)
