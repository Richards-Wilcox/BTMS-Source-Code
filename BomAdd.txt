Private Function BuildPODetailRSfromBOM(ByVal lngBOMID As Long) As ADODB.Recordset
    On Error GoTo Error_Handler
    Const cstrErrorSource As String = cstrApplicationName & "." & cstrModuleName & ".BuildPODetailRSfromBOM()"
    Dim llngErrNo As Long
    Dim lstrErrDescr As String
    Dim lstrErrSource As String
    llngErrNo = 0
    lstrErrDescr = ""
    lstrErrSource = ""

    Dim lstrSQL As String
    Dim lstrSchema As String
    Dim llngAffected As Long
    Dim lobjADORS As ADODB.Recordset
    Dim lobjBOMDetRS As ADODB.Recordset
    Dim lobjMainRS As ADODB.Recordset
    Dim lobjBOMRS As ADODB.Recordset
    Dim MeasSys As Integer
    Dim lsngLen As Single
    Dim ldblCost As Double
    Dim TempSprLen As Single
    Dim lstrPriceField As String

''''''''''Change in .NET''''''''''''''''
    Dim lstrPONumArray() As String
    Dim lintCount As Integer
    ReDim lstrPONumArray(0)
    Dim lstrDocuname As String

''''''''''END Change in .NET

    Select Case intCheck
        Case 1:  lstrPriceField = "G"
        Case 2:  lstrPriceField = "I"
        Case 3:  lstrPriceField = "N"
        Case 4:  lstrPriceField = "T"
        
        Case 5: lstrPriceField = "J"
        Case 6: lstrPriceField = "M"
        Case 7: lstrPriceField = "S"
        Case 8: lstrPriceField = "V"

        
    End Select

    lstrSQL = "Select bommeassys,bomdrqty from BoM where BOMID=" & lngBOMID
    Set lobjBOMRS = CnnBTMS.Execute(lstrSQL, llngAffected, adCmdText)
    If Not lobjBOMRS Is Nothing Then
        If lobjBOMRS.EOF = False And lobjBOMRS.BOF = False Then
            MeasSys = lobjBOMRS.Fields("BomMeasSys").Value
        End If
        Set lobjBOMRS = Nothing
    End If
    
    ' Get empty recordset
    lstrErrDescr = "Error getting empty ADO recordset."
    lstrSQL = "Select * from PODetails where 1=0"
    Set lobjADORS = New ADODB.Recordset
    lobjADORS.CursorType = adOpenForwardOnly
    lobjADORS.CursorLocation = adUseClient
    lobjADORS.Open lstrSQL, CnnBTMS, adOpenForwardOnly, adLockOptimistic
    Set lobjADORS.ActiveConnection = Nothing
    lstrErrDescr = ""
    
    lstrSQL = "Select * from BOMDetails where BOMID=" & lngBOMID
    'lstrSQL = lstrSQL & " and BOMPtQtyUnAvailable>0"
    Set lobjBOMDetRS = CnnBTMS.Execute(lstrSQL, llngAffected, adCmdText)
    If Not lobjBOMDetRS Is Nothing Then
        lobjBOMDetRS.ActiveConnection = Nothing
        If Not (lobjBOMDetRS.BOF And lobjBOMDetRS.EOF) Then
            lobjBOMDetRS.MoveFirst
            Do While lobjBOMDetRS.EOF = False
                lobjADORS.AddNew
                lobjADORS.Fields("POID").Value = mlngPOID
                lobjADORS.Fields("PONum").Value = mstrPOName
                lobjADORS.Fields("POPtNum").Value = IIf(IsNull(lobjBOMDetRS.Fields("BOMPtNum").Value), "", Trim(lobjBOMDetRS.Fields("BOMPtNum").Value))
                lobjADORS.Fields("POPtDesc").Value = IIf(IsNull(lobjBOMDetRS.Fields("BOMPtDesc").Value), "N/A", Trim(lobjBOMDetRS.Fields("BOMPtDesc").Value))
                lobjADORS.Fields("POPtMtCost").Value = IIf(IsNull(lobjBOMDetRS.Fields("BOMPtMtCost").Value), 0, lobjBOMDetRS.Fields("BOMPtMtCost").Value)
                lobjADORS.Fields("POPtQtyOrd").Value = IIf(IsNull(lobjBOMDetRS.Fields("BOMPttotamountReq").Value), 0, (lobjBOMDetRS.Fields("BOMPttotamountReq").Value))
                lobjADORS.Fields("POPtNotes").Value = ""
                lobjADORS.Fields("POCat").Value = GetPOCat(lobjADORS.Fields("POPtNum").Value)
                lobjADORS.Fields("POSubCategory").Value = GetSubCategory(lobjADORS.Fields("POPtNum").Value)
                
                lobjADORS.Fields("POPtQtyRcvd").Value = 0
                lobjADORS.Fields("POPtStatus").Value = "Active"
                lobjADORS.Fields("PODiscount").Value = IIf(IsNull(lobjBOMDetRS.Fields("BOMDiscount").Value), 0, lobjBOMDetRS.Fields("BOMDiscount").Value)
                lobjADORS.Fields("POPtMultID").Value = IIf(IsNull(lobjBOMDetRS.Fields("BOMPtMultID").Value), 0, lobjBOMDetRS.Fields("BOMPtMultID").Value)

''''''''''Change in .NET''''''''''''''''
'                lobjADORS.Fields("POPtCallPrice").Value = IIf(IsNull(lobjBOMDetRS.Fields("BOMCallPrice").Value), 0, lobjBOMDetRS.Fields("BOMCallPrice").Value)
'                If lobjADORS.Fields("POPtCallPrice").Value = True Then lobjADORS.Fields("POCallPriceExpDate").Value = IIf(IsNull(lobjBOMDetRS.Fields("BOMCallPriceExpDate").Value), Now() + 30, lobjBOMDetRS.Fields("BOMCallPriceExpDate").Value)
                lobjADORS.Fields("POPtCallPrice").Value = IIf(IsNull(lobjBOMDetRS.Fields("BOMCallPrice").Value), 0, lobjBOMDetRS.Fields("BOMCallPrice").Value)
                If lobjADORS.Fields("POPtCallPrice").Value = True And Mid(lobjBOMDetRS.Fields("BOMPtNum").Value, 1, 4) <> "COMM" Then
                        
                    lstrDocuname = lobjBOMDetRS.Fields("BOMNum")
                    lstrPONumArray(lintCount) = lobjBOMDetRS.Fields("BOMPtNum")
                    ReDim Preserve lstrPONumArray(UBound(lstrPONumArray) + 1)
                    lintCount = lintCount + 1
                        
                    lobjADORS.Fields("POCallPriceExpDate").Value = Null
                    lobjADORS.Fields("POPtCallPrice").Value = False
                    
                'non-stock item transfer call price date
                ElseIf lobjADORS.Fields("POPtCallPrice").Value = True And Mid(lobjBOMDetRS.Fields("BOMPtNum").Value, 1, 4) = "COMM" Then
                    lobjADORS.Fields("POCallPriceExpDate").Value = IIf(IsNull(lobjBOMDetRS.Fields("BOMCallPriceExpDate").Value), Now() + 30, lobjBOMDetRS.Fields("BOMCallPriceExpDate").Value)
                End If
                
''''''''''END Change in .NET
                
                lobjADORS.Fields("POPtUM").Value = IIf(IsNull(lobjBOMDetRS.Fields("BOMPtUM").Value), "EACH", Trim(lobjBOMDetRS.Fields("BOMPtUM").Value))
                lobjADORS.Fields("POPtWgt").Value = IIf(IsNull(lobjBOMDetRS.Fields("BOMPtShipWgt").Value), 0, lobjBOMDetRS.Fields("BOMPtShipWgt").Value)
                lobjADORS.Fields("POPtPhantomPN").Value = False
                
                 '-- Mod Hanobver  MultId changed to strMultId
'                lstrSQL = "Select PartNum, descimp, Length, shipweight, UM, SpecialMult, MultID, " & lstrPriceField & " from Main where PartNum=" & SQLString(lobjADORS.Fields("POPtNum").Value)
                lstrSQL = "Select PartNum, descimp, Length, shipweight, UM, SpecialMult, " & strMultId & ", " & lstrPriceField & " from Main where PartNum=" & SQLString(lobjADORS.Fields("POPtNum").Value)

                
                Set lobjMainRS = CnnBTMS.Execute(lstrSQL, llngAffected, adCmdText)
                If Not lobjMainRS Is Nothing Then
                    lobjMainRS.ActiveConnection = Nothing
                    If Not (lobjMainRS.BOF And lobjMainRS.EOF) Then
                    
                        ldblCost = Decrypt(GetSym(lobjBOMDetRS.Fields("BOMPtNum").Value), lobjMainRS.Fields(lstrPriceField).Value)
                        
                        If DBLValue(lobjMainRS.Fields("SpecialMult").Value) > 0 Then
                            ldblCost = DBLValue(lobjMainRS.Fields("SpecialMult").Value) * ldblCost
                        Else
                         '-- Mod Hanobver  MultId changed to strMultId

                            If Not Trim(lobjMainRS.Fields(strMultId).Value) = "" Then
                                ldblCost = ldblCost * (MultipliersCase(DBLValue(Trim(lobjMainRS.Fields(strMultId).Value))))
                            End If
                        End If
                        
                        lobjADORS.Fields("POPtMtCost").Value = ldblCost
                        lobjADORS.Fields("POPtUM").Value = lobjMainRS.Fields("UM").Value
                        lobjADORS.Fields("POPtWgt").Value = lobjMainRS.Fields("shipweight").Value
                        If MeasSys = 0 Then lobjADORS.Fields("POPtDesc").Value = IIf(IsNull(lobjMainRS.Fields("DescImp").Value), "N/A", Trim(lobjMainRS.Fields("DescImp").Value))
                        
                        If lobjADORS.Fields("POPtUM").Value = "LFT" Then
                            lobjADORS.Fields("POPtQtyOrd").Value = DBLValue(lobjADORS.Fields("POPtQtyOrd").Value) / 12
                        End If
                        If lobjBOMDetRS.Fields("BOMPtUM").Value = "INCHES" And lobjMainRS.Fields("UM").Value = "EACH" Or _
                                        Left(lobjADORS.Fields("POPtNum").Value, 4) = "433-" Then
'                        If lobjADORS.Fields("POPtNum").Value = "328-227" Or _
                                lobjADORS.Fields("POPtNum").Value = "426-0271" Or _
                                Left(lobjADORS.Fields("POPtNum").Value, 4) = "433-" Then
'                                lobjADORS.Fields("POPtNum").Value = "114-702" Or _
'                                lobjADORS.Fields("POPtNum").Value = "114-703" Or _
'                                lobjADORS.Fields("POPtNum").Value = "114-704" Or _
'                                lobjADORS.Fields("POPtNum").Value = "114-706" Or _
'                                lobjADORS.Fields("POPtNum").Value = "950-009" Or _
'                                lobjADORS.Fields("POPtNum").Value = "950-010" Or _
'                                lobjADORS.Fields("POPtNum").Value = "950-011" Or _
'                                lobjADORS.Fields("POPtNum").Value = "950-012" Or _
'                                lobjADORS.Fields("POPtNum").Value = "950-013" Or _
'                                lobjADORS.Fields("POPtNum").Value = "950-014" Or _
'                                lobjADORS.Fields("POPtNum").Value = "950-015" Or _
'                                lobjADORS.Fields("POPtNum").Value = "114-701" Or _
'                                lobjADORS.Fields("POPtNum").Value = "426-0272" Or _
'                                lobjADORS.Fields("POPtNum").Value = "950-016" Then
                            
                            If Left(lobjADORS.Fields("POPtNum").Value, 4) = "433-" Then
                                        
                                lsngLen = GetLengthFromMain(lobjADORS.Fields("POPtNum").Value)
                                lobjADORS.Fields("POPtWgt").Value = DBLValue(lobjMainRS.Fields("shipweight").Value) * lsngLen
                                lobjADORS.Fields("POPtMtCost").Value = ldblCost * lsngLen
                                'springs; weight and price go per inch in Main, even though the UM is EACH, need to multiply by the total lenght.
                            Else
                                lsngLen = lobjMainRS.Fields("Length").Value
                                'pvc and shaft; weight and price go per each in the main, no need to modify.
                            End If
                            Dim lArrBOM(1) As Long, lArrQuote(0) As Long
                            lArrBOM(0) = lngBOMID
                            lArrQuote(0) = 0
                            lobjADORS.Fields("POPtQtyOrd").Value = GetNumberOfPartsRequired(lobjADORS.Fields("POPtNum").Value, lsngLen, lArrBOM(), lArrQuote())
                        End If
                    Else
                        lstrSQL = "Select PartNum, DescImp from Main"
                        Set lobjMainRS = CnnBTMS.Execute(lstrSQL, llngAffected, adCmdText)
                        If Not lobjMainRS Is Nothing Then
                            lobjMainRS.ActiveConnection = Nothing
                            If Not (lobjMainRS.BOF And lobjMainRS.EOF) Then
                                lobjMainRS.MoveFirst 'section
                                lobjMainRS.Find "PartNum like " & SQLString(Left(lobjADORS.Fields("POPtNum").Value, 6) & "000")
                                If lobjMainRS.EOF = False And lobjMainRS.BOF = False Then
                                    If MeasSys = 0 Then lobjADORS.Fields("POPtDesc").Value = lobjMainRS.Fields("DescImp").Value
                                    lobjADORS.Fields("POPtWgt").Value = lobjBOMDetRS.Fields("BOMPtShipWgt").Value
                                    lobjADORS.Fields("POPtUM").Value = "EACH"
                                    lobjADORS.Fields("POPtPhantomPN").Value = True
                                    lobjADORS.Fields("POCat").Value = 0
                                    lobjADORS.Fields("POSubCategory").Value = GetSubCategory(lobjMainRS.Fields("PartNum").Value)
                                End If
                            End If
                        End If
                    End If
                    Set lobjMainRS = Nothing
                End If
                lobjADORS.Fields("POPtQtyBO").Value = lobjADORS.Fields("POPtQtyOrd").Value
'                If mblnAdd = True Then
'                    Call UpdateOnPOsForPN(lobjADORS.Fields("POPtNum").Value, lobjBOMDetRS.Fields("BOMPttotamountReq").Value)
'                Else
'                    Call UpdateOnPOsForPN(lobjADORS.Fields("POPtNum").Value, 0 - DBLValue(lobjBOMDetRS.Fields("BOMPttotamountReq").Value))
'                End If
                lobjBOMDetRS.MoveNext
            Loop
            lobjBOMDetRS.Close
        End If
    End If

''''''''''Change in .NET''''''''''''''''
    'part no. found on active POs
    Dim lstrData As String
    Dim i As Integer
    
    If lintCount >= 1 And mblnFlagRemoveTag = False Then

        For i = 0 To UBound(lstrPONumArray) - 1
            lstrData = lstrData & lstrPONumArray(i) & vbNewLine
        Next
        MsgBox lstrData, vbApplicationModal + vbInformation, "Values Converted"
        mblnflagMessageshown = True
     End If
''''''''''END Change in .NET
    Set lobjBOMDetRS = Nothing
    
    Set BuildPODetailRSfromBOM = lobjADORS
    Set lobjADORS = Nothing
    Exit Function

Error_Handler:
    llngErrNo = IIf(llngErrNo = 0, Err.Number, llngErrNo)
    lstrErrDescr = IIf(Len(lstrErrDescr) = 0, Err.Description, Err.Description)
    lstrErrSource = IIf(Len(Trim(Err.Source)) = 0, "", Err.Source & vbCrLf) & cstrErrorSource
    Call Err.Raise(llngErrNo, lstrErrSource, lstrErrDescr)
End Function


Private Sub AddToPO(ByVal lngDocID As Long, ByVal BoM As Boolean)
     On Error GoTo Error_Handler
    Const cstrErrorSource As String = cstrApplicationName & "." & cstrModuleName & ".AddToPO()"
    Dim llngErrNo As Long
    Dim lstrErrDescr As String
    Dim lstrErrSource As String
    llngErrNo = 0
    lstrErrDescr = ""
    lstrErrSource = ""

    Dim lstrSQL As String
    Dim lintMode As Integer
    Dim lstrWhereClause As String
    Dim lstrExcludeFieldList As String
    Dim llngAffected As Long
    Dim lobjADORS As ADODB.Recordset
    Dim lobjParentRS As ADODB.Recordset
    Dim lobjMainRS As ADODB.Recordset
    Dim lobjDOCRS As ADODB.Recordset
    Dim i As Integer, j As Integer
    Dim lsngLen As Single

''''''''''Change in .NET''''''''''''''''
    Dim lstrNumArray() As String
    Dim lintCount As Integer
    ReDim lstrNumArray(0)
    
    mblnflagMessageshown = False 'initial value
     
''''''''''END Change in .NET

    ' Copy the fields from the screen to an ADO Recordset
    If BoM = True Then
        Set lobjADORS = BuildPODetailRSfromBOM(lngDocID)
    Else
        Set lobjADORS = BuildPODetailRSfromQuote(lngDocID)
    End If
    
    CnnBTMS.BeginTrans
    ' Process through the recordset and insert or update rows
    If Not lobjADORS Is Nothing Then
        If Not (lobjADORS.BOF And lobjADORS.EOF) Then
            lobjADORS.MoveFirst
            lstrExcludeFieldList = ""
            
            Do While lobjADORS.EOF = False
                ' Check to see if the record exists in the
                ' QuoteDetails table already
                lobjADORS.Fields("POID").Value = mlngPOID
                lobjADORS.Fields("PONum").Value = mstrPOName

                lstrSQL = "select POPtNum, POPtQtyOrd, POPtMultID,PODiscount,POPtMtCost,POPtCallPrice from PODetails "
                lstrSQL = lstrSQL & "where POID=" & mlngPOID
                lstrSQL = lstrSQL & "and POPtNum=" & SQLString(lobjADORS.Fields("POPtNum").Value)
                Set lobjParentRS = CnnBTMS.Execute(lstrSQL, llngAffected, adCmdText)
                If Not lobjParentRS Is Nothing Then
                    If Not (lobjParentRS.BOF And lobjParentRS.EOF) Then
                        lobjParentRS.MoveFirst
                        ' Reset the Mode so that we do an update of the existing part
                        lintMode = MODE_EDIT
                        ' Average the cost and sum the qty's
                        If lobjADORS.Fields("POPtMultID").Value <> lobjParentRS.Fields("POPtMultID").Value Then
                            lobjADORS.Fields("POPtCallPrice").Value = True
                            lobjADORS.Fields("POCallPriceExpDate").Value = Format(Now() + 30, "Short Date")
                        End If
                        If DBLValue(lobjADORS.Fields("PODiscount").Value) + DBLValue(lobjParentRS.Fields("PODiscount").Value) > 0 Then
                            If DBLValue(lobjParentRS.Fields("PODiscount").Value) > 0 Then
                                If DBLValue(lobjADORS.Fields("PODiscount").Value) > DBLValue(lobjParentRS.Fields("PODiscount").Value) Or _
                                lobjADORS.Fields("PODiscount").Value = 0 Then

Private Sub cmdAccept_Click()

    On Error GoTo Error_Handler
    Const cstrErrorSource As String = cstrApplicationName & "." & cstrModuleName & ".cmdAccept_Click()"
    Dim llngErrNo As Long
    Dim lstrErrDescr As String
    Dim lstrErrSource As String
    llngErrNo = 0
    lstrErrDescr = ""
    lstrErrSource = ""
    CnnBTMS.BeginTrans
    Dim i As Integer, chek As Boolean
    For i = 2 To Grid2.Rows - 1
        If Grid2.IsSelected(i) = True Then
            If chek = False Then
                If MsgBox("Are you sure you wish to do this?", vbYesNo + vbExclamation, "You are updating the PO with this Document") = vbNo Then
                    CnnBTMS.RollbackTrans
                    Exit Sub
                End If
            End If
            mblnAdd = True
            chek = True
            If mlngPOID > 0 Then
                Screen.MousePointer = vbHourglass
                Me.Refresh
                Call UpdateOnPOsForPO(mlngPOID, True) 'first we need to remove the old qtys
                Call AddToPO(Grid2.TextMatrix(i, 1), IIf(Grid2.TextMatrix(i, 3) = "BoM", True, False))
                Call UpdateOnPOsForPO(mlngPOID, False) 'update OnPos withe the new qtys
            End If
        End If
    Next
    Screen.MousePointer = vbDefault
    If chek = False Then
        MsgBox "Please select Document to add.", vbApplicationModal + vbExclamation + vbOKOnly, "No Document Selected"
        CnnBTMS.RollbackTrans
        Exit Sub
    End If
    Call UpdatePOTotal
    CnnBTMS.CommitTrans
    Unload Me
    frmPO.mlngPOID = mlngPOID
    frmPO.mstrPONum = mstrPOName
    frmPO.Show
    Exit Sub

Error_Handler:
    Screen.MousePointer = vbDefault
    CnnBTMS.RollbackTrans
    llngErrNo = IIf(llngErrNo = 0, Err.Number, llngErrNo)
    lstrErrDescr = IIf(Len(lstrErrDescr) = 0, Err.Description, lstrErrDescr & vbCrLf & Err.Description)

    Dim lobjQuoteRS As ADODB.Recordset
