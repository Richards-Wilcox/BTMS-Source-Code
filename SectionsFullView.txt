Private Sub cmdAccept_Click()
    
    On Error GoTo Error_Handler
    Const cstrErrorSource As String = cstrApplicationName & "." & cstrModuleName & ".cmdAccept_Click()"
    Dim llngErrNo As Long
    Dim lstrErrDescr As String
    Dim lstrErrSource As String
    llngErrNo = 0
    lstrErrDescr = ""
    lstrErrSource = ""
    Dim Wdt As String
    

    If MeasU = 1 Then
        Wdt = GetImpString(216) '18feet
    Else
        aCon.Inch = 216
        aCon.mmcalc
        Wdt = CStr(Round(aCon.mm, 0)) & " mm"
    End If

    If DrWidth > 216 Then
        If cboMaterial(0).ItemData(cboMaterial(0).ListIndex) = 1 Or (cboMaterial(1).Text = "Glass" And Me.chkSealedUnit.Value = 1) Then 'glass - for single and double glazed units
            If MsgBox("Due to deflection, with doors over " & Wdt & " , Richards Wilcox does not recommend using of glass for glazing." & vbCrLf & _
                "Would you like to reselect material?", vbYesNo + vbApplicationModal, "Full View") = vbYes Then
                If cboMaterial(0).Enabled = True Then cboMaterial(0).SetFocus
                Exit Sub
            End If
        End If
    End If
                
    If Me.chkSealedUnit.Value = 1 Or cboThickness.ItemData(cboThickness.ListIndex) = 3 Then
        If MsgBox(" WARNING! " & vbCrLf & _
            "Because of the thickness of the glass chosen, the difference in weight between the thermal sections and the aluminum sections is enough to seriously affect the balance and operation of the door." & vbCrLf & _
            "Press No if you wish to contact Richards-Wilcox for a solution or to reselect material. " & vbCrLf & _
            "Press Yes to continue.", vbYesNo + vbApplicationModal, "Full View") = vbNo Then
                chkSealedUnit.Value = 0
                If cboMaterial(0).Enabled = True Then cboMaterial(0).SetFocus
                Exit Sub
        End If
    End If
    
    Call SaveFullView
    If chkECapsOnFullView.Value = 1 Then
        varDoorData.EndCapsOnFullView = True
    Else
        varDoorData.EndCapsOnFullView = False
    End If
    
    Dim L21 As Integer
    Dim L24 As Integer
    Dim lstrPN As String
    Dim lblnCP As Boolean
    L21 = ModSections.GetNumberOfLiteSectionsPerHeight(21)
    L24 = ModSections.GetNumberOfLiteSectionsPerHeight(24)
    
    If L21 > 0 Then
        lstrPN = GetFullViewPN(True)
        
        If lstrPN = "" Then
            MsgBox "Glazing option not found. Select Other: Glass or Other: Plastic option in the Material list."
            Exit Sub
            
        End If
        
        If isCallPrice(lstrPN) = True Then
'            MsgBox "The chosen Full View section is a " & Chr(34) & "Call Price" & Chr(34) & " item." & _
'                vbCrLf & "It will be highlighted on the BoM Details screen." & vbCrLf & _
'                    "Please call for pricing and enter the cost manually." & vbCrLf & vbCrLf & _
'                        "The section and glazing weights have been applied for springing.", vbApplicationModal + vbInformation, "Call Price Item"
            
           MsgBox "The chosen Full View section is a " & Chr(34) & "Call Price" & Chr(34) & " item and will be highlighted on the BoM Details screen" & vbCrLf & _
                    "Please call for pricing and enter the cost manually." & vbCrLf & _
                        "The Full View section weights have been applied for springing." & vbCrLf & _
                            "You may need to manually adjust the glazing weight.", vbApplicationModal + vbInformation, "Call Price Item"
                    
            lblnCP = True
        End If
    End If
    
    If lblnCP = False And L24 > 0 Then
        lstrPN = GetFullViewPN(False)
        If lstrPN = "" Then
            MsgBox "Glazing option not found. Select Other: Glass or Other: Plastic option in the Material list."
            Exit Sub
            
        End If
        
        If isCallPrice(lstrPN) = True Then
'            MsgBox "The chosen Full View section is a " & Chr(34) & "Call Price" & Chr(34) & " item." & _
'                vbCrLf & "It will be highlighted on the BoM Details screen." & vbCrLf & _
'                    "Please call for pricing and enter the cost manually." & vbCrLf & vbCrLf & _
'                        "The section and glazing weights have been applied for springing.", vbApplicationModal + vbInformation, "Call Price Item"

            MsgBox "The chosen Full View section is a " & Chr(34) & "Call Price" & Chr(34) & " item and will be highlighted on the BoM Details screen" & vbCrLf & _
                    "Please call for pricing and enter the cost manually." & vbCrLf & _
                        "The Full View section weights have been applied for springing." & vbCrLf & _
                            "You may need to manually adjust the glazing weight.", vbApplicationModal + vbInformation, "Call Price Item"
                    
        End If
    End If
    
    frmSectionsEntry.mblnCancel = False

Public Sub SaveFullView()

    On Error GoTo Error_Handler
    Const cstrErrorSource As String = cstrApplicationName & "." & cstrModuleName & ".SaveFullView()"
    Dim llngErrNo As Long
    Dim lstrErrDescr As String
    Dim lstrErrSource As String
    llngErrNo = 0
    lstrErrDescr = ""
    lstrErrSource = ""
    
    Dim lstrLiteType As String
    Dim i As Integer
    i = 0
    For i = 0 To IIf(frmSectionsFullView.chkSealedUnit.Value = 1, 1, 0)
        
        If i = 1 Then cboMaterial(1).ItemData(cboMaterial(1).ListIndex) = cboMaterial(0).ItemData(cboMaterial(0).ListIndex) ' 'single or 1/2"  - Acrylic + lexan "Finish" option not available - put through other plastic if wish to get tinted
        
        '1st digit - colour
        '7-anodized
        '8-white
        '9-by others
        If chkByOthers.Value = 1 Then
            lstrLiteType = lstrLiteType & "9"
        Else
            If optWhite.Value = True Then
                lstrLiteType = lstrLiteType & "8"
            ElseIf optAnodized.Value = True Then
                lstrLiteType = lstrLiteType & "7"
            End If
        End If
        
        '2nd digit - size
        '0-1/8
        '1-1/4
        If cboThickness.ItemData(cboThickness.ListIndex) = 1 Or cboThickness.ItemData(cboThickness.ListIndex) = 3 Then
            lstrLiteType = lstrLiteType & "0"
        ElseIf cboThickness.ItemData(cboThickness.ListIndex) = 2 Then
            lstrLiteType = lstrLiteType & "1"
        End If
        
        '3rd digit - material
        '0-glass
        '1-acrylic
        '2-lexan
        '3-other
        If chkByOthers.Value = 1 Then   'If they check the box to signify that they are supplying their own glass
            If cboMaterial(i).ItemData(cboMaterial(i).ListIndex) - 1 = 1 Or _
                cboMaterial(i).ItemData(cboMaterial(i).ListIndex) - 1 = 2 Or _
                    cboMaterial(i).ItemData(cboMaterial(i).ListIndex) - 1 = 4 Then 'acrylic = lexan = other:Plastic
                    lstrLiteType = lstrLiteType & "1"
            Else
                lstrLiteType = lstrLiteType & "0"   'glass, other
            End If
        Else
            If cboMaterial(i).ItemData(cboMaterial(i).ListIndex) - 1 = 3 Then 'other: Glass
                
                lstrLiteType = lstrLiteType & "0"
            
            ElseIf cboMaterial(i).ItemData(cboMaterial(i).ListIndex) - 1 = 4 Then 'other: Plastic
                
                lstrLiteType = lstrLiteType & "3"
            
            Else
            
                lstrLiteType = lstrLiteType & CStr(cboMaterial(i).ItemData(cboMaterial(i).ListIndex) - 1)

            End If
            
        End If
        
        '4th digit - finish
        '1-clear
        '2-tinted
        '3-tempered
        '4-other
        '0-n/a (for other than glass)
        
        'fixed in 2.0.18 - before 90019001 should be 90009000
'        If chkByOthers.Value = 1 Then   'If they check the box to signify that they are supplying their own glass, for springing purposes, we don't care what finish they used (clear, tempered, tinted or other).
'            lstrLiteType = lstrLiteType & "1"
'        Else
'            If cboMaterial(i).ItemData(cboMaterial(i).ListIndex) = 1 Then
'                lstrLiteType = lstrLiteType & CStr(cboFinish(i).ItemData(cboFinish(i).ListIndex))
'            Else
'                lstrLiteType = lstrLiteType & "0"
'            End If
'        End If




        
        
        'If cboMaterial(0).ItemData(cboMaterial(0).ListIndex) = 1 Or cboMaterial(0).ItemData(cboMaterial(0).ListIndex) = 4 Or cboMaterial(0).ItemData(cboMaterial(0).ListIndex) = 5 Then
        
            If chkByOthers.Value = 1 Then   'If they check the box to signify that they are supplying their own glass, for springing purposes, we don't care what finish they used (clear, tempered, tinted or other).
                lstrLiteType = lstrLiteType & "1"   '90019001
            Else
                If cboMaterial(i).ItemData(cboMaterial(i).ListIndex) = 1 Then
                    lstrLiteType = lstrLiteType & CStr(cboFinish(i).ItemData(cboFinish(i).ListIndex))
                Else
                    
                    If cboMaterial(i).ItemData(cboMaterial(i).ListIndex) = 4 Then
                        lstrLiteType = lstrLiteType & "4"
                    Else
                        lstrLiteType = lstrLiteType & "0"
                    End If
                    
                    
                End If
            End If
        'End If
        
        
        
    Next
    
    '1/2" acrylic + Lexan + Other: Plastic - repeat the base code
    
'    If cboMaterial(0).ItemData(cboMaterial(0).ListIndex) <> 2 Or cboMaterial(0).ItemData(cboMaterial(0).ListIndex) <> 3 Then
'        If frmSectionsFullView.chkSealedUnit.Value = 1 Then
'            lstrLiteType = lstrLiteType & lstrLiteType
'        End If
'    End If
    
    varDoorData.LiteType = CLng(lstrLiteType)
    
    Exit Sub

Error_Handler:
    llngErrNo = IIf(llngErrNo = 0, Err.Number, llngErrNo)
    lstrErrDescr = IIf(Len(lstrErrDescr) = 0, Err.Description, lstrErrDescr & vbCrLf & Err.Description)
    lstrErrSource = IIf(Len(Trim(Err.Source)) = 0, "", Err.Source & vbCrLf) & cstrErrorSource
    Call Err.Raise(llngErrNo, lstrErrSource, lstrErrDescr)

End Sub
