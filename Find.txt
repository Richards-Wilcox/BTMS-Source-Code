Private Function SearchGrid(ByVal mblnFirst As Boolean, ByVal pgrdData As VSFlexGrid, ByVal pstrFindWhat As String) As Long
    On Error GoTo Error_Handler
    Const cstrErrorSource As String = cstrApplicationName & "." & cstrModuleName & ".SearchGrid()"
    Dim llngErrNo As Long
    Dim lstrErrDescr As String
    Dim lstrErrSource As String
    llngErrNo = 0
    lstrErrDescr = ""
    lstrErrSource = ""
    
    Dim lintStartRow As Integer
    Dim i As Integer
    Dim lintPos As Integer
    Dim lintCol As Integer
    
    'set column where to search according to form name.
    Select Case lobjForm.Name
    Case "frmBOMList"
        lintCol = 1
    Case "frmPOList"
        lintCol = 2
    Case "frmQuoteList"
        lintCol = 2
    Case "frmPOBOMAdd"
         lintCol = 0
    End Select
    
'    If mblnFirst = True Then
'        lintStartRow = 1
'    Else
'         If lobjForm.Name = "frmPOBOMAdd" Then 'YS -- InHouse Search on Po Add remove
'                 lintStartRow = 2
'        Else
'                lintStartRow = pgrdData.Row + 1
'        End If
'    End If


    If mblnFirst = True Then
             If lobjForm.Name = "frmPOBOMAdd" Then 'YS -- InHouse Search on Po Add remove
        
                 lintStartRow = 2
             Else
                 
                 lintStartRow = 1
              End If
    Else

        lintStartRow = pgrdData.Row + 1


    End If


    
    For i = lintStartRow To pgrdData.Rows - 1
        lintPos = InStr(1, pgrdData.TextMatrix(i, lintCol), pstrFindWhat, vbTextCompare)
'
'        If lintPos > 0 Then
'            Exit For
'        End If

         If lobjForm.Name = "frmPOBOMAdd" Then 'YS -- InHouse Search on Po Add remove
              If lintPos = 1 Then  'Only find that begins with
                Exit For
              End If
         Else
                 If lintPos > 0 Then
                    Exit For
                End If
         
         End If



    Next
    
    If i = pgrdData.Rows Then i = -1 'no match
    SearchGrid = i
    
    Exit Function

Error_Handler:
    llngErrNo = IIf(llngErrNo = 0, Err.Number, llngErrNo)
    lstrErrDescr = IIf(Len(lstrErrDescr) = 0, Err.Description, lstrErrDescr & vbCrLf & Err.Description)
    lstrErrSource = IIf(Len(Trim(Err.Source)) = 0, "", Err.Source & vbCrLf) & cstrErrorSource
    Call Err.Raise(llngErrNo, lstrErrSource, lstrErrDescr)

End Function

Private Sub Form_Load()
    On Error GoTo Error_Handler
    Const cstrErrorSource As String = cstrApplicationName & "." & cstrModuleName & ".Form_Load()"
    Dim llngErrNo As Long
    Dim lstrErrDescr As String
    Dim lstrErrSource As String
    llngErrNo = 0
    lstrErrDescr = ""
    lstrErrSource = ""
    Dim currBPP As Long
    currBPP = GetDeviceCaps(hdc, BITSPIXEL)
    If currBPP <= 8 Then Call GoGrey(Me)
    Me.Move Screen.Width / 2 - Me.Width / 2, Screen.Height / 2 - Me.Height / 2
    cmdFind.Enabled = False
    cmdFindNext.Enabled = False
    LoadHelpText Me
    MousePos1.Init Me
    mblnInFocus = True
    Exit Sub

Error_Handler:
    llngErrNo = IIf(llngErrNo = 0, Err.Number, llngErrNo)
    lstrErrDescr = IIf(Len(lstrErrDescr) = 0, Err.Description, lstrErrDescr & vbCrLf & Err.Description)
    Call DisplayError(llngErrNo, IIf(Len(Trim(Err.Source)) = 0, "", Err.Source & vbCrLf) & cstrErrorSource, lstrErrDescr)
End Sub
