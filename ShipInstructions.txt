Private Sub SaveSpecialPartNew(ByVal lngBOMID As Long)
    On Error GoTo Error_Handler
    Const cstrErrorSource As String = cstrApplicationName & "." & cstrModuleName & ".SaveSpecialPartNew()"
    Dim llngErrNo As Long
    Dim lstrErrDescr As String
    Dim lstrErrSource As String
    llngErrNo = 0
    lstrErrDescr = ""
    lstrErrSource = ""
    
    Dim lobjBOMDetRS As ADODB.Recordset
    Dim llngAffected As Long
    Dim lstrSQL As String
    Dim lintCnt As Integer
    Dim lintTopNum As Integer
    'finds the highest non stock number added to the Bom and saves in the registry.
    
    lstrSQL = "select * from BOMDetails where bomid=" & lngBOMID
    Set lobjBOMDetRS = CnnBTMS.Execute(lstrSQL, llngAffected, adCmdText)
    If Not lobjBOMDetRS Is Nothing Then
        If Not (lobjBOMDetRS.EOF And lobjBOMDetRS.BOF) Then
            Do While lobjBOMDetRS.EOF = False
                If Mid(lobjBOMDetRS.Fields("BOMPtNum").Value, 1, 9) = "COMM-NSTK" Then
                    lintCnt = DBLValue(Mid(lobjBOMDetRS.Fields("BOMPtNum").Value, 11))
                    If lintCnt > lintTopNum Then lintTopNum = lintCnt
                End If
                lobjBOMDetRS.MoveNext
            Loop
        End If
    End If
    Set lobjBOMDetRS = Nothing
    
    If lintTopNum <= 0 Then lintTopNum = 0
    
    aReg.OpenKey LocalMachine, "\Software\Richards-Wilcox Canada\BTMS", True
    aReg.WriteValue "LastNumber", CStr(lintTopNum)
    aReg.CloseKey
    Exit Sub

Error_Handler:
    llngErrNo = IIf(llngErrNo = 0, Err.Number, llngErrNo)
    lstrErrDescr = IIf(Len(lstrErrDescr) = 0, Err.Description, lstrErrDescr & vbCrLf & Err.Description)
    lstrErrSource = IIf(Len(Trim(Err.Source)) = 0, "", Err.Source & vbCrLf) & cstrErrorSource
    Call Err.Raise(llngErrNo, lstrErrSource, lstrErrDescr)

End Sub

Private Sub Form_Load()

    On Error GoTo Error_Handler
    Const cstrErrorSource As String = cstrApplicationName & "." & cstrModuleName & ".Form_Load()"
    Dim llngErrNo As Long
    Dim lstrErrDescr As String
    llngErrNo = 0
    lstrErrDescr = ""
    
    Dim lstrSQL As String
    Dim lobjRS As ADODB.Recordset
    Dim llngAffected As Long
    Dim ldblPrice As Double
    Dim ldblPrice1 As Double

    Dim currBPP As Long
    currBPP = GetDeviceCaps(hdc, BITSPIXEL)
    If currBPP <= 8 Then Call GoGrey(Me)
    
'-- Mod Hanobver New MultId

'  If aCryp.GetOption(16, 7) = CKOptionOff Then
    If strMultId = "MultId" Then

    lstrSQL = "select PartNum, G,I,N,T , MultID, SpecialMult from Main where PartNum=" & SQLString("911-230") & _
    " union select PartNum, G,I,N,T , MultID, SpecialMult from Main where Partnum=" & SQLString("911-231")
    
    Else 'New MultId


    lstrSQL = "select PartNum, J,M,S,V , NewMultID, SpecialMult from Main where PartNum=" & SQLString("911-230") & _
    " union select PartNum, J,M,S,V , NewMultID, SpecialMult from Main where Partnum=" & SQLString("911-231")

  End If

    
    
    
    
    Set lobjRS = CnnBTMS.Execute(lstrSQL, llngAffected, adCmdText)
    If Not lobjRS Is Nothing Then
        If Not (lobjRS.EOF And lobjRS.BOF) Then
            lobjRS.MoveFirst
            Do While lobjRS.EOF = False
                Select Case intCheck
                    Case 1: ldblPrice = DBLValue(lobjRS.Fields("G").Value)
                    Case 2: ldblPrice = DBLValue(lobjRS.Fields("I").Value)
                    Case 3: ldblPrice = DBLValue(lobjRS.Fields("N").Value)
                    Case 4: ldblPrice = DBLValue(lobjRS.Fields("T").Value)
                    
                    Case 5: ldblPrice = DBLValue(lobjRS.Fields("J").Value)
                    Case 6: ldblPrice = DBLValue(lobjRS.Fields("M").Value)
                    Case 7: ldblPrice = DBLValue(lobjRS.Fields("S").Value)
                    Case 8: ldblPrice = DBLValue(lobjRS.Fields("V").Value)

                
                End Select
                ldblPrice = Decrypt(GetSym(lobjRS.Fields("PartNum").Value), ldblPrice)

                If lobjRS.Fields("SpecialMult").Value > 0 Then
                    ldblPrice = DBLValue(lobjRS.Fields("SpecialMult").Value) * ldblPrice
                Else
                      '-- Mod Hanobver  MultId changed to strMultId

                    If Not Trim(lobjRS.Fields(strMultId).Value) = "" Then
                    
                        ldblPrice = ldblPrice * (MultipliersCase(DBLValue(Trim(lobjRS.Fields(strMultId).Value))))
                    
                    End If
                End If
                ldblPrice1 = ldblPrice1 + ldblPrice
                lobjRS.MoveNext
            Loop
        End If
    End If
    
    chkRWBox.Caption = chkRWBox.Caption & " ($" & Format(ldblPrice1, "Fixed") & ")"
    aReg.OpenKey LocalMachine, "\Software\Richards-Wilcox Canada\BTMS", True
    chkRWBox.Value = DBLValue(aReg.ReadValue("RWBox"))
    aReg.CloseKey
    Me.Move (Screen.Width - Me.Width) / 2, (Screen.Height - Me.Height) / 2
    mstrBOMNum = ""
    cboStdInstr.Text = cboStdInstr.List(0)
    LoadHelpText Me
    MousePos1.Init Me
    mblnInFocus = True
    
    Exit Sub

Error_Handler:
    llngErrNo = IIf(llngErrNo = 0, Err.Number, llngErrNo)
    lstrErrDescr = IIf(Len(lstrErrDescr) = 0, Err.Description, Err.Description)
    Call DisplayError(llngErrNo, IIf(Len(Trim(Err.Source)) = 0, "", Err.Source & vbCrLf) & cstrErrorSource, lstrErrDescr)

End Sub
Private Sub ProcessQuote()
    On Error GoTo Error_Handler
    Const cstrErrorSource As String = cstrApplicationName & "." & cstrModuleName & ".ProcessQuote()"
    Dim llngErrNo As Long
    Dim lstrErrDescr As String
    Dim lstrErrSource As String
    llngErrNo = 0
    lstrErrDescr = ""
    lstrErrSource = ""
    
    If mlngBOMID < 1 Then Exit Sub
    
    Dim llngAffected As Long
    Dim lstrSQLQuery As String
    Dim lblnInTrans As Boolean
    
    lblnInTrans = False
    
    CnnBTMS.BeginTrans
    lblnInTrans = True
    
    mlngQuoteID = CreateQuote()
    Call AddToQuote
    
    'Update the Quote number on the BoM
    lstrSQLQuery = "Update BoM"
    lstrSQLQuery = lstrSQLQuery & " Set QuoteID=" & mlngQuoteID
    lstrSQLQuery = lstrSQLQuery & " , QuoteNum=" & SQLString(mstrBOMNum)
    lstrSQLQuery = lstrSQLQuery & " Where bomid=" & mlngBOMID
    Call CnnBTMS.Execute(lstrSQLQuery, llngAffected, adCmdText And adExecuteNoRecords)
        
    Call UpdateQuoteTotal(mlngQuoteID)
    Call UpdateQuoteTotalPrice(mlngQuoteID)
    CnnBTMS.CommitTrans
    
    lblnInTrans = False
    Exit Sub

Error_Handler:
    llngErrNo = IIf(llngErrNo = 0, Err.Number, llngErrNo)
    lstrErrDescr = IIf(Len(lstrErrDescr) = 0, Err.Description, lstrErrDescr & vbCrLf & Err.Description)
    lstrErrSource = IIf(Len(Trim(Err.Source)) = 0, "", Err.Source & vbCrLf) & cstrErrorSource
    If lblnInTrans = False Then CnnBTMS.RollbackTrans
    Call Err.Raise(llngErrNo, lstrErrSource, lstrErrDescr)
End Sub

Private Function BuildQuoteDetailRS() As ADODB.Recordset
    On Error GoTo Error_Handler
    Const cstrErrorSource As String = cstrApplicationName & "." & cstrModuleName & ".BuildQuoteDetailRS()"
    Dim llngErrNo As Long
    Dim lstrErrDescr As String
    Dim lstrErrSource As String
    llngErrNo = 0
    lstrErrDescr = ""
    lstrErrSource = ""

    Dim lstrSQL As String
    Dim lstrSchema As String
    Dim llngAffected As Long
    Dim lobjADORS As ADODB.Recordset
    Dim lobjBOMDetRS As ADODB.Recordset
    
    ' Get empty recordset
    lstrErrDescr = "Error getting empty ADO recordset."
    lstrSQL = "Select * from QuoteDetails where 1=0"
    Set lobjADORS = New ADODB.Recordset
    lobjADORS.CursorType = adOpenForwardOnly
    lobjADORS.CursorLocation = adUseClient
    lobjADORS.Open lstrSQL, CnnBTMS, adOpenForwardOnly, adLockOptimistic
    Set lobjADORS.ActiveConnection = Nothing
    lstrErrDescr = ""
       
    lstrSQL = "Select * from BOMDetails where BOMID=" & mlngBOMID
    Set lobjBOMDetRS = CnnBTMS.Execute(lstrSQL, llngAffected, adCmdText)
    If Not lobjBOMDetRS Is Nothing Then
        lobjBOMDetRS.ActiveConnection = Nothing
        If Not (lobjBOMDetRS.BOF And lobjBOMDetRS.EOF) Then
            lobjBOMDetRS.MoveFirst
            Do While lobjBOMDetRS.EOF = False
                If lobjBOMDetRS.Fields("BOMPtNum").Value <> "" And lobjBOMDetRS.Fields("BOMPttotamountReq").Value > 0 Then
                    lobjADORS.AddNew
                    lobjADORS.Fields("QuoteID").Value = mlngQuoteID
                    lobjADORS.Fields("QuoteNum").Value = mstrBOMNum
                    lobjADORS.Fields("QuotePtNum").Value = IIf(IsNull(lobjBOMDetRS.Fields("BOMPtNum").Value), "", Trim(lobjBOMDetRS.Fields("BOMPtNum").Value))
                    lobjADORS.Fields("QuotePtDesc").Value = IIf(IsNull(lobjBOMDetRS.Fields("BOMPtDesc").Value), "N/A", Trim(lobjBOMDetRS.Fields("BOMPtDesc").Value))
                    lobjADORS.Fields("QuotePtCallPrice").Value = IIf(IsNull(lobjBOMDetRS.Fields("BOMCallPrice").Value), 0, lobjBOMDetRS.Fields("BOMCallPrice").Value)
                    If lobjADORS.Fields("QuotePtCallPrice").Value = True Then lobjADORS.Fields("QuoteCallPriceExpDate").Value = IIf(IsNull(lobjBOMDetRS.Fields("BOMCallPriceExpDate").Value), Now() + 30, lobjBOMDetRS.Fields("BOMCallPriceExpDate").Value)
                    lobjADORS.Fields("QuotePtCost").Value = IIf(IsNull(lobjBOMDetRS.Fields("BOMPtMtCost").Value), 0, lobjBOMDetRS.Fields("BOMPtMtCost").Value)
                    lobjADORS.Fields("QuotePtNotes").Value = "" 'IIf(IsNull(lobjBOMDetRS.Fields("BOMPtNotes").Value), "", Trim(lobjBOMDetRS.Fields("BOMPtNotes").Value))

Private Function CreateQuote() As Long
    On Error GoTo Error_Handler
    Const cstrErrorSource As String = cstrApplicationName & "." & cstrModuleName & ".CreateQuote()"
    Dim llngErrNo As Long
    Dim lstrErrDescr As String
    Dim lstrErrSource As String
    llngErrNo = 0
    lstrErrDescr = ""
    lstrErrSource = ""

    Dim llngAffected As Long
    Dim lstrSQLQuery As String
    Dim lobjBOMRS As ADODB.Recordset
    Dim lobjQuoteRS As ADODB.Recordset
    Dim llngQuoteID As Long
    
    If Not mlngBOMID > 0 Then Exit Function
    
    llngQuoteID = 0
    lstrSQLQuery = "select * from BoM where BOMID=" & mlngBOMID
    Set lobjBOMRS = CnnBTMS.Execute(lstrSQLQuery, llngAffected, adCmdText)
    If Not lobjBOMRS Is Nothing Then
        lobjBOMRS.ActiveConnection = Nothing
        If Not (lobjBOMRS.BOF And lobjBOMRS.EOF) Then
            lobjBOMRS.MoveFirst
            Set lobjQuoteRS = BuildQuoteRS(lobjBOMRS)
            If Not lobjQuoteRS Is Nothing Then
                If Not (lobjQuoteRS.BOF And lobjQuoteRS.EOF) Then
                    lobjQuoteRS.MoveFirst
                    lstrErrDescr = "Error building insert statement for Quote"
                    lstrSQLQuery = Build_Insert(lobjQuoteRS, "Quote", "QuoteID")
                    lstrErrDescr = "Error inserting data into Quote"
                    
                    ' Insert the new Quote
                    Call CnnBTMS.Execute(lstrSQLQuery, 0, adCmdText And adExecuteNoRecords)
                    lstrErrDescr = ""
                    lobjQuoteRS.Close
                    
                    ' Find the QuoteID
                    lstrSQLQuery = "select QuoteID from Quote where QuoteNum=" & SQLString(mstrBOMNum)
                    Set lobjQuoteRS = CnnBTMS.Execute(lstrSQLQuery, llngAffected, adCmdText)
                    If Not lobjQuoteRS Is Nothing Then
                        lobjQuoteRS.ActiveConnection = Nothing
                        If Not (lobjQuoteRS.BOF And lobjQuoteRS.EOF) Then
                            lobjQuoteRS.MoveFirst
                            llngQuoteID = IIf(IsNull(lobjQuoteRS.Fields("QuoteID").Value), 0, lobjQuoteRS.Fields("QuoteID").Value)
                        End If
                        lobjQuoteRS.Close
                    End If
                End If
            End If
        End If
        lobjBOMRS.Close
    End If
    
    Set lobjQuoteRS = Nothing
    Set lobjBOMRS = Nothing
    CreateQuote = llngQuoteID
    Exit Function

Error_Handler:
    llngErrNo = IIf(llngErrNo = 0, Err.Number, llngErrNo)
    lstrErrDescr = IIf(Len(lstrErrDescr) = 0, Err.Description, lstrErrDescr & vbCrLf & Err.Description)
    lstrErrSource = IIf(Len(Trim(Err.Source)) = 0, "", Err.Source & vbCrLf) & cstrErrorSource
    Call Err.Raise(llngErrNo, lstrErrSource, lstrErrDescr)
End Function

Private Sub AddToQuote()
     On Error GoTo Error_Handler
    Const cstrErrorSource As String = cstrApplicationName & "." & cstrModuleName & ".AddToQuote()"
    Dim llngErrNo As Long
    Dim lstrErrDescr As String
    Dim lstrErrSource As String
    llngErrNo = 0
    lstrErrDescr = ""
    lstrErrSource = ""

    Dim lstrSQL As String
    Dim lintMode As Integer
    Dim lstrWhereClause As String
    Dim lstrExcludeFieldList As String
    Dim llngAffected As Long
    Dim lobjADORS As ADODB.Recordset
    Dim lobjParentRS As ADODB.Recordset
    
    ' Copy the fields from the screen to an ADO Recordset
    Set lobjADORS = BuildQuoteDetailRS()
    Call CopyLenDetails
    lintMode = MODE_ADD
    ' Process through the recordset and insert or update rows
    If Not lobjADORS Is Nothing Then
        If Not (lobjADORS.BOF And lobjADORS.EOF) Then
            lobjADORS.MoveFirst
            lstrExcludeFieldList = ""
            
            Do While lobjADORS.EOF = False
                lintMode = MODE_ADD
                ' Check to see if the record exists in the
                ' BOMDetails table already
                lstrSQL = "select QuotePtNum, QuotePtQty, QuoteTotAmountReq, QuoteDiscount, QuotePtMultID, QuotePtCost from QuoteDetails"
                lstrSQL = lstrSQL & " where QuoteID=" & mlngQuoteID
                lstrSQL = lstrSQL & " and QuotePtNum=" & SQLString(lobjADORS.Fields("QuotePtNum").Value)
                Set lobjParentRS = CnnBTMS.Execute(lstrSQL, llngAffected, adCmdText)
                If Not lobjParentRS Is Nothing Then
                    If Not (lobjParentRS.BOF And lobjParentRS.EOF) Then
                        lobjParentRS.MoveFirst
                        ' Reset the Mode so that we do an update of the existing part
                        lintMode = MODE_EDIT
                        ' Average the cost and sum the qty's
                        If lobjADORS.Fields("QuotePtMultID").Value <> lobjParentRS.Fields("QuotePtMultID").Value Then
                            lobjADORS.Fields("QuotePtCallPrice").Value = True
                            lobjADORS.Fields("QuoteCallPriceExpDate").Value = Format(Now() + 30, "Short Date")
                        End If
                        If DBLValue(lobjADORS.Fields("QuoteDiscount").Value) + DBLValue(lobjParentRS.Fields("QuotePtMultID").Value) > 0 Then
                            If DBLValue(lobjParentRS.Fields("QuoteDiscount").Value) > 0 Then
                                If DBLValue(lobjADORS.Fields("QuoteDiscount").Value) > DBLValue(lobjParentRS.Fields("QuoteDiscount").Value) Or _
                                DBLValue(lobjADORS.Fields("QuoteDiscount").Value) = 0 Then
                                    lobjADORS.Fields("QuoteDiscount").Value = lobjParentRS.Fields("QuoteDiscount").Value
                                End If
                            End If
                            lobjADORS.Fields("QuotePtCallPrice").Value = True
                            lobjADORS.Fields("QuoteCallPriceExpDate").Value = Format(Now() + 30, "Short Date")
                        End If
                        lobjADORS.Fields("QuotePtCost").Value = DBLValue(lobjADORS.Fields("Quotetotamountreq").Value) * DBLValue(lobjADORS.Fields("QuotePtCost").Value)
                        lobjADORS.Fields("QuotePtCost").Value = DBLValue(lobjADORS.Fields("QuotePtCost").Value) + DBLValue(lobjParentRS.Fields("Quotetotamountreq").Value) * DBLValue(lobjParentRS.Fields("QuotePtCost").Value)
                        lobjADORS.Fields("QuotePtCost").Value = DBLValue(lobjADORS.Fields("QuotePtCost").Value) / IIf(DBLValue(lobjParentRS.Fields("Quotetotamountreq").Value) + DBLValue(lobjADORS.Fields("Quotetotamountreq").Value) = 0, 1, DBLValue(lobjParentRS.Fields("Quotetotamountreq").Value) + DBLValue(lobjADORS.Fields("Quotetotamountreq").Value))
                        lobjADORS.Fields("QuotePtQty").Value = DBLValue(lobjADORS.Fields("QuotePtQty").Value) + DBLValue(lobjParentRS.Fields("QuotePtQty").Value)
                        lobjADORS.Fields("Quotetotamountreq").Value = DBLValue(lobjADORS.Fields("Quotetotamountreq").Value) + DBLValue(lobjParentRS.Fields("Quotetotamountreq").Value)
                    Else
                        lobjADORS.Fields("QuoteID").Value = mlngQuoteID
                        lstrErrDescr = "Error building insert statement for QuoteDetails"
                        lstrSQL = Build_Insert(lobjADORS, "QuoteDetails", lstrExcludeFieldList)
                        lstrErrDescr = "Error inserting data into QuoteDetails"
                        lobjParentRS.Close
                        Set lobjParentRS = Nothing
                        GoTo Label
                    End If
                    lobjParentRS.Close
                End If
                Set lobjParentRS = Nothing

                'Build the SQL Insert or Update statement
                If lintMode = MODE_ADD Then
                    lobjADORS.Fields("QuoteID").Value = mlngQuoteID
                    lstrErrDescr = "Error building insert statement for QuoteDetails"
                    lstrSQL = Build_Insert(lobjADORS, "QuoteDetails", lstrExcludeFieldList)
                    lstrErrDescr = "Error inserting data into QuoteDetails"
                Else
                    lstrErrDescr = "Error building update statement for QuoteDetails"
                    lstrWhereClause = "QuoteID=" & mlngQuoteID
                    lstrWhereClause = lstrWhereClause & " and QuotePtNum=" & SQLString(lobjADORS.Fields("QuotePtNum").Value)
                    lstrSQL = Build_Update(lobjADORS, "QuoteDetails", lstrWhereClause, "QuoteID, QuoteNum")
                    lstrErrDescr = "Error updating QuoteDetails"
                End If
Label:
                Call CnnBTMS.Execute(lstrSQL, 0, adCmdText And adExecuteNoRecords)
                lstrErrDescr = ""
                lobjADORS.MoveNext
            Loop
            
            lobjADORS.Close
        End If
    End If
    
    Set lobjADORS = Nothing
    
    Exit Sub

Error_Handler:
    llngErrNo = IIf(llngErrNo = 0, Err.Number, llngErrNo)
    lstrErrDescr = IIf(Len(lstrErrDescr) = 0, Err.Description, lstrErrDescr & vbCrLf & Err.Description)
    lstrErrSource = IIf(Len(Trim(Err.Source)) = 0, "", Err.Source & vbCrLf) & cstrErrorSource
    Call Err.Raise(llngErrNo, lstrErrSource, lstrErrDescr)
End Sub
