Option Explicit
'KN
Private Const cstrModuleName As String = "frmInvMenu"
Public mstrPass As String
Public mblnCancel As Boolean

Private Sub cmdInvReceipt_Click()

    On Error GoTo Error_Handler
    Const cstrErrorSource As String = cstrApplicationName & "." & cstrModuleName & ".cmdInvReceipt_Click()"
    Dim llngErrNo As Long
    Dim lstrErrDescr As String
    Dim lstrErrSource As String
    llngErrNo = 0
    lstrErrDescr = ""
    lstrErrSource = ""
    Screen.MousePointer = vbHourglass
    Unload Me
    frmInvReceipts.Show
    Screen.MousePointer = vbDefault
    Exit Sub

Error_Handler:
    Screen.MousePointer = vbDefault
    llngErrNo = IIf(llngErrNo = 0, Err.Number, llngErrNo)
    lstrErrDescr = IIf(Len(lstrErrDescr) = 0, Err.Description, Err.Description)
    Call DisplayError(llngErrNo, IIf(Len(Trim(Err.Source)) = 0, "", Err.Source & vbCrLf) & cstrErrorSource, lstrErrDescr)

End Sub

Private Sub cmdInvValuation_Click()
    On Error GoTo Error_Handler
    Const cstrErrorSource As String = cstrApplicationName & "." & cstrModuleName & ".cmdInvValuation_Click()"
    Dim llngErrNo As Long
    Dim lstrErrDescr As String
    Dim lstrErrSource As String
    llngErrNo = 0
    lstrErrDescr = ""
    lstrErrSource = ""
    
    Dim lstrREGPass As String
    aReg.OpenKey LocalMachine, "\Software\Computer-ID", True 'opens or creates key
    lstrREGPass = aReg.ReadValue("SetUpX")
    aReg.CloseKey

    Dim lobjFrmX As frmInvValuationPassword
    Set lobjFrmX = New frmInvValuationPassword
    Set lobjFrmX.lobjParent = Me
    lobjFrmX.Show vbModal
    Set lobjFrmX = Nothing
    
    If mblnCancel = True Then 'cancelled from password form
        Exit Sub
    Else
        If mstrPass = "" Then 'did not enter a password
            If lstrREGPass = "" Then 'did not set a password
                frmInvValuation.Show vbModal 'OK
            Else    'password set, cannot enter without entering one.
                MsgBox "Wrong password!", vbExclamation + vbApplicationModal, "Wrong Password."
            End If
        Else 'entered password (even if no password set, when they try to enter one they get the message.
            If StrComp(mstrPass, lstrREGPass, vbTextCompare) <> 0 Then
                MsgBox "Wrong password!", vbExclamation + vbApplicationModal, "Wrong Password."
            Else 'correct password
                frmInvValuation.Show vbModal
            End If
        End If
    End If
    
    Exit Sub

Error_Handler:
    Screen.MousePointer = vbDefault
    llngErrNo = IIf(llngErrNo = 0, Err.Number, llngErrNo)
    lstrErrDescr = IIf(Len(lstrErrDescr) = 0, Err.Description, lstrErrDescr & vbCrLf & Err.Description)
    Call DisplayError(llngErrNo, IIf(Len(Trim(Err.Source)) = 0, "", Err.Source & vbCrLf) & cstrErrorSource, lstrErrDescr)

End Sub

Private Sub Form_Load()

    On Error GoTo Error_Handler
    Const cstrErrorSource As String = cstrApplicationName & "." & cstrModuleName & ".Form_Load()"
    Dim llngErrNo As Long
    Dim lstrErrDescr As String
    llngErrNo = 0
    lstrErrDescr = ""
    Dim currBPP As Long
    currBPP = GetDeviceCaps(hdc, BITSPIXEL)
    If currBPP <= 8 Then Call GoGrey(Me)
'    Me.Width = 3855
'    Me.Height = 4485
    Me.Move Screen.Width / 2 - Me.Width / 2, Screen.Height / 2 - Me.Height / 2
    
    cmdInvReceipt.Enabled = gblnDistributor
    
    If IsMetricDefault = True Then
        OptMetric.Value = True
    Else
        OptImperial.Value = True
    End If
    
    Exit Sub

Error_Handler:
    llngErrNo = IIf(llngErrNo = 0, Err.Number, llngErrNo)
    lstrErrDescr = IIf(Len(lstrErrDescr) = 0, Err.Description, Err.Description)
    Call DisplayError(llngErrNo, IIf(Len(Trim(Err.Source)) = 0, "", Err.Source & vbCrLf) & cstrErrorSource, lstrErrDescr)

End Sub
