Private Sub cmdAllocate_Click()
    
    On Error GoTo Error_Handler
    Const cstrErrorSource As String = cstrApplicationName & "." & cstrModuleName & ".cmdAllocate_Click()"
    Dim llngErrNo As Long
    Dim lstrErrDescr As String
    Dim lstrErrSource As String
    Dim lstrSQL As String
    Dim llngAffected As Long
    llngErrNo = 0
    lstrErrDescr = ""
    lstrErrSource = ""
    'Screen.MousePointer = vbHourglass
    If mlngBOMID > 0 Then
        If cmdAllocate.Caption = "A&llocate" Then
            If CountDetails < 1 Then
                MsgBox "The BoM has no details.", vbExclamation + vbApplicationModal, "Cannot Allocate"
                Screen.MousePointer = vbDefault
                Exit Sub
            End If
            If CheckBOMforAllocation(mlngBOMID, mstrBOMUM, True) = True Then
                MsgBox "Cannot allocate this BoM.", vbExclamation + vbApplicationModal, "Cannot Allocate"
                Screen.MousePointer = vbDefault
                Exit Sub
            End If
            
            If frmBOMAlloc.grdData1.TextMatrix(1, 0) <> "" Then
                frmBOMAlloc.mlngDOCID = mlngBOMID
                frmBOMAlloc.mdblYTD = GetBoMTotal(mlngBOMID)
                frmBOMAlloc.mblnNow = True
                frmBOMAlloc.mintBackTag = 1
                mblnInFocus = False
                frmBOMAlloc.Show vbModal
                mblnInFocus = True
            Else
                If Me.lblReserve.Caption <> "" Then Call UnAllocateBOM(mlngBOMID)  'unreserve first
                Call AllocateBOM(mlngBOMID)
                Dim lsngYTD As Single
                
                If mlngCurrCust > 0 Then
                    lsngYTD = DBLValue(InputBox("Please enter the amount you wish to go towards the Customer's Year-to-Date Sales.", "Customer's YTD"))
                    Call SetCustYTD(mlngBOMID, lsngYTD)
                lstrSQL = "update Bom set BoMCustYTD=" & lsngYTD
                Call CnnBTMS.Execute(lstrSQL, llngAffected, adCmdText And adExecuteNoRecords)
                End If
                MsgBox "BoM has been allocated.", vbApplicationModal + vbOKOnly + vbInformation, "BoM Allocated"
                
            End If
        Else
            If CountDetails < 1 Then
                MsgBox "The BoM has no details.", vbExclamation + vbApplicationModal, "Cannot Un-Allocate"
                Screen.MousePointer = vbDefault
                Exit Sub
            End If
            Call UnAllocateBOM(mlngBOMID)
            If mlngCurrCust > 0 Then
                Call Un_SetCustYTD(mlngBOMID, GetCustBoMYTD(mlngBOMID))
            End If
            MsgBox "BoM un-allocated.", vbApplicationModal + vbOKOnly + vbInformation, "BoM Un-Allocated."
        End If
        
        Call LoadData
    End If
    Screen.MousePointer = vbDefault
    Exit Sub

Error_Handler:
    Screen.MousePointer = vbDefault
    llngErrNo = IIf(llngErrNo = 0, Err.Number, llngErrNo)
    lstrErrDescr = IIf(Len(lstrErrDescr) = 0, Err.Description, lstrErrDescr & vbCrLf & Err.Description)
    Call DisplayError(llngErrNo, IIf(Len(Trim(Err.Source)) = 0, "", Err.Source & vbCrLf) & cstrErrorSource, lstrErrDescr)
End Sub

Private Sub cmdReserve_Click()

    Dim llngErrNo As Long
    Dim lstrErrDescr As String
    Dim lstrErrSource As String

    lstrErrDescr = ""
    lstrErrSource = ""

            Exit Sub
        End If
        Call CheckBOMforAllocation(mlngBOMID, mstrBOMUM, False)

            frmBOMAlloc.mstrDocUM = mstrBOMUM

            frmBOMAlloc.mdblYTD = GetBoMTotal(mlngBOMID)
            mblnInFocus = False
            frmBOMAlloc.Show vbModal
            mblnInFocus = True
        Else
            frmReserveDate.mlngDOCID = mlngBOMID
            frmReserveDate.mintBackTag = 1
            frmReserveDate.mstrDocUM = mstrBOMUM
            frmReserveDate.mdblYTD = GetBoMTotal(mlngBOMID)
            mblnInFocus = False
            frmReserveDate.Show vbModal
            mblnInFocus = True
        End If
    Else
        If CountDetails < 1 Then
            MsgBox "The BoM has no details.", vbExclamation + vbApplicationModal, "Cannot Un-Reserve"
            Screen.MousePointer = vbDefault
            Exit Sub
        End If
        'Screen.MousePointer = vbHourglass
        Call UnAllocateBOM(mlngBOMID)
        If lblReserve.Caption = "" Then
            Call Un_SetCustYTD(mlngBOMID, GetBoMTotal(mlngBOMID))
        End If
        MsgBox "BoM un-reserved.", vbApplicationModal + vbOKOnly + vbInformation, "BoM Un-Reserved."

Private Sub cmdUpdateLbr_Click()
On Error GoTo Error_Handler
    Const cstrErrorSource As String = cstrApplicationName & "." & cstrModuleName & ".cmdUpdateLbr_Click()"
    Dim llngErrNo As Long
    Dim lstrErrDescr As String
    Dim lstrErrSource As String
    llngErrNo = 0
    lstrErrDescr = ""
    lstrErrSource = ""
    
    Dim lstrSQLQuery As String
    Dim TempLabRt As Double
    Dim Lbr As Double
    Dim OH As Double
    
    Dim llngAffected As Long
    
    Dim lobjBOMDetRS As ADODB.Recordset
    Dim lobjMainRS As ADODB.Recordset
    
    
    CnnBTMS.BeginTrans
    
    Screen.MousePointer = vbHourglass
       
     lstrSQLQuery = "select * from BOMDetails where BOMID=" & mlngBOMID
    lstrSQLQuery = lstrSQLQuery & " order by BOMptnum"
    Set lobjBOMDetRS = CnnBTMS.Execute(lstrSQLQuery, llngAffected, adCmdText)
    
    lstrSQLQuery = "SELECT PartNum, LbrCost, UM, Length FROM Main"
    Set lobjMainRS = New ADODB.Recordset
    lobjMainRS.CursorType = adOpenForwardOnly
    lobjMainRS.CursorLocation = adUseClient
    lobjMainRS.Open lstrSQLQuery, CnnBTMS, adOpenForwardOnly, adLockOptimistic
    
       
    If Not lobjBOMDetRS Is Nothing Then
        If Not (lobjBOMDetRS.BOF And lobjBOMDetRS.EOF) Then
            lobjBOMDetRS.MoveFirst
            
            Do While lobjBOMDetRS.EOF = False
                If Not lobjMainRS Is Nothing Then
                    lobjMainRS.ActiveConnection = Nothing
                    lobjMainRS.MoveFirst
                    lobjMainRS.Find "PartNum=" & SQLString(lobjBOMDetRS.Fields("BOMPtNum").Value)
                    If lobjMainRS.EOF = False And lobjMainRS.BOF = False Then
                        TempLabRt = DBLValue(lobjMainRS.Fields("LbrCost"))
                        
                        If lobjMainRS.Fields("UM").Value = "LFT" Then
                             TempLabRt = DBLValue(lobjMainRS.Fields("LbrCost")) / 12
                        End If
                        
                        If lobjMainRS.Fields("UM").Value = "EACH" And lobjBOMDetRS.Fields("BOMPtUM").Value = "INCHES" And Left(lobjBOMDetRS.Fields("BOMPtNum").Value, 4) <> "433-" Then
                            TempLabRt = lobjMainRS.Fields("LbrCost") / lobjMainRS.Fields("Length").Value
                        
                        End If
                        
                        Lbr = Lbr + (TempLabRt * lobjBOMDetRS.Fields("BOMPtTotAmountReq").Value)
                    End If
                End If
                lobjBOMDetRS.MoveNext
            Loop
        End If
        
    End If
      
    
                
    'get the admin saved Lbr rate, divide by 60, multiply by lbr.
    aReg.OpenKey LocalMachine, "\software\Richards-Wilcox Canada", True
    Lbr = IIf(DBLValue(aReg.ReadValue("LbrCost")) = 0, 0, DBLValue(aReg.ReadValue("LbrCost")) / 60 * Lbr)
    OH = Lbr * IIf(DBLValue(aReg.ReadValue("OHCost")) = 0, 0.2, DBLValue(aReg.ReadValue("OHCost")))
    aReg.CloseKey
    
    'Update the fields
    lstrSQLQuery = "update BoM Set BOMLabourCost=" & Replace(Lbr, ",", ".") & " , BOMOverheadCost=" & Replace(OH, ",", ".") & " where BomID=" & mlngBOMID
    Call CnnBTMS.Execute(lstrSQLQuery, 0, adCmdText And adExecuteNoRecords)
       
                    
    Set lobjMainRS = Nothing
    Set lobjBOMDetRS = Nothing
    
    CnnBTMS.CommitTrans
    
    Call LoadData
    
    Screen.MousePointer = vbDefault
    Exit Sub

Error_Handler:
    Screen.MousePointer = vbDefault
    CnnBTMS.RollbackTrans
    llngErrNo = IIf(llngErrNo = 0, Err.Number, llngErrNo)
    lstrErrDescr = IIf(Len(lstrErrDescr) = 0, Err.Description, lstrErrDescr & vbCrLf & Err.Description)
    Call DisplayError(llngErrNo, IIf(Len(Trim(Err.Source)) = 0, "", Err.Source & vbCrLf) & cstrErrorSource, lstrErrDescr)
End Sub

Private Sub SaveData()
    On Error GoTo Error_Handler
    Const cstrErrorSource As String = cstrApplicationName & "." & cstrModuleName & ".SaveData()"
    Dim llngErrNo As Long
    Dim lstrErrDescr As String
    Dim lstrErrSource As String
    llngErrNo = 0
    lstrErrDescr = ""
    lstrErrSource = ""

    Dim lstrSQL As String
    Dim lstrWhereClause As String
    Dim lblnInTrans As Boolean
    Dim lstrExcludeFieldList As String
    Dim llngAffected As Long
    Dim lobjADORS As ADODB.Recordset
    
    lblnInTrans = False
    
    If Not ValidateData(lobjADORS) Then Exit Sub
    ' Copy the fields from the screen to an ADO Recordset
    Set lobjADORS = BuildRS()
    
    ' Process through the recordset and insert or update rows
    If Not lobjADORS Is Nothing Then
        If Not (lobjADORS.BOF And lobjADORS.EOF) Then
            lobjADORS.MoveFirst
            CnnBTMS.BeginTrans
            lblnInTrans = True
            Do While lobjADORS.EOF = False
                If mintMode = MODE_ADD Then
                    lstrExcludeFieldList = "BOMID"
                    lstrErrDescr = "Error building insert statement for " & cstrTableName
                    lstrSQL = Build_Insert(lobjADORS, cstrTableName, lstrExcludeFieldList)
                    lstrErrDescr = "Error inserting data into " & cstrTableName
                    
                    mlngBOMID = GetID(cstrTableName, "BOMID", "BOMNum", Trim(lobjADORS.Fields("BOMNum").Value))
                Else
                    lstrExcludeFieldList = "BOMID, BOMNUM"
                    lstrErrDescr = "Error building update statement for " & cstrTableName
                    lstrWhereClause = "BOMID=" & mlngBOMID
                    lstrSQL = Build_Update(lobjADORS, cstrTableName, lstrWhereClause, lstrExcludeFieldList)
                    lstrErrDescr = "Error updating " & cstrTableName
                End If
                
                Call CnnBTMS.Execute(lstrSQL, 0, adCmdText And adExecuteNoRecords)
                lstrErrDescr = ""
                lobjADORS.MoveNext
            Loop
            
            lobjADORS.Close
            CnnBTMS.CommitTrans
            lblnInTrans = False
        End If
    End If
    
    Set lobjADORS = Nothing
    Exit Sub

Error_Handler:
    llngErrNo = IIf(llngErrNo = 0, Err.Number, llngErrNo)
    lstrErrDescr = IIf(Len(lstrErrDescr) = 0, Err.Description, Err.Description)
    lstrErrSource = IIf(Len(Trim(Err.Source)) = 0, "", Err.Source & vbCrLf) & cstrErrorSource
    If lblnInTrans = True Then CnnBTMS.RollbackTrans
    Call Err.Raise(llngErrNo, lstrErrSource, lstrErrDescr)
End Sub

Public Function ParseDocument(wField As Word.Field, WordDoc As Word.Document) As String
    On Error GoTo Error_Handler
    Const cstrErrorSource As String = cstrApplicationName & "." & cstrModuleName & ".GetNewResult()"
    Dim llngErrNo As Long
    Dim lstrErrDescr As String
    Dim lstrErrSource As String
    llngErrNo = 0
    lstrErrDescr = ""
    lstrErrSource = ""

    Dim lstrStopPos As Long
    Dim lstrWord As String
    Dim lstrWordValue As String
    Dim wRange As Word.Range
    
    ' These three lines strip down the field code to find
    ' out it's name
    lstrStopPos = InStrRev(wField.Code, "\*")
    lstrWord = Left(wField.Code, lstrStopPos - 3)
    lstrWord = Right(lstrWord, Len(lstrWord) - 14)
    
    ' Check this field hasn't already appeared in this
    ' document.
    Select Case UCase(lstrWord)
        Case "BOMNUM":          lstrWordValue = Trim(txtBOMNum.Text)
        Case "BOMDRMODEL":      lstrWordValue = Trim(lblDrModel.Caption)
        Case "BOMDRHEIGHT":     lstrWordValue = Trim(lblHeight.Caption)
        Case "BOMDRWIDTH":      lstrWordValue = Trim(lblWidth.Caption)
        Case "BOMFACECOLOUR":   lstrWordValue = Trim(lblDrColor.Caption)
        Case "BOMPRICE":        lstrWordValue = Trim(lblPrice.Caption)
        Case "BOMDRQTY":        lstrWordValue = Trim(lblDrQty.Caption)
        Case "BOMDRTYPE":       lstrWordValue = Trim(lblDrType.Caption)
            
        ' I don't simply want to insert a string -
        ' I wish to insert a table at the Product Field.
        Case "PRODUCTS"
                                        
            Dim lstrText As String
            Dim lstrSQLQuery As String
            Dim llngAffected As String
            lstrText = ""
            lstrSQLQuery = "select BOMPtNum, BOMPtDesc, BOMPtQtyReq, BOMPtQtyAvailable, BOMPtQtyUnAvailable,"
            lstrSQLQuery = lstrSQLQuery & " BOMPtMtCost, (BOMPtMtCost * BOMPtQtyReq), BOMPtUM"
            lstrSQLQuery = lstrSQLQuery & " from " & cstrTableName
            lstrSQLQuery = lstrSQLQuery & " where BOMnum=" & SQLString(Trim(mstrDocNum))
            lstrSQLQuery = lstrSQLQuery & " order by BOMptnum"
            Dim lobjBOMDetailRS As ADODB.Recordset
            Set lobjBOMDetailRS = CnnBTMS.Execute(lstrSQLQuery, llngAffected, adCmdText)
            lstrText = "Part No" & "~" & "Description" & "~" & "Req Qty" & "~" & "Order Qty" & "~" & "Cost" & "~" & "Total Cost" & vbCrLf
            If Not lobjBOMDetailRS Is Nothing Then
                lobjBOMDetailRS.ActiveConnection = Nothing
                If Not (lobjBOMDetailRS.BOF And lobjBOMDetailRS.EOF) Then
                    lobjBOMDetailRS.MoveFirst
                    lstrText = lstrText & lobjBOMDetailRS.GetString(adClipString, -1, "~", vbCrLf)
                End If
            End If
                        
            ' Get the range (location) of the product field
            Set wRange = wField.Code
            ' Delete the field, as any text will be inserted into the
            ' {} of the existing field.
            wField.Delete
            
            ' Enter our table information including headers.
            ' Ideally, I would get this data from an ADO recordset
            ' using GetString().
            With wRange
        
                .Text = lstrText
                .FormattedText.Font.Name = "Arial"
                .FormattedText.Font.Size = "8"
            
                ' Once the data is there, we can convert it to a table
                ' structure and format it to look pretty!
                .ConvertToTable "~", , , 50, wdTableFormatColorful2
                .Columns(1).Width = 110
                .Columns(2).Width = 140
            End With
            
            ' Send back blank string as field does not exist anymore
            lstrWordValue = ""
    
        Case Else
            
            ' Get the value of the field from the user
            lstrWordValue = lstrWord
    
    End Select
    
    ParseDocument = lstrWordValue
    Exit Function

Error_Handler:
    llngErrNo = IIf(llngErrNo = 0, Err.Number, llngErrNo)
    lstrErrDescr = IIf(Len(lstrErrDescr) = 0, Err.Description, Err.Description)
    lstrErrSource = IIf(Len(Trim(Err.Source)) = 0, "", Err.Source & vbCrLf) & cstrErrorSource
    Call Err.Raise(llngErrNo, lstrErrSource, lstrErrDescr)
End Function

                    lobjADORS.Close
                End If

            Set lobjADORS = Nothing
        End If
        lobjBOMDetRS.Close
    End If
    Set lobjBOMDetRS = Nothing
    
    
    ' ==== Copy the BoM Sections across ====
    lstrTableName = "BOMSection"
    
    ' Get empty recordset
    Set lobjADORS = New ADODB.Recordset
    lobjADORS.CursorType = adOpenForwardOnly
    lobjADORS.CursorLocation = adUseClient
    
    lstrSQLQuery = "select * from " & lstrTableName & " where BOMID=" & lngBOMID
    Set lobjBOMDetRS = CnnBTMS.Execute(lstrSQLQuery, llngAffected, adCmdText)
    If Not lobjBOMDetRS Is Nothing Then
        lobjBOMDetRS.ActiveConnection = Nothing
        If Not (lobjBOMDetRS.BOF And lobjBOMDetRS.EOF) Then
            lstrSQLQuery = "Select * from " & lstrTableName & " where 1=0"
            lobjADORS.Open lstrSQLQuery, CnnBTMS, adOpenForwardOnly, adLockOptimistic
            Set lobjADORS.ActiveConnection = Nothing
            
            lobjBOMDetRS.MoveFirst
            Do While lobjBOMDetRS.EOF = False
                lobjADORS.AddNew
                For Each lobjADOField In lobjBOMDetRS.Fields
                    If Not UCase(Trim(lobjADOField.Name)) = "BOMID" Then
                        If IsNull(lobjADOField.Value) = False Then
                            lobjADORS.Fields(lobjADOField.Name).Value = lobjADOField.Value
                        End If
                        If UCase(Trim(lobjADOField.Name)) = "BOMNUM" Then
                            lobjADORS.Fields("BOMNum").Value = lstrBOMNum
                        End If
                    Else
                        lobjADORS.Fields("BOMID").Value = llngNewBOMID
                    End If
                Next
                
                lobjBOMDetRS.MoveNext
            Loop
            
            ' Insert the BoM or BoM child record
            If Not lobjADORS Is Nothing Then
                If Not (lobjADORS.BOF And lobjADORS.EOF) Then
                    lobjADORS.MoveFirst
                    
                    Do While lobjADORS.EOF = False
                        lstrErrDescr = "Error building insert statement for " & lstrTableName
                        lstrSQLQuery = Build_Insert(lobjADORS, lstrTableName, "")
                        lstrErrDescr = "Error inserting data into " & lstrTableName
                        Call CnnBTMS.Execute(lstrSQLQuery, 0, adCmdText And adExecuteNoRecords)
                        lstrErrDescr = ""
                        lobjADORS.MoveNext
                    Loop
                    lobjADORS.Close
                End If
            End If
            Set lobjADORS = Nothing
        End If
        lobjBOMDetRS.Close
    End If
    Set lobjBOMDetRS = Nothing
    
    ' ==== Copy the Part Length Details across ====
    lstrTableName = "BOMPartLenDetails"
    
    ' Get empty recordset
    Set lobjADORS = New ADODB.Recordset
    lobjADORS.CursorType = adOpenForwardOnly
    lobjADORS.CursorLocation = adUseClient
    
    lstrSQLQuery = "select * from " & lstrTableName & " where docID=" & lngBOMID
    Set lobjBOMDetRS = CnnBTMS.Execute(lstrSQLQuery, llngAffected, adCmdText)
    If Not lobjBOMDetRS Is Nothing Then
        lobjBOMDetRS.ActiveConnection = Nothing
        If Not (lobjBOMDetRS.BOF And lobjBOMDetRS.EOF) Then
            lstrSQLQuery = "Select * from " & lstrTableName & " where 1=0"
            lobjADORS.Open lstrSQLQuery, CnnBTMS, adOpenForwardOnly, adLockOptimistic
            Set lobjADORS.ActiveConnection = Nothing
            
            lobjBOMDetRS.MoveFirst
            Do While lobjBOMDetRS.EOF = False
                lobjADORS.AddNew
                For Each lobjADOField In lobjBOMDetRS.Fields
                    If Not UCase(Trim(lobjADOField.Name)) = "DOCID" Then
                        If IsNull(lobjADOField.Value) = False Then
                            lobjADORS.Fields(lobjADOField.Name).Value = lobjADOField.Value
                        End If
                        If UCase(Trim(lobjADOField.Name)) = "DOCNUM" Then
                            lobjADORS.Fields("docNum").Value = lstrBOMNum
                        End If

                        ''''''''''Change in .NET''''''''''''''''
                        If UCase(Trim(lobjADOField.Name)) = "QTY" Then
                            ldblQtyReq = DBLValue(lobjADORS.Fields(lobjADOField.Name).Value)
                            ldblQtyReq = (ldblQtyReq / llngDrQty) * llngDrQtyNew
                            lobjADORS.Fields(lobjADOField.Name).Value = ldblQtyReq
                        ''''''''''END Change in .NET
                        End If
                    Else
                        lobjADORS.Fields("docID").Value = llngNewBOMID
                    End If
                Next
                
                lobjBOMDetRS.MoveNext
            Loop
            
            ' Insert the BoM or BoM child record
            If Not lobjADORS Is Nothing Then
                If Not (lobjADORS.BOF And lobjADORS.EOF) Then
                    lobjADORS.MoveFirst
                    
                    Do While lobjADORS.EOF = False
                        lstrErrDescr = "Error building insert statement for " & lstrTableName
                        lstrSQLQuery = Build_Insert(lobjADORS, lstrTableName, "")
                        lstrErrDescr = "Error inserting data into " & lstrTableName
                        Call CnnBTMS.Execute(lstrSQLQuery, 0, adCmdText And adExecuteNoRecords)
                        lstrErrDescr = ""
                        lobjADORS.MoveNext
                    Loop
                    lobjADORS.Close
                End If
            End If
            Set lobjADORS = Nothing
        End If
        lobjBOMDetRS.Close
    End If
    Set lobjBOMDetRS = Nothing

    Call DeleteTempBOM
    mstrDocNum = lstrBOMNum
    mlngBOMID = llngNewBOMID
   
    CnnBTMS.CommitTrans
    lblnInTrans = False
    Exit Function

Error_Handler:
    llngErrNo = IIf(llngErrNo = 0, Err.Number, llngErrNo)
    lstrErrDescr = IIf(Len(lstrErrDescr) = 0, Err.Description, lstrErrDescr & vbCrLf & Err.Description)
    lstrErrSource = IIf(Len(Trim(Err.Source)) = 0, "", Err.Source & vbCrLf) & cstrErrorSource
    If lblnInTrans = True Then CnnBTMS.RollbackTrans
    Call Err.Raise(llngErrNo, lstrErrSource, lstrErrDescr)
End Function

Private Sub UpdatePrices(ByVal llngBOMID As Long)

    On Error GoTo Error_Handler
    Const cstrErrorSource As String = cstrApplicationName & "." & cstrModuleName & ".UpdatePrices()"
    Dim llngErrNo As Long
    Dim lstrErrDescr As String
    Dim lstrErrSource As String
    llngErrNo = 0
    lstrErrDescr = ""
    lstrErrSource = ""

    Dim lstrPriceField As String
    Dim lstrSQL As String
    Dim lobjMainRS As ADODB.Recordset
    Dim lobjBOMDetRS As ADODB.Recordset
    Dim ldblCost As Double
    Dim ldblList As Double
    Dim TempLabRt As Double
    Dim Lbr As Double

''''''''''Change in .NET''''''''''''''''
    Dim OH As Double
''''''''''END Change in .NET
    
    Dim lstrBasePN As String
    
    Select Case intCheck
        Case 1:  lstrPriceField = "G"
        Case 2:  lstrPriceField = "I"
        Case 3:  lstrPriceField = "N"
        Case 4:  lstrPriceField = "T"
        
        Case 5:  lstrPriceField = "J"
        Case 6:  lstrPriceField = "M"
        Case 7:  lstrPriceField = "S"
        Case 8:  lstrPriceField = "V"

        
    End Select
    
     '-- Mod  2.0.21 Hanover  MultId changed to strMultId
     
    Select Case intCheck
        Case 1, 2, 3, 4: strMultId = "MultId"
        Case 5, 6, 7, 8: strMultId = "NewMultId"
        
    End Select

     
     
     
     
'    lstrSQL = "select PartNum," & lstrPriceField & ", MultID, SpecialMult, UM, LbrCost, Length from Main order by PartNum"
    lstrSQL = "select PartNum," & lstrPriceField & "," & strMultId & ", SpecialMult, UM, LbrCost, Length from Main order by PartNum"

    
    Set lobjMainRS = CnnBTMS.Execute(lstrSQL, 0, adCmdText)
    
    lstrSQL = "select * from BOMDetails where BOMID=" & llngBOMID
    Set lobjBOMDetRS = New ADODB.Recordset
    lobjBOMDetRS.CursorLocation = adUseClient
    lobjBOMDetRS.Open lstrSQL, CnnBTMS, adOpenKeyset, adLockOptimistic
    
    If Not lobjBOMDetRS Is Nothing Then
        'lobjBOMDetRS.ActiveConnection = Nothing
        If Not (lobjBOMDetRS.EOF And lobjBOMDetRS.BOF) Then
            
            If lobjBOMDetRS.Fields("BOMPtNum").Value <> "" Then
                lobjBOMDetRS.MoveFirst
                Do While lobjBOMDetRS.EOF = False
                    lobjMainRS.MoveFirst
                    lobjMainRS.Find "PartNum=" & SQLString(lobjBOMDetRS.Fields("BOMPtNum").Value)
                    If lobjMainRS.EOF = False And lobjMainRS.BOF = False Then
                        ldblCost = Decrypt(GetSym(lobjBOMDetRS.Fields("BOMPtNum").Value), DBLValue(lobjMainRS.Fields(lstrPriceField).Value))
                        ldblList = ldblCost
                        TempLabRt = DBLValue(lobjMainRS.Fields("LbrCost").Value)
                        If lobjMainRS.Fields("SpecialMult").Value > 0 Then
                            ldblCost = DBLValue(lobjMainRS.Fields("SpecialMult").Value) * ldblCost
                        Else
                        
                         '-- Mod Hanover  MultId changed to strMultId
                            If Not Trim(lobjMainRS.Fields(strMultId).Value) = "" Then
                                ldblCost = ldblCost * (MultipliersCase(DBLValue(Trim(lobjMainRS.Fields(strMultId).Value))))
                            End If
                        End If
                        'at this point ldblCost is decrypted cost with multipliers applied
                        
                        ''2.0.21 -- update with new multid
                        
                         lobjBOMDetRS.Fields("BOMPtMultId").Value = lobjMainRS.Fields(strMultId).Value
                        '''''''''''''''''''''''''''''''''''''''''
                        
                        lobjBOMDetRS.Fields("BOMPtMtCost").Value = ldblCost
                        lobjBOMDetRS.Fields("BOMPtlistCost").Value = ldblList
                        
                        If lobjMainRS.Fields("UM").Value = "LFT" Then
                            lobjBOMDetRS.Fields("BOMPtMtCost").Value = ldblCost / 12
                            lobjBOMDetRS.Fields("BOMPtlistCost").Value = ldblList / 12
                            TempLabRt = TempLabRt / 12
                        End If
                        
                        If GetUMfromMain(lobjBOMDetRS.Fields("BOMPtNum").Value) = "EACH" And GetUMfromInventory(lobjBOMDetRS.Fields("BOMPtNum").Value) = "INCHES" Then
                            'Left(lobjBOMDetRS.Fields("BOMPtNum").Value, 4) = "433-"
'                        If lobjBOMDetRS.Fields("BOMPtNum").Value = "426-0271" Or _
                                lobjBOMDetRS.Fields("BOMPtNum").Value = "328-227" Then
                               'lobjBOMDetRs.Fields("BOMPtNum").Value = "114-700" Or _
                               'lobjBOMDetRs.Fields("BOMPtNum").Value = "114-701" Or _
                                'lobjBOMDetRs.Fields("BOMPtNum").Value = "114-702" Or _
                                'lobjBOMDetRs.Fields("BOMPtNum").Value = "114-703" Or _
                                'lobjBOMDetRs.Fields("BOMPtNum").Value = "114-704" Or _
                                'lobjBOMDetRs.Fields("BOMPtNum").Value = "114-706" Or _
                                'lobjBOMDetRs.Fields("BOMPtNum").Value = "950-009" Or _
                                'lobjBOMDetRs.Fields("BOMPtNum").Value = "950-010" Or _
                                'lobjBOMDetRs.Fields("BOMPtNum").Value = "950-011" Or _
                                'lobjBOMDetRs.Fields("BOMPtNum").Value = "950-012" Or _
                                'lobjBOMDetRs.Fields("BOMPtNum").Value = "950-013" Or _
                                'lobjBOMDetRs.Fields("BOMPtNum").Value = "950-014" Or _
                                'lobjBOMDetRs.Fields("BOMPtNum").Value = "950-015" Or _
                                'lobjBOMDetRs.Fields("BOMPtNum").Value = "426-0272" Or _
                                'lobjBOMDetRs.Fields("BOMPtNum").Value = "950-016" Then     'PVC Bushing and shaft
                                                    
                         ''2.0.21 -- update with new multid
                         lobjBOMDetRS.Fields("BOMPtMultId").Value = lobjMainRS.Fields(strMultId).Value
                        '''''''''''''''''''''''''''''''''''''''''''''''''''''''''''''''''''''''
                            lobjBOMDetRS.Fields("BOMPtMtCost").Value = (ldblCost / DBLValue(lobjMainRS.Fields("Length").Value))
                            lobjBOMDetRS.Fields("BOMPtlistCost").Value = (ldblList / DBLValue(lobjMainRS.Fields("Length").Value))
                            TempLabRt = (TempLabRt / DBLValue(lobjMainRS.Fields("Length").Value))
                        End If
                        Lbr = Lbr + (TempLabRt * lobjBOMDetRS.Fields("BOMPtTotAmountReq").Value)
                        
                        lobjBOMDetRS.Update
                        
                    Else 'not found in main table
                        lobjMainRS.MoveFirst 'section
                        
                        lstrBasePN = Left(lobjBOMDetRS.Fields("BOMPtNum").Value, 6) & "000"
                        lobjMainRS.Find "PartNum like " & SQLString(lstrBasePN)
                        If lobjMainRS.EOF = False And lobjMainRS.BOF = False Then
                            ldblCost = Decrypt(GetSym(lstrBasePN), DBLValue(lobjMainRS.Fields(lstrPriceField).Value))

                            ldblList = ldblCost
                            If lobjMainRS.Fields("SpecialMult").Value > 0 Then
                                ldblCost = DBLValue(lobjMainRS.Fields("SpecialMult").Value) * ldblCost
                            Else
                             '-- Mod Hanover  MultId changed to strMultId
                                If Not Trim(lobjMainRS.Fields(strMultId).Value) = "" Then
                                    ldblCost = ldblCost * (MultipliersCase(DBLValue(Trim(lobjMainRS.Fields(strMultId).Value))))
                                End If
                            End If
                            
                         ''2.0.21 -- update with new multid
                         lobjBOMDetRS.Fields("BOMPtMultId").Value = lobjMainRS.Fields(strMultId).Value
                            '''''''''''''''''''''''''''''''''''''''''''''''''''

                            
                            Dim lsngDrWidth As Single
                            lsngDrWidth = GetBoMDrWidth(llngBOMID)
                            lobjBOMDetRS.Fields("BOMPtMtCost").Value = IIf(Right(lobjMainRS.Fields("PartNum").Value, 4) = "1000", ldblCost * (lsngDrWidth * 21) / 144, (lsngDrWidth * 24) / 144 * ldblCost)
                            lobjBOMDetRS.Fields("BOMPtlistCost").Value = IIf(Right(lobjMainRS.Fields("PartNum").Value, 4) = "1000", ldblList * (lsngDrWidth * 21) / 144, (lsngDrWidth * 24) / 144 * ldblList)
                            lobjBOMDetRS.Update
                        End If
                    End If
                    lobjBOMDetRS.MoveNext
                Loop
            End If
        End If
    End If
    Set lobjBOMDetRS = Nothing
    Set lobjMainRS = Nothing
        
    aReg.OpenKey LocalMachine, "\software\Richards-Wilcox Canada", True
    Lbr = IIf(DBLValue(aReg.ReadValue("LbrCost")) = 0, 0, DBLValue(aReg.ReadValue("LbrCost")) / 60 * Lbr)

''''''''''Change in .NET''''''''''''''''
    OH = DBLValue(aReg.ReadValue("OHCost"))
    aReg.CloseKey
    OH = Lbr * OH
'    lstrSQL = "update BoM Set BOMPriceModel=" & intCheck & " , BOMLabourCost=" & Replace(Lbr, ",", ".") & " where BomID=" & llngBOMID
    lstrSQL = "update BoM Set BOMPriceModel=" & intCheck & " , BOMLabourCost=" & Replace(Lbr, ",", ".") & " , BOMOverheadCost=" & Replace(OH, ",", ".") & " where BomID=" & llngBOMID

''''''''''END Change in .NET
    Call CnnBTMS.Execute(lstrSQL, 0, adCmdText And adExecuteNoRecords)
    Exit Sub

Error_Handler:
    llngErrNo = IIf(llngErrNo = 0, Err.Number, llngErrNo)
    lstrErrDescr = IIf(Len(lstrErrDescr) = 0, Err.Description, lstrErrDescr & vbCrLf & Err.Description)
    lstrErrSource = IIf(Len(Trim(Err.Source)) = 0, "", Err.Source & vbCrLf) & cstrErrorSource
    Call Err.Raise(llngErrNo, lstrErrSource, lstrErrDescr)

End Sub
