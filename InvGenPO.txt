Private Sub cmdAddtoPO_Click()
    On Error GoTo Error_Handler
    Const cstrErrorSource As String = cstrApplicationName & "." & cstrModuleName & ".cmdAddtoPO_Click()"
    Dim llngErrNo As Long
    Dim lstrErrDescr As String
    Dim lstrErrSource As String
    llngErrNo = 0
    lstrErrDescr = ""
    lstrErrSource = ""
    Screen.MousePointer = vbHourglass
    Dim i As Integer
    For i = 1 To grdData.Rows - 1
        If grdData.TextMatrix(i, 0) <> "" Then
            If grdData.Cell(flexcpChecked, i, 12) = flexChecked Then
                If DBLValue(grdData.TextMatrix(i, 11)) > 0 Then
                    Exit For
                Else
                    Screen.MousePointer = vbDefault
                    MsgBox "Selected items must have quantity greater than 0 to be put on a Purchase Order.", vbExclamation + vbApplicationModal, "Invalid Quantity Selected."
                    Exit Sub
                End If
            End If
        End If
    Next
    
    If i = grdData.Rows Then
        Screen.MousePointer = vbDefault
        MsgBox "No items selected.", vbInformation + vbApplicationModal, "No Selection."
        Exit Sub
    End If
    
    
    Dim frmX As frmPOAdd
    Set frmX = New frmPOAdd
    Set frmX.mobjParent = Me
    Screen.MousePointer = vbDefault
    mblnInFocus = False
    frmX.Show vbModal
    mblnInFocus = True
    Screen.MousePointer = vbHourglass
    
    Dim lstrSQL As String
    Dim lobjRS As ADODB.Recordset
    Dim llngAffected As Long
    
    If Not mstrData = "" Then
        Dim larrData As Variant
        larrData = Split(mstrData, "~")
        mobjPO.CancelFlag = IIf(larrData(1) = "True", True, False)
        
        mobjPO.PONum = Trim(larrData(0))
        mobjPO.POID = DBLValue(Trim(larrData(2)))
        CnnBTMS.BeginTrans
        
        Call UpdateOnPOsForPO(mobjPO.POID, True) 'first have to subtract that PO from the Inventory
        Call AddToPO
        Call UpdateOnPOsForPO(mobjPO.POID, False) 'then can add new quantities
        
        ' Find the number of PODetails
        Dim llngCount As Long
        llngCount = -1
        lstrSQL = "select Count(*) from PODetails where POID=" & mobjPO.POID
        Set lobjRS = CnnBTMS.Execute(lstrSQL, llngAffected, adCmdText)
        If Not lobjRS Is Nothing Then
            lobjRS.ActiveConnection = Nothing
            If Not (lobjRS.BOF And lobjRS.EOF) Then
                lobjRS.MoveFirst
                llngCount = IIf(IsNull(lobjRS.Fields(0).Value), 0, DBLValue(lobjRS.Fields(0).Value))
            End If
            lobjRS.Close
        End If
        Set lobjRS = Nothing
        
        ' Delete PO if no Children
        If llngCount <= 0 Then
            lstrSQL = "Delete * from PO where POID=" & mobjPO.POID
            Call CnnBTMS.Execute(lstrSQL, 0, adCmdText And adExecuteNoRecords)
        End If
        
        CnnBTMS.CommitTrans

    Else
        Screen.MousePointer = vbDefault
        Exit Sub
    End If
    For i = 1 To Me.grdData.Rows - 1
        lstrSQL = "select QtyOnpos from inventory where PartNum=" & SQLString(grdData.TextMatrix(i, 0))
        Set lobjRS = CnnBTMS.Execute(lstrSQL, llngAffected, adCmdText)
        If Not lobjRS Is Nothing Then
            If Not (lobjRS.EOF = True And lobjRS.BOF = True) Then
                grdData.TextMatrix(i, 8) = Format(lobjRS.Fields("QtyOnPos").Value, "Fixed")
            End If
        End If
    Next i
    grdData.Row = 1
    grdData.Col = 0
    Screen.MousePointer = vbDefault
    Exit Sub

Error_Handler:
    CnnBTMS.RollbackTrans
    Screen.MousePointer = vbDefault
    llngErrNo = IIf(llngErrNo = 0, Err.Number, llngErrNo)
    lstrErrDescr = IIf(Len(lstrErrDescr) = 0, Err.Description, lstrErrDescr & vbCrLf & Err.Description)
    Call DisplayError(llngErrNo, IIf(Len(Trim(Err.Source)) = 0, "", Err.Source & vbCrLf) & cstrErrorSource, lstrErrDescr)

End Sub

Private Sub Form_Load()
    On Error GoTo Error_Handler
    Const cstrErrorSource As String = cstrApplicationName & "." & cstrModuleName & ".Form_Load()"
    Dim llngErrNo As Long
    Dim lstrErrDescr As String
    Dim lstrErrSource As String
    llngErrNo = 0
    lstrErrDescr = ""
    lstrErrSource = ""

    Dim llngAffected As Long
    Dim lstrSQL As String
    Dim lobjStockRS As ADODB.Recordset
    Dim lobjPORS As ADODB.Recordset
    Dim lintCounter As Integer
    Dim lstrDescrField As String
    
    Screen.MousePointer = vbHourglass
    
    lstrDescrField = IIf(MeasU = 0, "DescMet", "DescImp")
    lstrSQL = "SELECT M.PartNum AS PN, M." & lstrDescrField & " AS Descr, M.DescOther AS DescrO"
    lstrSQL = lstrSQL & ", INV.InvUM AS UM, INV.QtyInStock AS InStock, Inv.QtyMin AS MinQty, Inv.RestockingLevel AS RestockingL"
    lstrSQL = lstrSQL & ", INV.QtyReserved AS ReservedQ, INV.QtyOnPOs AS OnPOs, INV.QtyOnOrder AS OnOrder, INV.QtyBkOrder AS BkO"
    lstrSQL = lstrSQL & ",0,0"
    lstrSQL = lstrSQL & ", M.UM AS UMM, M.Length AS Length, M.PhantomPN FROM Main M"
    lstrSQL = lstrSQL & " LEFT JOIN Inventory Inv ON INV.PartNum= M.PartNum"
    lstrSQL = lstrSQL & " where M.Obsolete=false ORDER BY M.PartNum"
    
    Set lobjStockRS = CnnBTMS.Execute(lstrSQL, llngAffected, adCmdText)

    lobjStockRS.Filter = "PhantomPN=False"
    
    If Not lobjStockRS Is Nothing Then
        If Not (lobjStockRS.EOF = True And lobjStockRS.BOF = True) Then
            lobjStockRS.MoveFirst
            grdData.VirtualData = False
            grdData.DataMode = flexDMFree
            Set grdData.DataSource = lobjStockRS
        End If
    End If

    lstrSQL = "select POID,PONum,POStatus from PO where POStatus='Active' and POPriceModel=" & intCheck
    Set lobjPORS = CnnBTMS.Execute(lstrSQL, llngAffected, adCmdText)
    If Not lobjPORS Is Nothing Then
        lobjPORS.ActiveConnection = Nothing
        If lobjPORS.BOF And lobjPORS.EOF Then
            Me.cmdAddtoPO.Enabled = False
        End If
    End If
    Set lobjPORS = Nothing

    Dim currBPP As Long
    currBPP = GetDeviceCaps(hdc, BITSPIXEL)
    If currBPP <= 8 Then Call GoGrey(Me)
    
    Call SetGridTitles

    lblInfo.Caption = "Put a check mark in the " & Chr(34) & "Selected" & Chr(34) & " column beside the items you wish to add to a Purchase Order." & vbCrLf & _
            "To select all items click on the " & Chr(34) & "Select All" & Chr(34) & " button."
    Me.Move Screen.Width / 2 - Me.Width / 2, Screen.Height / 2 - Me.Height / 2
    LoadHelpText Me
    MousePos1.Init Me
    Screen.MousePointer = vbDefault
    
    Exit Sub

Error_Handler:
    Screen.MousePointer = vbDefault
    llngErrNo = IIf(llngErrNo = 0, Err.Number, llngErrNo)
    lstrErrDescr = IIf(Len(lstrErrDescr) = 0, Err.Description, lstrErrDescr & vbCrLf & Err.Description)
    Call DisplayError(llngErrNo, IIf(Len(Trim(Err.Source)) = 0, "", Err.Source & vbCrLf) & cstrErrorSource, lstrErrDescr)
       
End Sub

    grdData.TextMatrix(0, 10) = "Bk Order"
    grdData.TextMatrix(0, 11) = "Recm'd for PO"

    grdData.ColWidth(3) = grdData.Width / 14.5
    grdData.ColWidth(4) = grdData.Width / 13.5

    grdData.ColWidth(6) = grdData.Width / 12.5
    grdData.ColWidth(7) = grdData.Width / 13.5

    grdData.ColWidth(9) = grdData.Width / 13.5
    grdData.ColWidth(10) = grdData.Width / 13.5

    grdData.ColFormat(7) = "0.00"
    grdData.ColFormat(8) = "0.00"

    grdData.ColFormat(11) = "0.00"

    grdData.Redraw = flexRDNone
    grdData.Col = 11
    For llngRow = 1 To grdData.Rows - 1
            grdData.Row = llngRow
            If Not (Trim(grdData.TextMatrix(llngRow, 0)) = "") Then
                grdData.CellBackColor = &HE0E0E0

                    ClearConObj
                    aCon.Inch = grdData.TextMatrix(llngRow, 4)
                    aCon.mmcalc
                    grdData.TextMatrix(llngRow, 4) = Round(aCon.mm, 0)
                    ClearConObj
                    aCon.Inch = grdData.TextMatrix(llngRow, 5)
                    aCon.mmcalc
                    grdData.TextMatrix(llngRow, 5) = Round(aCon.mm, 0)
                    ClearConObj
                    aCon.Inch = grdData.TextMatrix(llngRow, 6)
                    aCon.mmcalc
                    grdData.TextMatrix(llngRow, 6) = Round(aCon.mm, 0)
                    ClearConObj
                    aCon.Inch = grdData.TextMatrix(llngRow, 7)
                    aCon.mmcalc
                    grdData.TextMatrix(llngRow, 7) = Round(aCon.mm, 0)
                    ClearConObj
                    aCon.Inch = grdData.TextMatrix(llngRow, 8)
                    aCon.mmcalc
                    grdData.TextMatrix(llngRow, 8) = Round(aCon.mm, 0)
                    ClearConObj
                    aCon.Inch = grdData.TextMatrix(llngRow, 9)
                    aCon.mmcalc
                    grdData.TextMatrix(llngRow, 9) = Round(aCon.mm, 0)
                    ClearConObj
                    aCon.Inch = grdData.TextMatrix(llngRow, 10)
                    aCon.mmcalc
                    grdData.TextMatrix(llngRow, 10) = Round(aCon.mm, 0)
                    ClearConObj
                    aCon.Inch = grdData.TextMatrix(llngRow, 11)
                    aCon.mmcalc

Private Sub AddToPO()
     On Error GoTo Error_Handler
    Const cstrErrorSource As String = cstrApplicationName & "." & cstrModuleName & ".AddToPO()"
    Dim llngErrNo As Long
    Dim lstrErrDescr As String
    Dim lstrErrSource As String
    llngErrNo = 0
    lstrErrDescr = ""
    lstrErrSource = ""

    Dim lstrSQL As String
    Dim lintMode As Integer
    Dim lstrWhereClause As String
    Dim lstrExcludeFieldList As String
    Dim llngAffected As Long
    Dim lobjADORS As ADODB.Recordset
    Dim lobjParentRS As ADODB.Recordset
    Dim lobjMainRS As ADODB.Recordset
    Dim lblnCheck As Boolean
    Dim lobjRS As ADODB.Recordset
    Dim lobjTempRS As ADODB.Recordset
    Dim TempQty As Integer, TempLen As Single, TempBOM() As Long
    Dim TempSprLen As Single, i As Integer, j As Integer, TempQuote() As Long
    'this routine is being called when new po was just created and bom details are transfered
    ' or when adding to existing details

''''''''''Change in .NET''''''''''''''''
    Dim lstrNumArray() As String
    Dim lintCount As Integer
    ReDim lstrNumArray(0)
     
''''''''''END Change in .NET
    
    ' Copy the fields from the screen to an ADO Recordset
    Set lobjADORS = BuildPODetailRS
    
    ' Process through the recordset and insert or update rows
    If Not lobjADORS Is Nothing Then
        If Not (lobjADORS.BOF And lobjADORS.EOF) Then
            lobjADORS.MoveFirst
            lstrExcludeFieldList = ""
            
            Do While lobjADORS.EOF = False
                ' Check to see if the record exists in the
                ' BOMDetails table already
                lstrSQL = "select * from PODetails" 'POPtNum, POPtQtyOrd, POPtMtCost
                lstrSQL = lstrSQL & " where POID=" & mobjPO.POID
                lstrSQL = lstrSQL & " and POPtNum=" & SQLString(lobjADORS.Fields("POPtNum").Value)
                Set lobjParentRS = CnnBTMS.Execute(lstrSQL, llngAffected, adCmdText) 'existing PO details
                If Not lobjParentRS Is Nothing Then
                    If Not (lobjParentRS.BOF And lobjParentRS.EOF) Then
                        lobjParentRS.MoveFirst 'found record, add to existing detail
                        ' Reset the Mode so that we do an update of the existing part
                        lintMode = MODE_EDIT
                        ' Average the cost and sum the qty's
                        If lobjADORS.Fields("POPtMultID").Value <> lobjParentRS.Fields("POPtMultID").Value Then
                            lobjADORS.Fields("POPtCallPrice").Value = True
                            lobjADORS.Fields("POCallPriceExpDate").Value = Format(Now() + 30, "Short Date")
                        End If
                        If DBLValue(lobjADORS.Fields("PODiscount").Value) + DBLValue(lobjParentRS.Fields("PODiscount").Value) > 0 Then
                            If DBLValue(lobjParentRS.Fields("PODiscount").Value) > 0 Then
                                If DBLValue(lobjADORS.Fields("PODiscount").Value) > DBLValue(lobjParentRS.Fields("PODiscount").Value) Or _
                                lobjADORS.Fields("PODiscount").Value = 0 Then
                                    lobjADORS.Fields("PODiscount").Value = lobjParentRS.Fields("PODiscount").Value
                                End If
                            End If
                            lobjADORS.Fields("POPtCallPrice").Value = True
                            lobjADORS.Fields("POCallPriceExpDate").Value = Format(Now() + 30, "Short Date")

Private Sub grdData_KeyPressEdit(ByVal Row As Long, ByVal Col As Long, KeyAscii As Integer)
    On Error GoTo Error_Handler
    Const cstrErrorSource As String = cstrApplicationName & "." & cstrModuleName & ".grdData_KeyPressEdit()"
    Dim llngErrNo As Long
    Dim lstrErrDescr As String
    Dim lstrErrSource As String
    llngErrNo = 0
    lstrErrDescr = ""
    lstrErrSource = ""
    Dim lstrText As String
    lstrText = grdData.EditText
    
    If Len(lstrText) >= grdData.EditMaxLength Then
        Beep
        KeyAscii = 0
        Exit Sub
    End If
    
    If grdData.TextMatrix(grdData.Row, 3) <> "EACH" Then
        If DecimU = 1 Then
            If Not IsNumeric(Chr(KeyAscii)) And KeyAscii <> 46 And KeyAscii <> 8 Then
                Beep
                KeyAscii = 0
                Exit Sub
            End If
            If KeyAscii = 46 Then
                If Not grdData.EditSelLength = Len(lstrText) Then
                    If InStr(1, lstrText, Chr(46)) <> 0 Then
                        Beep
                        KeyAscii = 0
                        Exit Sub
                    End If
                End If
            End If
        Else
            If Not IsNumeric(Chr(KeyAscii)) And KeyAscii <> 44 And KeyAscii <> 8 Then
                Beep
                KeyAscii = 0
                Exit Sub
            End If
            If KeyAscii = 44 Then
                If Not grdData.EditSelLength = Len(lstrText) Then
                    If InStr(1, lstrText, Chr(44)) <> 0 Then
                        Beep
                        KeyAscii = 0
                        Exit Sub
                    End If
                End If
            End If
        End If
    Else
        If Not IsNumeric(Chr(KeyAscii)) Xor KeyAscii = 8 Then
            Beep
            KeyAscii = 0
            Exit Sub
        End If
    End If
    Exit Sub
Error_Handler:
    llngErrNo = IIf(llngErrNo = 0, Err.Number, llngErrNo)
    lstrErrDescr = IIf(Len(lstrErrDescr) = 0, Err.Description, lstrErrDescr & vbCrLf & Err.Description)
    Call DisplayError(llngErrNo, IIf(Len(Trim(Err.Source)) = 0, "", Err.Source & vbCrLf) & cstrErrorSource, lstrErrDescr)
End Sub
