Private Sub LoadData()
    On Error GoTo Error_Handler
    Const cstrErrorSource As String = cstrApplicationName & "." & cstrModuleName & ".LoadData()"
    Dim llngErrNo As Long
    Dim lstrErrDescr As String
    Dim lstrErrSource As String
    llngErrNo = 0
    lstrErrDescr = ""
    lstrErrSource = ""

    Dim lstrSQL As String
    Dim llngAffected As Long
    Dim lstrDescrField As String
    Dim lobjRS As ADODB.Recordset
    Dim i As Integer
    Dim lstrFilterString As String

''''''''''Change in .NET''''''''''''''''
    Dim lintONPOS As Integer
''''''''''END Change in .NET
        
    lstrDescrField = IIf(UCase(mstrUM) = "METRIC", "DescMet", "DescImp")
    lstrSQL = "select M.PartNum, M.Subcategory, " & IIf(UCase(mstrUM) = "METRIC", "M.DescMet", "M.DescImp") & " , M.DescOther "
    lstrSQL = lstrSQL & " ,INV.InvUM as UM, INV.QtyInStock as InStock,Inv.QtyInStock-Inv.QtyReserved as Available, Inv.QtyOnPOS as OnPOs"
    lstrSQL = lstrSQL & ",INV.QtyOnOrder  as OnOrder,INV.QtyReserved as ReservedQ, Inv.QtyMin as MinQty, Inv.RestockingLevel as RestockingL, M.UM as UMM, M.Length as Length,M.Obsolete as Obsolete,M.PkgQty as PkgQty,0,0, BOMGroupCategory From Main M"
    lstrSQL = lstrSQL & " LEFT JOIN Inventory Inv ON  INV.PartNum= M.PartNum where Obsolete=false"
    lstrSQL = lstrSQL & " order by M.PartNum"
    
    Set lobjRS = CnnBTMS.Execute(lstrSQL, llngAffected, adCmdText)
    
    If mstrBackTag = "frmSecNonStock" Then
        lobjRS.MoveFirst
        lstrFilterString = "Subcategory='Hinge' or Subcategory='Handles and Pulls' or Subcategory='Lock' or Subcategory='Roller' or (Subcategory='Section Extras' and Partnum='152-321') or "
        lstrFilterString = lstrFilterString & "(Subcategory='Section Extras' and Partnum='324-990') or "
        lstrFilterString = lstrFilterString & "(Subcategory='Section Extras' and Partnum='324-991') or "
        lstrFilterString = lstrFilterString & "(Subcategory='Section Extras' and Partnum='324-992') or "
        lstrFilterString = lstrFilterString & "(Subcategory='Section Extras' and Partnum='324-993') or "
        lstrFilterString = lstrFilterString & "(Subcategory='Section Extras' and Partnum='328-214') or "
        lstrFilterString = lstrFilterString & "(Subcategory='Section Extras' and Partnum='328-276') or "
        lstrFilterString = lstrFilterString & "(Subcategory='Section Extras' and Partnum='328-278') or "
        lstrFilterString = lstrFilterString & "(Subcategory='Section Extras' and Partnum='328-279') or "
        lstrFilterString = lstrFilterString & "(Subcategory='Section Extras' and Partnum='426-0310') or "
        lstrFilterString = lstrFilterString & "(Subcategory='Section Extras' and Partnum='426-0311') or "
        lstrFilterString = lstrFilterString & "(Subcategory='Section Extras' and Partnum='426-0312') or "
        lstrFilterString = lstrFilterString & "(Subcategory='Section Extras' and Partnum='426-0313') or "
        lstrFilterString = lstrFilterString & " Partnum='4' or "
        lstrFilterString = lstrFilterString & " Partnum='4-0069' or "
        lstrFilterString = lstrFilterString & "Subcategory='Truss' or (Subcategory='Bracket' and BOMGroupCategory <> 19)"
        lobjRS.Filter = lstrFilterString
    ElseIf mstrBackTag = "frmBOMDetail" Or mstrBackTag = "frmBOMDetailList" Or mstrBackTag = "frmQDetail" Or mstrBackTag = "frmQuoteDetail" Then
        lobjRS.MoveFirst
        lobjRS.Filter = "Partnum<>'4-0069' and partnum<>'4'"
    End If
    
    If Not lobjRS Is Nothing Then
        lobjRS.ActiveConnection = Nothing
        If Not (lobjRS.BOF And lobjRS.EOF) Then
            grdData.VirtualData = False
            grdData.DataMode = flexDMFree
            Set grdData.DataSource = lobjRS
        End If
    End If
        
    Select Case mstrBackTag
        Case "frmBOMDetail", "frmBOMDetailList"
            lstrSQL = "select BomPtNum, BOMPtQtyReq from bomdetails where bomid=" & mlngDOCID
            Set lobjRS = CnnBTMS.Execute(lstrSQL, llngAffected, adCmdText)
            For i = 1 To grdData.Rows - 1
                If Not lobjRS Is Nothing Then
                    If Not (lobjRS.EOF = True And lobjRS.BOF = True) Then
                        lobjRS.MoveFirst
                        lobjRS.Find "BOMPtNum=" & SQLString(grdData.TextMatrix(i, 0))
                        'show qty on the document
                        If lobjRS.EOF = False Then
                            grdData.TextMatrix(i, 16) = lobjRS.Fields("BOMPtQtyReq").Value
                        Else
                            grdData.TextMatrix(i, 16) = Format(0, "Fixed")
                        End If
                    Else
                        grdData.TextMatrix(i, 16) = Format(0, "Fixed")
                    End If
                End If
                'show recommended for order qty
                If DBLValue(grdData.TextMatrix(i, 5)) >= DBLValue(grdData.TextMatrix(i, 10)) Then
                    grdData.TextMatrix(i, 17) = Format(0, "Fixed")
                Else
                    grdData.TextMatrix(i, 17) = DBLValue(grdData.TextMatrix(i, 10)) + DBLValue(grdData.TextMatrix(i, 11)) - DBLValue(grdData.TextMatrix(i, 5)) - DBLValue(grdData.TextMatrix(i, 7)) - DBLValue(grdData.TextMatrix(i, 8))
                End If
                If DBLValue(grdData.TextMatrix(i, 17)) < 0 Then grdData.TextMatrix(i, 17) = Format(0, "Fixed")
                'translate pkgqty to inches (eg. PVC bushing)
                If grdData.TextMatrix(i, 4) = "INCHES" And grdData.TextMatrix(i, 12) = "EACH" Then
                    grdData.TextMatrix(i, 15) = DBLValue(grdData.TextMatrix(i, 15)) * DBLValue(grdData.TextMatrix(i, 13))
                End If

''''''''''Change in .NET''''''''''''''''
'                If grdData.TextMatrix(i, 7) = "" Then
'                    lintONPOS = GetQtyOnActivePOs(grdData.TextMatrix(i, 0))
'
'                    If mstrUM = "Metric" Then
'                        ClearConObj
'                        aCon.Inch = lintONPOS
'                        aCon.mmcalc
'                        lintONPOS = aCon.mm
'                    End If
'                    grdData.TextMatrix(i, 7) = lintONPOS
'                End If
'                If grdData.TextMatrix(i, 8) = "" Then
'                    lintONPOS = GetQtyOnOrder(grdData.TextMatrix(i, 0))
'
'                    If mstrUM = "Metric" Then
'                        ClearConObj
'                        aCon.Inch = lintONPOS
'                        aCon.mmcalc
'                        lintONPOS = aCon.mm
'                    End If
'                    grdData.TextMatrix(i, 8) = lintONPOS
'                End If
                If grdData.TextMatrix(i, 4) = "" Then
                    grdData.TextMatrix(i, 4) = IIf(mstrUM = "Metric", "MM", "INCHES")
                End If
''''''''''END Change in .NET

    Exit Sub

Error_Handler:

Exit Sub 'sometimes this routine produced unexpected errors if
'mouse is over the boundaries of the grid. Generating an error
'screen is not necessary
    llngErrNo = IIf(llngErrNo = 0, Err.Number, llngErrNo)
    lstrErrDescr = IIf(Len(lstrErrDescr) = 0, Err.Description, Err.Description)
    Call DisplayError(llngErrNo, IIf(Len(Trim(Err.Source)) = 0, "", Err.Source & vbCrLf) & cstrErrorSource, lstrErrDescr)
    
End Sub

Public Function SetGrid(ByVal llngRet As Long) As Boolean
    On Error GoTo Error_Handler
    Const cstrErrorSource As String = cstrApplicationName & "." & cstrModuleName & ".SetGrid()"
    Dim llngErrNo As Long
    Dim lstrErrDescr As String
    Dim lstrErrSource As String
    llngErrNo = 0
    lstrErrDescr = ""
    lstrErrSource = ""

    Dim i As Integer
    For i = 1 To grdData.Rows - 1
        grdData.IsSelected(i) = False
    Next
    
    grdData.IsSelected(llngRet) = True
    grdData.Row = llngRet
    grdData.TopRow = llngRet
    glngRow = grdData.Row
    grdData.AutoSearch = 1


    Exit Function

Error_Handler:
    llngErrNo = IIf(llngErrNo = 0, Err.Number, llngErrNo)
    lstrErrDescr = IIf(Len(lstrErrDescr) = 0, Err.Description, lstrErrDescr & vbCrLf & Err.Description)
    lstrErrSource = IIf(Len(Trim(Err.Source)) = 0, "", Err.Source & vbCrLf) & cstrErrorSource
    Call Err.Raise(llngErrNo, lstrErrSource, lstrErrDescr)

End Function
