Public Sub LoadSubCategory()

    On Error GoTo Error_Handler
    Const cstrErrorSource As String = cstrApplicationName & "." & cstrModuleName & ".LoadSubcategory()"
    Dim llngErrNo As Long
    Dim lstrErrDescr As String
    Dim lstrErrSource As String
    llngErrNo = 0
    lstrErrDescr = ""
    lstrErrSource = ""

    Dim llngAffected As Long
    Dim lstrSQLQuery As String
    Dim lobjSubCatRS As ADODB.Recordset
    cboSubCat.AddItem "Choose Sub Category..."
    cboSubCat.AddItem "ALL"
    
    lstrSQLQuery = "select subcategory from catsub order by subcategory"
    Set lobjSubCatRS = CnnBTMS.Execute(lstrSQLQuery, llngAffected, adCmdText)
    If Not lobjSubCatRS Is Nothing Then
        lobjSubCatRS.ActiveConnection = Nothing
        If Not (lobjSubCatRS.BOF And lobjSubCatRS.EOF) Then
            lobjSubCatRS.MoveFirst
            Dim lstrName As String
            Do While lobjSubCatRS.EOF = False
                lstrName = IIf(IsNull(lobjSubCatRS.Fields("subcategory").Value), "", Trim(lobjSubCatRS.Fields("subcategory").Value))

''''''''''Change in .NET''''''''''''''''
'                If Not lstrName = "" Then Call cboSubCat.AddItem(lstrName)
                If (lstrName <> "" And lstrName <> "Spring") Then Call cboSubCat.AddItem(lstrName)
''''''''''END Change in .NET
                lobjSubCatRS.MoveNext
            Loop
                lobjSubCatRS.Close
        End If
    End If
    Set lobjSubCatRS = Nothing
        
    cboSubCat.ListIndex = 0
    
    Exit Sub

Error_Handler:
    llngErrNo = IIf(llngErrNo = 0, Err.Number, llngErrNo)

Public Function GetTotalValue() As Double
    On Error GoTo Error_Handler
    Const cstrErrorSource As String = cstrApplicationName & "." & cstrModuleName & ".GetTotalValue()"
    Dim llngErrNo As Long
    Dim lstrErrDescr As String
    Dim lstrErrSource As String
    llngErrNo = 0
    lstrErrDescr = ""
    lstrErrSource = ""

    Dim lstrSQL As String
    Dim lobjINVRS As ADODB.Recordset
    Dim llngAffected As Long
    Dim lstrPriceField As String
    Dim ldblPrice As Double
    Dim ldblTotValue As Double
    
    Select Case intCheck
        Case 1: lstrPriceField = "G"
        Case 2: lstrPriceField = "I"
        Case 3: lstrPriceField = "N"
        Case 4: lstrPriceField = "T"
        
        Case 5: lstrPriceField = "J"
        Case 6: lstrPriceField = "M"
        Case 7: lstrPriceField = "S"
        Case 8: lstrPriceField = "V"

        
        
    End Select
    
    lstrSQL = "select Inv.PartNum as PN, INV.INVUM as UM, Main.UM as UMM, Main.Length as Length ,INV.QtyInStock as InStock, Main.Subcategory as SubCategory, Main." & lstrPriceField & " as Price from Inventory INV"
    lstrSQL = lstrSQL & " left join Main Main on Inv.PartNum = Main.PartNum"
    Set lobjINVRS = CnnBTMS.Execute(lstrSQL, llngAffected, adCmdText)
    If Not lobjINVRS Is Nothing Then
        If Not (lobjINVRS.EOF And lobjINVRS.BOF) Then
            Set lobjINVRS.ActiveConnection = Nothing
            If cboSubCat.ListIndex > 1 Then
                lobjINVRS.Filter = "Subcategory=" & SQLString(cboSubCat.List(cboSubCat.ListIndex))
            End If
            lobjINVRS.MoveFirst
            Do While lobjINVRS.EOF = False
                ldblPrice = Decrypt(GetSym(lobjINVRS.Fields("PN")), DBLValue(lobjINVRS.Fields("Price").Value))
                
                If lobjINVRS.Fields("UMM").Value = "LFT" Then
                    ldblPrice = (ldblPrice / 12) * DBLValue(lobjINVRS.Fields("InStock").Value)
                ElseIf lobjINVRS.Fields("UM").Value = "INCHES" And lobjINVRS.Fields("UMM").Value = "EACH" Then
                    ldblPrice = (ldblPrice / DBLValue(lobjINVRS.Fields("Length").Value)) * DBLValue(lobjINVRS.Fields("InStock").Value)
                Else
                    ldblPrice = ldblPrice * DBLValue(lobjINVRS.Fields("InStock").Value)
                End If

                ldblTotValue = ldblTotValue + ldblPrice
                
                lobjINVRS.MoveNext
            Loop
        End If
    End If
    Set lobjINVRS = Nothing
    GetTotalValue = ldblTotValue
    
    Exit Function

Error_Handler:
    llngErrNo = IIf(llngErrNo = 0, Err.Number, llngErrNo)
    lstrErrDescr = IIf(Len(lstrErrDescr) = 0, Err.Description, lstrErrDescr & vbCrLf & Err.Description)
    lstrErrSource = IIf(Len(Trim(Err.Source)) = 0, "", Err.Source & vbCrLf) & cstrErrorSource
    Call Err.Raise(llngErrNo, lstrErrSource, lstrErrDescr)

End Function
