Option Explicit
Const cstrModuleName As String = "frmSectionsSpecialLites"

Private Sub cmdAccept_Click()
    On Error GoTo Error_Handler
    Const cstrErrorSource As String = cstrApplicationName & "." & cstrModuleName & ".cmdAccept_Click()"
    Dim llngErrNo As Long
    Dim lstrErrDescr As String
    Dim lstrErrSource As String
    llngErrNo = 0
    lstrErrDescr = ""
    lstrErrSource = ""

    If DBLValue(txtPtQty.Text) = 0 Then
        MsgBox "Please enter quantity required.", vbOKOnly + vbApplicationModal + vbExclamation, "Required Quantity"
        txtPtQty.SetFocus
        Exit Sub
    End If
    
    If DBLValue(txtPtQty.Text) > (N21 + N24 - 1) * (GetMaxLitesPerSect) Then
        MsgBox "The quantity of lites entered is greater than this door size allows.", vbOKOnly + vbApplicationModal + vbExclamation, "Invalid Quantity"
        txtPtQty.SetFocus
        Exit Sub
    End If
    
    If DBLValue(txtPtCost.Text) = 0 Then
        MsgBox "Please enter cost for the item.", vbOKOnly + vbApplicationModal + vbExclamation, "Item Cost"
        txtPtCost.SetFocus
        Exit Sub
    End If
    
    If Trim(txtSprWgt.Text) = "" Then
'        If MsgBox("Would you like to enter springing weight for this item?", vbQuestion + vbApplicationModal + vbYesNo, "Springing Weight") = vbYes Then
        MsgBox "Please enter the springing weight for this item.", vbOKOnly + vbApplicationModal + vbExclamation, "Invalid Springing Weight"
        txtSprWgt.SetFocus
        Exit Sub
'        End If
    End If
    
    If Trim(txtPtDesc.Text) = "" Then
        MsgBox "Please enter description for this item.", vbApplicationModal + vbExclamation, "Item Description"
        txtPtDesc.SetFocus
        Exit Sub
    End If
    
    If MsgBox("Are you sure you want to add this item to your Bill of Materials?", vbYesNo + vbApplicationModal + vbQuestion, "Save Data") = vbYes Then
        Screen.MousePointer = vbHourglass
        Call SaveData
    End If
    frmSectionsEntry.mblnCancel = False
    Unload Me
    Screen.MousePointer = vbDefault
    Exit Sub

Error_Handler:
    Screen.MousePointer = vbDefault
    llngErrNo = IIf(llngErrNo = 0, Err.Number, llngErrNo)
    lstrErrDescr = IIf(Len(lstrErrDescr) = 0, Err.Description, lstrErrDescr & vbCrLf & Err.Description)
    Call DisplayError(llngErrNo, IIf(Len(Trim(Err.Source)) = 0, "", Err.Source & vbCrLf) & cstrErrorSource, lstrErrDescr)

End Sub

Private Sub SaveData()

    On Error GoTo Error_Handler
    Const cstrErrorSource As String = cstrApplicationName & "." & cstrModuleName & ".SaveData()"
    Dim llngErrNo As Long
    Dim lstrErrDescr As String
    Dim lstrErrSource As String
    llngErrNo = 0
    lstrErrDescr = ""
    lstrErrSource = ""

    Dim lintCount As Integer
    If isEmptyArraySpItems Then
        lintCount = 0
    Else
        lintCount = UBound(varSpItem()) + 1
    End If
    ReDim Preserve varSpItem(lintCount)
    
    varSpItem(lintCount).strPN = Trim(txtPN.Text)
    varSpItem(lintCount).strDescription = Trim(txtPtDesc.Text)
    varSpItem(lintCount).sngQty = DBLValue(txtPtQty.Text)
    varSpItem(lintCount).strUM = "EACH"
    varSpItem(lintCount).sngPrice = DBLValue(txtPtCost.Text)
    varSpItem(lintCount).sngShippingWgt = DBLValue(txtShipWgt.Text)
    varSpItem(lintCount).sngSpringingWgt = DBLValue(txtSprWgt.Text)
    If MeasU = 0 Then
        ClearConObj
        aCon.KG = DBLValue(txtShipWgt.Text)
        aCon.LBcalc
        varSpItem(lintCount).sngShippingWgt = DBLValue(aCon.LB)
        ClearConObj
        aCon.KG = DBLValue(txtSprWgt.Text)
        aCon.LBcalc
        varSpItem(lintCount).sngSpringingWgt = DBLValue(aCon.LB)
    End If
        
    varSpItem(lintCount).strNotes = Trim(txtPtNotes.Text)
    varSpItem(lintCount).intType = 1  'Lite
    
    If Left(txtPN.Text, 9) = "COMM-NSTK" Then CurrSpItem = CurrSpItem + 1
    Exit Sub

Error_Handler:
    llngErrNo = IIf(llngErrNo = 0, Err.Number, llngErrNo)
    lstrErrDescr = IIf(Len(lstrErrDescr) = 0, Err.Description, lstrErrDescr & vbCrLf & Err.Description)
    lstrErrSource = IIf(Len(Trim(Err.Source)) = 0, "", Err.Source & vbCrLf) & cstrErrorSource
    Call Err.Raise(llngErrNo, lstrErrSource, lstrErrDescr)

End Sub

Private Sub LoadData()
    On Error GoTo Error_Handler
    Const cstrErrorSource As String = cstrApplicationName & "." & cstrModuleName & ".LoadData()"
    Dim llngErrNo As Long
    Dim lstrErrDescr As String
    Dim lstrErrSource As String
    llngErrNo = 0
    lstrErrDescr = ""
    lstrErrSource = ""
'    aReg.OpenKey LocalMachine, "\Software\Richards-Wilcox Canada\BTMS", True 'opens or creates key
'
'    If CurrSpItem = aReg.ReadValue("LastNumber") Then CurrSpItem = CurrSpItem + 1
'    aReg.CloseKey
'''''''    txtPN.Text = "COMM-NSTK" & CurrSpItem   'GetSpecialPart("NSTK")
'''''''    txtPN.Enabled = False
'''''''    chkCall.Enabled = False
'''''''    chkCall.Value = 1
'''''''    lblUM.Caption = "EACH"


        Dim L21 As Integer
        Dim L24 As Integer
        Dim strBasePN As String
        Dim lstrLitePN As String
        Dim lstrString As String
        Dim lstrDescrField As String
        Dim lstrSQLQuery As String
        Dim lvarSectData As Variant
            
        L21 = ModSections.GetNumberOfLiteSectionsPerHeight(21)
        L24 = ModSections.GetNumberOfLiteSectionsPerHeight(24)


        txtPN.Enabled = False
        chkCall.Enabled = False
        chkCall.Value = 1
        lblUM.Caption = "EACH"
        
        txtShipWgt.Text = Format(CalcFullViewWgt(IIf(varDoorData.EndCapsOnFullView = True, True, False), GetMaxLitesPerSect), "0.00")
        
        txtSprWgt.Text = txtShipWgt.Text
        
        If L21 > 0 Then
            lstrLitePN = GetFullViewPN(True)
            lstrLitePN = GetPhantomInfo(lstrLitePN & IIf(varDoorData.EndCapsOnFullView = True, "-2", "-0"), DrWidth, IIf(MeasU = 1, False, True))
            lvarSectData = Split(lstrLitePN, "~")
            txtPN.Text = CStr(lvarSectData(0))
            
            
            
            
'            Call FillSummaryGrid(CStr(lvarSectData(0)), L21, "Sections", GetSubCategory(CStr(lvarSectData(0))), 0)
'            For X = 0 To UBound(varDoorData.arrSections)
'                If (varDoorData.arrSections(X).intNLites > 0 And varDoorData.LiteType > 10) Then
'                    If varDoorData.arrSections(X).intHeight = 21 Then varDoorData.arrSections(X).strPN = CStr(lvarSectData(0))
'                End If
'            Next X
        End If
        If L24 > 0 Then
            lstrLitePN = GetFullViewPN(False)
            lstrLitePN = GetPhantomInfo(lstrLitePN & IIf(varDoorData.EndCapsOnFullView = True, "-2", "-0"), DrWidth, IIf(MeasU = 1, False, True))
            lvarSectData = Split(lstrLitePN, "~")
            txtPN.Text = CStr(lvarSectData(0))
            
            
'            Call FillSummaryGrid(CStr(lvarSectData(0)), L24, "Sections", GetSubCategory(CStr(lvarSectData(0))), 0)
'            For X = 0 To UBound(varDoorData.arrSections)
'                If (varDoorData.arrSections(X).intNLites > 0 And varDoorData.LiteType > 10) Then
'                    If varDoorData.arrSections(X).intHeight = 24 Then varDoorData.arrSections(X).strPN = CStr(lvarSectData(0))
'                End If
'            Next X
        End If
        
        
       lstrDescrField = IIf(MeasU = 1, "DescImp", "DescMet")
       
       strBasePN = Left(CStr(lvarSectData(0)), 6)
       
       Dim lobjMainRS As ADODB.Recordset
       lstrSQLQuery = "SELECT PartNum, " & lstrDescrField & " FROM Main"
       Set lobjMainRS = New ADODB.Recordset
       lobjMainRS.CursorType = adOpenForwardOnly
       lobjMainRS.CursorLocation = adUseClient
       lobjMainRS.Open lstrSQLQuery, CnnBTMS, adOpenForwardOnly, adLockOptimistic
    
       
       If Not lobjMainRS Is Nothing Then
            lobjMainRS.ActiveConnection = Nothing
            lobjMainRS.MoveFirst
            lobjMainRS.Find "PartNum like " & SQLString(strBasePN)
            If lobjMainRS.EOF = False And lobjMainRS.BOF = False Then
                  txtPtDesc.Text = lobjMainRS.Fields(lstrDescrField).Value
            End If
       End If
        

    Exit Sub

Error_Handler:
    llngErrNo = IIf(llngErrNo = 0, Err.Number, llngErrNo)
    lstrErrDescr = IIf(Len(lstrErrDescr) = 0, Err.Description, lstrErrDescr & vbCrLf & Err.Description)
    lstrErrSource = IIf(Len(Trim(Err.Source)) = 0, "", Err.Source & vbCrLf) & cstrErrorSource
    Call Err.Raise(llngErrNo, lstrErrSource, lstrErrDescr)

End Sub
