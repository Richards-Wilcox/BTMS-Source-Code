Private Sub cmdCancel_Click()
    On Error GoTo Error_Handler
    Const cstrErrorSource As String = cstrApplicationName & "." & cstrModuleName & ".cmdCancel_Click()"
    Dim llngErrNo As Long
    Dim lstrErrDescr As String
    Dim lstrErrSource As String

    lstrErrDescr = ""
    lstrErrSource = ""

    lstrErrDescr = IIf(Len(lstrErrDescr) = 0, Err.Description, lstrErrDescr & vbCrLf & Err.Description)

End Sub

Private Sub cmdSave_Click()
    On Error GoTo Error_Handler
    Const cstrErrorSource As String = cstrApplicationName & "." & cstrModuleName & ".cmdSave_Click()"
    Dim llngErrNo As Long
    Dim lstrErrDescr As String
    Dim lstrErrSource As String
    llngErrNo = 0
    lstrErrDescr = ""
    lstrErrSource = ""
    
    aReg.OpenKey LocalMachine, "\Software\Richards-Wilcox Canada", True
    
    'thermatite section
    aReg.WriteValue "BulkThermatiteSections", IIf(DBLValue(txtBTS.Text) = 0, 1, txtBTS.Text)
    aReg.WriteValue "ThermatiteSections", IIf(DBLValue(txtBTS.Text) = 0, 1, txtBTS.Text)
    aReg.WriteValue "ThermalumeSections", IIf(DBLValue(txtBTS.Text) = 0, 1, txtBTS.Text)
    aReg.WriteValue "FullViewSections", IIf(DBLValue(txtBTS.Text) = 0, 1, txtBTS.Text)
    
    'bulk springs
    aReg.WriteValue "BulkSprings", IIf(DBLValue(txtBS.Text) = 0, 1, txtBS.Text)
    
    'Galvanized + alumn + lites/PVC
    aReg.WriteValue "GalvanizedComponents", IIf(DBLValue(txtOA.Text) = 0, 1, txtOA.Text)
    aReg.WriteValue "AluminumComponents", IIf(DBLValue(txtOA.Text) = 0, 1, txtOA.Text)
    aReg.WriteValue "LitesPVCMisc", IIf(DBLValue(txtOA.Text) = 0, 1, txtOA.Text)
    
    'steel H/W + Alumn HW + Misc
    aReg.WriteValue "SteelHardware", IIf(DBLValue(txtBH.Text) = 0, 1, txtBH.Text)
    aReg.WriteValue "AluminumHardware", IIf(DBLValue(txtBH.Text) = 0, 1, txtBH.Text)
    aReg.WriteValue "Miscellaneous", IIf(DBLValue(txtBH.Text) = 0, 1, txtBH.Text)
    
    'Dyna-Hoist Operator
    aReg.WriteValue "DynaHoist", IIf(DBLValue(txtOp.Text) = 0, 1, txtOp.Text)
    
    aReg.CloseKey
    Unload Me
    
    Exit Sub

Error_Handler:
    llngErrNo = IIf(llngErrNo = 0, Err.Number, llngErrNo)
    lstrErrDescr = IIf(Len(lstrErrDescr) = 0, Err.Description, lstrErrDescr & vbCrLf & Err.Description)
    Call DisplayError(llngErrNo, IIf(Len(Trim(Err.Source)) = 0, "", Err.Source & vbCrLf) & cstrErrorSource, lstrErrDescr)

End Sub

Private Sub Form_Load()

    txtOp.Text = Format(IIf(aReg.ReadValue("DynaHoist") = "1", 0, aReg.ReadValue("DynaHoist")), "0.000")

Private Function Validation(lobjControl As Object) As Boolean
    On Error GoTo Error_Handler
    Const cstrErrorSource As String = cstrApplicationName & "." & cstrModuleName & ".Validation()"
    Dim llngErrNo As Long
    Dim lstrErrDescr As String
    Dim lstrErrSource As String
    llngErrNo = 0
    lstrErrDescr = ""
    lstrErrSource = ""
    
    Validation = True
    If TypeOf lobjControl Is TextBox Then
        If DBLValue(lobjControl.Text) >= 1 Then Validation = False
    End If
    
    Exit Function

Error_Handler:
    llngErrNo = IIf(llngErrNo = 0, Err.Number, llngErrNo)
    lstrErrDescr = IIf(Len(lstrErrDescr) = 0, Err.Description, lstrErrDescr & vbCrLf & Err.Description)
    lstrErrSource = IIf(Len(Trim(Err.Source)) = 0, "", Err.Source & vbCrLf) & cstrErrorSource
    Call Err.Raise(llngErrNo, lstrErrSource, lstrErrDescr)

End Function
