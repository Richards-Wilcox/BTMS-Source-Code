Public Sub KillProcess(NameProcess As String)
    Const PROCESS_ALL_ACCESS = 0
    Const TH32CS_SNAPPROCESS As Long = 2&
    Const PROCESS_TERMINATE = &H1&
    Dim uProcess  As PROCESSENTRY32
    Dim RProcessFound As Long
    Dim hSnapShot As Long
    Dim SzExename As String
    Dim ExitCode As Long
    Dim MyProcess As Long
    Dim AppKill As Boolean
    Dim AppCount As Integer
    Dim i As Integer
    Dim WinDirEnv As String
        
       If NameProcess <> "" Then
          AppCount = 0

          uProcess.dwSize = Len(uProcess)
          hSnapShot = CreateToolhelpSnapshot(TH32CS_SNAPPROCESS, 0&)
          RProcessFound = ProcessFirst(hSnapShot, uProcess)
          Do
            i = InStr(1, uProcess.szExeFile, Chr(0))
            SzExename = LCase$(Left$(uProcess.szExeFile, i - 1))
            WinDirEnv = Environ("Windir") + "\"
            WinDirEnv = LCase$(WinDirEnv)
        
            If Right$(SzExename, Len(NameProcess)) = LCase$(NameProcess) Then
               AppCount = AppCount + 1
               MyProcess = OpenProcess(PROCESS_TERMINATE, False, uProcess.th32ProcessID)
               AppKill = TerminateProcess(MyProcess, ExitCode)
               Call CloseHandle(MyProcess)
            End If
            RProcessFound = ProcessNext(hSnapShot, uProcess)
          Loop While RProcessFound
          Call CloseHandle(hSnapShot)
       End If

End Sub

Public Function GetTempDir() As String
' Returns the temp directory from the OS system
    Dim sBuffer As String
    Dim HoldBuffer As String
    Dim iBuffLen As Long
    Dim iReturn As Long
    Const BUFFER_LENGTH = 256

    iBuffLen = BUFFER_LENGTH

Public Function RoundUp(ByVal dblVal As Double) As Double
    On Error GoTo Error_Handler
    Const cstrErrorSource As String = cstrApplicationName & "." & ".RoundUp()"
    Dim llngErrNo As Long
    Dim lstrErrDescr As String
    Dim lstrErrSource As String
    llngErrNo = 0
    lstrErrDescr = ""
    lstrErrSource = ""

    Dim lstrValue As String
    Dim ldblValue2 As Double
    Dim lintPos As Integer
    
    RoundUp = dblVal
    lstrValue = dblVal
    lintPos = InStr(1, lstrValue, ".")
    If lintPos = 0 Then lintPos = InStr(1, lstrValue, ",")
    If lintPos > 0 Then lstrValue = Left(lstrValue, lintPos - 1)
    ldblValue2 = DBLValue(lstrValue)
    If dblVal > ldblValue2 Then RoundUp = ldblValue2 + 1
    Exit Function

Error_Handler:
    llngErrNo = IIf(llngErrNo = 0, Err.Number, llngErrNo)
    lstrErrDescr = IIf(Len(lstrErrDescr) = 0, Err.Description, lstrErrDescr & vbCrLf & Err.Description)
    lstrErrSource = IIf(Len(Trim(Err.Source)) = 0, "", Err.Source & vbCrLf) & cstrErrorSource
    Call Err.Raise(llngErrNo, lstrErrSource, lstrErrDescr)
End Function

Public Function IsTaskRunning(sWindowName As String) As Boolean
    
    Call EnumWindows(AddressOf fEnumWindowsCallBack, 0&) 'makes a list of all open windows.
    If IsRWConverterRunning = 1 Then IsTaskRunning = True
    
End Function

Private Function fEnumWindowsCallBack(ByVal hwnd As Long, ByVal lParam As Long) As Long
    Dim lReturn     As Long
    Dim lExStyle    As Long
    Dim bNoOwner    As Boolean
    Dim sWindowText As String
    '
    ' This callback function is called by Windows (from
    ' the EnumWindows API call) for EVERY window that exists.
    
    ' Windows that:
    '   -   are not this app's
    '   -   are visible
    '   -   do not have a parent
    '   -   have no owner and are not Tool windows OR
    '       have an owner and are App windows
    '
    If IsWindowVisible(hwnd) Then
        If GetParent(hwnd) = 0 Then
            bNoOwner = (GetWindow(hwnd, GW_OWNER) = 0)
            lExStyle = GetWindowLong(hwnd, GWL_EXSTYLE)
            
            If (((lExStyle And WS_EX_TOOLWINDOW) = 0) And bNoOwner) Or _
                ((lExStyle And WS_EX_APPWINDOW) And Not bNoOwner) Then
                '
                ' Get the window's caption.
                '
                sWindowText = Space$(256)
                lReturn = GetWindowText(hwnd, sWindowText, Len(sWindowText))
                If Left(sWindowText, 12) = "RW Converter" Then
                        'set the variable to 1 if instance of the converter was found running.
                    IsRWConverterRunning = 1
                    fEnumWindowsCallBack = False 'do not re-call function
                    Exit Function
                End If
            End If
        End If
    End If
    fEnumWindowsCallBack = True 'makes the function call itself

End Function

Public Function SetTopMostWindow(hwnd As Long, Topmost As Boolean) _
   As Long

   If Topmost = True Then 'Make the window topmost
      SetTopMostWindow = SetWindowPos(hwnd, HWND_TOPMOST, 0, 0, 0, _
         0, FLAGS)
   Else
      SetTopMostWindow = SetWindowPos(hwnd, HWND_NOTOPMOST, 0, 0, _
         0, 0, FLAGS)
      SetTopMostWindow = False
   End If
End Function
