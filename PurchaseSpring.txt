Private Sub addItems_PusherSpring()

    On Error GoTo Error_Handler
    Const cstrErrorSource As String = cstrApplicationName & "." & cstrModuleName & ".addItems_PusherSpring()"
    Dim llngErrNo As Long
    Dim lstrErrDescr As String
    Dim lstrErrSource As String
    llngErrNo = 0
    lstrErrDescr = ""
    lstrErrSource = ""

    Dim PusherSpringQTY As Integer
    Dim i As Integer
    Dim ReturnIndexPS As Integer
    
    Select Case cboPSpring.ItemData(cboPSpring.ListIndex)
'        Case 1
        Case 2
 
        ' bef Mod 2.021
         ' mstrPSPN = "7-3141"
         
         'After Mod 2.0.21
            mstrPSPN = "322-314"  'Pusher spring only
            PusherSpringQTY = 1
'        Case 2
        Case 1

            mstrPSPN = "7-3142"  'Pusher AsslyPkg.
            PusherSpringQTY = 1
        End Select
        'populates golobal var to display in summary form
    frmTrackEntry.ReturnIndexPS = cboPSpring.ItemData(cboPSpring.ListIndex)
    
    For i = frmTrackEntry.Grid2.Row To frmTrackEntry.Grid2.Rows - 1
        If frmTrackEntry.Grid2.Text = mstrPSPN Then
            Exit Sub
        Else
            Call SetTrackOptionsGrid(mstrPSPN, PusherSpringQTY, "Track", GetSubCategory(mstrPSPN), 0)
            Exit Sub
        End If
    Next
    
    Exit Sub

Error_Handler:
    llngErrNo = IIf(llngErrNo = 0, Err.Number, llngErrNo)
    lstrErrDescr = IIf(Len(lstrErrDescr) = 0, Err.Description, Err.Description)
    lstrErrSource = IIf(Len(Trim(Err.Source)) = 0, "", Err.Source & vbCrLf) & cstrErrorSource
    Call Err.Raise(llngErrNo, lstrErrSource, lstrErrDescr)
    
End Sub
