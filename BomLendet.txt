Private Sub cmdSave_Click()
    On Error GoTo Error_Handler
    Const cstrErrorSource As String = cstrApplicationName & "." & cstrModuleName & ".cmdSave_Click()"
    Dim llngErrNo As Long
    Dim lstrErrDescr As String
    Dim lstrErrSource As String
    llngErrNo = 0
    lstrErrDescr = ""
    lstrErrSource = ""
    
    If (txtLen1(0).Text = txtLen1(1).Text) And DBLValue(txtLen1(0).Text) > 0 And DBLValue(txtLen1(1).Text) > 0 Then
        MsgBox "The lengths you entered are equal." & vbCrLf & "Consider modifying quantity of the first length instead of entering it as a separate line. ", vbInformation + vbOKOnly + vbApplicationModal, "Validation error."
        Exit Sub
    End If

        Exit Sub
    End If
    
    If MsgBox("Save Data?", vbApplicationModal + vbYesNo + vbQuestion, "Save Data.") = vbYes Then Call SaveData
    
    cmdBack_Click
    Exit Sub
    
'    mintMode = MODE_VIEW
'    Call ClearScreen
'    Call LoadData
'
'    Exit Sub

Error_Handler:
    llngErrNo = IIf(llngErrNo = 0, Err.Number, llngErrNo)
    lstrErrDescr = IIf(Len(lstrErrDescr) = 0, Err.Description, lstrErrDescr & vbCrLf & Err.Description)
    Call DisplayError(llngErrNo, IIf(Len(Trim(Err.Source)) = 0, "", Err.Source & vbCrLf) & cstrErrorSource, lstrErrDescr)

End Sub

Private Sub LoadData()
    On Error GoTo Error_Handler
    Const cstrErrorSource As String = cstrApplicationName & "." & cstrModuleName & ".LoadData()"
    Dim llngErrNo As Long
    Dim lstrErrDescr As String
    Dim lstrErrSource As String
    llngErrNo = 0
    lstrErrDescr = ""
    lstrErrSource = ""

    Dim lstrSQL As String
    Dim llngAffected As Long
    Dim lobjLenRS As ADODB.Recordset
    Dim i As Integer
    Dim TableName As String
    
    txtPN.Text = Me.mstrDOCPtNum
    
    If Me.mstrDocType = "BoM" Then
        TableName = "BOMPartLenDetails"
    Else
        TableName = "QuotePartLenDetails"
    End If
    
    lstrSQL = "select * from " & TableName & " where Docid=" & mlngDOCID
    lstrSQL = lstrSQL & " and PartNum=" & SQLString(mstrDOCPtNum)
    
    Set lobjLenRS = CnnBTMS.Execute(lstrSQL, llngAffected, adCmdText)
    If Not lobjLenRS Is Nothing Then
        If Not (lobjLenRS.EOF And lobjLenRS.BOF) Then
            lobjLenRS.MoveFirst
            Do While lobjLenRS.EOF = False
                If UCase(mstrDocUM) = "METRIC" Then
                    ClearConObj
                    aCon.Inch = DBLValue(lobjLenRS.Fields("Length").Value)
                    aCon.mmcalc
                    txtLen1(i).Text = Round(aCon.mm, 0)
                Else
                    txtLen1(i).Text = lobjLenRS.Fields("Length").Value
                End If
                txtQty(i).Text = lobjLenRS.Fields("Qty").Value
                msngLen(i) = lobjLenRS.Fields("Length").Value
                i = i + 1
                lobjLenRS.MoveNext
            Loop
        End If
    End If
    
    Call SetControls
    Exit Sub

Error_Handler:
    llngErrNo = IIf(llngErrNo = 0, Err.Number, llngErrNo)
    lstrErrDescr = IIf(Len(lstrErrDescr) = 0, Err.Description, lstrErrDescr & vbCrLf & Err.Description)
    lstrErrSource = IIf(Len(Trim(Err.Source)) = 0, "", Err.Source & vbCrLf) & cstrErrorSource
    Call Err.Raise(llngErrNo, lstrErrSource, lstrErrDescr)

End Sub

Private Sub SaveData()
    On Error GoTo Error_Handler
    Const cstrErrorSource As String = cstrApplicationName & "." & cstrModuleName & ".SaveData()"
    Dim llngErrNo As Long
    Dim lstrErrDescr As String
    Dim lstrErrSource As String
    llngErrNo = 0
    lstrErrDescr = ""
    lstrErrSource = ""

    Dim lstrSQL As String
    Dim lobjADORS As ADODB.Recordset
    Dim llngAffected As Long
    Dim i As Integer
    Dim lsngNewLen(1) As Single
    Dim lstrTableName As String
    
    lstrTableName = IIf(UCase(mstrDocType) = "BOM", "BOMPartLenDetails", "QuotePartLenDetails")
    
    If UCase(mstrDocUM) = "METRIC" Then
        If DBLValue(txtLen1(0).Text) > 0 Then
            Call ClearConObj
            aCon.mm = DBLValue(txtLen1(0).Text)
            aCon.Inchcalc
            lsngNewLen(0) = aCon.Inch
        Else
            lsngNewLen(0) = 0
        End If
        If DBLValue(txtLen1(1).Text) > 0 Then
            Call ClearConObj
            aCon.mm = DBLValue(txtLen1(1).Text)
            aCon.Inchcalc
            lsngNewLen(1) = aCon.Inch
        Else
            lsngNewLen(1) = 0
        End If
    Else
        lsngNewLen(0) = DBLValue(txtLen1(0).Text)
        lsngNewLen(1) = DBLValue(txtLen1(1).Text)
    End If
    
    If DBLValue(txtQty(0).Text) = 0 Then
        lsngNewLen(0) = 0
    End If
    
    If DBLValue(txtQty(1).Text) = 0 Then
        lsngNewLen(1) = 0
    End If
    
    CnnBTMS.BeginTrans
    
    If msngLen(0) = 0 And msngLen(1) = 0 Then 'no record
        If lsngNewLen(1) = 0 Then  'one insert only
            
            lstrSQL = "Select * from " & lstrTableName & " where 1=0"
            Set lobjADORS = New ADODB.Recordset
            lobjADORS.CursorType = adOpenForwardOnly
            lobjADORS.CursorLocation = adUseClient
            lobjADORS.Open lstrSQL, CnnBTMS, adOpenForwardOnly, adLockOptimistic
            Set lobjADORS.ActiveConnection = Nothing
            lobjADORS.AddNew
            lobjADORS.Fields("PartNum").Value = txtPN.Text
            lobjADORS.Fields("Length").Value = lsngNewLen(0)
            lobjADORS.Fields("Qty").Value = DBLValue(txtQty(0).Text)
    
            lobjADORS.Fields("DocNum").Value = mstrDocName
            lobjADORS.Fields("DocID").Value = mlngDOCID
            lobjADORS.Update
            lstrSQL = Build_Insert(lobjADORS, lstrTableName, "CustNum")
            Call CnnBTMS.Execute(lstrSQL, llngAffected, adCmdText And adExecuteNoRecords)
        Else ' both insert
            For i = 0 To 1
                lstrSQL = "Select * from " & lstrTableName & " where 1=0"
                Set lobjADORS = New ADODB.Recordset
                lobjADORS.CursorType = adOpenForwardOnly
                lobjADORS.CursorLocation = adUseClient
                lobjADORS.Open lstrSQL, CnnBTMS, adOpenForwardOnly, adLockOptimistic
                Set lobjADORS.ActiveConnection = Nothing
                lobjADORS.AddNew
                lobjADORS.Fields("PartNum").Value = txtPN.Text
                lobjADORS.Fields("Length").Value = lsngNewLen(i)
                lobjADORS.Fields("Qty").Value = DBLValue(txtQty(i).Text)
        
                lobjADORS.Fields("DocNum").Value = mstrDocName
                lobjADORS.Fields("DocID").Value = mlngDOCID
                lobjADORS.Update
                lstrSQL = Build_Insert(lobjADORS, lstrTableName, "CustNum")
                Call CnnBTMS.Execute(lstrSQL, llngAffected, adCmdText And adExecuteNoRecords)
            Next
        End If
    ElseIf msngLen(1) = 0 And msngLen(0) > 0 Then 'only one record
        If lsngNewLen(1) > 0 Then  'first update, second insert
            lstrSQL = "update " & lstrTableName & " set length=" & Replace(lsngNewLen(0), ",", ".")
            lstrSQL = lstrSQL & " ,qty=" & DBLValue(txtQty(0).Text)
            lstrSQL = lstrSQL & " Where docid=" & mlngDOCID
            lstrSQL = lstrSQL & " and partnum=" & SQLString(txtPN.Text)
            lstrSQL = lstrSQL & " and Length=" & Replace(msngLen(0), ",", ".")
            Call CnnBTMS.Execute(lstrSQL, llngAffected, adCmdText)
            
            lstrSQL = "Select * from " & lstrTableName & " where 1=0"
            Set lobjADORS = New ADODB.Recordset
            lobjADORS.CursorType = adOpenForwardOnly
            lobjADORS.CursorLocation = adUseClient
            lobjADORS.Open lstrSQL, CnnBTMS, adOpenForwardOnly, adLockOptimistic
            Set lobjADORS.ActiveConnection = Nothing
            lobjADORS.AddNew
            lobjADORS.Fields("PartNum").Value = txtPN.Text
            lobjADORS.Fields("Length").Value = lsngNewLen(1)
            lobjADORS.Fields("Qty").Value = DBLValue(txtQty(1).Text)

            lobjADORS.Fields("DocNum").Value = mstrDocName
            lobjADORS.Fields("DocID").Value = mlngDOCID
            lobjADORS.Update
            lstrSQL = Build_Insert(lobjADORS, lstrTableName, "CustNum")
            Call CnnBTMS.Execute(lstrSQL, llngAffected, adCmdText And adExecuteNoRecords)
        
        Else 'first update
            lstrSQL = "update " & lstrTableName & " set length=" & Replace(lsngNewLen(0), ",", ".")
            lstrSQL = lstrSQL & " ,qty=" & DBLValue(txtQty(0).Text)
            lstrSQL = lstrSQL & " Where docid=" & mlngDOCID
            lstrSQL = lstrSQL & " and partnum=" & SQLString(txtPN.Text)
            lstrSQL = lstrSQL & " and Length=" & Replace(msngLen(0), ",", ".")
            Call CnnBTMS.Execute(lstrSQL, llngAffected, adCmdText)
        
        End If
    ElseIf msngLen(1) > 0 And msngLen(0) > 0 Then 'two records
        
        If lsngNewLen(1) = 0 Then 'first update, second delete
            lstrSQL = "update " & lstrTableName & " set length=" & Replace(lsngNewLen(0), ",", ".")
            lstrSQL = lstrSQL & " ,qty=" & DBLValue(txtQty(0).Text)
            lstrSQL = lstrSQL & " Where docid=" & mlngDOCID
            lstrSQL = lstrSQL & " and partnum=" & SQLString(txtPN.Text)
            lstrSQL = lstrSQL & " and Length=" & Replace(msngLen(0), ",", ".")
            Call CnnBTMS.Execute(lstrSQL, llngAffected, adCmdText)
            
            lstrSQL = "Delete * from " & lstrTableName & " where "
            lstrSQL = lstrSQL & " docid=" & mlngDOCID
            lstrSQL = lstrSQL & " and partnum=" & SQLString(txtPN.Text)
            lstrSQL = lstrSQL & " and Length=" & Replace(msngLen(1), ",", ".")
            Call CnnBTMS.Execute(lstrSQL, llngAffected, adCmdText And adExecuteNoRecords)
        ElseIf lsngNewLen(0) = 0 Then 'first delete, second update
            
            lstrSQL = "Delete * from " & lstrTableName & " where "
            lstrSQL = lstrSQL & " docid=" & mlngDOCID
            lstrSQL = lstrSQL & " and partnum=" & SQLString(txtPN.Text)
            lstrSQL = lstrSQL & " and Length=" & Replace(msngLen(0), ",", ".")
            Call CnnBTMS.Execute(lstrSQL, llngAffected, adCmdText And adExecuteNoRecords)
            
            lstrSQL = "update " & lstrTableName & " set length=" & Replace(lsngNewLen(1), ",", ".")
            lstrSQL = lstrSQL & " ,qty=" & DBLValue(txtQty(1).Text)
            lstrSQL = lstrSQL & " Where docid=" & mlngDOCID
            lstrSQL = lstrSQL & " and partnum=" & SQLString(txtPN.Text)
            lstrSQL = lstrSQL & " and Length=" & Replace(msngLen(1), ",", ".")
            Call CnnBTMS.Execute(lstrSQL, llngAffected, adCmdText)
            
        Else 'both update
            For i = 0 To 1
                lstrSQL = "update " & lstrTableName & " set length=" & Replace(lsngNewLen(i), ",", ".")
                lstrSQL = lstrSQL & " ,qty=" & DBLValue(txtQty(i).Text)
                lstrSQL = lstrSQL & " Where docid=" & mlngDOCID
                lstrSQL = lstrSQL & " and partnum=" & SQLString(txtPN.Text)
                lstrSQL = lstrSQL & " and Length=" & Replace(msngLen(i), ",", ".")
                Call CnnBTMS.Execute(lstrSQL, llngAffected, adCmdText)
            Next
        End If
        
    End If
    
    'update bom(quote)details table with the new qty
    Dim ldblNewQty As Double 'will be in mm or inches
    If UCase(mstrDocUM) = "METRIC" Then
        ldblNewQty = (DBLValue(txtLen1(0).Text) * DBLValue(txtQty(0).Text)) + (DBLValue(txtLen1(1).Text) * DBLValue(txtQty(1).Text))
    Else
        ldblNewQty = (lsngNewLen(0) * DBLValue(txtQty(0).Text)) + (lsngNewLen(1) * DBLValue(txtQty(1).Text))
    End If
    
    If UCase(mstrDocType) = "BOM" Then
        lstrSQL = "update bomdetails set bomptqtyreq=" & Replace(ldblNewQty, ",", ".")
        lstrSQL = lstrSQL & ",bompttotamountreq=" & Replace((lsngNewLen(0) * DBLValue(txtQty(0).Text)) + (lsngNewLen(1) * DBLValue(txtQty(1).Text)), ",", ".") 'lsngNewLen(0) + lsngNewLen(1) bug in 2.0.16 needs to multiply with the new qty entered
        lstrSQL = lstrSQL & " where bomid=" & mlngDOCID
        lstrSQL = lstrSQL & " and BomPtNum=" & SQLString(txtPN.Text)
    Else
        lstrSQL = "update Quotedetails set Quoteptqty=" & Replace(ldblNewQty, ",", ".")
        lstrSQL = lstrSQL & ",quotetotamountreq=" & Replace((lsngNewLen(0) * DBLValue(txtQty(0).Text)) + (lsngNewLen(1) * DBLValue(txtQty(1).Text)), ",", ".")
        lstrSQL = lstrSQL & " where Quoteid=" & mlngDOCID
        lstrSQL = lstrSQL & " and quotePtNum=" & SQLString(txtPN.Text)
    End If
    Call CnnBTMS.Execute(lstrSQL, llngAffected, adCmdText)
    
    CnnBTMS.CommitTrans
    Exit Sub

Error_Handler:
    CnnBTMS.RollbackTrans
    llngErrNo = IIf(llngErrNo = 0, Err.Number, llngErrNo)
    lstrErrDescr = IIf(Len(lstrErrDescr) = 0, Err.Description, lstrErrDescr & vbCrLf & Err.Description)
    lstrErrSource = IIf(Len(Trim(Err.Source)) = 0, "", Err.Source & vbCrLf) & cstrErrorSource
    Call Err.Raise(llngErrNo, lstrErrSource, lstrErrDescr)

End Sub
