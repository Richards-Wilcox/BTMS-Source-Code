Private Sub SaveData()
    On Error GoTo Error_Handler
    Const cstrErrorSource As String = cstrApplicationName & "." & cstrModuleName & ".SaveData()"
    Dim llngErrNo As Long
    Dim lstrErrDescr As String
    Dim lstrErrSource As String
    llngErrNo = 0
    lstrErrDescr = ""
    lstrErrSource = ""

    Dim llngAffected As Long
    Dim lstrSQLQuery As String
    Dim llngQuoteTempID As Long
    
    If mblnQuoteChange = True Or mblnPOChange = True Then
        mintMode = MODE_VIEW
        Call SetControls(mintMode)
        If mlngBOMID = 0 Then Exit Sub
    
        If MsgBox("Are you sure you wish to do this?", vbApplicationModal + vbYesNo + vbExclamation, "You are updating the PO and/or the Quote with this BoM") = vbNo Then
            Call LoadData
            Exit Sub
        End If
        Screen.MousePointer = vbHourglass
        If mblnQuoteChange = True Then
            If cmbQuote.ListIndex < 0 Then cmbQuote.ListIndex = 0
            llngQuoteTempID = mlngQuoteID
        End If
        If mblnPOChange = True Then
            If cmbPO.ListIndex < 0 Then cmbPO.ListIndex = 0
        End If
        
        Screen.MousePointer = vbHourglass
        CnnBTMS.BeginTrans
        If mlngQuoteID + cmbQuote.ItemData(cmbQuote.ListIndex) > 0 Then
            Call ProcessQuote(cmbQuote.ItemData(cmbQuote.ListIndex), mlngQuoteID, mlngBOMID)
        End If
        If mlngPOID + cmbPO.ItemData(cmbPO.ListIndex) > 0 Then Call ProcessPO(cmbPO.ItemData(cmbPO.ListIndex), mlngPOID, mlngBOMID)
        
        lstrSQLQuery = "Update BoM Set "
        If mblnQuoteChange = True Then
            lstrSQLQuery = lstrSQLQuery & "QuoteNum=" & SQLString(Trim(cmbQuote.List(cmbQuote.ListIndex)))
            lstrSQLQuery = lstrSQLQuery & ", QuoteID=" & cmbQuote.ItemData(cmbQuote.ListIndex)
            If mblnPOChange = True Then lstrSQLQuery = lstrSQLQuery & " and "
        End If
        
        If mblnPOChange = True Then
            lstrSQLQuery = lstrSQLQuery & "PONum=" & SQLString(Trim(cmbPO.List(cmbPO.ListIndex)))
            lstrSQLQuery = lstrSQLQuery & ", POID=" & cmbPO.ItemData(cmbPO.ListIndex)
        End If
        
        lstrSQLQuery = lstrSQLQuery & " where bomid=" & mlngBOMID
        Call CnnBTMS.Execute(lstrSQLQuery, llngAffected, adCmdText And adExecuteNoRecords)
        CnnBTMS.CommitTrans
        
        If mlngQuoteID + cmbQuote.ItemData(cmbQuote.ListIndex) > 0 Then
            Call UpdateLabourOnQuote(IIf(mlngQuoteID > 0, mlngQuoteID, llngQuoteTempID))
            Call UpdateQuoteTotalPrice(mlngQuoteID)
        End If
        
        Call LoadData
        Screen.MousePointer = vbDefault
    End If
    Exit Sub

Error_Handler:
    llngErrNo = IIf(llngErrNo = 0, Err.Number, llngErrNo)
    lstrErrDescr = IIf(Len(lstrErrDescr) = 0, Err.Description, lstrErrDescr & vbCrLf & Err.Description)
    lstrErrSource = IIf(Len(Trim(Err.Source)) = 0, "", Err.Source & vbCrLf) & cstrErrorSource
    CnnBTMS.RollbackTrans
    Screen.MousePointer = vbDefault
    Call Err.Raise(llngErrNo, lstrErrSource, lstrErrDescr)
End Sub

Private Sub LoadData()
    On Error GoTo Error_Handler
    Const cstrErrorSource As String = cstrApplicationName & "." & cstrModuleName & ".LoadData()"
    Dim llngErrNo As Long
    Dim lstrErrDescr As String
    Dim lstrErrSource As String
    llngErrNo = 0
    lstrErrDescr = ""
    lstrErrSource = ""
    
    mintMode = MODE_VIEW
    If mlngBOMID <= 0 Then Exit Sub
    
    Dim llngAffected As Long
    Dim lstrSQLQuery As String
    Dim lobjBOMRS As ADODB.Recordset
    
    lstrSQLQuery = "select * from BoM where BOMID=" & mlngBOMID
    Set lobjBOMRS = CnnBTMS.Execute(lstrSQLQuery, llngAffected, adCmdText)
    If Not lobjBOMRS Is Nothing Then
        lobjBOMRS.ActiveConnection = Nothing
        If Not (lobjBOMRS.BOF And lobjBOMRS.EOF) Then
            lobjBOMRS.MoveFirst
            Dim lstrFieldName As String
            lstrFieldName = "BOMID"
            mlngBOMID = lobjBOMRS.Fields(lstrFieldName).Value
            lstrFieldName = "POID"
            mlngPOID = IIf(IsNull(lobjBOMRS.Fields(lstrFieldName).Value), 0, lobjBOMRS.Fields(lstrFieldName).Value)
            lstrFieldName = "QuoteID"
            mlngQuoteID = IIf(IsNull(lobjBOMRS.Fields(lstrFieldName).Value), 0, lobjBOMRS.Fields(lstrFieldName).Value)
            lstrFieldName = "BOMNum"
            lblBOMNum.Caption = IIf(IsNull(lobjBOMRS.Fields(lstrFieldName).Value), mstrDocNum, Trim(lobjBOMRS.Fields(lstrFieldName).Value))
            lstrFieldName = "BOMDate"
            lblDate.Caption = IIf(IsNull(lobjBOMRS.Fields(lstrFieldName).Value), "", Format(lobjBOMRS.Fields(lstrFieldName).Value, cstrDateMask))
            lstrFieldName = "CustName"
            lblName.Caption = IIf(IsNull(lobjBOMRS.Fields(lstrFieldName).Value), "", Trim(lobjBOMRS.Fields(lstrFieldName).Value))
            lobjBOMRS.Close
        End If
    End If
    Set lobjBOMRS = Nothing
    
    Call LoadQuotes
    Call LoadPOs
    
    Call SetControls(mintMode)
    Exit Sub

Error_Handler:
    llngErrNo = IIf(llngErrNo = 0, Err.Number, llngErrNo)
    lstrErrDescr = IIf(Len(lstrErrDescr) = 0, Err.Description, Err.Description)
    lstrErrSource = IIf(Len(Trim(Err.Source)) = 0, "", Err.Source & vbCrLf) & cstrErrorSource
    Call Err.Raise(llngErrNo, lstrErrSource, lstrErrDescr)
End Sub

Private Sub SetControls(ByVal intMode)
    On Error GoTo Error_Handler
    Const cstrErrorSource As String = cstrApplicationName & "." & cstrModuleName & ".SetControls()"
    Dim llngErrNo As Long
    Dim lstrErrDescr As String
    Dim lstrErrSource As String
    llngErrNo = 0
    lstrErrDescr = ""
    lstrErrSource = ""

    cmdAddQuote.Caption = "New &Quote"
    cmdAddPO.Caption = "New &PO"
    
    Select Case intMode
        Case MODE_EDIT
                        cmdEdit.Caption = "Cancel &Edit"
                        cmdEdit.Enabled = True
                        cmdSave.Enabled = True
                        cmdBack.Enabled = False
                        cmdAddQuote.Enabled = False
                        cmdAddPO.Enabled = False
                        
                        If cmbPO.ListIndex >= 0 Then cmbPO.Enabled = True
                        If cmbQuote.ListIndex >= 0 Then cmbQuote.Enabled = True
                        
                        cmbPO.BackColor = &HFFFFFF
                        cmbQuote.BackColor = &HFFFFFF
        Case Else
                        cmdEdit.Caption = "&Edit BoM"
                        cmdEdit.Enabled = True
                        cmdSave.Enabled = False
                        cmdBack.Enabled = True
                        cmdAddQuote.Enabled = False
                        cmdAddPO.Enabled = False
                        
                        cmbPO.Enabled = False
                        cmbQuote.Enabled = False
                        
                        cmbPO.BackColor = &HE0E0E0
                        cmbQuote.BackColor = &HE0E0E0
                        
                        If mlngQuoteID = 0 Then
                            cmdAddQuote.Enabled = True
                        Else
                            If mblnEditable = False Then
                                cmdEdit.Enabled = False
                                'cmdAddQuote.Enabled = False
                            Else
'                                cmdAddQuote.Enabled = True
'                                cmdAddQuote.Caption = "Goto &Quote"
                            End If
                            cmdAddQuote.Enabled = True
                            cmdAddQuote.Caption = "Goto &Quote"
                        End If
                        
                        If mlngPOID = 0 Then
                            cmdAddPO.Enabled = True
                        Else
                            If mblnEditable = False Then
                                cmdEdit.Enabled = False
                                'cmdAddPO.Enabled = False
                            Else
                                'cmdAddPO.Enabled = True
                                'cmdAddPO.Caption = "Goto &PO"
                            End If
                            cmdAddPO.Enabled = True
                            cmdAddPO.Caption = "Goto &PO"
                        End If
    End Select
    
    If cmbPO.ListCount > 0 And Not cmbPO.List(cmbPO.ListIndex) = "" Then
        cmdAddQuote.Enabled = False
    End If
    If cmbQuote.ListCount > 0 And Not cmbQuote.List(cmbQuote.ListIndex) = "" Then
        cmdAddPO.Enabled = False
    End If
    
    If UCase(Trim(lblBOMNum.Caption)) = "TEMPBOM" Then
        cmbPO.Enabled = False
        cmdAddPO.Enabled = False
        cmbQuote.Enabled = False
        cmdAddQuote.Enabled = False
    End If
    
    If gblnDistributor = False Then
        cmbPO.Enabled = False
        cmdAddPO.Enabled = False
    End If
    Exit Sub

Error_Handler:
    llngErrNo = IIf(llngErrNo = 0, Err.Number, llngErrNo)
    lstrErrDescr = IIf(Len(lstrErrDescr) = 0, Err.Description, Err.Description)
    lstrErrSource = IIf(Len(Trim(Err.Source)) = 0, "", Err.Source & vbCrLf) & cstrErrorSource
    Call Err.Raise(llngErrNo, lstrErrSource, lstrErrDescr)
End Sub

    Dim llngErrNo As Long
    Dim lstrErrDescr As String
    Dim lstrErrSource As String
    llngErrNo = 0
    lstrErrDescr = ""
    lstrErrSource = ""
    
    If lngBOMID < 1 Then Exit Sub
    If lngCurrQuoteID > 0 And lngCurrQuoteID = lngOldQuoteID Then Exit Sub
    
    Dim llngAffected As Long
    Dim lstrSQLQuery As String
    Dim lblnInTrans As Boolean
    Dim lstrQuoteNum As String

    CnnBTMS.BeginTrans

    Set lobjQuoteRS = CnnBTMS.Execute(lstrSQLQuery, llngAffected, adCmdText)
    If Not lobjQuoteRS Is Nothing Then
        lobjQuoteRS.ActiveConnection = Nothing
        If Not (lobjQuoteRS.BOF And lobjQuoteRS.EOF) Then

            Set lobjQuoteRS = CnnBTMS.Execute(lstrSQLQuery, llngAffected, adCmdText)
            If Not lobjQuoteRS Is Nothing Then
                lobjQuoteRS.ActiveConnection = Nothing
                If Not (lobjQuoteRS.BOF And lobjQuoteRS.EOF) Then
                    lobjQuoteRS.MoveFirst
                    mobjQuote.QuoteNum = lobjQuoteRS.Fields("QuoteNum").Value
                End If
            End If
        End If
        Set lobjQuoteRS = Nothing
    End If
    
    If lngCurrQuoteID = 0 And lngOldQuoteID = 0 Then
        ' Prompt For missing Information
        
        Dim lblnBreak As Boolean
    
        Do Until lblnBreak
        DoEvents
            mstrData = ""
            Dim frmX As frmBOMQuote
            Set frmX = New frmBOMQuote
            Set frmX.objParent = Me
            Screen.MousePointer = vbDefault

                    Exit Sub
                End If
                lstrQuoteNum = Trim(larrData(0))
                
                '  check for duplicates
                lstrSQLQuery = "SELECT QuoteNum FROM Quote"
                Set lobjQuoteRS = CnnBTMS.Execute(lstrSQLQuery, llngAffected, adCmdText)
                If Not lobjQuoteRS Is Nothing Then
                    lobjQuoteRS.ActiveConnection = Nothing

                        lobjQuoteRS.MoveFirst
                        lobjQuoteRS.Find "QuoteNum=" & SQLString(lstrQuoteNum)
                        If lobjQuoteRS.BOF = False And lobjQuoteRS.EOF = False Then
                            MsgBox "The Quote name already exists, please chose another.", vbInformation + vbApplicationModal, "Quote Name Exists"
                        Else
                            lblnBreak = True
                            mobjQuote.QuoteNum = Trim(larrData(0))
                            lngCurrQuoteID = CreateQuote(lngBOMID)
                            mobjQuote.QuoteID = lngCurrQuoteID
                            
                        End If
                    Else

                        mobjQuote.QuoteNum = Trim(larrData(0))
                        lngCurrQuoteID = CreateQuote(lngBOMID)
                        mobjQuote.QuoteID = lngCurrQuoteID
                    End If
                    lobjQuoteRS.Close
                End If
                Set lobjQuoteRS = Nothing
            End If
            If lstrQuoteNum = "" Then
                MsgBox "Please enter a Quote Number", vbApplicationModal + vbInformation + vbOKOnly, "Quote Error"
                lblnBreak = False

Private Function CreateQuote(ByVal lngBOMID As Long) As Long
    On Error GoTo Error_Handler
    Const cstrErrorSource As String = cstrApplicationName & "." & cstrModuleName & ".CreateQuote()"
    Dim llngErrNo As Long
    Dim lstrErrDescr As String
    Dim lstrErrSource As String
    llngErrNo = 0
    lstrErrDescr = ""
    lstrErrSource = ""

    Dim llngAffected As Long
    Dim lstrSQLQuery As String
    Dim lobjBOMRS As ADODB.Recordset
    Dim lobjQuoteRS As ADODB.Recordset
    Dim llngQuoteID As Long
    
    If Not mlngBOMID > 0 Then Exit Function
    
    llngQuoteID = 0
    lstrSQLQuery = "select * from BoM where BOMID=" & lngBOMID
    Set lobjBOMRS = CnnBTMS.Execute(lstrSQLQuery, llngAffected, adCmdText)
    If Not lobjBOMRS Is Nothing Then
        lobjBOMRS.ActiveConnection = Nothing
        If Not (lobjBOMRS.BOF And lobjBOMRS.EOF) Then
            lobjBOMRS.MoveFirst
            Set lobjQuoteRS = BuildQuoteRS(lobjBOMRS)
            If Not lobjQuoteRS Is Nothing Then
                If Not (lobjQuoteRS.BOF And lobjQuoteRS.EOF) Then
                    lobjQuoteRS.MoveFirst
                    lstrErrDescr = "Error building insert statement for Quote"
                    lstrSQLQuery = Build_Insert(lobjQuoteRS, "Quote", "QuoteID")
                    lstrErrDescr = "Error inserting data into Quote"
                    
                    ' Insert the new Quote
                    Call CnnBTMS.Execute(lstrSQLQuery, 0, adCmdText And adExecuteNoRecords)
                    lstrErrDescr = ""
                    lobjQuoteRS.Close
                    
                    ' Find the QuoteID
                    lstrSQLQuery = "select QuoteID from Quote where QuoteNum=" & SQLString(mobjQuote.QuoteNum)
                    Set lobjQuoteRS = CnnBTMS.Execute(lstrSQLQuery, llngAffected, adCmdText)
                    If Not lobjQuoteRS Is Nothing Then
                        lobjQuoteRS.ActiveConnection = Nothing
                        If Not (lobjQuoteRS.BOF And lobjQuoteRS.EOF) Then
                            lobjQuoteRS.MoveFirst
                            llngQuoteID = IIf(IsNull(lobjQuoteRS.Fields("QuoteID").Value), 0, lobjQuoteRS.Fields("QuoteID").Value)
                        End If
                        lobjQuoteRS.Close
                    End If
                End If
            End If
        End If
        lobjBOMRS.Close
    End If
    
    Set lobjQuoteRS = Nothing
    Set lobjBOMRS = Nothing
    CreateQuote = llngQuoteID
    Exit Function

Error_Handler:
    llngErrNo = IIf(llngErrNo = 0, Err.Number, llngErrNo)
    lstrErrDescr = IIf(Len(lstrErrDescr) = 0, Err.Description, lstrErrDescr & vbCrLf & Err.Description)
    lstrErrSource = IIf(Len(Trim(Err.Source)) = 0, "", Err.Source & vbCrLf) & cstrErrorSource
    Call Err.Raise(llngErrNo, lstrErrSource, lstrErrDescr)
End Function

Private Sub AddToQuote(ByVal lngQuoteID As Long, ByVal lngBOMID As Long)

    Dim lstrErrSource As String
    llngErrNo = 0
    lstrErrDescr = ""
    lstrErrSource = ""

    Dim lstrSQL As String
    Dim lintMode As Integer
    Dim lstrWhereClause As String
    Dim lstrExcludeFieldList As String
    Dim llngAffected As Long
    Dim lobjADORS As ADODB.Recordset
    Dim lobjParentRS As ADODB.Recordset

''''''''''Change in .NET''''''''''''''''
    Dim lstrNumArray() As String
    Dim lintCount As Integer
    ReDim lstrNumArray(0)
    
    mblnflagMessageshown = False 'initial value
     
''''''''''END Change in .NET
    
    ' Copy the fields from the screen to an ADO Recordset
    Set lobjADORS = BuildQuoteDetailRS(lngBOMID)
    Call CopyLenDetails(mobjQuote.QuoteID, mobjQuote.QuoteNum)
    
    lintMode = MODE_ADD
    ' Process through the recordset and insert or update rows
    If Not lobjADORS Is Nothing Then
        If Not (lobjADORS.BOF And lobjADORS.EOF) Then
            lobjADORS.MoveFirst
            lstrExcludeFieldList = ""
            
            Do While lobjADORS.EOF = False
                lintMode = MODE_ADD
                ' Check to see if the record exists in the
                ' BOMDetails table already
                lstrSQL = "select QuotePtNum, QuotePtQty, QuoteTotAmountReq, QuotePtCallPrice, QuoteDiscount, QuotePtMultID, QuotePtCost from QuoteDetails"
                lstrSQL = lstrSQL & " where QuoteID=" & lngQuoteID
                lstrSQL = lstrSQL & " and QuotePtNum=" & SQLString(lobjADORS.Fields("QuotePtNum").Value)
                Set lobjParentRS = CnnBTMS.Execute(lstrSQL, llngAffected, adCmdText)
                If Not lobjParentRS Is Nothing Then
                    If Not (lobjParentRS.BOF And lobjParentRS.EOF) Then
                        lobjParentRS.MoveFirst
                        ' Reset the Mode so that we do an update of the existing part
                        lintMode = MODE_EDIT
                        ' Average the cost and sum the qty's
                        If lobjADORS.Fields("QuotePtMultID").Value <> lobjParentRS.Fields("QuotePtMultID").Value Then
                            lobjADORS.Fields("QuotePtCallPrice").Value = True
                            lobjADORS.Fields("QuoteCallPriceExpDate").Value = Format(Now() + 30, "Short Date")
                        End If
                        If lobjADORS.Fields("QuoteDiscount").Value + lobjParentRS.Fields("QuotePtMultID").Value > 0 Then
                            If lobjParentRS.Fields("QuoteDiscount").Value > 0 Then
                                If lobjADORS.Fields("QuoteDiscount").Value > lobjParentRS.Fields("QuoteDiscount").Value Or _
                                lobjADORS.Fields("QuoteDiscount").Value = 0 Then
                                    lobjADORS.Fields("QuoteDiscount").Value = lobjParentRS.Fields("QuoteDiscount").Value
                                End If
                            End If
'                            lobjADORS.Fields("QuotePtCallPrice").Value = True
'                            lobjADORS.Fields("QuoteCallPriceExpDate").Value = Format(Now() + 30, "Short Date")
                        End If

''''''''''Change in .NET''''''''''''''''
'                        lobjADORS.Fields("QuotePtCost").Value = DBLValue(lobjADORS.Fields("Quotetotamountreq").Value) * DBLValue(lobjADORS.Fields("QuotePtCost").Value)
'                        lobjADORS.Fields("QuotePtCost").Value = DBLValue(lobjADORS.Fields("QuotePtCost").Value) + DBLValue(lobjParentRS.Fields("Quotetotamountreq").Value) * DBLValue(lobjParentRS.Fields("QuotePtCost").Value)

                        If lobjADORS.Fields("QuotePtCallPrice").Value <> lobjParentRS.Fields("QuotePtCallPrice").Value Then

                                lstrNumArray(lintCount) = lobjADORS.Fields("QuotePtNum")
                                ReDim Preserve lstrNumArray(UBound(lstrNumArray) + 1)
                                lintCount = lintCount + 1
                            End If

''''''''''END Change in .NET

                    lobjADORS.Fields("QuoteID").Value = lngQuoteID

                    lstrErrDescr = "Error updating QuoteDetails"
                End If

                lstrErrDescr = ""
                lobjADORS.MoveNext
            Loop

Private Sub RemoveFromQuote(ByVal lngQuoteID As Long, ByVal lngBOMID As Long)
     On Error GoTo Error_Handler
    Const cstrErrorSource As String = cstrApplicationName & "." & cstrModuleName & ".RemoveFromQuote()"
    Dim llngErrNo As Long
    Dim lstrErrDescr As String
    Dim lstrErrSource As String
    llngErrNo = 0
    lstrErrDescr = ""
    lstrErrSource = ""

    Dim lstrSQL As String
    Dim lstrWhereClause As String
    Dim lstrExcludeFieldList As String
    Dim llngAffected As Long
    Dim lobjADORS As ADODB.Recordset
    Dim lobjParentRS As ADODB.Recordset

'''''''''Change in .NET''''''''''''''''
    mblnFlagRemoveTag = True    'keep track of remove tag so that it won't present same message box in BuildQuoteDetailRS routine has.
''''''''''END Change in .NET
    
    ' Copy the fields from the screen to an ADO Recordset
    Set lobjADORS = BuildQuoteDetailRS(lngBOMID)
    
    ' Process through the recordset and remove  rows
    If Not lobjADORS Is Nothing Then
        If Not (lobjADORS.BOF And lobjADORS.EOF) Then
            lobjADORS.MoveFirst
            lstrExcludeFieldList = ""
            
            Do While lobjADORS.EOF = False
                ' Check to see if the record exists in the
                ' BOMDetails table already
                lobjADORS.Fields("QuoteID").Value = lngQuoteID

                lstrSQL = "select QuotePtNum,QuoteTotAmountReq, QuotePtQty, QuotePtCost from QuoteDetails"
                lstrSQL = lstrSQL & " where QuoteID=" & lngQuoteID
                lstrSQL = lstrSQL & " and QuotePtNum=" & SQLString(lobjADORS.Fields("QuotePtNum").Value)
                Set lobjParentRS = CnnBTMS.Execute(lstrSQL, llngAffected, adCmdText)
                If Not lobjParentRS Is Nothing Then
                    If Not (lobjParentRS.BOF And lobjParentRS.EOF) Then
                        lobjParentRS.MoveFirst
                        lobjADORS.Fields("QuotePtQty").Value = DBLValue(lobjParentRS.Fields("QuotePtQty").Value) - DBLValue(lobjADORS.Fields("QuotePtQty").Value)
                        lobjADORS.Fields("QuoteTotAmountReq").Value = DBLValue(lobjParentRS.Fields("QuoteTotAmountReq").Value) - DBLValue(lobjADORS.Fields("QuoteTotAmountReq").Value)
                        If DBLValue(lobjADORS.Fields("QuotePtQty").Value) < 0 Then lobjADORS.Fields("QuotePtQty").Value = 0
                        If DBLValue(lobjADORS.Fields("QuoteTotAmountReq").Value) < 0 Then lobjADORS.Fields("QuoteTotAmountReq").Value = 0
                    End If
                    lobjParentRS.Close
                End If
                Set lobjParentRS = Nothing

                If lobjADORS.Fields("QuotePtQty").Value > 0 Then
                    lstrErrDescr = "Error building update statement for QuoteDetails"
                    lstrWhereClause = "QuoteID=" & lngQuoteID
                    lstrWhereClause = lstrWhereClause & " and QuotePtNum=" & SQLString(lobjADORS.Fields("QuotePtNum").Value)
                    lstrSQL = Build_Update(lobjADORS, "QuoteDetails", lstrWhereClause, lstrExcludeFieldList)
                    lstrErrDescr = "Error updating QuoteDetails"
                Else
                    lstrErrDescr = "Error building delete statement for QuoteDetails"
                    lstrSQL = "Delete * from QuoteDetails where QuoteID=" & lngQuoteID
                    lstrSQL = lstrSQL & " and QuotePtNum=" & SQLString(lobjADORS.Fields("QuotePtNum").Value)
                    lstrErrDescr = "Error deleting QuoteDetails"
                End If
                
                Call CnnBTMS.Execute(lstrSQL, 0, adCmdText And adExecuteNoRecords)
                lstrErrDescr = ""
                lobjADORS.MoveNext
            Loop
            
            lobjADORS.Close
            Set lobjParentRS = Nothing
        End If
    End If
    Call RemovePartLenDetails(lngBOMID)
    
    Set lobjADORS = Nothing
    
''''''''''Change in .NET''''''''''''''''
    mblnFlagRemoveTag = False
''''''''''END Change in .NET
    Exit Sub

Error_Handler:
    llngErrNo = IIf(llngErrNo = 0, Err.Number, llngErrNo)
    lstrErrDescr = IIf(Len(lstrErrDescr) = 0, Err.Description, lstrErrDescr & vbCrLf & Err.Description)
    lstrErrSource = IIf(Len(Trim(Err.Source)) = 0, "", Err.Source & vbCrLf) & cstrErrorSource
    Call Err.Raise(llngErrNo, lstrErrSource, lstrErrDescr)
End Sub

Private Function CreatePO(ByVal lngBOMID As Long) As Long
    On Error GoTo Error_Handler

    Dim llngErrNo As Long

    Dim lstrErrSource As String
    llngErrNo = 0

    lstrErrSource = ""

    Dim llngAffected As Long
    Dim lstrSQLQuery As String
    Dim lobjBOMRS As ADODB.Recordset
    Dim lobjPORS As ADODB.Recordset
    Dim llngPOID As Long
    
    If Not mlngBOMID > 0 Then Exit Function
    
    llngPOID = 0
    lstrSQLQuery = "select * from BoM where BOMID=" & lngBOMID
    Set lobjBOMRS = CnnBTMS.Execute(lstrSQLQuery, llngAffected, adCmdText)
    If Not lobjBOMRS Is Nothing Then
        lobjBOMRS.ActiveConnection = Nothing
        If Not (lobjBOMRS.BOF And lobjBOMRS.EOF) Then
            lobjBOMRS.MoveFirst
            Set lobjPORS = BuildPORS(lobjBOMRS)
            If Not lobjPORS Is Nothing Then
                If Not (lobjPORS.BOF And lobjPORS.EOF) Then
                    lobjPORS.MoveFirst
                    lstrErrDescr = "Error building insert statement for PO"
                    lstrSQLQuery = Build_Insert(lobjPORS, "PO", "POID")
                    lstrErrDescr = "Error inserting data into PO"
                    
                    ' Insert the new PO
                    Call CnnBTMS.Execute(lstrSQLQuery, 0, adCmdText And adExecuteNoRecords)
                    lstrErrDescr = ""
                    lobjPORS.Close
                    
                    ' Find the POID
                    lstrSQLQuery = "select POID from PO where PONum=" & SQLString(mobjPO.PONum)
                    Set lobjPORS = CnnBTMS.Execute(lstrSQLQuery, llngAffected, adCmdText)
                    If Not lobjPORS Is Nothing Then
                        lobjPORS.ActiveConnection = Nothing
                        If Not (lobjPORS.BOF And lobjPORS.EOF) Then
                            lobjPORS.MoveFirst
                            llngPOID = IIf(IsNull(lobjPORS.Fields("POID").Value), 0, lobjPORS.Fields("POID").Value)
                        End If
                        lobjPORS.Close
                    End If
                End If
            End If
        End If
        lobjBOMRS.Close
    End If
    
    Set lobjPORS = Nothing
    Set lobjBOMRS = Nothing
    CreatePO = llngPOID
    Exit Function

Error_Handler:
    llngErrNo = IIf(llngErrNo = 0, Err.Number, llngErrNo)

End Function

Private Sub CopyLenDetails(ByVal QuoteID As Long, ByVal QuoteName As String)
    On Error GoTo Error_Handler
    Const cstrErrorSource As String = cstrApplicationName & "." & cstrModuleName & ".CopyLenDetails()"
    Dim llngErrNo As Long
    Dim lstrErrDescr As String
    Dim lstrErrSource As String
    llngErrNo = 0
    lstrErrDescr = ""
    lstrErrSource = ""

    Dim lstrSQL As String
    Dim llngAffected As Long
    Dim lobjRS As ADODB.Recordset
    Dim lobjLenRS As ADODB.Recordset
    
    lstrSQL = "select * from BOMPartLenDetails where docid=" & mlngBOMID
    lstrSQL = lstrSQL & " and docnum=" & SQLString(mstrDocNum)
    
    Set lobjRS = CnnBTMS.Execute(lstrSQL, llngAffected, adCmdText)
    
    lstrSQL = "select * from QuotePartLenDetails where 1=0"
    
    Set lobjLenRS = New ADODB.Recordset
    lobjLenRS.CursorLocation = adUseClient
    lobjLenRS.Open lstrSQL, CnnBTMS, adOpenKeyset, adLockOptimistic
    
    If Not lobjRS Is Nothing Then
        If Not (lobjRS.EOF And lobjRS.BOF) Then
            lobjRS.MoveFirst
            Do While lobjRS.EOF = False
                lobjLenRS.AddNew
                lobjLenRS.Fields("PartNum").Value = lobjRS.Fields("PartNum").Value
                lobjLenRS.Fields("Length").Value = lobjRS.Fields("Length").Value
                lobjLenRS.Fields("Qty").Value = lobjRS.Fields("Qty").Value
                lobjLenRS.Fields("DocNum").Value = QuoteName
                lobjLenRS.Fields("DocID").Value = QuoteID
                lobjLenRS.Fields("bomID").Value = mlngBOMID
                lobjLenRS.Update
                
                lobjRS.MoveNext
            Loop
        End If
    End If
    Set lobjRS = Nothing
    Exit Sub

Error_Handler:
    llngErrNo = IIf(llngErrNo = 0, Err.Number, llngErrNo)
    lstrErrDescr = IIf(Len(lstrErrDescr) = 0, Err.Description, lstrErrDescr & vbCrLf & Err.Description)
    lstrErrSource = IIf(Len(Trim(Err.Source)) = 0, "", Err.Source & vbCrLf) & cstrErrorSource
    Call Err.Raise(llngErrNo, lstrErrSource, lstrErrDescr)

End Sub
