Private Sub SaveData()

    On Error GoTo Error_Handler
    Const cstrErrorSource As String = cstrApplicationName & "." & cstrModuleName & ".SaveData()"
    Dim llngErrNo As Long
    Dim lstrErrDescr As String
    Dim lstrErrSource As String
    llngErrNo = 0
    lstrErrDescr = ""
    lstrErrSource = ""
    
    Dim lstrSQL As String
    Dim lstrWhereClause As String
    Dim lstrExcludeFieldList As String
    Dim llngAffected As Long
    Dim lobjADORS As ADODB.Recordset
    If mintMode = MODE_ADD Then
        lstrSQL = "select custname from customerinformation where custname=" & SQLString(Trim(txtCName.Text))
        Set lobjADORS = CnnBTMS.Execute(lstrSQL, llngAffected, adCmdText)
        If Not lobjADORS Is Nothing Then
            If Not (lobjADORS.EOF = True And lobjADORS.BOF = True) Then
                MsgBox "Customer name already exists.", vbApplicationModal + vbOKOnly + vbInformation, "Duplicate Customer Name"
                Exit Sub
            End If
        End If
    End If
    ' Copy the fields from the screen to an ADO Recordset
    
    Set lobjADORS = BuildRS()
    If Not ValidateData(lobjADORS) Then Exit Sub
    
    ' Process through the recordset and insert or update rows
    If Not lobjADORS Is Nothing Then
        If Not (lobjADORS.BOF And lobjADORS.EOF) Then
            lobjADORS.MoveFirst
            Do While lobjADORS.EOF = False
                If mintMode = MODE_ADD Then
                    lstrErrDescr = "Error building insert statement for " & cstrTableName
                    lstrSQL = Build_Insert(lobjADORS, cstrTableName, "CustNum")
                    lstrErrDescr = "Error inserting data into " & cstrTableName
                    Call CnnBTMS.Execute(lstrSQL, 0, adCmdText And adExecuteNoRecords)
                    
                    mlngCustNum = GetID(cstrTableName, "CustNum", "CustName", Trim(lobjADORS.Fields("CustName").Value))
                Else
                    lstrErrDescr = "Error building update statement for " & cstrTableName
                    lstrWhereClause = "CustNum=" & mlngCustNum
                    lstrSQL = Build_Update(lobjADORS, cstrTableName, lstrWhereClause)
                    lstrErrDescr = "Error updating " & cstrTableName
                    Call CnnBTMS.Execute(lstrSQL, 0, adCmdText And adExecuteNoRecords)
                End If
                lstrErrDescr = ""
                
                mstrCustName = Trim(lobjADORS.Fields("CustName").Value)
                lobjADORS.MoveNext
            Loop
            
            lobjADORS.Close
        End If
    End If
    
    Set lobjADORS = Nothing
    Call LoadData
    
    Exit Sub

Error_Handler:

    llngErrNo = IIf(llngErrNo = 0, Err.Number, llngErrNo)
    lstrErrDescr = IIf(Len(lstrErrDescr) = 0, Err.Description, Err.Description)
    lstrErrSource = IIf(Len(Trim(Err.Source)) = 0, "", Err.Source & vbCrLf) & cstrErrorSource
    Call Err.Raise(llngErrNo, lstrErrSource, lstrErrDescr)

End Sub

Private Sub txtMarkup_KeyPress(KeyAscii As Integer)

    On Error GoTo Error_Handler
    Const cstrErrorSource As String = cstrApplicationName & "." & cstrModuleName & ".txtmarkup_KeyPress()"
    Dim llngErrNo As Long
    Dim lstrErrDescr As String
    llngErrNo = 0
    lstrErrDescr = ""
    
'    If ValNumericText(txtMarkup, KeyAscii, Len(txtMarkup.Text)) = False Then
'        Beep
'        KeyAscii = 0
'    End If

    'Khurram - Nov 18, 2003
    
    'instead of making change in the global defined function : ' ValNumericText '
    'copy the lines of code from the function and allow -ve value (KeyAsscii = 45)
    
    If DecimU = 1 Then
        If Not IsNumeric(Chr(KeyAscii)) And KeyAscii <> 46 And KeyAscii <> 8 And KeyAscii <> 45 Then
            Beep
            KeyAscii = 0
            Exit Sub
        End If
    Else
        If Not IsNumeric(Chr(KeyAscii)) And KeyAscii <> 44 And KeyAscii <> 8 And KeyAscii <> 45 Then
            Beep
            KeyAscii = 0
            Exit Sub
        End If
    End If
    
    If KeyAscii = 44 Then
        If Not txtMarkup.SelLength = Len(txtMarkup.Text) Then
            If InStr(1, txtMarkup.Text, Chr(46)) <> 0 Or InStr(1, txtMarkup.Text, Chr(44)) <> 0 Then
                Beep
                KeyAscii = 0
                Exit Sub
            End If
        End If
    End If
    
    If KeyAscii = 45 Then
        If Not txtMarkup.SelLength = Len(txtMarkup.Text) Then
            If InStr(1, txtMarkup.Text, Chr(45)) <> 0 Then
                Beep
                KeyAscii = 0
                Exit Sub
            End If
        End If
    End If
    
    If KeyAscii = 44 Then
        If Not txtMarkup.SelLength = Len(txtMarkup.Text) Then
            If InStr(1, txtMarkup.Text, Chr(46)) <> 0 Or InStr(1, txtMarkup.Text, Chr(44)) <> 0 Then
                Beep
                KeyAscii = 0
                Exit Sub
            End If
        End If
    End If
    Exit Sub

Error_Handler:
    llngErrNo = IIf(llngErrNo = 0, Err.Number, llngErrNo)
    lstrErrDescr = IIf(Len(lstrErrDescr) = 0, Err.Description, Err.Description)
    Call DisplayError(llngErrNo, IIf(Len(Trim(Err.Source)) = 0, "", Err.Source & vbCrLf) & cstrErrorSource, lstrErrDescr)

End Sub

Private Sub LoadData()

    On Error GoTo Error_Handler
    Const cstrErrorSource As String = cstrApplicationName & "." & cstrModuleName & ".LoadData()"
    Dim llngErrNo As Long

    Dim lstrErrSource As String
    llngErrNo = 0
    lstrErrDescr = ""
    lstrErrSource = ""
    
    Dim llngAffected As Long
    Dim lstrSQLQuery As String
    Dim lobjQuoteRS As ADODB.Recordset
    
    mintMode = MODE_VIEW
    Call ClearScreen
    
    lstrSQLQuery = " select * from  CustomerInformation where CustNum=" & mlngCustNum
    Set lobjQuoteRS = CnnBTMS.Execute(lstrSQLQuery, llngAffected, adCmdText)
    If Not lobjQuoteRS Is Nothing Then
        lobjQuoteRS.ActiveConnection = Nothing
        If Not (lobjQuoteRS.BOF And lobjQuoteRS.EOF) Then
            lobjQuoteRS.MoveFirst
            mlngCustNum = lobjQuoteRS.Fields("CustNum").Value
            
            lblCustNum.Caption = IIf(IsNull(lobjQuoteRS.Fields("CustName").Value), "", lobjQuoteRS.Fields("CustNum").Value)
    
            txtCName.Text = IIf(IsNull(lobjQuoteRS.Fields("CustName").Value), "", Trim(lobjQuoteRS.Fields("CustName").Value))
            txtCName.MaxLength = lobjQuoteRS.Fields("CustName").DefinedSize
    
            txtMarkup.Text = Format(lobjQuoteRS.Fields("CustMarkup").Value, cstrDiscountMask)
            txtMarkup.MaxLength = 6
    
            txtStAddress1.Text = IIf(IsNull(lobjQuoteRS.Fields("CustStrAddr1").Value), "", Trim(lobjQuoteRS.Fields("CustStrAddr1").Value))
            txtStAddress1.MaxLength = lobjQuoteRS.Fields("CustStrAddr1").DefinedSize
    
            txtStAddress2.Text = IIf(IsNull(lobjQuoteRS.Fields("CustStrAddr2").Value), "", Trim(lobjQuoteRS.Fields("CustStrAddr2").Value))
            txtStAddress2.MaxLength = lobjQuoteRS.Fields("CustStrAddr2").DefinedSize
    
            txtStAddress3.Text = IIf(IsNull(lobjQuoteRS.Fields("CustStrAddr3").Value), "", Trim(lobjQuoteRS.Fields("CustStrAddr3").Value))
            txtStAddress3.MaxLength = lobjQuoteRS.Fields("CustStrAddr3").DefinedSize
    
            txtCity.Text = IIf(IsNull(lobjQuoteRS.Fields("CustCity").Value), "", Trim(lobjQuoteRS.Fields("CustCity").Value))
            txtCity.MaxLength = lobjQuoteRS.Fields("CustCity").DefinedSize

            txtProvince.MaxLength = lobjQuoteRS.Fields("CustProv").DefinedSize
    
            txtCountry.Text = IIf(IsNull(lobjQuoteRS.Fields("CustCountry").Value), "", Trim(lobjQuoteRS.Fields("CustCountry").Value))
            txtCountry.MaxLength = lobjQuoteRS.Fields("CustCountry").DefinedSize

            txtPCode.MaxLength = lobjQuoteRS.Fields("CustPostal").DefinedSize
    
            txtContact.Text = IIf(IsNull(lobjQuoteRS.Fields("CustContact").Value), "", Trim(lobjQuoteRS.Fields("CustContact").Value))
            txtContact.MaxLength = lobjQuoteRS.Fields("CustContact").DefinedSize
    
            txtExtn.Text = IIf(IsNull(lobjQuoteRS.Fields("CustExt").Value), "", Trim(lobjQuoteRS.Fields("CustExt").Value))
            txtExtn.MaxLength = lobjQuoteRS.Fields("CustExt").DefinedSize
    
            txtPhone.Text = IIf(IsNull(lobjQuoteRS.Fields("CustPhone").Value), "", Trim(lobjQuoteRS.Fields("CustPhone").Value))
            txtPhone.MaxLength = lobjQuoteRS.Fields("CustPhone").DefinedSize
    
            txtFax.Text = IIf(IsNull(lobjQuoteRS.Fields("CustFax").Value), "", Trim(lobjQuoteRS.Fields("CustFax").Value))
            txtFax.MaxLength = lobjQuoteRS.Fields("CustFax").DefinedSize

Private Function BuildRS() As ADODB.Recordset

    On Error GoTo Error_Handler
    Const cstrErrorSource As String = cstrApplicationName & "." & cstrModuleName & ".BuildRS()"
    Dim llngErrNo As Long
    Dim lstrErrDescr As String
    Dim lstrErrSource As String
    llngErrNo = 0
    lstrErrDescr = ""
    lstrErrSource = ""

    Dim lstrSQL As String
    Dim lstrSchema As String
    Dim llngAffected As Long
    Dim lobjADORS As ADODB.Recordset
    
    ' Get empty recordset
    lstrErrDescr = "Error getting empty ADO recordset."
    lstrSQL = "Select * from " & cstrTableName & " where 1=0"
    Set lobjADORS = New ADODB.Recordset
    lobjADORS.CursorType = adOpenForwardOnly
    lobjADORS.CursorLocation = adUseClient
    lobjADORS.Open lstrSQL, CnnBTMS, adOpenForwardOnly, adLockOptimistic
    lobjADORS.AddNew
    Set lobjADORS.ActiveConnection = Nothing
    lstrErrDescr = ""
    
    lobjADORS.Fields("CustName").Value = Trim(txtCName.Text)
    lobjADORS.Fields("CustMarkup").Value = DBLValue(txtMarkup.Text)
    lobjADORS.Fields("CustStrAddr1").Value = IIf(txtStAddress1.Text = "", "", txtStAddress1.Text)
    lobjADORS.Fields("CustStrAddr2").Value = IIf(txtStAddress2.Text = "", "", txtStAddress2.Text)
    lobjADORS.Fields("CustStrAddr3").Value = IIf(txtStAddress3.Text = "", "", txtStAddress3.Text)
    lobjADORS.Fields("CustCity").Value = IIf(txtCity.Text = "", "", txtCity.Text)
    lobjADORS.Fields("CustProv").Value = IIf(txtProvince.Text = "", "", txtProvince.Text)
    lobjADORS.Fields("CustCountry").Value = IIf(txtCountry.Text = "", "", txtCountry.Text)
    lobjADORS.Fields("CustPostal").Value = IIf(txtPCode.Text = "", "", txtPCode.Text)
    lobjADORS.Fields("CustContact").Value = IIf(txtContact.Text = "", "", txtContact.Text)
    lobjADORS.Fields("CustPhone").Value = IIf(txtPhone.Text = "", "", txtPhone.Text)
    lobjADORS.Fields("CustExt").Value = Trim(txtExtn.Text)
    lobjADORS.Fields("CustFax").Value = IIf(txtFax.Text = "", "", txtFax.Text)
    lobjADORS.Fields("CustEmail").Value = IIf(txtEmail.Text = "", "", txtEmail.Text)
    lobjADORS.Fields("CustExtraInfo").Value = IIf(txtNotes.Text = "", "", txtNotes.Text)
    lobjADORS.Fields("CustYTD").Value = DBLValue(txtYTD.Text)
    lobjADORS.Fields("CustActive").Value = IIf(chkActive.Value = 1, True, False)
    
    Set BuildRS = lobjADORS
    Set lobjADORS = Nothing
    Exit Function

Error_Handler:

End Function
