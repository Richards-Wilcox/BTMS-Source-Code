Private Sub cmdReserve_Click()
    On Error GoTo Error_Handler
    Const cstrErrorSource As String = cstrApplicationName & "." & cstrModuleName & ".cmdReserve_Click()"
    Dim llngErrNo As Long
    Dim lstrErrDescr As String
    Dim lstrErrSource As String
    llngErrNo = 0
    lstrErrDescr = ""
    lstrErrSource = ""

    If cmdReserve.Caption = "&Reserve" Then
        If CountDetails < 1 Then
            MsgBox "The Quote has no details.", vbExclamation + vbApplicationModal, "Cannot Reserve"
            Exit Sub
        End If
        Call CheckQuoteforAllocation(mlngQuoteID, mstrQuoteUM, False) 'will populate the frmBOMAlloc grid
        If frmBOMAlloc.grdData1.TextMatrix(1, 0) <> "" Then
            frmBOMAlloc.mblnNow = False
            frmBOMAlloc.mlngDOCID = mlngQuoteID
            frmBOMAlloc.mstrDocUM = mstrQuoteUM
            frmBOMAlloc.mintBackTag = 2
            frmBOMAlloc.mdblYTD = DBLValue(txtPrice.Text)
            mblnInFocus = False
            frmBOMAlloc.Show vbModal
            mblnInFocus = True
        Else
            frmReserveDate.mlngDOCID = mlngQuoteID
            frmReserveDate.mintBackTag = 2
            frmReserveDate.mstrDocUM = mstrQuoteUM
            frmReserveDate.mdblYTD = DBLValue(txtPrice.Text)
            
            frmReserveDate.Show vbModal
        End If
    Else
        If CountDetails < 1 Then
            MsgBox "The Quote has no details.", vbExclamation + vbApplicationModal, "Cannot Un-Reserve"
            Exit Sub
        End If
        Call UnAllocateQuote(mlngQuoteID)
        If lblReserve.Caption = "" Then
            Call Un_SetCustYTDQ(mlngQuoteID, DBLValue(txtPrice.Text))
        End If
        MsgBox "Quote un-reserved!", vbApplicationModal + vbOKOnly + vbInformation, "Quote Un-Reserved."
    
    End If
    Call LoadData
    Exit Sub

Error_Handler:
    llngErrNo = IIf(llngErrNo = 0, Err.Number, llngErrNo)
    lstrErrDescr = IIf(Len(lstrErrDescr) = 0, Err.Description, lstrErrDescr & vbCrLf & Err.Description)
    Call DisplayError(llngErrNo, IIf(Len(Trim(Err.Source)) = 0, "", Err.Source & vbCrLf) & cstrErrorSource, lstrErrDescr)

End Sub

Private Sub cmdAllocate_Click()
    
    On Error GoTo Error_Handler
    Const cstrErrorSource As String = cstrApplicationName & "." & cstrModuleName & ".cmdAllocate_Click()"
    Dim llngErrNo As Long
    Dim lstrErrDescr As String
    Dim lstrErrSource As String
    llngErrNo = 0
    lstrErrDescr = ""
    lstrErrSource = ""
    
    If mlngQuoteID > 0 Then

        If cmdAllocate.Caption = "A&llocate" Then
            If CountDetails < 1 Then
                MsgBox "The Quote has no details.", vbExclamation + vbApplicationModal, "Cannot Allocate"
                Exit Sub
            End If
    
            If CheckQuoteforAllocation(mlngQuoteID, mstrQuoteUM, True) = True Then
                MsgBox "Cannot allocate this Quote.", vbExclamation + vbApplicationModal, "Cannot Allocate Quote"
                Exit Sub
            End If
            
            If frmBOMAlloc.grdData1.TextMatrix(1, 0) <> "" Then
                frmBOMAlloc.mlngDOCID = mlngQuoteID
                frmBOMAlloc.mblnNow = True
                frmBOMAlloc.mintBackTag = 2
                frmBOMAlloc.mdblYTD = DBLValue(txtPrice.Text)
                mblnInFocus = False
                frmBOMAlloc.Show vbModal
                mblnInFocus = True
            Else
                If lblReserve.Caption <> "" Then Call UnAllocateQuote(mlngQuoteID) 'unreserve first
                Call AllocateQuote(mlngQuoteID)
                Call SetCustYTDQ(mlngQuoteID, DBLValue(txtPrice.Text))
                MsgBox "Quote has been allocated.", vbApplicationModal + vbOKOnly + vbInformation, "Quote Allocated"
                Call LoadData
            End If

        Else
            If CountDetails < 1 Then
                MsgBox "The Quote has no details.", vbExclamation + vbApplicationModal, "Cannot Un-Allocate"
                Exit Sub
            End If
            Call UnAllocateQuote(mlngQuoteID)
            Call Un_SetCustYTDQ(mlngQuoteID, DBLValue(txtPrice.Text))
            MsgBox "Quote un-allocated!", vbApplicationModal + vbOKOnly + vbInformation, "Quote Un-Allocated."
            Call LoadData
        End If
    End If
    Exit Sub

Error_Handler:
    llngErrNo = IIf(llngErrNo = 0, Err.Number, llngErrNo)
    lstrErrDescr = IIf(Len(lstrErrDescr) = 0, Err.Description, lstrErrDescr & vbCrLf & Err.Description)
    Call DisplayError(llngErrNo, IIf(Len(Trim(Err.Source)) = 0, "", Err.Source & vbCrLf) & cstrErrorSource, lstrErrDescr)
End Sub

Private Sub LoadName()

    Dim llngErrNo As Long
    Dim lstrErrDescr As String
    Dim lstrErrSource As String
    llngErrNo = 0
    lstrErrDescr = ""
    lstrErrSource = ""

    Dim llngAffected As Long
    Dim llngIndex As Long
    Dim lstrSQLQuery As String
    Dim lobjCustRS As ADODB.Recordset
    
    llngIndex = 0
    Call cmbName.AddItem("")
    cmbName.ListIndex = llngIndex
    ReDim Preserve arrCustNum(0)
    arrCustNum(0) = 0
    mlngCurrCust = 0
    
    lstrSQLQuery = "select custnum,custname from customerinformation where CustActive=True"
    Set lobjCustRS = CnnBTMS.Execute(lstrSQLQuery, llngAffected, adCmdText)
    If Not lobjCustRS Is Nothing Then
        lobjCustRS.ActiveConnection = Nothing
        If Not (lobjCustRS.BOF And lobjCustRS.EOF) Then
            ReDim Preserve arrCustNum(lobjCustRS.RecordCount + 1)
            lobjCustRS.MoveFirst
            Dim lstrName As String
            Do While lobjCustRS.EOF = False
                lstrName = IIf(IsNull(lobjCustRS.Fields("custname").Value), "", Trim(lobjCustRS.Fields("custname").Value))

                    llngIndex = llngIndex + 1
                    arrCustNum(llngIndex) = lobjCustRS.Fields("custnum").Value
                End If
                lobjCustRS.MoveNext
            Loop
            lobjCustRS.Close
        End If
    End If
    Set lobjCustRS = Nothing
    Exit Sub

Private Sub SaveData()
    On Error GoTo Error_Handler
    Const cstrErrorSource As String = cstrApplicationName & "." & cstrModuleName & ".SaveData()"
    Dim llngErrNo As Long
    Dim lstrErrDescr As String
    Dim lstrErrSource As String
    llngErrNo = 0
    lstrErrDescr = ""
    lstrErrSource = ""

    Dim lstrSQL As String
    Dim lstrWhereClause As String
    Dim lstrExcludeFieldList As String
    Dim llngAffected As Long
    Dim lobjADORS As ADODB.Recordset
    
    ' Copy the fields from the screen to an ADO Recordset
    Set lobjADORS = BuildRS()
    
    If Not ValidateData(lobjADORS) Then Exit Sub
    
    ' Process through the recordset and insert or update rows
    If Not lobjADORS Is Nothing Then
        If Not (lobjADORS.BOF And lobjADORS.EOF) Then
            lobjADORS.MoveFirst
            Do While lobjADORS.EOF = False
                If mintMode = MODE_ADD Then
                    lstrErrDescr = "Error building insert statement for " & cstrTableName
                    lstrSQL = Build_Insert(lobjADORS, cstrTableName, "QuoteID")
                    lstrErrDescr = "Error inserting data into " & cstrTableName
                    Call CnnBTMS.Execute(lstrSQL, 0, adCmdText And adExecuteNoRecords)
                    
                    mlngQuoteID = GetID(cstrTableName, "QuoteID", "QuoteNum", Trim(lobjADORS.Fields("QuoteNum").Value))
                Else
                    lstrErrDescr = "Error building update statement for " & cstrTableName
                    lstrWhereClause = "QuoteNum=" & SQLString(mstrDocNum)
                    lstrSQL = Build_Update(lobjADORS, cstrTableName, lstrWhereClause, "QuoteID, QuoteNum")
                    lstrErrDescr = "Error updating " & cstrTableName
                    Call CnnBTMS.Execute(lstrSQL, 0, adCmdText And adExecuteNoRecords)
                End If
                lstrErrDescr = ""
                
                mstrDocNum = Trim(lobjADORS.Fields("QuoteNum").Value)
                lobjADORS.MoveNext
            Loop
            
            lobjADORS.Close
        End If
    End If
    
    Set lobjADORS = Nothing
    Call LoadData
    Exit Sub

Error_Handler:
    llngErrNo = IIf(llngErrNo = 0, Err.Number, llngErrNo)
    lstrErrDescr = IIf(Len(lstrErrDescr) = 0, Err.Description, Err.Description)
    lstrErrSource = IIf(Len(Trim(Err.Source)) = 0, "", Err.Source & vbCrLf) & cstrErrorSource
    Call Err.Raise(llngErrNo, lstrErrSource, lstrErrDescr)
End Sub

Private Sub CalcAmounts()
    
    On Error GoTo Error_Handler
    Const cstrErrorSource As String = cstrApplicationName & "." & cstrModuleName & ".CalcAmounts()"
    Dim llngErrNo As Long
    Dim lstrErrDescr As String
    Dim lstrErrSource As String
    llngErrNo = 0
    lstrErrDescr = ""
    lstrErrSource = ""
    Dim ldblTemp As Double
    
    'format
    txtNetPrice.Text = Format(DBLValue(txtNetPrice.Text), cstrDollarMask)
    txtMarkup.Text = Format(DBLValue(txtMarkup.Text), cstrDollarMask)
    txtValue1.Text = Format(DBLValue(txtValue1.Text), cstrDollarMask)
    txtValue2.Text = Format(DBLValue(txtValue2.Text), cstrDollarMask)
    txtValue3.Text = Format(DBLValue(txtValue3.Text), cstrDollarMask)
    txtValue4.Text = Format(DBLValue(txtValue4.Text), cstrDollarMask)
    txtLabour.Text = Format(DBLValue(txtLabour.Text), cstrDollarMask)
    txtOHCost.Text = Format(DBLValue(txtOHCost.Text), cstrDollarMask)

    txtSubTotMarkup.Text = Format(DBLValue(txtSubTotMarkup.Text), cstrDollarMask)
    txtShipping.Text = Format(DBLValue(txtShipping.Text), cstrDollarMask)
    txtMisc.Text = Format(DBLValue(txtMisc.Text), cstrDollarMask)
    txtPrice.Text = Format(DBLValue(txtPrice.Text), cstrDollarMask)
    txtNetConverted.Text = Format(DBLValue(txtNetConverted.Text), cstrDollarMask)
    txtSubTotal.Text = Format(DBLValue(txtSubTotal.Text), cstrDollarMask)
    
    txtPctValue(0).Text = Format(DBLValue(txtPctValue(0).Text), cstrPctMask)
    txtPctValue(1).Text = Format(DBLValue(txtPctValue(1).Text), cstrPctMask)
    txtPctValue(2).Text = Format(DBLValue(txtPctValue(2).Text), cstrPctMask)
    txtPctValue(3).Text = Format(DBLValue(txtPctValue(3).Text), cstrPctMask)
    txtPctNetConverted.Text = Format(DBLValue(txtPctNetConverted.Text), cstrPctMask)
    
    'calculate
    txtNetConverted.Text = Format(IIf(DBLValue(txtPctNetConverted.Text) = 0, DBLValue(txtNetPrice.Text) + DBLValue(txtLabour.Text) + DBLValue(txtOHCost.Text), (DBLValue(txtNetPrice.Text) + DBLValue(txtLabour.Text) + DBLValue(txtOHCost.Text)) * DBLValue(txtPctNetConverted.Text)), cstrDollarMask)
    
    If Not DBLValue(txtNetConverted.Text) = 0 Then
        txtValue1.Text = Format(DBLValue(txtNetConverted.Text) * DBLValue(txtPctValue(0).Text), cstrDollarMask)
        txtValue2.Text = Format(DBLValue(txtNetConverted.Text) * DBLValue(txtPctValue(1).Text), cstrDollarMask)
        txtValue3.Text = Format(DBLValue(txtNetConverted.Text) * DBLValue(txtPctValue(2).Text), cstrDollarMask)
        txtValue4.Text = Format(DBLValue(txtNetConverted.Text) * DBLValue(txtPctValue(3).Text), cstrDollarMask)
    End If
    
    txtSubTotal.Text = Format(DBLValue(txtNetConverted.Text) + DBLValue(txtValue1.Text) + DBLValue(txtValue2.Text) + DBLValue(txtValue3.Text) + DBLValue(txtValue4.Text) + DBLValue(txtMisc.Text), cstrDollarMask)
    
    txtMarkup.Text = Format(DBLValue(txtSubTotal.Text) * DBLValue(txtPctMarkup.Text), cstrDollarMask)
    
    txtSubTotMarkup.Text = Format(DBLValue(txtSubTotal.Text) + DBLValue(txtMarkup.Text), cstrDollarMask)
    
    ldblTemp = DBLValue(txtMarkup.Text) + DBLValue(txtShipping.Text) + DBLValue(txtSubTotal.Text)
    txtPrice.Text = Format(ldblTemp, cstrDollarMask)
    
    Exit Sub
    
Error_Handler:
    llngErrNo = IIf(llngErrNo = 0, Err.Number, llngErrNo)
    lstrErrDescr = IIf(Len(lstrErrDescr) = 0, Err.Description, Err.Description)
    lstrErrSource = IIf(Len(Trim(Err.Source)) = 0, "", Err.Source & vbCrLf) & cstrErrorSource
    Call Err.Raise(llngErrNo, lstrErrSource, lstrErrDescr)
End Sub

Private Sub txtPctMarkup_KeyPress(KeyAscii As Integer)
    On Error GoTo Error_Handler
    Const cstrErrorSource As String = cstrApplicationName & "." & cstrModuleName & ".txtPctMarkup_KeyPress()"
    Dim llngErrNo As Long
    Dim lstrErrDescr As String
    llngErrNo = 0
    lstrErrDescr = ""
    
'    If ValNumericText(txtPctMarkup, KeyAscii, Len(txtPctMarkup.Text)) = False Then
'        Beep
'        KeyAscii = 0
'    End If
    
    'Khurram - Nov 18, 2003
    
    'instead of making change in the global defined function : ' ValNumericText '
    'copy the lines of code from the function and allow -ve value (KeyAsscii = 45)
    
    If DecimU = 1 Then
        If Not IsNumeric(Chr(KeyAscii)) And KeyAscii <> 46 And KeyAscii <> 8 And KeyAscii <> 45 Then
            Beep
            KeyAscii = 0
            Exit Sub
        End If
    Else
        If Not IsNumeric(Chr(KeyAscii)) And KeyAscii <> 44 And KeyAscii <> 8 And KeyAscii <> 45 Then
            Beep
            KeyAscii = 0
            Exit Sub
        End If
    End If
    
    If KeyAscii = 44 Then
        If Not txtPctMarkup.SelLength = Len(txtPctMarkup.Text) Then
            If InStr(1, txtPctMarkup.Text, Chr(46)) <> 0 Or InStr(1, txtPctMarkup.Text, Chr(44)) <> 0 Then
                Beep
                KeyAscii = 0
                Exit Sub
            End If
        End If
    End If
    
    If KeyAscii = 46 Then
        If Not txtPctMarkup.SelLength = Len(txtPctMarkup.Text) Then
            If InStr(1, txtPctMarkup.Text, Chr(44)) <> 0 Or InStr(1, txtPctMarkup.Text, Chr(46)) <> 0 Then
                Beep
                KeyAscii = 0
                Exit Sub
            End If
        End If
    End If
    
    If KeyAscii = 45 Then
        If Not txtPctMarkup.SelLength = Len(txtPctMarkup.Text) Then
            If InStr(1, txtPctMarkup.Text, Chr(45)) <> 0 Then
                Beep
                KeyAscii = 0
                Exit Sub
            End If
        End If
    End If

    Exit Sub

Error_Handler:
    llngErrNo = IIf(llngErrNo = 0, Err.Number, llngErrNo)

Private Sub ProcessPO(ByVal lngCurrPOID As Long, _
                      ByVal lngOldPOID As Long, _
                      ByVal lngQuoteID As Long)
    On Error GoTo Error_Handler
    Const cstrErrorSource As String = cstrApplicationName & "." & cstrModuleName & ".ProcessPO()"
    Dim llngErrNo As Long
    Dim lstrErrDescr As String
    Dim lstrErrSource As String
    llngErrNo = 0
    lstrErrDescr = ""
    lstrErrSource = ""

    If lngQuoteID < 1 Then Exit Sub
    If lngCurrPOID > 0 And lngCurrPOID = lngOldPOID Then Exit Sub
    
    Dim llngAffected As Long
    Dim lstrSQLQuery As String
    Dim lblnInTrans As Boolean
    
    lblnInTrans = False
    
            
    ' Create new PO
    CnnBTMS.BeginTrans
    lblnInTrans = True
    If lngCurrPOID = 0 And lngOldPOID = 0 Then
        ' Prompt For missing Information
        '   Distributer Number, Shipping Address, Requested Date
        mstrData = ""
        Dim frmX As frmBOMPO
        Set frmX = New frmBOMPO
        Set frmX.objParent = Me
        Screen.MousePointer = vbDefault
        mblnInFocus = False
        frmX.Show vbModal
        mblnInFocus = True
        Screen.MousePointer = vbHourglass
    
        If Not mstrData = "" Then
            Dim larrData As Variant
            larrData = Split(mstrData, "~")
            mobjPO.CancelFlag = IIf(larrData(1) = "True", True, False)
            
            If mobjPO.CancelFlag = True Then
                CnnBTMS.RollbackTrans
                Exit Sub
            End If
            mobjPO.PONum = Trim(larrData(0))

''''''''''Change in .NET''''''''''''''''
'            mobjPO.ReqDate = Trim(larrData(2))

''''''''''END Change in .NET
            mobjPO.DistNum = DBLValue(Trim(larrData(3)))
            mobjPO.ShipNum = DBLValue(Trim(larrData(4)))
            lngCurrPOID = CreatePO(lngQuoteID)
        End If
    End If
    If lngOldPOID > 0 Then
        Call UpdateOnPOsForPO(lngOldPOID, True) 'remove from Iventory before removing from PO.
        Call RemoveFromPO(lngOldPOID, lngQuoteID)
    End If
    If lngCurrPOID > 0 Then
        Call AddToPO(lngCurrPOID, lngQuoteID)
        Call UpdateOnPOsForPO(lngCurrPOID, False) 'you can only create new po, cannot add,
                                                    'therefore do not have to remove from ONPos first
    End If
    
    'Update the PO number on the Quote
    mlngPOID = lngCurrPOID
    lstrSQLQuery = "Update Quote"
    lstrSQLQuery = lstrSQLQuery & " Set POID=" & mlngPOID
    lstrSQLQuery = lstrSQLQuery & " , PONum=" & SQLString(mstrPONum)
    lstrSQLQuery = lstrSQLQuery & " , " & IIf(mlngPOID = 0, "QuoteStatus='Active'", "QuoteStatus='Issued'")
    lstrSQLQuery = lstrSQLQuery & " Where QuoteID=" & mlngQuoteID
    Call CnnBTMS.Execute(lstrSQLQuery, llngAffected, adCmdText And adExecuteNoRecords)
    
    'Update the PO number on the BoM
'    mlngPOID = lngCurrPOID
'    lstrSQLQuery = "Update BoM"
'    lstrSQLQuery = lstrSQLQuery & " Set POID=" & mlngPOID
'    lstrSQLQuery = lstrSQLQuery & " , PONum=" & SQLString(mstrPONum)
'    lstrSQLQuery = lstrSQLQuery & " Where QuoteID=" & mlngQuoteID
'    Call CnnBTMS.Execute(lstrSQLQuery, llngAffected, adCmdText And adExecuteNoRecords)
    
    CnnBTMS.CommitTrans
    lblnInTrans = False
    
    Call LoadData
    Exit Sub

Error_Handler:
    llngErrNo = IIf(llngErrNo = 0, Err.Number, llngErrNo)
    lstrErrDescr = IIf(Len(lstrErrDescr) = 0, Err.Description, lstrErrDescr & vbCrLf & Err.Description)
    lstrErrSource = IIf(Len(Trim(Err.Source)) = 0, "", Err.Source & vbCrLf) & cstrErrorSource
    If lblnInTrans = True Then CnnBTMS.RollbackTrans
    Call Err.Raise(llngErrNo, lstrErrSource, lstrErrDescr)
End Sub

Private Function CreatePO(ByVal lngQuoteID As Long) As Long
    On Error GoTo Error_Handler
    Const cstrErrorSource As String = cstrApplicationName & "." & cstrModuleName & ".CreatePO()"
    Dim llngErrNo As Long
    Dim lstrErrDescr As String
    Dim lstrErrSource As String
    llngErrNo = 0
    lstrErrDescr = ""
    lstrErrSource = ""

    Dim llngAffected As Long
    Dim lstrSQLQuery As String
    Dim lobjQuoteRS As ADODB.Recordset
    Dim lobjPORS As ADODB.Recordset
    Dim llngPOID As Long
    
    If Not mlngQuoteID > 0 Then Exit Function
    
    llngPOID = 0
    lstrSQLQuery = "select * from Quote where QuoteID=" & lngQuoteID
    Set lobjQuoteRS = CnnBTMS.Execute(lstrSQLQuery, llngAffected, adCmdText)
    If Not lobjQuoteRS Is Nothing Then
        lobjQuoteRS.ActiveConnection = Nothing
        If Not (lobjQuoteRS.BOF And lobjQuoteRS.EOF) Then
            lobjQuoteRS.MoveFirst
            Set lobjPORS = BuildPORS(lobjQuoteRS)
            If Not lobjPORS Is Nothing Then
                If Not (lobjPORS.BOF And lobjPORS.EOF) Then
                    lobjPORS.MoveFirst
                    lstrErrDescr = "Error building insert statement for PO"
                    lstrSQLQuery = Build_Insert(lobjPORS, "PO", "POID")
                    lstrErrDescr = "Error inserting data into PO"
                    
                    ' Insert the new PO
                    Call CnnBTMS.Execute(lstrSQLQuery, 0, adCmdText And adExecuteNoRecords)
                    lstrErrDescr = ""
                    lobjPORS.Close
                    
                    ' Find the POID
                    lstrSQLQuery = "select POID from PO where PONum=" & SQLString(mobjPO.PONum)
                    Set lobjPORS = CnnBTMS.Execute(lstrSQLQuery, llngAffected, adCmdText)
                    If Not lobjPORS Is Nothing Then
                        lobjPORS.ActiveConnection = Nothing
                        If Not (lobjPORS.BOF And lobjPORS.EOF) Then
                            lobjPORS.MoveFirst
                            llngPOID = IIf(IsNull(lobjPORS.Fields("POID").Value), 0, lobjPORS.Fields("POID").Value)
                        End If
                        lobjPORS.Close
                    End If
                End If
            End If
        End If
        lobjQuoteRS.Close
    End If
    
    Set lobjPORS = Nothing
    Set lobjQuoteRS = Nothing
    CreatePO = llngPOID
    Exit Function

Error_Handler:
    llngErrNo = IIf(llngErrNo = 0, Err.Number, llngErrNo)
    lstrErrDescr = IIf(Len(lstrErrDescr) = 0, Err.Description, lstrErrDescr & vbCrLf & Err.Description)
    lstrErrSource = IIf(Len(Trim(Err.Source)) = 0, "", Err.Source & vbCrLf) & cstrErrorSource
    Call Err.Raise(llngErrNo, lstrErrSource, lstrErrDescr)
End Function

                lobjADORS.Fields("PONum").Value = mobjPO.PONum

                If Not lobjParentRS Is Nothing Then
                    If Not (lobjParentRS.BOF And lobjParentRS.EOF) Then
                        lobjParentRS.MoveFirst
                        ' Reset the Mode so that we do an update of the existing part
                        lintMode = MODE_EDIT
                        ' Average the cost and sum the qty's

                            lobjADORS.Fields("POPtCallPrice").Value = True

                                lobjADORS.Fields("PODiscount").Value = 0 Then
                                    lobjADORS.Fields("PODiscount").Value = lobjParentRS.Fields("PODiscount").Value
                                End If
                            End If
                            lobjADORS.Fields("POPtCallPrice").Value = True
                            lobjADORS.Fields("POCallPriceExpDate").Value = Format(Now() + 30, "Short Date")
                        End If

''''''''''Change in .NET''''''''''''''''
                        'only QTY needed to be added if pn exists on PO. Price needed to be live.
'                        lobjADORS.Fields("POPtMtCost").Value = lobjParentRS.Fields("POPtMtCost").Value '(lobjADORS.Fields("POPtQtyOrd").Value * lobjADORS.Fields("POPtMtCost").Value + lobjParentRS.Fields("POPtQtyOrd").Value * lobjParentRS.Fields("POPtMtCost").Value) / (lobjParentRS.Fields("POPtQtyOrd").Value + lobjADORS.Fields("POPtQtyOrd").Value)

                                lsngLen = GetLengthFromMain(lobjADORS.Fields("POPtNum").Value)
                            Else
                                lstrSQL = "Select PartNum, Length from Main where PartNum=" & SQLString(lobjADORS.Fields("POPtNum").Value)

                                        lsngLen = DBLValue(lobjMainRS.Fields("Length").Value)
                                    End If
                                End If
                                Set lobjMainRS = Nothing
                            End If
                            
                            Dim TempLen As Single, lobjTempBOMRS As ADODB.Recordset, lobjTempBOMDetRS As ADODB.Recordset

                    lstrSQL = Build_Insert(lobjADORS, "PODetails", lstrExcludeFieldList)

            lobjADORS.Close

                            End If
                            
                            Dim lobjTempBOMRS As ADODB.Recordset, lobjTempBOMDetRS As ADODB.Recordset
                            'now find all documents belonging to that PO, except the one being removed
                            lstrSQL = "Select QuoteID as DOCID, " & "'Quote'" & " from Quote where POID= " & lngPOID & " and  QuoteID <> " & lngQuoteID
                            lstrSQL = lstrSQL & " union select BOMID as DOCID, " & "'BoM'" & " from BoM where POID= " & lngPOID
                            'lstrSQL = lstrSQL & " Union select QuoteID as DOCID, " & "'Quote'" & " from Quote where QuoteID <> " & lngQuoteID
                            Dim lArrBOM() As Long, lArrQuote() As Long
                            Set lobjQuoteRS = CnnBTMS.Execute(lstrSQL, llngAffected, adCmdText)
                            If Not lobjQuoteRS Is Nothing Then
                                lobjQuoteRS.ActiveConnection = Nothing
                                If Not (lobjQuoteRS.BOF And lobjQuoteRS.EOF) Then
                                    lobjQuoteRS.MoveFirst
                                    ReDim lArrBOM(0)
                                    ReDim lArrQuote(0)
                                    i = 0
                                    j = 0
                                    Do While lobjQuoteRS.EOF = False
                                        If lobjQuoteRS.Fields(1).Value = "Quote" Then
                                            ReDim Preserve lArrQuote(j)
                                            lArrQuote(j) = lobjQuoteRS.Fields("DocID").Value
                                        Else
                                            ReDim Preserve lArrBOM(i)
                                            lArrBOM(i) = lobjQuoteRS.Fields("DocID").Value
                                        End If
                                        Set lobjRS = CnnBTMS.Execute(lstrSQL, llngAffected, adCmdText)
                                        i = i + 1
                                        j = j + 1
                                        lobjQuoteRS.MoveNext
                                    Loop
                                    lobjADORS.Fields("POPtQtyOrd").Value = DBLValue(lobjParentRS.Fields("POPtQtyOrd").Value) - ModLaunch.GetNumberOfPartsRequired(lobjADORS.Fields("POPtNum").Value, lsngLen, lArrBOM(), lArrQuote()) ' RoundUp((TempLen) / lsngLen)
                                Else
                                    lobjADORS.Fields("POPtQtyOrd").Value = DBLValue(lobjParentRS.Fields("POPtQtyOrd").Value) - DBLValue(lobjADORS.Fields("POPtQtyOrd").Value)
                                End If
                                Set lobjRS = Nothing
                            End If
                            
                        Else
                            lobjADORS.Fields("POPtQtyOrd").Value = lobjParentRS.Fields("POPtQtyOrd").Value - lobjADORS.Fields("POPtQtyOrd").Value
                        End If
                        'if the user decreased the qty on the PO, then when removing the original document and that item was only on that document, total could be less than 0.
                        If DBLValue(lobjADORS.Fields("POPtQtyOrd").Value) < 0 Then lobjADORS.Fields("POPtQtyOrd").Value = 0
                    End If
                    lobjParentRS.Close
                End If
                Set lobjParentRS = Nothing

                If lobjADORS.Fields("POPtQtyOrd").Value > 0 Then
                    lstrErrDescr = "Error building update statement for PODetails"
                    lstrWhereClause = "POID=" & lngPOID
                    lstrWhereClause = lstrWhereClause & " and POPtNum=" & SQLString(lobjADORS.Fields("POPtNum").Value)
                    lstrSQL = Build_Update(lobjADORS, "PODetails", lstrWhereClause, lstrExcludeFieldList)
                    lstrErrDescr = "Error updating PODetails"
                Else
                    lstrErrDescr = "Error building delete statement for PODetails"
                    lstrSQL = "Delete * from PODetails where POID=" & lngPOID
                    lstrSQL = lstrSQL & " and POPtNum=" & SQLString(lobjADORS.Fields("POPtNum").Value)
                    lstrErrDescr = "Error deleting PODetails"
                End If
                
                Call CnnBTMS.Execute(lstrSQL, 0, adCmdText And adExecuteNoRecords)
                lstrErrDescr = ""
                lobjADORS.MoveNext
            Loop
            
            lobjADORS.Close
        End If
    End If
    
''''''''''Change in .NET''''''''''''''''
    mblnFlagRemoveTag = False
''''''''''END Change in .NET

    Set lobjADORS = Nothing
    
    ' Find the number of PODetails
    Dim llngCount As Long
    llngCount = -1
    
    lstrSQL = "select Count(*) from PODetails where POID=" & lngPOID
    Set lobjADORS = CnnBTMS.Execute(lstrSQL, llngAffected, adCmdText)
    If Not lobjADORS Is Nothing Then
        lobjADORS.ActiveConnection = Nothing
        If Not (lobjADORS.BOF And lobjADORS.EOF) Then
            lobjADORS.MoveFirst
            llngCount = IIf(IsNull(lobjADORS.Fields(0).Value), 0, lobjADORS.Fields(0).Value)
        End If
        lobjADORS.Close
    End If
    Set lobjADORS = Nothing
    
    ' Delete PO if no Children
    If llngCount <= 0 Then
        lstrSQL = "Delete * from PO where POID=" & lngPOID
        Call CnnBTMS.Execute(lstrSQL, 0, adCmdText And adExecuteNoRecords)
    End If
    Exit Sub

Error_Handler:
    llngErrNo = IIf(llngErrNo = 0, Err.Number, llngErrNo)
    lstrErrDescr = IIf(Len(lstrErrDescr) = 0, Err.Description, lstrErrDescr & vbCrLf & Err.Description)
    lstrErrSource = IIf(Len(Trim(Err.Source)) = 0, "", Err.Source & vbCrLf) & cstrErrorSource
    Call Err.Raise(llngErrNo, lstrErrSource, lstrErrDescr)
End Sub
