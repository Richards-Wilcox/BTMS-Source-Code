Private Sub cmdReset_Click()

    On Error GoTo Error_Handler
    Const cstrErrorSource As String = cstrApplicationName & "." & cstrModuleName & ".cmdReset_Click()"
    Dim llngErrNo As Long
    Dim lstrErrDescr As String
    llngErrNo = 0
    lstrErrDescr = ""
    Screen.MousePointer = vbHourglass
    Dim i As Integer, Temp21 As Integer, Temp24 As Integer
    
    'clear the sections
    '''''''''''''''''''''''''''''''''''''''''''''
    For i = 1 To 4
        lblInfo(i).Caption = ""
    Next
    fraSection(0).Visible = False
    For i = 1 To SecNum - 1
        Unload fraSection(i)
        If i < SecNum - 1 Then Unload txtLite(i)
    Next
    chkCutTop.Value = 0
    chkCutBtm.Value = 0
    txtCutTop.Text = ""
    txtCutBtm.Text = ""
    cutT = 0
    cutB = 0
    varDoorData.CutBtm = 0
    varDoorData.CutTop = 0
    fraDoor.Enabled = True
    Cut = 0
    SecH = 0
    Command5.Enabled = True
    'reload sections
    ''''''''''''''''''''''''''''''''''''''
    If IsUnrestricted = True And DrHgt > 264 Then
        Call SectionsNumCombinationOversized
    Else
        Call SectionsNumCombination   'get the # of 24 and 21 inch sections
    End If
    SecH = N21 * 21 + N24 * 24
    SecNum = N21 + N24
    Temp21 = N21
    Temp24 = N24
    
    txtLite(0).Move fraSection(0).Left + fraSection(0).Width + 20, fraSection(0).Top
    txtLite(0).Text = IIf(varDoorData.arrSections(0).intNLites > 0, varDoorData.arrSections(0).intNLites, "")
    fraSection(0).Visible = True
    For i = 1 To SecNum - 1 'load section frames
        Load fraSection(i)
        Load txtLite(i)
        fraSection(i).Height = fraSection(0).Height
        txtLite(i).Height = fraSection(i).Height
        fraSection(i).Width = fraSection(0).Width
        fraSection(i).Move fraSection(0).Left, fraSection(i - 1).Top + fraSection(i).Height * 1.1
        txtLite(i).Move fraSection(i).Left + fraSection(i).Width + 20, fraSection(i).Top
        fraSection(i).Visible = True
        txtLite(i).Text = IIf(varDoorData.arrSections(i).intNLites > 0, varDoorData.arrSections(i).intNLites, "")
        txtLite(i).Visible = True 'IIf(i = SecNum - 1, False, True)
        If i = SecNum - 1 Then Unload txtLite(i)
    Next
    
    For i = N21 + N24 - 1 To 0 Step -1 'colour sections
        If Temp24 > 0 Then
            fraSection(i).BackColor = &H0& 'black
            Temp24 = Temp24 - 1
        Else
            fraSection(i).BackColor = &HFFFFFF    'white
            Temp21 = Temp21 - 1
        End If
    Next
    
    fraDoor.Height = fraSection(0).Height * SecNum * 1.2
    fraDoor.Enabled = True
    
    If MeasU = 0 Then
        ClearConObj
        aCon.Inch = DrHgt
        aCon.mmcalc
        lblInfo(0).Caption = Round(aCon.mm, 0)
    Else
        lblInfo(0).Caption = DrHgt
    End If
    
    If MeasU = 0 Then
        ClearConObj
        aCon.Inch = SecH
        aCon.mmcalc
        lblInfo(1).Caption = Round(aCon.mm, 0)
    Else
        lblInfo(1).Caption = SecH
    End If
    
    Cut = (DrHgt) - SecH
    If Cut > 0 Then Cut = Round(Cut, 0) + RoundFraction(Cut - Int(Cut))
    If MeasU = 0 Then
        ClearConObj
        aCon.Inch = Abs(Cut)
        aCon.mmcalc
        lblInfo(2).Caption = Round(aCon.mm, 0)
    Else
        lblInfo(2).Caption = Format(Abs(Cut), "Fixed")
    End If
    lblInfo(3).Caption = N21
    lblInfo(4).Caption = N24
    
    cutT = GetCut(False)
    cutB = GetCut(True)
    
    'there is a difference between the drhgt and hight in sections, have to add that difference to the cut.
    'when calculating cut, only height in sections is taken into consideration.
    If RoundUp(DrHgt) > DrHgt Then
        If IIf(fraSection(0).BackColor = &HFFFFFF, 21, 24) - cutT + (RoundUp(DrHgt) - DrHgt) <= 15 Then
            cutT = cutT + (RoundUp(DrHgt) - DrHgt)
            If cutT < 1 Then
                If cutB > 0 Then cutB = cutB + cutT
            End If
            cutT = 0 'ignore cuts smaller than an inch.
        Else
            cutB = cutB + (RoundUp(DrHgt) - DrHgt) 'if top is cut to maximum, bottom will have to be cut.
            If cutB < 1 Then
                If cutT > 0 Then cutT = cutB + cutT
            End If
            cutB = 0
        End If
    End If

    If cutB > 0 Then
        chkCutBtm.Value = 1
        txtCutBtm.Enabled = True
        If MeasU = 1 Then
            txtCutBtm.Text = IIf(cutB > 0, Format(cutB, "Fixed"), "")
        Else
            aCon.Inch = cutB
            aCon.mmcalc
            txtCutBtm.Text = Round(aCon.mm, 0)
        End If
    Else
        txtCutBtm.Enabled = False
    End If
    
    If cutT > 0 Then
        chkCutTop.Value = 1
        txtCutTop.Enabled = True
        If MeasU = 1 Then
            txtCutTop.Text = IIf(cutT > 0, Format(cutT, "Fixed"), "")
        Else
            aCon.Inch = cutT
            aCon.mmcalc
            txtCutTop.Text = Round(aCon.mm, 0)
        End If
    Else
        txtCutTop.Enabled = False
    End If
    Screen.MousePointer = vbDefault
    Exit Sub

Error_Handler:

    Screen.MousePointer = vbDefault
    llngErrNo = IIf(llngErrNo = 0, Err.Number, llngErrNo)
    lstrErrDescr = IIf(Len(lstrErrDescr) = 0, Err.Description, Err.Description)
    Call DisplayError(llngErrNo, IIf(Len(Trim(Err.Source)) = 0, "", Err.Source & vbCrLf) & cstrErrorSource, lstrErrDescr)

End Sub

Private Sub cmdAccept_Click()

    On Error GoTo Error_Handler
    Const cstrErrorSource As String = cstrApplicationName & "." & cstrModuleName & ".cmdAccept_Click()"
    Dim llngErrNo As Long
    Dim lstrErrDescr As String
    llngErrNo = 0
    lstrErrDescr = ""

    'evaluate users input
    Dim S As String
    Dim Resp As String
    Dim i As Integer
    Dim EnteredTopCut As Single, EnteredBtmCut As Single, lstrC As String, strMsg As String, MetCut As Integer
    EnteredBtmCut = 0
    EnteredBtmCut = 0
    For i = 0 To fraSection.Count - 1
        If fraSection(i).BackColor = &H959595 Then
            MsgBox "Please fill all empty placeholders.", vbApplicationModal + vbExclamation, "Invalid Configuration."
            Exit Sub
        End If
    Next
    
    If (chkCutBtm.Value = 0 And chkCutTop.Value = 0) And Cut < -1 Then
        MsgBox "This configuration is invalid without cutting sections. Please select cut routine or reconfigure.", vbApplicationModal + vbExclamation, "Invalid Configuration."
        Exit Sub
    End If
    
    If chkCutTop.Value = 1 And DBLValue(txtCutTop.Text) = 0 Then
        MsgBox "Please enter a valid cut amount for the top section or uncheck this option.", vbApplicationModal + vbExclamation, "Invalid Configuration."
        Exit Sub
    End If
    
    If chkCutBtm.Value = 1 And DBLValue(txtCutBtm.Text) = 0 Then
        MsgBox "Please enter a valid cut amount for the bottom section or uncheck this option.", vbApplicationModal + vbExclamation, "Invalid Configuration."
        Exit Sub
    End If
    
    fraDoor.Enabled = True
    
    If SecH < DrHgt Then 'not enough height Cut > 0
        Resp = MsgBox("The hole has not been filled. Chose Yes if you wish to add a new section or No to rearange the existing configuration.", vbYesNo + vbApplicationModal + vbExclamation, "Invalid Configuration.")
        If Resp = vbYes Then AddSection
    
    ElseIf Cut > -1 Then 'optimal solution, no need to cut. ( acceptable within 1 inch )
        'MsgBox "This is the optimal solution."
        Screen.MousePointer = vbHourglass
        fraDoor.Enabled = False
        cutT = 0
        cutB = 0
        varDoorData.CutBtm = 0
        varDoorData.CutTop = 0
        Call CheckForUnusedSections
        Call PopulateSectionArray
        Call FillGrd
        Call SaveLites
        Call SetLiteInfo
        frmSectionsEntry.Show
        Unload Me
    
    Else 'need to cut
        
        EnteredTopCut = cutT 'DBLValue(txtCutTop.Text) 'in inches
        EnteredBtmCut = cutB 'DBLValue(txtCutBtm.Text) 'in inches
        
        If MeasU = 1 Then
            lstrC = "inches"
        Else
'            If DBLValue(txtCutTop.Text) > 0 Then
'                aCon.mm = DBLValue(txtCutTop.Text)
'                aCon.Inchcalc
'                EnteredTopCut = aCon.Inch
'            End If
'            If DBLValue(txtCutBtm.Text) > 0 Then
'                aCon.mm = DBLValue(txtCutBtm.Text)
'                aCon.Inchcalc
'                EnteredBtmCut = aCon.Inch
'            End If
            lstrC = "mm"
        End If
        
        Dim BH As Integer, TH As Integer 'black bottom
        BH = IIf(fraSection(fraSection.Count - 1).BackColor = &H0&, 24, 21)
        TH = IIf(fraSection(0).BackColor = &H0&, 24, 21)
            
        BH = BH - 13 '13 is min section height (bottom)
        TH = TH - 15 '15 is min section height (top)
        'first see if the difference (cut) is a valid cut amount
        'and find if individual top and bottom cut amounts are legal
               
        If (Abs(Cut) > BH + TH) Or (EnteredTopCut > TH) Or (EnteredBtmCut > BH) Then
            MsgBox "The cut amount is too great, please reconfigure sections.", vbApplicationModal + vbExclamation, "Invalid Configuration."
            Exit Sub
        End If
           
        Dim lblnCondition As Boolean
        If MeasU = 0 Then
            lblnCondition = RoundUp(EnteredTopCut + EnteredBtmCut) <> RoundUp(Abs(Cut))
        Else
            lblnCondition = Format(EnteredTopCut + EnteredBtmCut, "Fixed") <> Format(Abs(Cut), "Fixed")
        End If
        If lblnCondition Then 'if entered cut amounts are not equal to the difference.
            MsgBox "Please configure the cut amount(s) to be equal to " & lblInfo(2).Caption & Chr(32) & lstrC, vbApplicationModal + vbExclamation, "Invalid Configuration."
            Exit Sub
'        Else
'            strMsg = "Please cut "
'            If chkCutTop.Value = 1 Then
'                 strMsg = strMsg & txtCutTop.Text & Chr(32) & lstrC & " off the top section"
'            End If
'            If chkCutBtm.Value = 1 Then
'                If chkCutTop.Value = 1 Then
'                    strMsg = strMsg & " and "
'                End If
'                strMsg = strMsg & txtCutBtm.Text & Chr(32) & lstrC & " off the bottom section"
'            End If
            
        End If
        Screen.MousePointer = vbHourglass
        varDoorData.CutBtm = EnteredBtmCut
        varDoorData.CutTop = EnteredTopCut
        fraDoor.Enabled = False
        Call CheckForUnusedSections
        Call PopulateSectionArray
        Call FillGrd
        Call SetLiteInfo
        frmSectionsEntry.Show
        Unload Me
    End If
    Screen.MousePointer = vbDefault
    Exit Sub

Error_Handler:
    Screen.MousePointer = vbDefault
    llngErrNo = IIf(llngErrNo = 0, Err.Number, llngErrNo)
    lstrErrDescr = IIf(Len(lstrErrDescr) = 0, Err.Description, Err.Description)
    Call DisplayError(llngErrNo, IIf(Len(Trim(Err.Source)) = 0, "", Err.Source & vbCrLf) & cstrErrorSource, lstrErrDescr)

End Sub

    Dim llngErrNo As Long
    Dim lstrErrDescr As String
    Dim lstrErrSource As String
    llngErrNo = 0
    lstrErrDescr = ""
    lstrErrSource = ""
    Dim i As Integer
    Dim lintMinDrH As Integer
    Dim AddedSection As Integer
    
    For i = 0 To fraSection.Count - 1
        If fraSection(i).BackColor = &H959595 Then

Public Sub Add21Section()

    On Error GoTo Error_Handler
    Const cstrErrorSource As String = cstrApplicationName & "." & cstrModuleName & ".Add21Section()"
    Dim llngErrNo As Long
    Dim lstrErrDescr As String
    Dim lstrErrSource As String
    llngErrNo = 0
    lstrErrDescr = ""
    lstrErrSource = ""

    SecH = SecH + 21
    N21 = N21 + 1
    Cut = (DrHgt) - SecH
    If Cut > 0 Then Cut = Round(Cut, 0) + RoundFraction(Cut - Int(Cut))
    If MeasU = 0 Then
        ClearConObj
        aCon.Inch = DrHgt
        aCon.mmcalc
        lblInfo(0).Caption = Round(aCon.mm, 0)
    Else
        lblInfo(0).Caption = DrHgt
    End If
    
    If MeasU = 0 Then
        ClearConObj
        aCon.Inch = SecH
        aCon.mmcalc
        lblInfo(1).Caption = Round(aCon.mm, 0)
    Else
        lblInfo(1).Caption = SecH
    End If
    
'    Cut = RoundUp(DrHgt) - SecH
'    Cut = Round(Cut, 0) + RoundFraction(Cut - Int(Cut))
    If MeasU = 0 Then
        ClearConObj
        aCon.Inch = Abs(Cut)
        aCon.mmcalc
        lblInfo(2).Caption = Round(aCon.mm, 0)
    Else
        lblInfo(2).Caption = Abs(Format(Cut, "Fixed"))
    End If

    lblInfo(3).Caption = N21
    cmdAccept.Enabled = True

    Exit Sub

Error_Handler:
    llngErrNo = IIf(llngErrNo = 0, Err.Number, llngErrNo)
    lstrErrDescr = IIf(Len(lstrErrDescr) = 0, Err.Description, Err.Description)
    lstrErrSource = IIf(Len(Trim(Err.Source)) = 0, "", Err.Source & vbCrLf) & cstrErrorSource
    Call Err.Raise(llngErrNo, lstrErrSource, lstrErrDescr)

End Sub

Private Sub AddSection()

    On Error GoTo Error_Handler
    Const cstrErrorSource As String = cstrApplicationName & "." & cstrModuleName & ".AddSection()"
    Dim llngErrNo As Long
    Dim lstrErrDescr As String
    Dim lstrErrSource As String
    llngErrNo = 0
    lstrErrDescr = ""
    lstrErrSource = ""

    Dim Scount As Integer
    Scount = fraSection.Count
    Load fraSection(Scount) 'add one section
    Load txtLite(Scount - 1)
    If Cut > 21 Then
        fraSection(Scount).BackColor = &H0&
        Add24Section
    Else
        fraSection(Scount).BackColor = &HFFFFFF
        Add21Section
    End If
    
    fraSection(Scount).Height = fraSection(0).Height
    fraSection(Scount).Width = fraSection(0).Width
    fraSection(Scount).Move fraSection(0).Left, fraSection(Scount - 1).Top + fraSection(Scount).Height * 1.1
    fraSection(Scount).Visible = True
    txtLite(Scount - 1).Height = txtLite(0).Height
    txtLite(Scount - 1).Width = txtLite(0).Width
    txtLite(Scount - 1).Move fraSection(Scount).Left + fraSection(Scount).Width + 20, fraSection(Scount - 1).Top
    txtLite(Scount - 1).Visible = True
    SecNum = SecNum + 1
    fraDoor.Height = fraSection(0).Height * SecNum * 1.2

    Exit Sub

Error_Handler:
    llngErrNo = IIf(llngErrNo = 0, Err.Number, llngErrNo)
    lstrErrDescr = IIf(Len(lstrErrDescr) = 0, Err.Description, Err.Description)
    lstrErrSource = IIf(Len(Trim(Err.Source)) = 0, "", Err.Source & vbCrLf) & cstrErrorSource
    Call Err.Raise(llngErrNo, lstrErrSource, lstrErrDescr)

End Sub

    On Error GoTo Error_Handler
    Const cstrErrorSource As String = cstrApplicationName & "." & cstrModuleName & ".txtCutBtm_Validate()"
    Dim llngErrNo As Long
    Dim lstrErrDescr As String
    Dim lstrErrSource As String
    llngErrNo = 0
    lstrErrDescr = ""
    lstrErrSource = ""
    
    txtCutBtm.Text = Format(DBLValue(txtCutBtm.Text), "Fixed")
    
    If MeasU = 1 Then
        If DBLValue(txtCutBtm.Text) > 14 Then
            MsgBox "The maximum cut for any section is 14 inches", vbApplicationModal + vbExclamation, "Invalid Configuration."
            txtCutBtm.SelStart = 0
            txtCutBtm.SelLength = Len(txtCutBtm.Text)
            Cancel = True
        End If
    Else
        aCon.mm = DBLValue(txtCutBtm.Text)
        aCon.Inchcalc
        If aCon.Inch > 14 Then
            MsgBox "The maximum cut for any section is 279 mm", vbApplicationModal + vbExclamation, "Invalid Configuration."
            txtCutBtm.SelStart = 0
            txtCutBtm.SelLength = Len(txtCutBtm.Text)
            Cancel = True
        End If
    End If
    
    lblRemainder.Caption = Format(DBLValue(lblInfo(2)) - (DBLValue(txtCutTop.Text) + DBLValue(txtCutBtm.Text)), "Fixed")
    
    If DrType = "T45" Then
        If chkCutBtm.Value = 1 And DBLValue(txtCutBtm.Text) <= 1 Then
            If MeasU = 1 Then
                MsgBox "Minimum cut amount from the bottom section must be higher than 1 inch.", vbApplicationModal + vbExclamation, "Invalid Configuration."
                txtCutBtm.SelStart = 0
                txtCutBtm.SelLength = Len(txtCutBtm.Text)
                Cancel = True
                Exit Sub
            Else
                MsgBox "Minimum cut amount from the bottom section must be higher than 25 mm. ", vbApplicationModal + vbExclamation, "Invalid Configuration."
                txtCutBtm.SelStart = 0
                txtCutBtm.SelLength = Len(txtCutBtm.Text)
                Cancel = True
                Exit Sub
            End If
        End If
    End If
    Exit Sub

Error_Handler:
    llngErrNo = IIf(llngErrNo = 0, Err.Number, llngErrNo)
    lstrErrDescr = IIf(Len(lstrErrDescr) = 0, Err.Description, lstrErrDescr & vbCrLf & Err.Description)

Private Sub txtCutTop_Validate(Cancel As Boolean)

    On Error GoTo Error_Handler
    Const cstrErrorSource As String = cstrApplicationName & "." & cstrModuleName & ".txtCutTop_Validate()"
    Dim llngErrNo As Long
    Dim lstrErrDescr As String
    Dim lstrErrSource As String
    llngErrNo = 0
    lstrErrDescr = ""
    lstrErrSource = ""
    
    txtCutTop.Text = Format(DBLValue(txtCutTop.Text), "Fixed")
    
    If MeasU = 1 Then
        If DBLValue(txtCutTop.Text) > 14 Then
            MsgBox "The maximum cut for any section is 14 inches", vbApplicationModal + vbExclamation, "Invalid Configuration."
            txtCutTop.SelStart = 0
            txtCutTop.SelLength = Len(txtCutTop.Text)
            Cancel = True
            Exit Sub
        End If
    Else
        aCon.mm = DBLValue(txtCutTop.Text)
        aCon.Inchcalc
        If aCon.Inch > 14 Then
            MsgBox "The maximum cut for any section is 279 mm", vbApplicationModal + vbExclamation, "Invalid Configuration."
            txtCutTop.SelStart = 0
            txtCutTop.SelLength = Len(txtCutTop.Text)
            Cancel = True
            Exit Sub
        End If
    End If
    
    lblRemainder.Caption = Format(DBLValue(lblInfo(2)) - (DBLValue(txtCutTop.Text) + DBLValue(txtCutBtm.Text)), "Fixed")
    
    If DrType = "T45" Then
        If chkCutTop.Value = 1 And DBLValue(txtCutTop.Text) <= 3 Then
            If MeasU = 1 Then
                MsgBox "Minimum cut amount from the top section must be higher than 3 inches.", vbApplicationModal + vbExclamation, "Invalid Configuration."
                txtCutTop.SelStart = 0
                txtCutTop.SelLength = Len(txtCutTop.Text)
                Cancel = True
                Exit Sub
            Else
                MsgBox "Minimum cut amount from the top section must be higher than 76 mm. ", vbApplicationModal + vbExclamation, "Invalid Configuration."
                txtCutTop.SelStart = 0
                txtCutTop.SelLength = Len(txtCutTop.Text)
                Cancel = True
                Exit Sub
            End If
        End If
    End If
    
    Exit Sub

Error_Handler:
    llngErrNo = IIf(llngErrNo = 0, Err.Number, llngErrNo)
    lstrErrDescr = IIf(Len(lstrErrDescr) = 0, Err.Description, lstrErrDescr & vbCrLf & Err.Description)
    Call DisplayError(llngErrNo, IIf(Len(Trim(Err.Source)) = 0, "", Err.Source & vbCrLf) & cstrErrorSource, lstrErrDescr)

End Sub

    If isEmptySectionsArray = False Then

            frmSectionsEntry.txt24LiteSections.Text = lint24LiteSections

            For i = 0 To UBound(varDoorData.arrSections()) - 1
                varDoorData.arrSections(i).intNLites = IIf(Me.txtLite(i).Text = "", 0, DBLValue(txtLite(i).Text))
            Next
            'If i < UBound(varDoorData.arrSections()) Then

''''''''''Change in .NET''''''''''''''''
'            frmSectionsEntry.txtLitesPerSection.Enabled = True
''''''''''END Change in .NET
            
            frmSectionsEntry.cmdShowAllLites.Enabled = False
            For i = 0 To UBound(varDoorData.arrSections) - 1
            
                lintLitesPerSection = IIf(varDoorData.arrSections(i).intNLites = 0, lintLitesPerSection, varDoorData.arrSections(i).intNLites)
                If i > 0 Then
                    If varDoorData.arrSections(i - 1).intNLites > 0 Then

                        End If
                    End If
                End If
                frmSectionsEntry.txtLitesPerSection.Text = lintLitesPerSection
            Next i
            'If lint21LiteSections = lint24LiteSections > 0 Then frmSectionsEntry.cboLiteType.ListIndex = 1
