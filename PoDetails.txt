Private Sub LoadData()
    On Error GoTo Error_Handler
    Const cstrErrorSource As String = cstrApplicationName & "." & cstrModuleName & ".LoadData()"
    Dim llngErrNo As Long
    Dim lstrErrDescr As String
    Dim lstrErrSource As String
    llngErrNo = 0
    lstrErrDescr = ""
    lstrErrSource = ""

    Dim llngAffected As Long
    Dim lstrSQLQuery As String
    Dim lobjPORS As ADODB.Recordset
    
    lstrSQLQuery = "select PONum, POStatus, POReqDate, POMeasSys"
    lstrSQLQuery = lstrSQLQuery & " from PO where POID=" & mlngPOID
    Set lobjPORS = CnnBTMS.Execute(lstrSQLQuery, llngAffected, adCmdText)
    If Not lobjPORS Is Nothing Then
        lobjPORS.ActiveConnection = Nothing
        If Not (lobjPORS.BOF And lobjPORS.EOF) Then
            lobjPORS.MoveFirst
            Dim lstrFieldName As String
            lstrFieldName = "PONum"
            lblPONum.Caption = IIf(IsNull(lobjPORS.Fields(lstrFieldName).Value), mstrPONum, Trim(lobjPORS.Fields(lstrFieldName).Value))
            lstrFieldName = "POMeasSys"
            mstrPOUM = IIf(lobjPORS.Fields(lstrFieldName).Value = 1, "Imperial", "Metric")
            lstrFieldName = "POStatus"
            lblStatus.Caption = IIf(IsNull(lobjPORS.Fields(lstrFieldName).Value), "", Trim(lobjPORS.Fields(lstrFieldName).Value))
            
            If lblStatus.Caption = "Issued" Or lblStatus.Caption = "Complete" Then
                cmdDelete.Enabled = False
                cmdAdd.Enabled = False
            End If
            
            lstrFieldName = "POReqDate"
            lblTotal.Caption = Format(Trim(lobjPORS.Fields(lstrFieldName).Value), cstrDateMask)
            
            lobjPORS.Close
            
            Call LoadGrid
        End If
    End If
    
    lstrSQLQuery = "select sum(POPtQtyOrd * POPtMtCost) from PODetails where POID=" & mlngPOID
    Set lobjPORS = CnnBTMS.Execute(lstrSQLQuery, llngAffected, adCmdText)
    If Not lobjPORS Is Nothing Then
        lobjPORS.ActiveConnection = Nothing
        If Not (lobjPORS.BOF And lobjPORS.EOF) Then
            If Not IsNull(lobjPORS.Fields(0).Value) Then
                lblPrice.Caption = Format(DBLValue(lobjPORS.Fields(0).Value), "Fixed")
            Else
                lblPrice.Caption = Format(0, "Fixed")
            End If
        End If
    End If
    
    lstrSQLQuery = "select sum(POPtQtyOrd * POPtWgt) from PODetails where POID=" & mlngPOID
    Set lobjPORS = CnnBTMS.Execute(lstrSQLQuery, llngAffected, adCmdText)
    If Not lobjPORS Is Nothing Then
        lobjPORS.ActiveConnection = Nothing
        If Not (lobjPORS.BOF And lobjPORS.EOF) Then
            If Not IsNull(lobjPORS.Fields(0).Value) Then
                lblWgt.Caption = Format(DBLValue(lobjPORS.Fields(0).Value), "Fixed")
            Else
                lblWgt.Caption = Format(0, "Fixed")
            End If
        End If
    End If
    Set lobjPORS = Nothing
    Exit Sub

Error_Handler:
    llngErrNo = IIf(llngErrNo = 0, Err.Number, llngErrNo)
    lstrErrDescr = IIf(Len(lstrErrDescr) = 0, Err.Description, Err.Description)
    lstrErrSource = IIf(Len(Trim(Err.Source)) = 0, "", Err.Source & vbCrLf) & cstrErrorSource
    Call Err.Raise(llngErrNo, lstrErrSource, lstrErrDescr)
    
End Sub

Private Sub LoadGrid()
    On Error GoTo Error_Handler
    Const cstrErrorSource As String = cstrApplicationName & "." & cstrModuleName & ".LoadGrid()"
    Dim llngErrNo As Long
    Dim lstrErrDescr As String
    Dim lstrErrSource As String
    llngErrNo = 0
    lstrErrDescr = ""
    lstrErrSource = ""

    Dim llngAffected As Long
    Dim lstrSQLQuery As String
    Dim ldbPOPtCost As Double
           
    grdData.Clear
    lstrSQLQuery = "select POPtNum, POPtDesc, POPtQtyOrd, POPtQtyBO, POPtQtyRcvd, POPtStatus, POPtUM, POPtWgt, POPtMtCost*POPtQtyOrd"
    lstrSQLQuery = lstrSQLQuery & " from PODetails"
    lstrSQLQuery = lstrSQLQuery & " where POID=" & mlngPOID
    lstrSQLQuery = lstrSQLQuery & " order by POPtNum"
    Set mobjPODetailRS = CnnBTMS.Execute(lstrSQLQuery, llngAffected, adCmdText)
    If Not mobjPODetailRS Is Nothing Then
        mobjPODetailRS.ActiveConnection = Nothing
        If Not (mobjPODetailRS.BOF And mobjPODetailRS.EOF) Then

    Call CheckForZeros

    grdData.Col = 2
    
    Exit Sub

Error_Handler:
    llngErrNo = IIf(llngErrNo = 0, Err.Number, llngErrNo)
    lstrErrDescr = IIf(Len(lstrErrDescr) = 0, Err.Description, Err.Description)
    lstrErrSource = IIf(Len(Trim(Err.Source)) = 0, "", Err.Source & vbCrLf) & cstrErrorSource
    Call Err.Raise(llngErrNo, lstrErrSource, lstrErrDescr)
End Sub

Private Sub grdData_AfterDataRefresh()

    On Error GoTo Error_Handler
    Const cstrErrorSource As String = cstrApplicationName & "." & cstrModuleName & ".grdData_AfterDataRefresh()"
    Dim llngErrNo As Long
    Dim lstrErrDescr As String
    Dim lstrErrSource As String
    llngErrNo = 0
    lstrErrDescr = ""
    lstrErrSource = ""

    ' Close the ADO connection for getting the Data
    If Not mobjPODetailRS Is Nothing Then
        If mobjPODetailRS.State = adStateOpen Then mobjPODetailRS.Close
        Set mobjPODetailRS = Nothing
    End If
    
    Set grdData.DataSource = Nothing
    If grdData.Rows < cintMinRows Then grdData.Rows = cintMinRows
    Exit Sub

Error_Handler:
    llngErrNo = IIf(llngErrNo = 0, Err.Number, llngErrNo)
    lstrErrDescr = IIf(Len(lstrErrDescr) = 0, Err.Description, Err.Description)
    lstrErrSource = IIf(Len(Trim(Err.Source)) = 0, "", Err.Source & vbCrLf) & cstrErrorSource
    Call Err.Raise(llngErrNo, lstrErrSource, lstrErrDescr)
End Sub
