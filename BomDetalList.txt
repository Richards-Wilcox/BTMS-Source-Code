Private Sub cmdDelete_Click()
    On Error GoTo Error_Handler
    Const cstrErrorSource As String = cstrApplicationName & "." & cstrModuleName & ".cmdDelete_Click()"
    Dim llngErrNo As Long
    Dim lstrErrDescr As String
    Dim lstrErrSource As String
    llngErrNo = 0
    lstrErrDescr = ""
    lstrErrSource = ""

    Dim llngRow As Long
    Dim lstrPartNum As String
    Dim llngAffected As Long
    Dim i As Integer
    llngRow = grdData.RowSel
    lstrPartNum = Trim(grdData.TextMatrix(llngRow, 1))
    
    If mlngBOMID > 0 And lstrPartNum <> "" Then
        If MsgBox("Are you sure you wish to delete the selected item?", vbApplicationModal + vbYesNo + vbQuestion, "Delete BoM Part") = vbYes Then
            Dim lstrSQLQuery As String
            lstrSQLQuery = "delete * from BOMdetails"
            lstrSQLQuery = lstrSQLQuery & " where BOMID=" & mlngBOMID
            lstrSQLQuery = lstrSQLQuery & " and BOMptnum=" & SQLString(lstrPartNum)
            Call CnnBTMS.Execute(lstrSQLQuery, llngAffected, adCmdText And adExecuteNoRecords)
            
            lstrSQLQuery = "delete * from BOMPartLendetails"
            lstrSQLQuery = lstrSQLQuery & " where DocID=" & mlngBOMID
            lstrSQLQuery = lstrSQLQuery & " and partnum=" & SQLString(lstrPartNum)
            Call CnnBTMS.Execute(lstrSQLQuery, llngAffected, adCmdText And adExecuteNoRecords)
            
            Call UpdateBOMTotal(mlngBOMID)
            Call LoadGrid
        End If
    End If
    Exit Sub

Private Sub SetGridTitles()
    On Error GoTo Error_Handler
    Const cstrErrorSource As String = cstrApplicationName & "." & cstrModuleName & ".SetGridTitles()"
    Dim llngErrNo As Long
    Dim lstrErrDescr As String
    Dim lstrErrSource As String
    llngErrNo = 0
    lstrErrDescr = ""
    lstrErrSource = ""
    
    grdData.ColHidden(0) = True
    If CheckShowPrices = False Then
        grdData.ColHidden(8) = True
        grdData.ColHidden(9) = True
        grdData.ColWidth(1) = grdData.Width / 7 '1700
        grdData.ColWidth(2) = grdData.Width / 4 '3100
        grdData.ColWidth(3) = grdData.Width / 9 '900
        grdData.ColWidth(4) = grdData.Width / 9 '900
        grdData.ColWidth(5) = grdData.Width / 9 '900
        grdData.ColWidth(6) = grdData.Width / 9 '900
        grdData.ColWidth(7) = grdData.Width / 9 '900
        grdData.ColWidth(8) = grdData.Width / 9 '1000
        grdData.ColWidth(9) = grdData.Width / 9 '1000
    Else
        grdData.ColHidden(8) = False
        grdData.ColHidden(9) = False
        grdData.ColWidth(1) = grdData.Width / 8 '1700
        grdData.ColWidth(2) = grdData.Width / 5 '3100
        grdData.ColWidth(3) = grdData.Width / 11 '900
        grdData.ColWidth(4) = grdData.Width / 11 '900
        grdData.ColWidth(5) = grdData.Width / 11 '900
        grdData.ColWidth(6) = grdData.Width / 11 '900
        grdData.ColWidth(7) = grdData.Width / 11 '900
        grdData.ColWidth(8) = grdData.Width / 11 '1000
        grdData.ColWidth(9) = grdData.Width / 11 '1000
    End If
    
    grdData.ColFormat(3) = "###0.00"
'    GrdData.ColFormat(4) = "###0.00"
'    GrdData.ColFormat(5) = "###0.00"
'    GrdData.ColFormat(6) = "###0.00"
    grdData.ColFormat(9) = "###0.00"
    grdData.ColFormat(8) = "###0.00"
    
    grdData.Row = 0
    grdData.Col = 1
    grdData.Text = "Part Number"
    grdData.Col = 2
    grdData.Text = "Description"
    grdData.Col = 3
    grdData.Text = "Req. Qty"
    grdData.Col = 4
    grdData.Text = "Stock Qty"
    grdData.Col = 5
    grdData.Text = "Order Qty"
    grdData.Col = 6
    grdData.Text = "Res. Qty"
    grdData.Col = 7
    grdData.Text = "UM"
    grdData.Col = 8
    grdData.Text = "Unit Cost (" & Chr(36) & ")"
    grdData.Col = 9
    grdData.Text = "Total Cost (" & Chr(36) & ")"
    
    Dim i As Integer
    Dim Rows As Integer

''''''''''Change in .NET''''''''''''''''
    If mstrAddedPN <> "" Then
        For i = 1 To Me.grdData.Rows - 1
            If Me.grdData.TextMatrix(i, 1) = mstrAddedPN Then Exit For
        Next
        mlngRow = i
        grdData.Row = mlngRow
        mstrAddedPN = ""
    Else
    
''''''''''END Change in .NET
    
        For i = 1 To Me.grdData.Rows - 1
            If Me.grdData.TextMatrix(i, 1) <> "" Then
                Rows = Rows + 1
            End If
        Next
            
        If mlngRow = 0 Or mlngRow = 1 Then
            grdData.Row = 1
        Else
            If mlngRow > Rows Then
                mlngRow = mlngRow - 1
            End If
            grdData.Row = mlngRow
        End If
''''''''''Change in .NET''''''''''''''''
    End If

''''''''''END Change in .NET
    If grdData.RowIsVisible(mlngRow) = False Then
        grdData.TopRow = mlngRow
    End If

        
    grdData.Col = 2
    
    Exit Sub

Error_Handler:
    llngErrNo = IIf(llngErrNo = 0, Err.Number, llngErrNo)
    lstrErrDescr = IIf(Len(lstrErrDescr) = 0, Err.Description, Err.Description)
    lstrErrSource = IIf(Len(Trim(Err.Source)) = 0, "", Err.Source & vbCrLf) & cstrErrorSource

End Sub

Private Sub LoadData()
    On Error GoTo Error_Handler
    Const cstrErrorSource As String = cstrApplicationName & "." & cstrModuleName & ".LoadData()"
    Dim llngErrNo As Long
    Dim lstrErrDescr As String
    Dim lstrErrSource As String
    llngErrNo = 0
    lstrErrDescr = ""
    lstrErrSource = ""

    Dim llngAffected As Long
    Dim lstrSQLQuery As String
    Dim lobjBOMRS As ADODB.Recordset
    
    mintMode = MODE_VIEW
    
    lstrSQLQuery = "select BOMNum, BOMDate, CustName, QuoteID, POID, BOMAlloc"
    lstrSQLQuery = lstrSQLQuery & " from BoM where BOMID=" & mlngBOMID
    Set lobjBOMRS = CnnBTMS.Execute(lstrSQLQuery, llngAffected, adCmdText)
    If Not lobjBOMRS Is Nothing Then
        lobjBOMRS.ActiveConnection = Nothing
        If Not (lobjBOMRS.BOF And lobjBOMRS.EOF) Then
            lobjBOMRS.MoveFirst
            Dim lstrFieldName As String
            lstrFieldName = "BOMNum"
            lblBOMNum.Caption = IIf(IsNull(lobjBOMRS.Fields(lstrFieldName).Value), mstrDocNum, Trim(lobjBOMRS.Fields(lstrFieldName).Value))
            lstrFieldName = "BOMDate"
            lblDate.Caption = IIf(IsNull(lobjBOMRS.Fields(lstrFieldName).Value), "", Format(lobjBOMRS.Fields(lstrFieldName).Value, cstrDateMask))
            lstrFieldName = "CustName"
            lblName.Caption = IIf(IsNull(lobjBOMRS.Fields(lstrFieldName).Value), "", Trim(lobjBOMRS.Fields(lstrFieldName).Value))
            lstrFieldName = "POID"
            mlngPOID = IIf(IsNull(lobjBOMRS.Fields(lstrFieldName).Value), 0, lobjBOMRS.Fields(lstrFieldName).Value)
            lstrFieldName = "QuoteID"
            mlngQuoteID = IIf(IsNull(lobjBOMRS.Fields(lstrFieldName).Value), 0, lobjBOMRS.Fields(lstrFieldName).Value)
            lstrFieldName = "BOMAlloc"
            mblnAlloc = IIf(lobjBOMRS.Fields(lstrFieldName).Value, True, False)
            lobjBOMRS.Close
            
            Call LoadGrid
        End If
    End If
    Set lobjBOMRS = Nothing
    
    Call SetControls(mintMode)
    Exit Sub

Error_Handler:
    llngErrNo = IIf(llngErrNo = 0, Err.Number, llngErrNo)
    lstrErrDescr = IIf(Len(lstrErrDescr) = 0, Err.Description, Err.Description)
    lstrErrSource = IIf(Len(Trim(Err.Source)) = 0, "", Err.Source & vbCrLf) & cstrErrorSource
    Call Err.Raise(llngErrNo, lstrErrSource, lstrErrDescr)
End Sub

Private Sub LoadGrid()
    On Error GoTo Error_Handler
    Const cstrErrorSource As String = cstrApplicationName & "." & cstrModuleName & ".LoadGrid()"
    Dim llngErrNo As Long
    Dim lstrErrDescr As String
    Dim lstrErrSource As String
    llngErrNo = 0
    lstrErrDescr = ""
    lstrErrSource = ""

    Dim llngAffected As Long
    Dim lstrSQLQuery As String
    Dim lobjRS As ADODB.Recordset
    
    grdData.Clear
    lstrSQLQuery = "select BOMPtNum, BOMPtDesc, BOMPtQtyReq,"
    lstrSQLQuery = lstrSQLQuery & " (select inventory.QtyInStock from inventory where inventory.PartNum = BOMDetails.BOMPtNum),"
    lstrSQLQuery = lstrSQLQuery & " (select inventory.QtyOnOrder from inventory where inventory.PartNum = BOMDetails.BOMPtNum),"
    lstrSQLQuery = lstrSQLQuery & " (select inventory.QtyReserved from inventory where inventory.PartNum = BOMDetails.BOMPtNum) ,BOMPtUM, BOMPtMtCost, "
    lstrSQLQuery = lstrSQLQuery & " (BOMPtMtCost * BOMPttotamountReq )"
    lstrSQLQuery = lstrSQLQuery & " from BOMdetails"
    lstrSQLQuery = lstrSQLQuery & " where BOMID=" & mlngBOMID
    lstrSQLQuery = lstrSQLQuery & " order by BOMptnum"
    Set mobjBOMDetailRS = CnnBTMS.Execute(lstrSQLQuery, llngAffected, adCmdText)
    If Not mobjBOMDetailRS Is Nothing Then
        mobjBOMDetailRS.ActiveConnection = Nothing
        If Not (mobjBOMDetailRS.BOF And mobjBOMDetailRS.EOF) Then
            grdData.VirtualData = False
            grdData.DataMode = flexDMFree
            Set grdData.DataSource = mobjBOMDetailRS
        End If
    End If
        
    Call SetGridTitles
    Call CheckForZeros
    Exit Sub

Error_Handler:
    llngErrNo = IIf(llngErrNo = 0, Err.Number, llngErrNo)
    lstrErrDescr = IIf(Len(lstrErrDescr) = 0, Err.Description, Err.Description)
    lstrErrSource = IIf(Len(Trim(Err.Source)) = 0, "", Err.Source & vbCrLf) & cstrErrorSource
    Call Err.Raise(llngErrNo, lstrErrSource, lstrErrDescr)
End Sub

Private Sub grdData_BeforeSort(ByVal Col As Long, Order As Integer)
    On Error GoTo Error_Handler
    Const cstrErrorSource As String = cstrApplicationName & "." & cstrModuleName & ".grdData_BeforeSort()"
    Dim llngErrNo As Long
    Dim lstrErrDescr As String
    Dim lstrErrSource As String
    llngErrNo = 0
    lstrErrDescr = ""
    lstrErrSource = ""

    ' no flags? apply custom sort
    If grdData.ExplorerBar > &H1000& Then Exit Sub
    
    ' save selection
    Dim R&, c&, rs&, cs&
    grdData.GetSelection R, c, rs, cs
    grdData.Redraw = flexRDNone
    
    ' apply sort to non-empty range
    Dim Row%
    For Row = grdData.Rows - 1 To grdData.FixedRows Step -1
        If Len(grdData.TextMatrix(Row, Col)) Then Exit For
    Next
    If Row > grdData.FixedRows Then
        Dim llngCol As Long
        llngCol = Col
        grdData.Select grdData.FixedRows, llngCol, Row, llngCol
        grdData.Sort = Order
    End If
    
    ' restore selection
    grdData.Select R, c, rs, cs
    grdData.Redraw = flexRDDirect
    
    ' cancel default sort
    Order = 0
    Exit Sub

Error_Handler:
    llngErrNo = IIf(llngErrNo = 0, Err.Number, llngErrNo)
    lstrErrDescr = IIf(Len(lstrErrDescr) = 0, Err.Description, Err.Description)
    lstrErrSource = IIf(Len(Trim(Err.Source)) = 0, "", Err.Source & vbCrLf) & cstrErrorSource
    Call Err.Raise(llngErrNo, lstrErrSource, lstrErrDescr)
End Sub

Private Sub CheckForZeros()
    On Error GoTo Error_Handler
    Const cstrErrorSource As String = cstrApplicationName & "." & cstrModuleName & ".CheckForZeros()"
    Dim llngErrNo As Long
    Dim lstrErrDescr As String
    Dim lstrErrSource As String
    llngErrNo = 0
    lstrErrDescr = ""
    lstrErrSource = ""

    Dim i As Integer, j As Integer
    ClearConObj

''''''''''Change in .NET''''''''''''''''
    Label4.Visible = False
''''''''''END Change in .NET
    
    For i = 1 To Me.grdData.Rows - 1
        If (DBLValue(Me.grdData.TextMatrix(i, 3)) = 0 Or _
            DBLValue(Me.grdData.TextMatrix(i, 8)) = 0) And Me.grdData.TextMatrix(i, 1) <> "" Then
            Me.grdData.Cell(flexcpBackColor, i, 0, i, Me.grdData.Cols - 1) = vbRed
            Me.grdData.Cell(flexcpForeColor, i, 0, i, Me.grdData.Cols - 1) = vbWhite

''''''''''Change in .NET''''''''''''''''
            Label4.Visible = True
''''''''''END Change in .NET
            
        End If
        
        If grdData.TextMatrix(i, 7) = "INCHES" And UCase(mstrBOMUM) = "METRIC" Then
            grdData.TextMatrix(i, 7) = "MM"
            If DBLValue(grdData.TextMatrix(i, 4)) > 0 Then
                aCon.Inch = DBLValue(grdData.TextMatrix(i, 4))
                aCon.mmcalc
                grdData.TextMatrix(i, 4) = Round(aCon.mm, 0)
            End If
            If DBLValue(grdData.TextMatrix(i, 5)) > 0 Then
                aCon.Inch = DBLValue(grdData.TextMatrix(i, 5))
                aCon.mmcalc
                grdData.TextMatrix(i, 5) = Round(aCon.mm, 0)
            End If
            If DBLValue(grdData.TextMatrix(i, 6)) > 0 Then
                aCon.Inch = DBLValue(grdData.TextMatrix(i, 6))
                aCon.mmcalc
                grdData.TextMatrix(i, 6) = Round(aCon.mm, 0)
            End If
            If DBLValue(grdData.TextMatrix(i, 8)) > 0 Then
                Dim Var As Double
                Var = DBLValue(grdData.TextMatrix(i, 8)) * 0.03937008
                grdData.TextMatrix(i, 8) = Var
                If Var < 1 Then
                    grdData.TextMatrix(i, 8) = Format(Var, "0.000000")
                Else
                    grdData.TextMatrix(i, 8) = Format(Var, "###0.00")
                End If
            Else
                grdData.TextMatrix(i, 8) = Format(grdData.TextMatrix(i, 8), "###0.00")
            End If
        Else
'            If DBLValue(GrdData.TextMatrix(i, 8)) < 1 Then
'                GrdData.TextMatrix(i, 8) = Format(GrdData.TextMatrix(i, 8), "###0.0000")
'            Else
'                GrdData.TextMatrix(i, 8) = Format(GrdData.TextMatrix(i, 8), "###0.00")
'            End If
        End If
    Next

    Exit Sub

Error_Handler:
    llngErrNo = IIf(llngErrNo = 0, Err.Number, llngErrNo)
    lstrErrDescr = IIf(Len(lstrErrDescr) = 0, Err.Description, lstrErrDescr & vbCrLf & Err.Description)
    lstrErrSource = IIf(Len(Trim(Err.Source)) = 0, "", Err.Source & vbCrLf) & cstrErrorSource
    Call Err.Raise(llngErrNo, lstrErrSource, lstrErrDescr)

End Sub
