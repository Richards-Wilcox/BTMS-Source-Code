Private Sub cmdDelete_Click()
    On Error GoTo Error_Handler
    Const cstrErrorSource As String = cstrApplicationName & "." & cstrModuleName & ".cmdDelete_Click()"
    Dim llngErrNo As Long
    Dim lstrErrDescr As String
    Dim lstrErrSource As String
    llngErrNo = 0
    lstrErrDescr = ""
    lstrErrSource = ""

    Dim llngRow As Long
    Dim lstrQuoteNum As String
    Dim lstrPartNum As String
    Dim llngAffected As Long
    
    llngRow = grdData.RowSel
    lstrQuoteNum = Trim(mstrDocNum)
    lstrPartNum = Trim(grdData.TextMatrix(llngRow, 1))
    If lstrPartNum <> "" Then
        If mlngQuoteID > 0 Then
            If mlngPOID > 0 Then
                MsgBox "This Quote is attached to a PO.", vbApplicationModal + vbOKOnly + vbExclamation, "Cannot Delete Quote Part"

                    lstrSQLQuery = "delete * from quotedetails"
                    lstrSQLQuery = lstrSQLQuery & " where quoteid=" & mlngQuoteID
                    lstrSQLQuery = lstrSQLQuery & " and quoteptnum=" & SQLString(lstrPartNum)
                    Call CnnBTMS.Execute(lstrSQLQuery, llngAffected, adCmdText And adExecuteNoRecords)
                    Call UpdateQuoteTotal(mlngQuoteID)
                    Call UpdateQuoteTotalPrice(mlngQuoteID)
                    Call LoadGrid
                End If
            End If
        End If
    End If
    Screen.MousePointer = vbDefault
    Exit Sub

    Screen.MousePointer = vbDefault

    Dim lstrErrDescr As String
    Dim lstrErrSource As String

    lstrErrSource = ""
    
    grdData.ColHidden(0) = True
    If CheckShowPrices = False Then
        grdData.ColHidden(5) = True
        grdData.ColHidden(6) = True
        grdData.ColWidth(1) = grdData.Width / 5 ' 2600
        grdData.ColWidth(2) = grdData.Width / 2.5  '4000
        grdData.ColWidth(3) = grdData.Width / 7  '1000
        grdData.ColWidth(4) = grdData.Width / 7 '800

        grdData.ColWidth(4) = grdData.Width / 8  '800
        grdData.ColWidth(5) = grdData.Width / 8 '1050
        grdData.ColWidth(6) = grdData.Width / 8 '1050
    End If
    
    grdData.Row = 0
    grdData.Col = 1
    grdData.Text = "Part Number"
    grdData.Col = 2
    grdData.Text = "Description"
    grdData.Col = 3
    grdData.Text = "Qty"
    grdData.Col = 4
    grdData.Text = "UM"
    grdData.Col = 5
    grdData.Text = "Unit Cost (" & Chr(36) & ")"
    grdData.Col = 6
    grdData.Text = "Total Cost (" & Chr(36) & ")"
    grdData.ColFormat(3) = "###0.00"
    grdData.ColFormat(5) = "###0.00"
    grdData.ColFormat(6) = "###0.00"
    
    Dim i As Integer
    Dim Rows As Integer
''''''''''Change in .NET''''''''''''''''
    If mstrAddedPN <> "" Then
        For i = 1 To Me.grdData.Rows - 1
            If Me.grdData.TextMatrix(i, 1) = mstrAddedPN Then Exit For
        Next
        mlngRow = i
        grdData.Row = mlngRow
        mstrAddedPN = ""
    Else
    
''''''''''END Change in .NET
        For i = 1 To Me.grdData.Rows - 1
            If Me.grdData.TextMatrix(i, 1) <> "" Then
                Rows = Rows + 1
            End If
        Next
            
        If mlngRow = 0 Or mlngRow = 1 Then
            grdData.Row = 1
        Else
            If mlngRow > Rows Then
                mlngRow = mlngRow - 1
            End If
            grdData.Row = mlngRow
        End If

''''''''''Change in .NET''''''''''''''''
    End If
''''''''''END Change in .NET
        
    If grdData.RowIsVisible(mlngRow) = False Then
        grdData.TopRow = mlngRow
    End If
    
    grdData.Col = 2
    
    Exit Sub

Error_Handler:
    llngErrNo = IIf(llngErrNo = 0, Err.Number, llngErrNo)
    lstrErrDescr = IIf(Len(lstrErrDescr) = 0, Err.Description, Err.Description)
    lstrErrSource = IIf(Len(Trim(Err.Source)) = 0, "", Err.Source & vbCrLf) & cstrErrorSource

Private Sub LoadData()
    On Error GoTo Error_Handler
    Const cstrErrorSource As String = cstrApplicationName & "." & cstrModuleName & ".LoadData()"
    Dim llngErrNo As Long
    Dim lstrErrDescr As String
    Dim lstrErrSource As String
    llngErrNo = 0
    lstrErrDescr = ""
    lstrErrSource = ""

    Dim llngAffected As Long
    Dim lstrSQLQuery As String
    Dim lobjQuoteRS As ADODB.Recordset
    
    mintMode = MODE_VIEW
    
    lstrSQLQuery = "select QuoteNum, QuoteStatus, QuoteDealerCost, QuoteMeasSys, QuoteAlloc, POID"
    lstrSQLQuery = lstrSQLQuery & " from quote where quoteid=" & mlngQuoteID
    Set lobjQuoteRS = CnnBTMS.Execute(lstrSQLQuery, llngAffected, adCmdText)
    If Not lobjQuoteRS Is Nothing Then
        lobjQuoteRS.ActiveConnection = Nothing
        If Not (lobjQuoteRS.BOF And lobjQuoteRS.EOF) Then
            lobjQuoteRS.MoveFirst
            Dim lstrFieldName As String
            lstrFieldName = "QuoteNum"
            lblQuoteNum.Caption = IIf(IsNull(lobjQuoteRS.Fields(lstrFieldName).Value), mstrDocNum, Trim(lobjQuoteRS.Fields(lstrFieldName).Value))
            lstrFieldName = "QuoteMeasSys"
            mstrQuoteUM = IIf(lobjQuoteRS.Fields(lstrFieldName).Value = 1, "Imperial", "Metric")
            lstrFieldName = "QuoteStatus"
            lblStatus.Caption = IIf(IsNull(lobjQuoteRS.Fields(lstrFieldName).Value), "", Trim(lobjQuoteRS.Fields(lstrFieldName).Value))
            lstrFieldName = "QuoteDealerCost"
            lblTotal.Caption = Format(IIf(IsNull(lobjQuoteRS.Fields(lstrFieldName).Value), "", Trim(lobjQuoteRS.Fields(lstrFieldName).Value)), cstrDollarMask)
            lstrFieldName = "POID"
            mlngPOID = IIf(IsNull(lobjQuoteRS.Fields(lstrFieldName).Value), 0, lobjQuoteRS.Fields(lstrFieldName).Value)
            mblnAlloc = lobjQuoteRS.Fields("QuoteAlloc").Value
            
            lobjQuoteRS.Close
            
            Call LoadGrid
        End If
    End If
    Set lobjQuoteRS = Nothing
    Call SetControls(mintMode)
    Exit Sub

Error_Handler:
    llngErrNo = IIf(llngErrNo = 0, Err.Number, llngErrNo)
    lstrErrDescr = IIf(Len(lstrErrDescr) = 0, Err.Description, Err.Description)
    lstrErrSource = IIf(Len(Trim(Err.Source)) = 0, "", Err.Source & vbCrLf) & cstrErrorSource
    Call Err.Raise(llngErrNo, lstrErrSource, lstrErrDescr)
End Sub

Private Sub grdData_BeforeSort(ByVal Col As Long, Order As Integer)
    On Error GoTo Error_Handler
    Const cstrErrorSource As String = cstrApplicationName & "." & cstrModuleName & ".grdData_BeforeSort()"
    Dim llngErrNo As Long
    Dim lstrErrDescr As String
    Dim lstrErrSource As String
    llngErrNo = 0
    lstrErrDescr = ""
    lstrErrSource = ""

    ' no flags? apply custom sort
    If grdData.ExplorerBar > &H1000& Then Exit Sub
    
    ' save selection
    Dim R&, c&, rs&, cs&
    grdData.GetSelection R, c, rs, cs
    grdData.Redraw = flexRDNone
    
    ' apply sort to non-empty range
    Dim Row%
    For Row = grdData.Rows - 1 To grdData.FixedRows Step -1
        If Len(grdData.TextMatrix(Row, Col)) Then Exit For
    Next
    If Row > grdData.FixedRows Then
        Dim llngCol As Long
        llngCol = Col
        grdData.Select grdData.FixedRows, llngCol, Row, llngCol
        grdData.Sort = Order
    End If
    
    ' restore selection
    grdData.Select R, c, rs, cs
    grdData.Redraw = flexRDDirect
    
    ' cancel default sort
    Order = 0
    Exit Sub

Error_Handler:
    llngErrNo = IIf(llngErrNo = 0, Err.Number, llngErrNo)
    lstrErrDescr = IIf(Len(lstrErrDescr) = 0, Err.Description, Err.Description)
    lstrErrSource = IIf(Len(Trim(Err.Source)) = 0, "", Err.Source & vbCrLf) & cstrErrorSource

        If (DBLValue(Me.grdData.TextMatrix(i, 3)) = 0 Or _

''''''''''Change in .NET''''''''''''''''
            Label4.Visible = True
''''''''''END Change in .NET
            'Next
        End If
        If grdData.TextMatrix(i, 4) = "MM" Then

                End If
            End If
