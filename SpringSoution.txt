Private Sub cboCable_Click()

On Error GoTo Error_Handler
    Const cstrErrorSource As String = cstrApplicationName & "." & cstrModuleName & ".cboCable_Click()"
    Dim llngErrNo As Long
    Dim lstrErrDescr As String
    Dim lstrErrSource As String
    llngErrNo = 0
    lstrErrDescr = ""
    lstrErrSource = ""
    
    Dim blRound  As Boolean, blSquare As Boolean
    
    Call DrumInfoFromTable  'after the selection of cable, drum information get calculated with new springs solutions
          
    ReDim varDoorData.arrSpringSolution(0)
    If frmSprSolutions.chkRound = 1 Then
        blRound = Springs("Round", True)
    End If
    If frmSprSolutions.chkSquare = 1 Then
        If blRound = False Then
            blSquare = Springs("Square", True)
        Else
            blSquare = Springs("Square", False)
        End If
    End If
    If blSquare = False And blRound = False Then
        Call NoSolutionsFound
    End If
    If blSquare = True Or blRound = True Then
        Call FoundSolutions
    End If
     
    Exit Sub

Error_Handler:
    llngErrNo = IIf(llngErrNo = 0, Err.Number, llngErrNo)
    lstrErrDescr = IIf(Len(lstrErrDescr) = 0, Err.Description, lstrErrDescr & vbCrLf & Err.Description)
    Call DisplayError(llngErrNo, IIf(Len(Trim(Err.Source)) = 0, "", Err.Source & vbCrLf) & cstrErrorSource, lstrErrDescr)
End Sub

Private Sub PositionSprings(ByVal lintInd As Integer)

    On Error GoTo Error_Handler
    Const cstrErrorSource As String = cstrApplicationName & "." & cstrModuleName & ".PositionSprings()"
    Dim llngErrNo As Long
    Dim lstrErrDescr As String
    Dim lstrErrSource As String
    llngErrNo = 0
    lstrErrDescr = ""
    lstrErrSource = ""
    Dim NSpr As Integer
    Dim Small As Integer
    Dim Big As Integer
    'sets the default number of left and right wound springs
    
    NSpr = UBound(varDoorData.arrSpringSolution(lintInd).arrSpring()) + 1
    'wiresize
    
    VSFlexGrid2.TextMatrix(1, 2) = varDoorData.arrSpringSolution(lintInd).arrSpring(0).WireSize
    VSFlexGrid2.TextMatrix(2, 2) = varDoorData.arrSpringSolution(lintInd).arrSpring(0).WireSize
    VSFlexGrid2.TextMatrix(3, 2) = IIf(varDoorData.arrSpringSolution(lintInd).Mismatched = True, varDoorData.arrSpringSolution(lintInd).arrSpring(UBound(varDoorData.arrSpringSolution(lintInd).arrSpring())).WireSize, "")
    VSFlexGrid2.TextMatrix(4, 2) = IIf(varDoorData.arrSpringSolution(lintInd).Mismatched = True, varDoorData.arrSpringSolution(lintInd).arrSpring(UBound(varDoorData.arrSpringSolution(lintInd).arrSpring())).WireSize, "")
    
    'PN
    VSFlexGrid2.TextMatrix(1, 0) = varDoorData.arrSpringSolution(lintInd).arrSpring(0).strPN
    VSFlexGrid2.TextMatrix(2, 0) = Swap(varDoorData.arrSpringSolution(lintInd).arrSpring(0).strPN, True)
    VSFlexGrid2.TextMatrix(3, 0) = IIf(varDoorData.arrSpringSolution(lintInd).Mismatched = True, varDoorData.arrSpringSolution(lintInd).arrSpring(UBound(varDoorData.arrSpringSolution(lintInd).arrSpring())).strPN, "")
    VSFlexGrid2.TextMatrix(4, 0) = IIf(varDoorData.arrSpringSolution(lintInd).Mismatched = True, Swap(varDoorData.arrSpringSolution(lintInd).arrSpring(UBound(varDoorData.arrSpringSolution(lintInd).arrSpring())).strPN, True), "")
    
    'Qty
    If varDoorData.arrSpringSolution(lintInd).Mismatched = False Then
        If NSpr / 2 = NSpr \ 2 Then 'even
            'LH
            VSFlexGrid2.TextMatrix(1, 3) = NSpr / 2
            VSFlexGrid2.TextMatrix(1, 1) = "LH"
            'RH
            VSFlexGrid2.TextMatrix(2, 3) = NSpr / 2
            VSFlexGrid2.TextMatrix(2, 1) = "RH"
        Else
            'LH
            VSFlexGrid2.TextMatrix(1, 3) = RoundUp(NSpr / 2)
            VSFlexGrid2.TextMatrix(1, 1) = "LH"
            'RH
            VSFlexGrid2.TextMatrix(2, 3) = NSpr \ 2
            VSFlexGrid2.TextMatrix(2, 1) = "RH"
        End If
    Else
        
        Small = GetSprQtyPerGauge(lintInd, varDoorData.arrSpringSolution(lintInd).arrSpring(0).WireSize)
        Big = GetSprQtyPerGauge(lintInd, varDoorData.arrSpringSolution(lintInd).arrSpring(UBound(varDoorData.arrSpringSolution(lintInd).arrSpring())).WireSize)
        Dim LeftIPPT1 As Single
        Dim RightIPPT1 As Single
        Dim LeftIPPT2 As Single
        Dim RightIPPT2 As Single
        Dim i As Integer
        Dim LS As Integer, LB As Integer, rs As Integer, RB As Integer
        Dim SmallIPPT As Single
        Dim BigIPPT As Single
        Dim Diff As Single
        
        If NSpr / 2 = NSpr \ 2 Then 'even
                    
            'LH Small
            VSFlexGrid2.TextMatrix(1, 3) = RoundDown(Small / 2)
            VSFlexGrid2.TextMatrix(1, 1) = "LH"
            'LH Big
            VSFlexGrid2.TextMatrix(2, 3) = Small - DBLValue(VSFlexGrid2.TextMatrix(1, 3))
            VSFlexGrid2.TextMatrix(2, 1) = "RH"
            
            'RH small
            VSFlexGrid2.TextMatrix(3, 3) = RoundUp(NSpr / 2) - RoundDown(Small / 2)
            VSFlexGrid2.TextMatrix(3, 1) = "LH"
            'RH Big
            VSFlexGrid2.TextMatrix(4, 3) = Big - DBLValue(VSFlexGrid2.TextMatrix(3, 3))
            VSFlexGrid2.TextMatrix(4, 1) = "RH"

        Else
            Dim Temp As Single
            Temp = Small / 2
            If Temp = Int(Temp) Then
                LS = Temp + 1
            Else
                LS = RoundUp(Temp)
            End If
        
            LB = RoundUp(NSpr / 2) - LS
            rs = Small - LS
            RB = Big - LB
            
            'LH
            VSFlexGrid2.TextMatrix(1, 3) = LS
            VSFlexGrid2.TextMatrix(1, 1) = "LH"
            VSFlexGrid2.TextMatrix(2, 3) = rs
            VSFlexGrid2.TextMatrix(2, 1) = "RH"

            'RH
            VSFlexGrid2.TextMatrix(3, 3) = LB
            VSFlexGrid2.TextMatrix(3, 1) = "LH"
            VSFlexGrid2.TextMatrix(4, 3) = RB
            VSFlexGrid2.TextMatrix(4, 1) = "RH"
        End If
        
    End If
        
    Exit Sub

Error_Handler:
    llngErrNo = IIf(llngErrNo = 0, Err.Number, llngErrNo)
    lstrErrDescr = IIf(Len(lstrErrDescr) = 0, Err.Description, Err.Description)
    lstrErrSource = IIf(Len(Trim(Err.Source)) = 0, "", Err.Source & vbCrLf) & cstrErrorSource
    Call Err.Raise(llngErrNo, lstrErrSource, lstrErrDescr)

End Sub

             varDoorData.aDrum.strStamping = "D575-120" Or _
              varDoorData.aDrum.strStamping = "D850-132" Or _
               varDoorData.aDrum.strStamping = "D1100-216" Then
        
        If DrWgt <= 704 Then
            cboCable.List(0) = IIf(MeasU = 1, "1/8" & Chr(34) & " Cable", "3.2mm Cable")
            cboCable.ItemData(cboCable.NewIndex) = 1
            cboCable.List(1) = IIf(MeasU = 1, "3/16" & Chr(34) & " Cable", "4.8mm Cable")
            cboCable.ItemData(cboCable.NewIndex) = 2
        ElseIf DrWgt > 704 And DrWgt <= 1480 Then
            cboCable.List(0) = IIf(MeasU = 1, "3/16" & Chr(34) & " Cable", "4.8mm Cable")
            cboCable.ItemData(cboCable.NewIndex) = 2
        End If
        
    ElseIf varDoorData.aDrum.strStamping = "D800-384" Or _
            varDoorData.aDrum.strStamping = "D800-120" Or _
            varDoorData.aDrum.strStamping = "D1150-236" Or _
            varDoorData.aDrum.strStamping = "D1350-336" Or _
             varDoorData.aDrum.strStamping = "D6375-164" Then
                
        If DrWgt <= 1480 Then
            cboCable.List(0) = IIf(MeasU = 1, "3/16" & Chr(34) & " Cable", "4.8mm Cable")
            cboCable.ItemData(cboCable.NewIndex) = 2
            cboCable.List(1) = IIf(MeasU = 1, "1/4" & Chr(34) & " Cable", "6.4mm Cable")
            cboCable.ItemData(cboCable.NewIndex) = 3
        ElseIf DrWgt > 1480 And DrWgt <= 2560 Then
            cboCable.List(0) = IIf(MeasU = 1, "1/4" & Chr(34) & " Cable", "6.4mm Cable")
            cboCable.ItemData(cboCable.NewIndex) = 3
        End If
    End If
    
    If cboCable.ListCount > 0 Then
        cableSafetyFactor_5_1 = False
        cboCable.ListIndex = 0
    Else
        cableSafetyFactor_5_1 = True
    End If
    
    Exit Function

Error_Handler:
    llngErrNo = IIf(llngErrNo = 0, Err.Number, llngErrNo)
    lstrErrDescr = IIf(Len(lstrErrDescr) = 0, Err.Description, lstrErrDescr & vbCrLf & Err.Description)
    lstrErrSource = IIf(Len(Trim(Err.Source)) = 0, "", Err.Source & vbCrLf) & cstrErrorSource
    Call Err.Raise(llngErrNo, lstrErrSource, lstrErrDescr)
End Function
