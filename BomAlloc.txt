Private Sub cmdAddtoPO_Click()
    On Error GoTo Error_Handler
    Const cstrErrorSource As String = cstrApplicationName & "." & cstrModuleName & ".cmdAddtoPO_Click()"
    Dim llngErrNo As Long
    Dim lstrErrDescr As String
    Dim lstrErrSource As String
    llngErrNo = 0
    lstrErrDescr = ""
    lstrErrSource = ""
    
    Dim i As Integer
    For i = 1 To grdData1.Rows - 1
        If grdData1.TextMatrix(i, 0) <> "" Then
            If grdData1.Cell(flexcpChecked, i, 14) = flexChecked Then
                If DBLValue(grdData1.TextMatrix(i, 13)) > 0 Then
                    Exit For
                Else
                    MsgBox "Selected items must have quantity greater than 0 to be put on a Purchase Order.", vbExclamation + vbApplicationModal, "Invalid Quantity Selected."
                    Exit Sub
                End If
            End If
        End If
    Next
    
    If i = grdData1.Rows Then
        MsgBox "No items selected.", vbInformation + vbApplicationModal, "No Selection."
        Screen.MousePointer = vbDefault
        Exit Sub
    End If
    
    Dim frmX As frmPOAdd
    Set frmX = New frmPOAdd
    Set frmX.mobjParent = Me
    frmX.mlngDOCID = mlngDOCID
    frmX.mintBackTag = mintBackTag
    Screen.MousePointer = vbDefault
    mblnInFocus = False
    frmX.Show vbModal
    mblnInFocus = True
    Screen.MousePointer = vbHourglass
    
    Dim lstrSQL As String
    Dim lobjRS As ADODB.Recordset
    Dim llngAffected As Long
    
    If Not mstrData = "" Then
        Dim larrData As Variant
        larrData = Split(mstrData, "~")
        mobjPO.CancelFlag = IIf(larrData(1) = "True", True, False)
        
        If mobjPO.CancelFlag = True Then
            Screen.MousePointer = vbDefault
            Exit Sub
        End If
        'mobjPO.PriceLevel = intCheck
        mobjPO.PONum = Trim(larrData(0))
        mobjPO.POID = DBLValue(Trim(larrData(2)))
        Call ProcessPO(mobjPO.POID)
    Else
        Screen.MousePointer = vbDefault
        Exit Sub
    End If
    
    For i = 1 To grdData1.Rows - 1
        lstrSQL = "select QtyOnpos, INVUM from inventory where PartNum=" & SQLString(grdData1.TextMatrix(i, 0))
        Set lobjRS = CnnBTMS.Execute(lstrSQL, llngAffected, adCmdText)
        If Not lobjRS Is Nothing Then
            If Not (lobjRS.EOF = True And lobjRS.BOF = True) Then
                If MeasU = 0 And lobjRS.Fields("INVUM").Value = "INCHES" Then
                    ClearConObj
                    aCon.Inch = lobjRS.Fields("QtyOnPos").Value
                    aCon.mmcalc
                    grdData1.TextMatrix(i, 8) = Round(aCon.mm, 0)
                Else
                    grdData1.TextMatrix(i, 8) = FormatNumber(lobjRS.Fields("QtyOnPos").Value, 2, vbTrue, vbTrue)
                End If
            End If
        End If
    Next i
    
    If mobjPO.POID > 0 Then
        cmdPO.Enabled = False
        cmdAddtoPO.Enabled = False
    End If
    grdData1.Row = 1
    grdData1.Col = 0
    
    Screen.MousePointer = vbDefault
    
    Exit Sub

Error_Handler:
    Screen.MousePointer = vbDefault
    llngErrNo = IIf(llngErrNo = 0, Err.Number, llngErrNo)
    lstrErrDescr = IIf(Len(lstrErrDescr) = 0, Err.Description, lstrErrDescr & vbCrLf & Err.Description)
    Call DisplayError(llngErrNo, IIf(Len(Trim(Err.Source)) = 0, "", Err.Source & vbCrLf) & cstrErrorSource, lstrErrDescr)

End Sub

Private Sub ProcessPO(ByVal lngCurrPOID As Long)
    
    On Error GoTo Error_Handler
    Const cstrErrorSource As String = cstrApplicationName & "." & cstrModuleName & ".ProcessPO()"
    Dim llngErrNo As Long
    Dim lstrErrDescr As String
    Dim lstrErrSource As String
    llngErrNo = 0
    lstrErrDescr = ""
    lstrErrSource = ""

    If mlngDOCID < 1 Then Exit Sub
    
    Dim llngAffected As Long
    Dim lstrSQLQuery As String
    Dim lblnInTrans As Boolean
    Dim lbtMeasPO As Byte
    Dim lbtMeasBOM As Byte
    Dim i As Integer
    Dim lobjPORS As ADODB.Recordset
    lblnInTrans = False
    
    ' Create new PO
    CnnBTMS.BeginTrans
    lblnInTrans = True
    
    If lngCurrPOID = 0 Then
        ' Prompt For missing Information
        '   Distributor Number, Shipping Address, Requested Date
        mstrData = ""
        Dim frmX As frmBOMPO
        Set frmX = New frmBOMPO
        Set frmX.objParent = Me
        Screen.MousePointer = vbDefault
        mblnInFocus = False
        frmX.Show vbModal
        mblnInFocus = True
        Screen.MousePointer = vbHourglass
        
        If Not mstrData = "" Then
            Dim larrData As Variant
            larrData = Split(mstrData, "~")
            mobjPO.CancelFlag = IIf(larrData(1) = "True", True, False)
            
            If mobjPO.CancelFlag = True Then
                CnnBTMS.RollbackTrans
                Exit Sub
            End If
            
            mobjPO.PONum = Trim(larrData(0))

''''''''''Change in .NET''''''''''''''''
'            mobjPO.ReqDate = Trim(larrData(2))

''''''''''END Change in .NET
            mobjPO.DistNum = DBLValue(Trim(larrData(3)))
            mobjPO.ShipNum = DBLValue(Trim(larrData(4)))
            mobjPO.POID = IIf(mobjPO.POID = 0, CreatePO(), mobjPO.POID)
        End If
    End If
            
    If mobjPO.POID > 0 Then
        Call UpdateOnPOsForPO(mobjPO.POID, True) 'first we need to remove the old qtys
        Call AddToPO(mobjPO.POID, mlngDOCID) 'add new qtys
        Call UpdateOnPOsForPO(mobjPO.POID, False) 'update OnPos withe the new qtys
    End If
    
    'Update the PO number on the BoM
    lstrSQLQuery = IIf(mintBackTag = 1, "Update BoM", "Update Quote")
    lstrSQLQuery = lstrSQLQuery & " Set POID=" & mobjPO.POID
    lstrSQLQuery = lstrSQLQuery & ", PONum=" & SQLString(mobjPO.PONum)
    lstrSQLQuery = lstrSQLQuery & IIf(mintBackTag = 1, " Where bomid=", " Where quoteid=") & mlngDOCID
    Call CnnBTMS.Execute(lstrSQLQuery, llngAffected, adCmdText And adExecuteNoRecords)
        
    ' Find the number of PODetails
    Dim llngCount As Long
    llngCount = -1
    lstrSQLQuery = "select Count(*) from PODetails where POID=" & mobjPO.POID
    Set lobjPORS = CnnBTMS.Execute(lstrSQLQuery, llngAffected, adCmdText)
    If Not lobjPORS Is Nothing Then
        lobjPORS.ActiveConnection = Nothing
        If Not (lobjPORS.BOF And lobjPORS.EOF) Then
            lobjPORS.MoveFirst
            llngCount = IIf(IsNull(lobjPORS.Fields(0).Value), 0, lobjPORS.Fields(0).Value)
        End If
        lobjPORS.Close
    End If
    Set lobjPORS = Nothing
    
    ' Delete PO if no Children
    If llngCount <= 0 Then
        lstrSQLQuery = "Delete * from PO where POID=" & mobjPO.POID
        Call CnnBTMS.Execute(lstrSQLQuery, 0, adCmdText And adExecuteNoRecords)
    End If
    
    CnnBTMS.CommitTrans
    lblnInTrans = False
    
    Exit Sub

Error_Handler:
    llngErrNo = IIf(llngErrNo = 0, Err.Number, llngErrNo)
    lstrErrDescr = IIf(Len(lstrErrDescr) = 0, Err.Description, lstrErrDescr & vbCrLf & Err.Description)
    lstrErrSource = IIf(Len(Trim(Err.Source)) = 0, "", Err.Source & vbCrLf) & cstrErrorSource
    If lblnInTrans = True Then CnnBTMS.RollbackTrans
    Call Err.Raise(llngErrNo, lstrErrSource, lstrErrDescr)
End Sub

Private Function BuildPODetailRS(ByVal lngBOMID As Long) As ADODB.Recordset
    On Error GoTo Error_Handler
    Const cstrErrorSource As String = cstrApplicationName & "." & cstrModuleName & ".BuildPODetailRS()"
    Dim llngErrNo As Long
    Dim lstrErrDescr As String
    Dim lstrErrSource As String
    llngErrNo = 0
    lstrErrDescr = ""
    lstrErrSource = ""

    Dim lstrSQL As String
    Dim lstrSchema As String
    Dim llngAffected As Long
    Dim lobjADORS As ADODB.Recordset
    Dim lobjBOMDetRS As ADODB.Recordset
    Dim lobjMainRS As ADODB.Recordset
    Dim lobjBOMRS As ADODB.Recordset
    Dim MeasSys As Integer
    Dim lsngLen As Single
    Dim ldblCost As Double
    Dim TempSprLen As Single
    Dim lstrPriceField As String
    Dim i As Integer
    
    Select Case intCheck
        Case 1:  lstrPriceField = "G"
        Case 2:  lstrPriceField = "I"
        Case 3:  lstrPriceField = "N"
        Case 4:  lstrPriceField = "T"
        
        Case 5: lstrPriceField = "J"
        Case 6: lstrPriceField = "M"
        Case 7: lstrPriceField = "S"
        Case 8: lstrPriceField = "V"

    End Select

    lstrSQL = "Select bommeassys,bomdrqty from BoM where BOMID=" & lngBOMID
    Set lobjBOMRS = CnnBTMS.Execute(lstrSQL, llngAffected, adCmdText)
    If Not lobjBOMRS Is Nothing Then
        If lobjBOMRS.EOF = False And lobjBOMRS.BOF = False Then
            MeasSys = lobjBOMRS.Fields("BomMeasSys").Value
        End If
        Set lobjBOMRS = Nothing
    End If
    
    ' Get empty recordset
    lstrErrDescr = "Error getting empty ADO recordset."
    lstrSQL = "Select * from PODetails where 1=0"
    Set lobjADORS = New ADODB.Recordset
    lobjADORS.CursorType = adOpenForwardOnly
    lobjADORS.CursorLocation = adUseClient
    lobjADORS.Open lstrSQL, CnnBTMS, adOpenForwardOnly, adLockOptimistic
    Set lobjADORS.ActiveConnection = Nothing
    lstrErrDescr = ""
    
    For i = 1 To grdData1.Rows - 1
        If grdData1.Cell(flexcpChecked, i, 14) = flexChecked Then
            lstrSQL = "Select * from BOMDetails where BOMID=" & lngBOMID & " and BOMPtNum=" & SQLString(grdData1.TextMatrix(i, 0))
            Set lobjBOMDetRS = CnnBTMS.Execute(lstrSQL, llngAffected, adCmdText)
            If Not lobjBOMDetRS Is Nothing Then
                lobjBOMDetRS.ActiveConnection = Nothing
                If Not (lobjBOMDetRS.BOF And lobjBOMDetRS.EOF) Then
                    lobjBOMDetRS.MoveFirst
                    Do While lobjBOMDetRS.EOF = False
                        lobjADORS.AddNew
                        lobjADORS.Fields("POID").Value = mobjPO.POID
                        lobjADORS.Fields("PONum").Value = mobjPO.PONum
                        lobjADORS.Fields("POPtNum").Value = IIf(IsNull(lobjBOMDetRS.Fields("BOMPtNum").Value), "", Trim(lobjBOMDetRS.Fields("BOMPtNum").Value))
                        lobjADORS.Fields("POPtDesc").Value = grdData1.TextMatrix(i, 1)
                        lobjADORS.Fields("POPtMtCost").Value = IIf(IsNull(lobjBOMDetRS.Fields("BOMPtMtCost").Value), 0, DBLValue(lobjBOMDetRS.Fields("BOMPtMtCost").Value))
                        If grdData1.TextMatrix(i, 3) = "MM" Then
                            ClearConObj
                            aCon.mm = DBLValue(grdData1.TextMatrix(i, 13))
                            aCon.Inchcalc
                            lobjADORS.Fields("POPtQtyOrd").Value = aCon.Inch
                        Else
                            lobjADORS.Fields("POPtQtyOrd").Value = DBLValue(grdData1.TextMatrix(i, 13))
                        End If
                        lobjADORS.Fields("POPtNotes").Value = ""
                        lobjADORS.Fields("POPtQtyRcvd").Value = 0
                        lobjADORS.Fields("POCat").Value = GetPOCat(lobjADORS.Fields("POPtNum").Value)
                        lobjADORS.Fields("POSubCategory").Value = GetSubCategory(lobjADORS.Fields("POPtNum").Value)
                        lobjADORS.Fields("POPtStatus").Value = "Active"
                        lobjADORS.Fields("PODiscount").Value = IIf(IsNull(lobjBOMDetRS.Fields("BOMDiscount").Value), 0, lobjBOMDetRS.Fields("BOMDiscount").Value)
                        lobjADORS.Fields("POPtMultID").Value = IIf(IsNull(lobjBOMDetRS.Fields("BOMPtMultID").Value), 0, lobjBOMDetRS.Fields("BOMPtMultID").Value)
                        lobjADORS.Fields("POPtCallPrice").Value = IIf(IsNull(lobjBOMDetRS.Fields("BOMCallPrice").Value), 0, lobjBOMDetRS.Fields("BOMCallPrice").Value)
                        If lobjADORS.Fields("POPtCallPrice").Value = True Then lobjADORS.Fields("POCallPriceExpDate").Value = IIf(IsNull(lobjBOMDetRS.Fields("BOMCallPriceExpDate").Value), Now() + 30, lobjBOMDetRS.Fields("BOMCallPriceExpDate").Value)
                        lobjADORS.Fields("POPtUM").Value = grdData1.TextMatrix(i, 3)
                        lobjADORS.Fields("POPtWgt").Value = IIf(IsNull(lobjBOMDetRS.Fields("BOMPtShipWgt").Value), 0, DBLValue(lobjBOMDetRS.Fields("BOMPtShipWgt").Value))
                        lobjADORS.Fields("POPtPhantomPN").Value = False
                        
                         '-- Mod Hanobver  MultId changed to strMultId
'                        lstrSQL = "Select PartNum, descimp, Length, shipweight, UM, SpecialMult, MultID, " & lstrPriceField & " from Main where PartNum=" & SQLString(lobjADORS.Fields("POPtNum").Value)
                        lstrSQL = "Select PartNum, descimp, Length, shipweight, UM, SpecialMult, " & strMultId & " ," & lstrPriceField & " from Main where PartNum=" & SQLString(lobjADORS.Fields("POPtNum").Value)
 
                        
                        Set lobjMainRS = CnnBTMS.Execute(lstrSQL, llngAffected, adCmdText)
                        If Not lobjMainRS Is Nothing Then
                            lobjMainRS.ActiveConnection = Nothing
                            If Not (lobjMainRS.BOF And lobjMainRS.EOF) Then
                            
                                ldblCost = Decrypt(GetSym(lobjBOMDetRS.Fields("BOMPtNum").Value), lobjMainRS.Fields(lstrPriceField).Value)
                                
                                If lobjMainRS.Fields("SpecialMult").Value > 0 Then
                                    ldblCost = lobjMainRS.Fields("SpecialMult").Value * ldblCost
                                Else
                                 '-- Mod Hanobver  MultId changed to strMultId
                                    If Not Trim(lobjMainRS.Fields(strMultId).Value) = "" Then
                                        ldblCost = ldblCost * (MultipliersCase(DBLValue(Trim(lobjMainRS.Fields(strMultId).Value))))
                                    End If
                                End If
                                
                                lobjADORS.Fields("POPtMtCost").Value = ldblCost
                                lobjADORS.Fields("POPtUM").Value = lobjMainRS.Fields("UM").Value
                                lobjADORS.Fields("POPtWgt").Value = DBLValue(lobjMainRS.Fields("shipweight").Value)
                                If MeasSys = 0 Then lobjADORS.Fields("POPtDesc").Value = IIf(IsNull(lobjMainRS.Fields("DescImp").Value), "N/A", Trim(lobjMainRS.Fields("DescImp").Value))
                                
                                If lobjADORS.Fields("POPtUM").Value = "LFT" Then
                                    lobjADORS.Fields("POPtQtyOrd").Value = DBLValue(lobjADORS.Fields("POPtQtyOrd").Value) / 12
                                End If
                                If lobjBOMDetRS.Fields("BOMPtUM").Value = "INCHES" And lobjMainRS.Fields("UM").Value = "EACH" Or _
                                        Left(lobjADORS.Fields("POPtNum").Value, 4) = "433-" Then
'                                If lobjADORS.Fields("POPtNum").Value = "328-227" Or _
                                        lobjADORS.Fields("POPtNum").Value = "426-0271" Or _
                                        Left(lobjADORS.Fields("POPtNum").Value, 4) = "433-" Then
        '                                lobjADORS.Fields("POPtNum").Value = "114-702" Or _
        '                                lobjADORS.Fields("POPtNum").Value = "114-703" Or _
        '                                lobjADORS.Fields("POPtNum").Value = "114-704" Or _
        '                                lobjADORS.Fields("POPtNum").Value = "114-706" Or _
        '                                lobjADORS.Fields("POPtNum").Value = "950-009" Or _
        '                                lobjADORS.Fields("POPtNum").Value = "950-010" Or _
        '                                lobjADORS.Fields("POPtNum").Value = "950-011" Or _
        '                                lobjADORS.Fields("POPtNum").Value = "950-012" Or _
        '                                lobjADORS.Fields("POPtNum").Value = "950-013" Or _
        '                                lobjADORS.Fields("POPtNum").Value = "950-014" Or _
        '                                lobjADORS.Fields("POPtNum").Value = "950-015" Or _
        '                                lobjADORS.Fields("POPtNum").Value = "114-701" Or _
        '                                lobjADORS.Fields("POPtNum").Value = "426-0272" Or _
        '                                lobjADORS.Fields("POPtNum").Value = "950-016" Then
                                    If Left(lobjADORS.Fields("POPtNum").Value, 4) = "433-" Then
                                        
                                        lsngLen = GetLengthFromMain(lobjADORS.Fields("POPtNum").Value)
                                        lobjADORS.Fields("POPtWgt").Value = DBLValue(lobjMainRS.Fields("shipweight").Value) * lsngLen
                                        lobjADORS.Fields("POPtMtCost").Value = ldblCost * lsngLen
                                        'springs; weight and price go per inch in Main, even though the UM is EACH, need to multiply by the total lenght.
                                    Else
                                        lsngLen = DBLValue(lobjMainRS.Fields("Length").Value)
                                        'pvc and shaft; weight and price go per each in the main, no need to modify.
                                    End If
                                    Dim lArrBOM(0) As Long, lArrQuote(0) As Long
                                    lArrBOM(0) = lngBOMID
                                    lArrQuote(0) = 0
                                    lobjADORS.Fields("POPtQtyOrd").Value = GetNumberOfPartsRequired(lobjADORS.Fields("POPtNum").Value, lsngLen, lArrBOM(), lArrQuote())
                
                                End If
                            End If
                            Set lobjMainRS = Nothing
                        End If
                        lobjADORS.Fields("POPtQtyBO").Value = DBLValue(lobjADORS.Fields("POPtQtyOrd").Value)
                        lobjBOMDetRS.MoveNext
                    Loop
                    lobjBOMDetRS.Close
                End If
            End If
            Set lobjBOMDetRS = Nothing
'        Else
'            Exit For
        End If
    Next
    Set BuildPODetailRS = lobjADORS
    Set lobjADORS = Nothing
    Exit Function

Error_Handler:
    llngErrNo = IIf(llngErrNo = 0, Err.Number, llngErrNo)
    lstrErrDescr = IIf(Len(lstrErrDescr) = 0, Err.Description, Err.Description)
    lstrErrSource = IIf(Len(Trim(Err.Source)) = 0, "", Err.Source & vbCrLf) & cstrErrorSource
    Call Err.Raise(llngErrNo, lstrErrSource, lstrErrDescr)
End Function

Private Function BuildPODetailRSQ(ByVal lngQuoteID As Long) As ADODB.Recordset

    lstrErrSource = ""

    Dim lstrSQL As String
    Dim lstrSchema As String
    Dim llngAffected As Long

    Dim lobjQuoteRS As ADODB.Recordset

    Dim TempSprLen As Single

Private Function CreatePO() As Long
    On Error GoTo Error_Handler
    Const cstrErrorSource As String = cstrApplicationName & "." & cstrModuleName & ".CreatePO()"
    Dim llngErrNo As Long
    Dim lstrErrDescr As String
    Dim lstrErrSource As String
    llngErrNo = 0
    lstrErrDescr = ""
    lstrErrSource = ""

    Dim llngAffected As Long
    Dim lstrSQLQuery As String
    Dim lobjBOMRS As ADODB.Recordset
    Dim lobjPORS As ADODB.Recordset
    Dim llngPOID As Long
    
    If Not mlngDOCID > 0 Then Exit Function
    
    llngPOID = 0
    If mintBackTag = 1 Then
        lstrSQLQuery = "select * from BoM where BOMID=" & mlngDOCID
    Else
        lstrSQLQuery = "select * from Quote where QuoteID=" & mlngDOCID
    End If
    Set lobjBOMRS = CnnBTMS.Execute(lstrSQLQuery, llngAffected, adCmdText)
    If Not lobjBOMRS Is Nothing Then
        lobjBOMRS.ActiveConnection = Nothing
        If Not (lobjBOMRS.BOF And lobjBOMRS.EOF) Then
            lobjBOMRS.MoveFirst
            Set lobjPORS = BuildPORS(lobjBOMRS)
            If Not lobjPORS Is Nothing Then
                If Not (lobjPORS.BOF And lobjPORS.EOF) Then
                    lobjPORS.MoveFirst
                    lstrErrDescr = "Error building insert statement for PO"
                    lstrSQLQuery = Build_Insert(lobjPORS, "PO", "POID")
                    lstrErrDescr = "Error inserting data into PO"
                    
                    ' Insert the new PO
                    Call CnnBTMS.Execute(lstrSQLQuery, 0, adCmdText And adExecuteNoRecords)
                    lstrErrDescr = ""
                    lobjPORS.Close
                    
                    ' Find the POID
                    lstrSQLQuery = "select POID from PO where PONum=" & SQLString(mobjPO.PONum)
                    Set lobjPORS = CnnBTMS.Execute(lstrSQLQuery, llngAffected, adCmdText)
                    If Not lobjPORS Is Nothing Then
                        lobjPORS.ActiveConnection = Nothing
                        If Not (lobjPORS.BOF And lobjPORS.EOF) Then
                            lobjPORS.MoveFirst
                            llngPOID = IIf(IsNull(lobjPORS.Fields("POID").Value), 0, lobjPORS.Fields("POID").Value)
                        End If
                        lobjPORS.Close
                    End If
                End If
            End If
        End If
        lobjBOMRS.Close
    End If
    
    Set lobjPORS = Nothing
    Set lobjBOMRS = Nothing
    CreatePO = llngPOID
    Exit Function

Error_Handler:
    llngErrNo = IIf(llngErrNo = 0, Err.Number, llngErrNo)
    lstrErrDescr = IIf(Len(lstrErrDescr) = 0, Err.Description, lstrErrDescr & vbCrLf & Err.Description)
    lstrErrSource = IIf(Len(Trim(Err.Source)) = 0, "", Err.Source & vbCrLf) & cstrErrorSource
    Call Err.Raise(llngErrNo, lstrErrSource, lstrErrDescr)
End Function

                End If

        End If

    Me.chkDescLocal.Value = 0
    If Me.mblnNow = True Then
        Label1.Caption = "The " & IIf(mintBackTag = 1, "BoM", "Quote") & " cannot be allocated." & vbCrLf & "The following items have insufficient stock."
        Me.cmdAddtoPO.Visible = False
        Me.cmdPO.Visible = False
        cmdReserve.Visible = False
        Me.cmdSelectAll.Visible = False
        Me.cmdUnSelectAll.Visible = False
    Else
        Label1.Caption = "Put a check mark in the " & Chr(34) & "Selected" & Chr(34) & " column beside the items you wish to add to a Purchase Order." & vbCrLf & _
            "To select all items click on the " & Chr(34) & "Select All" & Chr(34) & " button."
        If mintBackTag = 1 Then
            lstrSQL = "select POID from BoM where BOMID=" & mlngDOCID
        Else
            lstrSQL = "select POID from Quote where QuoteID=" & mlngDOCID
        End If
        Set lobjPORS = CnnBTMS.Execute(lstrSQL, llngAffected, adCmdText)
        If Not lobjPORS Is Nothing Then
            If Not (lobjPORS.EOF And lobjPORS.BOF) Then
               mobjPO.POID = IIf(IsNull(lobjPORS.Fields("POID").Value), 0, lobjPORS.Fields("POID").Value)
            End If
        End If
        Set lobjPORS = Nothing
        
        If mobjPO.POID > 0 Then
            cmdPO.Enabled = False
            cmdAddtoPO.Enabled = False
        End If
        
        lstrSQL = "select POID,PONum,POStatus from PO where POStatus='Active' and POPriceModel=" & intCheck
        Set lobjPORS = CnnBTMS.Execute(lstrSQL, llngAffected, adCmdText)
        If Not lobjPORS Is Nothing Then
            lobjPORS.ActiveConnection = Nothing
            If lobjPORS.BOF And lobjPORS.EOF Then
                cmdAddtoPO.Enabled = False
            End If
        End If
        Set lobjPORS = Nothing
    End If
    
    Call SetGridTitles
    
    Exit Sub

Error_Handler:
    llngErrNo = IIf(llngErrNo = 0, Err.Number, llngErrNo)
    lstrErrDescr = IIf(Len(lstrErrDescr) = 0, Err.Description, lstrErrDescr & vbCrLf & Err.Description)
    Call DisplayError(llngErrNo, IIf(Len(Trim(Err.Source)) = 0, "", Err.Source & vbCrLf) & cstrErrorSource, lstrErrDescr)

End Sub
