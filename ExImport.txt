Private Sub cmdImport_Click()
    On Error GoTo Error_Handler
    Const cstrErrorSource As String = cstrApplicationName & "." & cstrModuleName & ".cmdImport_Click()"
    Dim llngErrNo As Long
    Dim lstrErrDescr As String
    Dim lstrErrSource As String

    lstrErrSource = ""

    ComDialog1.DefaultExt = ".rwd"
    ComDialog1.CancelError = True
    ComDialog1.ShowOpen
    mstrFilepath = ComDialog1.FileName
    If Not mstrFilepath = "" Then
        lstrDocName = Mid(ComDialog1.FileTitle, 1, Len(ComDialog1.FileTitle) - 4)
    
        Do
            lblnEx = DocNameExists(lstrDocName)
            If lblnEx Then

                    vbCrLf & "Please select a different name.", "Select Different Name"))

            End If

Private Sub LoadData()

    On Error GoTo Error_Handler
    Const cstrErrorSource As String = cstrApplicationName & "." & cstrModuleName & ".LoadData()"
    Dim llngErrNo As Long
    Dim lstrErrDescr As String
    Dim lstrErrSource As String
    llngErrNo = 0
    lstrErrDescr = ""
    lstrErrSource = ""

    Dim llngAffected As Long
    Dim llngRecords As Long
    Dim lstrSQLQuery As String
    Dim mobjADORS As ADODB.Recordset
    
    grdData.Clear
    llngRecords = 0
    lstrSQLQuery = "SELECT 'BoM' as DocType, BoM.BOMNum as DocID, BoM.BOMDate as aDate,"
    lstrSQLQuery = lstrSQLQuery & " BoM.BOMID as ID From BoM  UNION"
    lstrSQLQuery = lstrSQLQuery & " SELECT 'Quote' as DocType, QUOTE.QuoteNum as DocID, QUOTE.QuoteDate as aDate,"
    lstrSQLQuery = lstrSQLQuery & " QUOTE.QuoteID as ID From QUOTE  UNION"
    lstrSQLQuery = lstrSQLQuery & " SELECT 'PO' as DocType, PO.PONum as DocID, PO.PODate as aDate,"
    lstrSQLQuery = lstrSQLQuery & " PO.POID as ID From PO WHERE POSTatus='Active' order by DocID"
    Set mobjADORS = CnnBTMS.Execute(lstrSQLQuery, llngAffected, adCmdText)
    If Not mobjADORS Is Nothing Then
        mobjADORS.ActiveConnection = Nothing
        If Not (mobjADORS.BOF And mobjADORS.EOF) Then
            llngRecords = mobjADORS.RecordCount
            grdData.VirtualData = False
            grdData.DataMode = flexDMFree
            Set grdData.DataSource = mobjADORS
        End If
    End If
    Call SetGridTitles
    Exit Sub

Error_Handler:
    llngErrNo = IIf(llngErrNo = 0, Err.Number, llngErrNo)
    lstrErrDescr = IIf(Len(lstrErrDescr) = 0, Err.Description, Err.Description)
    lstrErrSource = IIf(Len(Trim(Err.Source)) = 0, "", Err.Source & vbCrLf) & cstrErrorSource
    Call Err.Raise(llngErrNo, lstrErrSource, lstrErrDescr)
End Sub

        lintPOS1 = InStr(1, lstrLine, "|")

            Else

                                    Else
                                        lobjRS.Fields(i).Value = Mid(lstrLine, 1, lintPOS1 - 1)
                                    End If
                                End If
                            Else
                                lobjRS.Fields(i).Value = llngID
                            End If
                        End If
                        lstrLine = Mid(lstrLine, lintPOS1 + 1)
                        i = i + 1
                    End If
                Loop
            Next
            lobjRS.Update
        End If
''''''''''Change in .NET''''''''''''''''
    End If
''''''''''END Change in .NET

    Close #1
    If lstrDocType = "PO" Then
        Call UpdateOnPOsForPO(llngID, False)
    End If
    lobjRS.Close
    Set lobjRS = Nothing
    CnnBTMS.CommitTrans
    
    Screen.MousePointer = vbDefault
    Exit Function

Error_Handler:
    Screen.MousePointer = vbDefault
    CnnBTMS.RollbackTrans
    llngErrNo = IIf(llngErrNo = 0, Err.Number, llngErrNo)
    lstrErrDescr = IIf(Len(lstrErrDescr) = 0, Err.Description, lstrErrDescr & vbCrLf & Err.Description)
    lstrErrSource = IIf(Len(Trim(Err.Source)) = 0, "", Err.Source & vbCrLf) & cstrErrorSource
    Call Err.Raise(llngErrNo, lstrErrSource, lstrErrDescr)

End Function

Private Function DocNameExists(ByVal pstrName As String) As Boolean

    lstrSQL = "Select BomNum from BoM where BomNum = " & SQLString(pstrName)
    Set lobjRS = CnnBTMS.Execute(lstrSQL, 0, adCmdText)

        If Not (lobjRS.EOF And lobjRS.BOF) Then

        If Not (lobjRS.EOF And lobjRS.BOF) Then

        End If
    End If

    Set lobjRS = CnnBTMS.Execute(lstrSQL, 0, adCmdText)
    If Not lobjRS Is Nothing Then
        If Not (lobjRS.EOF And lobjRS.BOF) Then
            DocNameExists = True
        End If
    End If
    
    
    Exit Function

Private Sub grdData_BeforeSort(ByVal Col As Long, Order As Integer)
    On Error GoTo Error_Handler
    Const cstrErrorSource As String = cstrApplicationName & "." & cstrModuleName & ".grdData_BeforeSort()"
    Dim llngErrNo As Long
    Dim lstrErrDescr As String
    Dim lstrErrSource As String
    llngErrNo = 0
    lstrErrDescr = ""
    lstrErrSource = ""

    
    ' no flags? apply custom sort
    If grdData.ExplorerBar > &H1000& Then Exit Sub
    
    ' save selection
    Dim R&, c&, rs&, cs&
    grdData.GetSelection R, c, rs, cs
    grdData.Redraw = flexRDNone
    
    ' apply sort to non-empty range
    Dim Row%
    For Row = grdData.Rows - 1 To grdData.FixedRows Step -1
        If Len(grdData.TextMatrix(Row, Col)) Then Exit For
    Next
    If Row > grdData.FixedRows Then
        Dim llngCol As Long
        llngCol = Col
        grdData.Select grdData.FixedRows, llngCol, Row, llngCol
        grdData.Sort = Order
    End If
    
    ' restore selection
    grdData.Select R, c, rs, cs
    grdData.Redraw = flexRDDirect
    
    ' cancel default sort
    Order = 0
    Exit Sub

Error_Handler:
    llngErrNo = IIf(llngErrNo = 0, Err.Number, llngErrNo)
    lstrErrDescr = IIf(Len(lstrErrDescr) = 0, Err.Description, Err.Description)
    lstrErrSource = IIf(Len(Trim(Err.Source)) = 0, "", Err.Source & vbCrLf) & cstrErrorSource
    Call Err.Raise(llngErrNo, lstrErrSource, lstrErrDescr)
End Sub
