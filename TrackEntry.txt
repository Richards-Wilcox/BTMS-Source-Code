Private Sub cboShafts_Click()

' Sets SFD for S.S & Keyed T.S.
' Sets Coupler for S.S
' Sets Aux Shaft for S.S

    On Error GoTo Error_Handler
    Const cstrErrorSource As String = cstrApplicationName & "." & cstrModuleName & ".SubName()"
    Dim llngErrNo As Long
    Dim lstrErrDescr As String
    Dim lstrErrSource As String
    llngErrNo = 0
    lstrErrDescr = ""
    lstrErrSource = ""
    
    chkFlangeBearing.Value = 0
    chkSpringFailure.Enabled = True
       chkSpringFailure.Value = 0  ' 2.0.21
            
    ' If Solid Shaft
    If Not (cboShafts.Text = "Tube Shaft") And Not (cboShafts.Text = "Galvanized Tube Shaft") And Not (cboShafts.Text = "Keyed Tube Shaft") Then
            chkCoupler.Enabled = True
            If CECompliant = 1 Then
                If DrType = "T45" Or DrType = "T35" Then
                    chkSpringFailure.Value = 1
                    chkSpringFailure.Enabled = False
                End If
            End If
            
            If cboShafts.ItemData(cboShafts.ListIndex) = 5 Then chkFlangeBearing.Value = 1  ' 1-1/4 S.S
        
        
    ElseIf cboShafts.Text = "Keyed Tube Shaft" Then  ' 2.0.21 -  enable sfd
      
        chkCoupler.Value = 0
        chkCoupler.Enabled = False
'        chkSpringFailure.Value = 0
        chkSpringFailure.Enabled = True
        
        
        
    Else  ' Tube Shafts other than Keyed T.S
        chkCoupler.Value = 0
        chkCoupler.Enabled = False
        chkSpringFailure.Value = 0
        chkSpringFailure.Enabled = False
    End If
    
    
    
    
    If DrWidth >= 207 And Not (cboShafts.Text = "Tube Shaft") And Not (cboShafts.Text = "Galvanized Tube Shaft") And Not (cboShafts.Text = "Keyed Tube Shaft") Then
            chkCoupler.Value = 1   'by default, cut shaft
            chkCoupler.Enabled = False
    Else
        chkCoupler.Enabled = True
        chkCoupler.Value = 0
    End If
    
    If cboShafts.Text = "Keyed Tube Shaft" Then
        chkCoupler.Value = 0
        chkCoupler.Enabled = False
    End If
    
    'Auxillary shaft
    ' Enables for S.S
    
    If (cboShafts.Text = "Tube Shaft") Or (cboShafts.Text = "Galvanized Tube Shaft") Or (cboShafts.Text = "Keyed Tube Shaft") Then
        ChkAuxShaft.Enabled = False
    Else
        ChkAuxShaft.Enabled = True
    End If
    
    
'    chkSpringFailure.Enabled = True
'    If CECompliant = 1 Then
'        If DrType = "T45" Then
'            chkSpringFailure.Value = 1
'            chkSpringFailure.Enabled = False
'        End If
'    End If
    
    Exit Sub

Error_Handler:
    llngErrNo = IIf(llngErrNo = 0, Err.Number, llngErrNo)
    lstrErrDescr = IIf(Len(lstrErrDescr) = 0, Err.Description, lstrErrDescr & vbCrLf & Err.Description)
    Call DisplayError(llngErrNo, IIf(Len(Trim(Err.Source)) = 0, "", Err.Source & vbCrLf) & cstrErrorSource, lstrErrDescr)
End Sub

Private Sub Form_Resize()
    On Error GoTo Error_Handler
    Const cstrErrorSource As String = cstrApplicationName & "." & cstrModuleName & ".Form_Resize()"
    Dim llngErrNo As Long
    Dim lstrErrDescr As String
    Dim lstrErrSource As String
    llngErrNo = 0
    lstrErrDescr = ""
    lstrErrSource = ""
    
    'NOTE: It's best to put code here becuase any msg box in paint event execute twice if you move the msgbox around.
    Me.Move (Screen.Width - Me.Width) / 2, (Screen.Height - Me.Height) / 2
    
    If mblnDeadEndMsg = True Then MsgBox "The weight of the door has exceeded the maximum door weight capacity limitation (1000 lbs or 453.5 KG." & vbNewLine & _
                                "Please choose 3-inch hardware or consult with Richards-Wilcox Canada for further assistance.", vbApplicationModal + vbInformation, "Invalid Shaft"
'    mblnDeadEndMsg = False  'set back to its default
    
    Exit Sub

Error_Handler:
    llngErrNo = IIf(llngErrNo = 0, Err.Number, llngErrNo)
    lstrErrDescr = IIf(Len(lstrErrDescr) = 0, Err.Description, lstrErrDescr & vbCrLf & Err.Description)
    Call DisplayError(llngErrNo, IIf(Len(Trim(Err.Source)) = 0, "", Err.Source & vbCrLf) & cstrErrorSource, lstrErrDescr)


End Sub

Private Sub Form_Unload(Cancel As Integer)

    If Not gobjAgent Is Nothing Then
        If gobjAgent.ActorIsVisible = True Then
            gobjAgent.ActorStopAll
            gobjAgent.HideAgent
        End If
    End If

End Sub

Private Sub MousePos1_MouseOver(ControlName As String, Index As Long, CtrlData As String, PosX As Long, PosY As Long)
    On Error GoTo Error_Handler
    Const cstrErrorSource As String = cstrApplicationName & "." & cstrModuleName & ".MousePos1_MouseOver()"
    Dim llngErrNo As Long
    Dim lstrErrDescr As String
    Dim lstrErrSource As String

End Sub

                Rise = DBLValue(txtRise.Text)
                Run = DBLValue(txtRun.Text)

                    tang = Rise / Run
                    SlopHAng = Atn(tang) * 180 / 3.14159265358979
                End If
            Else

    frmGrid.Grid1.Col = 0

    Exit Sub
