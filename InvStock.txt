Public Sub LoadSubCategory()

    On Error GoTo Error_Handler
    Const cstrErrorSource As String = cstrApplicationName & "." & cstrModuleName & ".LoadSubcategory()"
    Dim llngErrNo As Long
    Dim lstrErrDescr As String
    Dim lstrErrSource As String
    llngErrNo = 0
    lstrErrDescr = ""
    lstrErrSource = ""

    Dim llngAffected As Long
    Dim lstrSQLQuery As String
    Dim lobjSubCatRS As ADODB.Recordset

''''''''''Change in .NET''''''''''''''''
    cboSubCat.AddItem "ALL"
    
''''''''''END Change in .NET
    lstrSQLQuery = "select subcategory from catsub order by subcategory"
    Set lobjSubCatRS = CnnBTMS.Execute(lstrSQLQuery, llngAffected, adCmdText)
    If Not lobjSubCatRS Is Nothing Then
        lobjSubCatRS.ActiveConnection = Nothing
        If Not (lobjSubCatRS.BOF And lobjSubCatRS.EOF) Then
            lobjSubCatRS.MoveFirst
            Dim lstrName As String
            Do While lobjSubCatRS.EOF = False
                lstrName = IIf(IsNull(lobjSubCatRS.Fields("subcategory").Value), "", Trim(lobjSubCatRS.Fields("subcategory").Value))
                If Not lstrName = "" Then Call cboSubCat.AddItem(lstrName)
                lobjSubCatRS.MoveNext
            Loop
                lobjSubCatRS.Close
        End If
    End If
    Set lobjSubCatRS = Nothing
        
    cboSubCat.ListIndex = 0
    
    Exit Sub

Error_Handler:
    llngErrNo = IIf(llngErrNo = 0, Err.Number, llngErrNo)
    lstrErrDescr = IIf(Len(lstrErrDescr) = 0, Err.Description, lstrErrDescr & vbCrLf & Err.Description)
    lstrErrSource = IIf(Len(Trim(Err.Source)) = 0, "", Err.Source & vbCrLf) & cstrErrorSource
    Call Err.Raise(llngErrNo, lstrErrSource, lstrErrDescr)
End Sub

Private Sub LoadData()
    On Error GoTo Error_Handler
    Const cstrErrorSource As String = cstrApplicationName & "." & cstrModuleName & ".LoadData()"
    Dim llngErrNo As Long
    Dim lstrErrDescr As String
    Dim lstrErrSource As String
    llngErrNo = 0
    lstrErrDescr = ""
    lstrErrSource = ""

    Dim llngAffected As Long
    Dim llngRecords As Long
    Dim lstrSQLQuery As String
    
    mintMode = MODE_VIEW
    grdData.Clear
        
    llngRecords = 0
    
    lstrSQLQuery = "Select INV.PartNum,MAIN.PartNLocal,"
    lstrSQLQuery = lstrSQLQuery & IIf(MeasU = 0, "MAIN.DescMet", "MAIN.DescImp")
    lstrSQLQuery = lstrSQLQuery & ",MAIN.DescOther,INV.INVUM,IIf(IsNull(INV.QtyInStock),0,INV.QtyInStock),0, Main.SubCategory "
    lstrSQLQuery = lstrSQLQuery & " From Inventory INV"
    lstrSQLQuery = lstrSQLQuery & " LEFT JOIN MAIN ON INV.PartNum = MAIN.PartNum"
    lstrSQLQuery = lstrSQLQuery & " order by INV.PartNum"
    Set mobjADORS = CnnBTMS.Execute(lstrSQLQuery, llngAffected, adCmdText)

    If Not mobjADORS Is Nothing Then
        mobjADORS.ActiveConnection = Nothing
        If Not (mobjADORS.BOF And mobjADORS.EOF) Then
            ''''''''''Change in .NET''''''''''''''''
            If Me.cboSubCat.ListIndex > 0 Then
            ''''''''''END Change in .NET
                mobjADORS.Filter = "Subcategory=" & SQLString(cboSubCat.List(cboSubCat.ListIndex))
            ''''''''''Change in .NET''''''''''''''''
            End If
            ''''''''''END Change in .NET
            llngRecords = mobjADORS.RecordCount
            grdData.VirtualData = False
            grdData.DataMode = flexDMFree
            Set grdData.DataSource = mobjADORS
        End If
    End If

    Dim i As Integer
    If MeasU = 0 Then
        ClearConObj
        For i = 1 To grdData.Rows - 1
            If grdData.TextMatrix(i, 5) = "INCHES" Then
                If DBLValue(grdData.TextMatrix(i, 6)) > 0 Then
                    aCon.Inch = DBLValue(grdData.TextMatrix(i, 6))
                    aCon.mmcalc
                    grdData.TextMatrix(i, 6) = Round(aCon.mm, 0)
                End If
                grdData.TextMatrix(i, 5) = "MM"
            End If
        Next
    End If
    
    fraGrid.Caption = " " & llngRecords & " Records "
    Call SetGridTitles
    Call SetControls
    Exit Sub

Error_Handler:
    llngErrNo = IIf(llngErrNo = 0, Err.Number, llngErrNo)
    lstrErrDescr = IIf(Len(lstrErrDescr) = 0, Err.Description, Err.Description)
    lstrErrSource = IIf(Len(Trim(Err.Source)) = 0, "", Err.Source & vbCrLf) & cstrErrorSource
    Call Err.Raise(llngErrNo, lstrErrSource, lstrErrDescr)
End Sub

Private Sub grdData_DblClick()
    On Error GoTo Error_Handler
    Const cstrErrorSource As String = cstrApplicationName & "." & cstrModuleName & ".grdData_DblClick()"
    Dim llngErrNo As Long
    Dim lstrErrDescr As String
    Dim lstrErrSource As String
    llngErrNo = 0
    lstrErrDescr = ""
    lstrErrSource = ""
    
    If mintMode = MODE_EDIT And mlngRow = 0 Then
        Dim llngRow As Long
        Dim ldblNewValue As Double
        
        Select Case mlngCol
            Case 7  ' Received Qty
                    If MsgBox("Do you wish to do this?", vbApplicationModal + vbExclamation + vbYesNo, "This will set the counted value equal to the in-stock value") = vbYes Then
                        grdData.Redraw = flexRDNone
                        For llngRow = 1 To grdData.Rows - 1
                            If Trim(grdData.TextMatrix(llngRow, 1)) = "" Then Exit For
                            grdData.TextMatrix(llngRow, 7) = grdData.TextMatrix(llngRow, 6)
                        Next llngRow
                        grdData.Redraw = flexRDDirect
                    End If
        End Select
    End If
    
    Exit Sub

    Call DisplayError(llngErrNo, IIf(Len(Trim(Err.Source)) = 0, "", Err.Source & vbCrLf) & cstrErrorSource, lstrErrDescr)

Private Sub grdData_KeyPressEdit(ByVal Row As Long, ByVal Col As Long, KeyAscii As Integer)
    On Error GoTo Error_Handler
    Const cstrErrorSource As String = cstrApplicationName & "." & cstrModuleName & ".grdData_KeyPressEdit()"
    Dim llngErrNo As Long
    Dim lstrErrDescr As String
    Dim lstrErrSource As String
    llngErrNo = 0
    lstrErrDescr = ""
    lstrErrSource = ""
    Dim lstrText As String
    lstrText = grdData.EditText
    
    If Len(lstrText) >= grdData.EditMaxLength Then
        Beep
        KeyAscii = 0
        Exit Sub
    End If
    
    If grdData.TextMatrix(grdData.Row, 5) <> "EACH" Then
        If DecimU = 1 Then
            If Not IsNumeric(Chr(KeyAscii)) And KeyAscii <> 46 And KeyAscii <> 8 Then
                Beep
                KeyAscii = 0
                Exit Sub
            End If
            If KeyAscii = 46 Then
                If Not grdData.EditSelLength = Len(lstrText) Then
                    If InStr(1, lstrText, Chr(46)) <> 0 Then
                        Beep
                        KeyAscii = 0
                        Exit Sub
                    End If
                End If
            End If
        Else
            If Not IsNumeric(Chr(KeyAscii)) And KeyAscii <> 44 And KeyAscii <> 8 Then
                Beep
                KeyAscii = 0
                Exit Sub
            End If
            If KeyAscii = 44 Then
                If Not grdData.EditSelLength = Len(lstrText) Then
                    If InStr(1, lstrText, Chr(44)) <> 0 Then
                        Beep
                        KeyAscii = 0
                        Exit Sub
                    End If
                End If
            End If
        End If
    Else
        If Not IsNumeric(Chr(KeyAscii)) Xor KeyAscii = 8 Then
            Beep
            KeyAscii = 0
            Exit Sub
        End If
    End If
    Exit Sub
Error_Handler:
    llngErrNo = IIf(llngErrNo = 0, Err.Number, llngErrNo)
    lstrErrDescr = IIf(Len(lstrErrDescr) = 0, Err.Description, lstrErrDescr & vbCrLf & Err.Description)

Private Sub SaveData()
    On Error GoTo Error_Handler
    Const cstrErrorSource As String = cstrApplicationName & "." & cstrModuleName & ".SaveData()"
    Dim llngErrNo As Long
    Dim lstrErrDescr As String
    Dim lstrErrSource As String
    llngErrNo = 0
    lstrErrDescr = ""
    lstrErrSource = ""

    Dim lstrSQL As String
    Dim lstrWhereClause As String
    Dim lstrExcludeFieldList As String
    Dim llngAffected As Long
    Dim lobjADORS As ADODB.Recordset
    
    Screen.MousePointer = vbHourglass
    
    ' Copy the fields from the screen to an ADO Recordset
    Set lobjADORS = BuildRS()
    
    Dim lintMode As Integer
     
    ' Process through the recordset and insert or update rows
    If Not lobjADORS Is Nothing Then
        If Not (lobjADORS.BOF And lobjADORS.EOF) Then
            lobjADORS.MoveFirst
            lstrExcludeFieldList = "InventoryID"
            Do While lobjADORS.EOF = False
                If GetEventState(QS_ALLEVENTS) <> 0 Then DoEvents
                ' Check to see if the record exists in the
                ' Inventory table already
                Dim lobjParentRS As ADODB.Recordset
                lintMode = MODE_ADD
                lstrSQL = "select PartNum from Inventory where PartNum=" & SQLString(lobjADORS.Fields("PartNum").Value)
                Set lobjParentRS = CnnBTMS.Execute(lstrSQL, llngAffected, adCmdText)
                If Not lobjParentRS Is Nothing Then
                    If Not (lobjParentRS.BOF And lobjParentRS.EOF) Then
                        lobjParentRS.MoveFirst
                        ' Reset the Mode so that we do an update of the existing part
                        lintMode = MODE_EDIT
                    End If
                    lobjParentRS.Close
                End If
                Set lobjParentRS = Nothing
            
                If lintMode = MODE_ADD Then
                    lstrErrDescr = "Error building insert statement for Inventory"
                    lstrSQL = Build_Insert(lobjADORS, "Inventory", "")
                    lstrErrDescr = "Error inserting data into Inventory"
                Else
                    lstrErrDescr = "Error building update statement for Inventory"
                    lstrWhereClause = "PartNum=" & SQLString(lobjADORS.Fields("PartNum").Value)
                    lstrSQL = Build_Update(lobjADORS, "Inventory", lstrWhereClause, "")
                    lstrErrDescr = "Error updating Inventory"
                End If
                
                Call CnnBTMS.Execute(lstrSQL, 0, adCmdText And adExecuteNoRecords)
                lstrErrDescr = ""
                lobjADORS.MoveNext
            Loop
            
            lobjADORS.Close
        End If
    End If
    
    Set lobjADORS = Nothing
    
    Call LoadData
    Screen.MousePointer = vbDefault
    Exit Sub

Error_Handler:
    llngErrNo = IIf(llngErrNo = 0, Err.Number, llngErrNo)
    lstrErrDescr = IIf(Len(lstrErrDescr) = 0, Err.Description, Err.Description)
    lstrErrSource = IIf(Len(Trim(Err.Source)) = 0, "", Err.Source & vbCrLf) & cstrErrorSource
    Screen.MousePointer = vbDefault
    Call Err.Raise(llngErrNo, lstrErrSource, lstrErrDescr)
End Sub

Private Function BuildRS() As ADODB.Recordset
    On Error GoTo Error_Handler
    Const cstrErrorSource As String = cstrApplicationName & "." & cstrModuleName & ".BuildRS()"
    Dim llngErrNo As Long
    Dim lstrErrDescr As String
    Dim lstrErrSource As String
    llngErrNo = 0
    lstrErrDescr = ""
    lstrErrSource = ""

    Dim lstrSQL As String
    Dim lstrSchema As String
    Dim llngAffected As Long
    Dim lobjADORS As ADODB.Recordset
    Dim llngCtr As Long
    
    ' Get empty recordset
    lstrErrDescr = "Error getting empty ADO recordset."
    lstrSQL = "Select PartNum,QtyInStock,PrevCountValue,PrevCountDate from Inventory where 1=0"
    Set lobjADORS = New ADODB.Recordset
    lobjADORS.CursorType = adOpenForwardOnly
    lobjADORS.CursorLocation = adUseClient
    lobjADORS.Open lstrSQL, CnnBTMS, adOpenForwardOnly, adLockOptimistic
    Set lobjADORS.ActiveConnection = Nothing
    lstrErrDescr = ""
    
    For llngCtr = 1 To grdData.Rows - 1
        If Trim(grdData.TextMatrix(llngCtr, 1)) = "" Then Exit For
        If DBLValue(grdData.TextMatrix(llngCtr, 6)) <> DBLValue(grdData.TextMatrix(llngCtr, 7)) Then
            lobjADORS.AddNew
            lobjADORS.Fields("PartNum").Value = Trim(grdData.TextMatrix(llngCtr, 1))
            lobjADORS.Fields("PrevCountValue").Value = DBLValue(grdData.TextMatrix(llngCtr, 6))
            lobjADORS.Fields("QtyInStock").Value = DBLValue(grdData.TextMatrix(llngCtr, 7))
            If grdData.TextMatrix(llngCtr, 5) = "MM" Then
                If DBLValue(grdData.TextMatrix(llngCtr, 6)) > 0 Then
                    ClearConObj
                    aCon.mm = DBLValue(grdData.TextMatrix(llngCtr, 6))
                    aCon.Inchcalc
                    lobjADORS.Fields("PrevCountValue").Value = aCon.Inch
                End If
                If DBLValue(grdData.TextMatrix(llngCtr, 7)) > 0 Then
                    ClearConObj
                    aCon.mm = DBLValue(grdData.TextMatrix(llngCtr, 7))
                    aCon.Inchcalc
                    lobjADORS.Fields("QtyInStock").Value = aCon.Inch
                End If
            End If
            lobjADORS.Fields("PrevCountDate").Value = Now()
            
        End If
    Next llngCtr
    Set BuildRS = lobjADORS
    Set lobjADORS = Nothing
    Exit Function

Error_Handler:
    llngErrNo = IIf(llngErrNo = 0, Err.Number, llngErrNo)
    lstrErrDescr = IIf(Len(lstrErrDescr) = 0, Err.Description, Err.Description)
    lstrErrSource = IIf(Len(Trim(Err.Source)) = 0, "", Err.Source & vbCrLf) & cstrErrorSource
    Call Err.Raise(llngErrNo, lstrErrSource, lstrErrDescr)
End Function
