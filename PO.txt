Private Sub cmdIssue_Click()

    On Error GoTo Error_Handler
    Const cstrErrorSource As String = cstrApplicationName & "." & cstrModuleName & ".cmdIssue_Click()"
    Dim llngErrNo As Long
    Dim lstrErrDescr As String
    Dim lstrErrSource As String
    llngErrNo = 0
    lstrErrDescr = ""
    lstrErrSource = ""
    Screen.MousePointer = vbHourglass
    Dim lstrOut As Boolean
    Dim lintYesNo As Integer
    Dim CompletePO As Byte
    Dim CancelFlag As Boolean
    Dim blnInTrans As Boolean
    
    blnInTrans = False
    If cmdIssue.Caption = "&Complete PO" Then CompletePO = 1
    If lblStatus.Caption = "Active" Then 'Issue PO
        If CountDetails < 1 Then
            Screen.MousePointer = vbDefault
            MsgBox "The Purchase Order has no details.", vbExclamation + vbApplicationModal, "Cannot Issue"
            Exit Sub
        End If
    
        Dim frmX As frmPOIssue
        Set frmX = New frmPOIssue
        frmX.mlngPOID = mlngPOID
        frmX.mstrPOName = mstrPONum
        Set frmX.objParent = Me
        Screen.MousePointer = vbDefault
        mblnInFocus = False
        frmX.Show vbModal
        mblnInFocus = True
        If Not mstrData = "" Then
            Dim larrData As Variant
            larrData = Split(mstrData, "~")
            CancelFlag = IIf(larrData(1) = "True", True, False)
            
            If CancelFlag = True Then
                Call LoadData(False)
                Screen.MousePointer = vbDefault
                Exit Sub
            End If
        End If
        
    ElseIf lblStatus.Caption = "Issued" Then 'complete PO
        CnnBTMS.BeginTrans
        blnInTrans = True
        'If CheckForOustandingItems = True Then
        Screen.MousePointer = vbDefault

''''''''''Change in .NET''''''''''''''''
        
'        lintYesNo = MsgBox("Would you like to complete this Purchase Order by receiving all outstanding items?" & _
            "Click NO to cancel the Order.", vbYesNoCancel + vbApplicationModal + vbQuestion, "Complete Purchase Order")
        lintYesNo = MsgBox("If you received all of the remaining ordered parts on this order click Yes." & vbCrLf & _
            "If you need to cancel the receiving of all of the remaining parts on this order, click No." & vbCrLf & _
            "To cancel this operation, click Cancel.", vbYesNoCancel + vbApplicationModal + vbQuestion, "Complete Purchase Order")

''''''''''END Change in .NET
        
        If lintYesNo = vbCancel Then
                CnnBTMS.RollbackTrans
                Exit Sub
        ElseIf lintYesNo = vbNo Then 'cancel order
            Call ChangePOStatus(mlngPOID, "Complete")
            If IsInvMaint = 1 And CheckPORunInventory(mlngPOID) = True Then Call CancelInventory(mlngPOID)
        Else 'receive all outstanding items
            If IsInvMaint = 1 And CheckPORunInventory(mlngPOID) = True Then Call ReceiveAll(mlngPOID)
            Call ChangePOStatus(mlngPOID, "Complete")
        End If
        'End If
        CnnBTMS.CommitTrans
    ElseIf lblStatus.Caption = "Complete" Then 'Undo PO
        If DateValue(Now()) - DateValue(Me.lblComplete.Caption) > 1 Then
            Screen.MousePointer = vbDefault
            MsgBox "More than one day passed since this PO was completed." & vbCrLf & _
                "Cannot undo.", vbExclamation + vbApplicationModal, "Cannot Undo PO"
        Else
            If CanUndo(mlngPOID) = True Then
            
                CnnBTMS.BeginTrans
                blnInTrans = True
                If MsgBox("Are you sure you want to Re-Issue this Purchase Order?" & vbCrLf & "This will reset all items to Outstanding status.", vbExclamation + vbYesNo + vbApplicationModal, "Re-Issue PO") = vbYes Then Call ChangePOStatus(mlngPOID, "ReIssue")
                If mlngTag > 0 Then 'And mlBytCompletePO = 0
                    If IsInvMaint = 1 And CheckPORunInventory(mlngPOID) = True Then Call ResetInventory(mlngPOID)
                    Call ResetPOValues(mlngPOID, "ReIssue")
                End If
                CnnBTMS.CommitTrans
            Else
                Screen.MousePointer = vbDefault
                MsgBox "Cannot Undo this PO, the related documents were allocated.", vbApplicationModal + vbInformation, "Cannot Undo"
                Exit Sub
            End If
        End If
    End If
    blnInTrans = False
    Call LoadData(False)
    Screen.MousePointer = vbDefault
    Exit Sub

Error_Handler:
    Screen.MousePointer = vbDefault
    If blnInTrans = True Then CnnBTMS.RollbackTrans
    llngErrNo = IIf(llngErrNo = 0, Err.Number, llngErrNo)
    lstrErrDescr = IIf(Len(lstrErrDescr) = 0, Err.Description, Err.Description)
    Call DisplayError(llngErrNo, IIf(Len(Trim(Err.Source)) = 0, "", Err.Source & vbCrLf) & cstrErrorSource, lstrErrDescr)

End Sub

Private Sub ReceiveAll(ByVal plngPOID As Long)

    On Error GoTo Error_Handler
    Const cstrErrorSource As String = cstrApplicationName & "." & cstrModuleName & ".ReceiveAll()"
    Dim llngErrNo As Long
    Dim lstrErrDescr As String
    Dim lstrErrSource As String
    llngErrNo = 0
    lstrErrDescr = ""
    lstrErrSource = ""

    Dim lstrSQL As String
    Dim lstrSchema As String
    Dim llngAffected As Long
    Dim lobjPODetRS As ADODB.Recordset
    Dim lobjINVRS As ADODB.Recordset
    Dim ldblReceived As Double
    
    'Get empty recordset
    lstrErrDescr = "Error getting empty ADO recordset."
    lstrSQL = "Select * from PODetails where POID=" & plngPOID
    Set lobjPODetRS = New ADODB.Recordset
    lobjPODetRS.CursorType = adOpenForwardOnly
    lobjPODetRS.CursorLocation = adUseClient
    lobjPODetRS.Open lstrSQL, CnnBTMS, adOpenForwardOnly, adLockOptimistic
    lstrErrDescr = ""
        
    If Not lobjPODetRS Is Nothing Then
        If Not (lobjPODetRS.EOF And lobjPODetRS.BOF) Then
            Do While lobjPODetRS.EOF = False
                ldblReceived = DBLValue(lobjPODetRS.Fields("POPtQtyOrd").Value)
                If IsInvMaint = 1 And CheckPORunInventory(mlngPOID) = True Then
                    lstrSQL = "select PartNum,InvUM,QtyInStock,QtyBkOrder,QtyOnOrder from Inventory"
                    Set lobjINVRS = New ADODB.Recordset
                    lobjINVRS.CursorType = adOpenForwardOnly
                    lobjINVRS.CursorLocation = adUseClient
                    lobjINVRS.Open lstrSQL, CnnBTMS, adOpenForwardOnly, adLockOptimistic

                    lobjINVRS.MoveFirst
                    lobjINVRS.Find "PartNum=" & SQLString(lobjPODetRS.Fields("POPtNum").Value)
                    If lobjINVRS.EOF = False And lobjINVRS.BOF = False Then
                        If lobjPODetRS.Fields("POPtStatus").Value = "Outstanding" Then
                            If WasPOPreviouslyReceived(mlngPOID) = True Then
                                If lobjPODetRS.Fields("POPtUM").Value = "EACH" And lobjINVRS.Fields("InvUM").Value = "INCHES" Then
                                    lobjINVRS.Fields("QtyInStock").Value = DBLValue(lobjINVRS.Fields("QtyInStock").Value) - DBLValue(lobjPODetRS.Fields("POPtQtyRcvd").Value) * GetLengthFromMain(lobjPODetRS.Fields("POPtNum").Value) + DBLValue(lobjPODetRS.Fields("POPtQtyOrd").Value) * GetLengthFromMain(lobjPODetRS.Fields("POPtNum").Value)
                                    lobjINVRS.Fields("QtyOnOrder").Value = DBLValue(lobjINVRS.Fields("QtyOnOrder").Value) - DBLValue(lobjPODetRS.Fields("POPtQtybo").Value) * GetLengthFromMain(lobjPODetRS.Fields("POPtNum").Value)
                                    lobjINVRS.Fields("QtyBkOrder").Value = DBLValue(lobjINVRS.Fields("QtyBkOrder").Value) - DBLValue(lobjPODetRS.Fields("POPtQtybo").Value) * GetLengthFromMain(lobjPODetRS.Fields("POPtNum").Value)
                                ElseIf lobjPODetRS.Fields("POPtUM").Value = "LFT" Then
                                    lobjINVRS.Fields("QtyInStock").Value = DBLValue(lobjINVRS.Fields("QtyInStock").Value) - DBLValue(lobjPODetRS.Fields("POPtQtyRcvd").Value) * 12 + DBLValue(lobjPODetRS.Fields("POPtQtyOrd").Value) * 12
                                    lobjINVRS.Fields("QtyOnOrder").Value = DBLValue(lobjINVRS.Fields("QtyOnOrder").Value) - DBLValue(lobjPODetRS.Fields("POPtQtybo").Value) * 12
                                    lobjINVRS.Fields("QtyBkOrder").Value = DBLValue(lobjINVRS.Fields("QtyBkOrder").Value) - DBLValue(lobjPODetRS.Fields("POPtQtybo").Value) * 12
                                Else
                                    lobjINVRS.Fields("QtyInStock").Value = DBLValue(lobjINVRS.Fields("QtyInStock").Value) - DBLValue(lobjPODetRS.Fields("POPtQtyRcvd").Value) + DBLValue(lobjPODetRS.Fields("POPtQtyOrd").Value)
                                    lobjINVRS.Fields("QtyOnOrder").Value = DBLValue(lobjINVRS.Fields("QtyOnOrder").Value) - DBLValue(lobjPODetRS.Fields("POPtQtybo").Value)
                                    lobjINVRS.Fields("QtyBkOrder").Value = DBLValue(lobjINVRS.Fields("QtyBkOrder").Value) - DBLValue(lobjPODetRS.Fields("POPtQtybo").Value)
                                End If
                            Else
                                If lobjPODetRS.Fields("POPtUM").Value = "EACH" And lobjINVRS.Fields("InvUM").Value = "INCHES" Then
                                    lobjINVRS.Fields("QtyInStock").Value = DBLValue(lobjINVRS.Fields("QtyInStock").Value) + DBLValue(lobjPODetRS.Fields("POPtQtyOrd").Value) * GetLengthFromMain(lobjPODetRS.Fields("POPtNum").Value)
                                    lobjINVRS.Fields("QtyOnOrder").Value = DBLValue(lobjINVRS.Fields("QtyOnOrder").Value) - DBLValue(lobjPODetRS.Fields("POPtQtyOrd").Value) * GetLengthFromMain(lobjPODetRS.Fields("POPtNum").Value)
                                ElseIf lobjPODetRS.Fields("POPtUM").Value = "LFT" Then
                                    lobjINVRS.Fields("QtyInStock").Value = DBLValue(lobjINVRS.Fields("QtyInStock").Value) + DBLValue(lobjPODetRS.Fields("POPtQtyOrd").Value) * 12
                                    lobjINVRS.Fields("QtyOnOrder").Value = DBLValue(lobjINVRS.Fields("QtyOnOrder").Value) - DBLValue(lobjPODetRS.Fields("POPtQtyOrd").Value) * 12
                                Else
                                    lobjINVRS.Fields("QtyInStock").Value = DBLValue(lobjINVRS.Fields("QtyInStock").Value) + DBLValue(lobjPODetRS.Fields("POPtQtyOrd").Value)
                                    lobjINVRS.Fields("QtyOnOrder").Value = DBLValue(lobjINVRS.Fields("QtyOnOrder").Value) - DBLValue(lobjPODetRS.Fields("POPtQtyOrd").Value)
                                End If
                            End If
                        End If
                    
                        lobjINVRS.Update
                    End If
                    If lobjPODetRS.Fields("POPtStatus").Value = "Outstanding" Then
                        lobjPODetRS.Fields("POPtQtyBO").Value = 0

Private Sub SaveData()
    On Error GoTo Error_Handler
    Const cstrErrorSource As String = cstrApplicationName & "." & cstrModuleName & ".SaveData()"
    Dim llngErrNo As Long
    Dim lstrErrDescr As String
    Dim lstrErrSource As String
    llngErrNo = 0
    lstrErrDescr = ""
    lstrErrSource = ""

    Dim lstrSQL As String
    Dim lstrWhereClause As String
    Dim lstrExcludeFieldList As String
    Dim llngAffected As Long
    Dim lobjADORS As ADODB.Recordset
    Dim lobjDistRS As ADODB.Recordset
    Dim lobjShipToRS As ADODB.Recordset
    Dim lstrTableName As String
    Dim lblnInTrans As Boolean
    ' Copy the fields from the screen to an ADO Recordset
    
    lstrTableName = "Distributor"
    Set lobjADORS = BuildRS(lstrTableName)
    
    If Not ValidateData(lobjADORS) Then Exit Sub
    
    CnnBTMS.BeginTrans
    lblnInTrans = True
    If Not lobjADORS Is Nothing Then
        If Not (lobjADORS.BOF And lobjADORS.EOF) Then
            lobjADORS.MoveFirst
            lstrExcludeFieldList = "DistNum, DistEmail, DistFax, DistLogo"
            Do While lobjADORS.EOF = False
                lstrErrDescr = "Error building update statement for Distributor"
                lstrWhereClause = "DistName=" & SQLString(cmbName.Text)
                lstrSQL = Build_Update(lobjADORS, lstrTableName, lstrWhereClause, lstrExcludeFieldList)
                lstrErrDescr = "Error updating Distributor"
                
                Call CnnBTMS.Execute(lstrSQL, 0, adCmdText And adExecuteNoRecords)
                lstrErrDescr = ""
                
                lobjADORS.MoveNext
            Loop
            
            lobjADORS.Close
        End If
    End If
    
    Set lobjADORS = Nothing

    lstrTableName = "ShipTo"
    Set lobjADORS = BuildRS(lstrTableName)
    If Not lobjADORS Is Nothing Then
        If Not (lobjADORS.BOF And lobjADORS.EOF) Then
            lobjADORS.MoveFirst
            lstrExcludeFieldList = "ShipToNum"
            Dim lstrFieldList As String
            lstrFieldList = "ShipToName, ShipToDistrNum, ShipToContact, ShipToStrAddress1, ShipToStrAddress2, ShipToStrAddress3, ShipToCity, ShipToProv, ShipToPostal, ShipToPhone, ShipToExt, ShipToCountry"
            Do While lobjADORS.EOF = False
                lstrErrDescr = "Error building update statement for ShipTo"
                If cboAddress.ListIndex >= 0 Then
                    lstrWhereClause = "ShipToNum=" & cboAddress.ItemData(cboAddress.ListIndex)
                    lstrSQL = Build_Update(lobjADORS, lstrTableName, lstrWhereClause, lstrExcludeFieldList)
                    lstrErrDescr = "Error updating ShipTo"
            
                    Call CnnBTMS.Execute(lstrSQL, 0, adCmdText And adExecuteNoRecords)
                Else
                    lstrSQL = Build_Insert(lobjADORS, lstrTableName, lstrExcludeFieldList)
                    lstrErrDescr = "Error updating ShipTo"
            
                    Call CnnBTMS.Execute(lstrSQL, 0, adCmdText And adExecuteNoRecords)
                    Call LoadNameTo(cmbName.ItemData(cmbName.ListIndex))
                    cboAddress.ListIndex = cboAddress.NewIndex
                End If
                lstrErrDescr = ""
                lobjADORS.MoveNext
            Loop
            
            lobjADORS.Close
        End If
    End If
    
    Set lobjADORS = Nothing
    lstrTableName = "PO"
    Set lobjADORS = BuildRS(lstrTableName)
    ' Process through the recordset and insert or update rows
    If Not lobjADORS Is Nothing Then
        If Not (lobjADORS.BOF And lobjADORS.EOF) Then
            lobjADORS.MoveFirst
            lstrExcludeFieldList = "POID"
            Do While lobjADORS.EOF = False
                If mintMode = MODE_ADD Then
                    lstrErrDescr = "Error building insert statement for " & lstrTableName
                    lstrSQL = Build_Insert(lobjADORS, lstrTableName, lstrExcludeFieldList)
                    lstrErrDescr = "Error inserting data into " & lstrTableName
                    Call CnnBTMS.Execute(lstrSQL, 0, adCmdText And adExecuteNoRecords)
                    
                    mlngPOID = GetID(lstrTableName, "POID", "PONum", Trim(lobjADORS.Fields("PONum").Value))
                Else
                    
                    lstrExcludeFieldList = lstrExcludeFieldList & ",PONum"
                    lstrErrDescr = "Error building update statement for " & lstrTableName
                    lstrWhereClause = "POID=" & mlngPOID
                    lstrSQL = Build_Update(lobjADORS, lstrTableName, lstrWhereClause, lstrExcludeFieldList)
                    lstrErrDescr = "Error updating " & lstrTableName
                
                    Call CnnBTMS.Execute(lstrSQL, 0, adCmdText And adExecuteNoRecords)
                    lstrErrDescr = ""
                End If
                mstrPONum = Trim(lobjADORS.Fields("PONum").Value)
                lobjADORS.MoveNext
            Loop
            
            lobjADORS.Close
        End If
    End If
    
    Set lobjADORS = Nothing
    CnnBTMS.CommitTrans
    lblnInTrans = False
    Call LoadData(False)
    Exit Sub

Error_Handler:
    llngErrNo = IIf(llngErrNo = 0, Err.Number, llngErrNo)
    lstrErrDescr = IIf(Len(lstrErrDescr) = 0, Err.Description, Err.Description)
    lstrErrSource = IIf(Len(Trim(Err.Source)) = 0, "", Err.Source & vbCrLf) & cstrErrorSource
    If lblnInTrans = True Then CnnBTMS.RollbackTrans
    Call Err.Raise(llngErrNo, lstrErrSource, lstrErrDescr)
End Sub

Public Function ParseDocument(wField As Word.Field, WordDoc As Word.Document) As String
    On Error GoTo Error_Handler
    Const cstrErrorSource As String = cstrApplicationName & "." & cstrModuleName & ".ParseDocument()"
    Dim llngErrNo As Long
    Dim lstrErrDescr As String
    Dim lstrErrSource As String
    llngErrNo = 0
    lstrErrDescr = ""
    lstrErrSource = ""

    Dim lstrStopPos As Long
    Dim lstrWord As String
    Dim lstrWordValue As String
    Dim wRange As Word.Range
    
    ' These three lines strip down the field code to find
    ' out it's name
    lstrStopPos = InStrRev(wField.Code, "\*")
    lstrWord = Left(wField.Code, lstrStopPos - 3)
    lstrWord = Right(lstrWord, Len(lstrWord) - 14)
    
    ' Check this field hasn't already appeared in this
    ' document.
    Select Case UCase(lstrWord)
        Case "PONUM":           lstrWordValue = Trim(txtPONum.Text)
        Case "ISSUEDATE":       lstrWordValue = Trim(lblIssued.Caption)
        Case "REQDATE":         lstrWordValue = Trim(txtReqDate.Caption)
        Case "POSTATUS":        lstrWordValue = Trim(lblStatus.Caption)
        Case "PONOTES":         lstrWordValue = Trim(txtNotes.Text)
        Case "FROMNAME":        lstrWordValue = Trim(cmbName.List(cmbName.ListIndex))
        Case "FROMADDR1":       lstrWordValue = Trim(txtAddr1.Text)
        Case "FROMADDR2":       lstrWordValue = Trim(txtAddr2.Text)
        Case "FROMADDR3":       lstrWordValue = Trim(txtAddr3.Text)
        Case "FROMCITY":        lstrWordValue = Trim(txtCity.Text)
        Case "FROMPROV":        lstrWordValue = Trim(txtProv.Text)
        Case "FROMCOUNTRY":     lstrWordValue = Trim(txtCountry.Text)
        Case "FROMCONTACT":     lstrWordValue = Trim(txtContact.Text)
        Case "FROMPHONE":       lstrWordValue = Trim(txtPhone.Text)
        Case "TONAME":          lstrWordValue = Trim(txtName.Text)
        Case "TOADDR1":         lstrWordValue = Trim(cboAddress.Text)
        Case "TOADDR2":         lstrWordValue = Trim(txtAddr2To.Text)
        Case "TOADDR3":         lstrWordValue = Trim(txtAddr3To.Text)
        Case "TOCITY":          lstrWordValue = Trim(txtCityTo.Text)
        Case "TOPROV":          lstrWordValue = Trim(txtProvTo.Text)
        Case "TOCOUNTRY":       lstrWordValue = Trim(txtCountryTo.Text)
        Case "TOCONTACT":       lstrWordValue = Trim(txtContactTo.Text)
        Case "TOPHONE":         lstrWordValue = Trim(txtPhoneTo.Text)
        
        ' I don't simply want to insert a string -
        ' I wish to insert a table at the Product Field.
        Case "PODETAILS"
                                        
            Dim lstrText As String
            Dim lstrSQLQuery As String
            Dim llngAffected As String
            lstrText = ""
            lstrSQLQuery = "select POPtNum, POPtDesc, POPtQtyOrd,"
            lstrSQLQuery = lstrSQLQuery & " POPtMtCost, (POPtMtCost * POPtQtyOrd)"
            lstrSQLQuery = lstrSQLQuery & " from POdetails"
            lstrSQLQuery = lstrSQLQuery & " where POID=" & mlngPOID
            lstrSQLQuery = lstrSQLQuery & " order by POPtNum"
            Dim lobjPODetailRS As ADODB.Recordset
            Set lobjPODetailRS = CnnBTMS.Execute(lstrSQLQuery, llngAffected, adCmdText)
            lstrText = "Part No" & "~" & "Description" & "~" & "Order Qty" & "~" & "Cost" & "~" & "Total Cost" & vbCrLf
            If Not lobjPODetailRS Is Nothing Then
                lobjPODetailRS.ActiveConnection = Nothing
                If Not (lobjPODetailRS.BOF And lobjPODetailRS.EOF) Then
                    lobjPODetailRS.MoveFirst
                    lstrText = lstrText & lobjPODetailRS.GetString(adClipString, -1, "~", vbCrLf)
                End If
            End If
                        
            ' Get the range (location) of the product field
            Set wRange = wField.Code
            ' Delete the field, as any text will be inserted into the
            ' {} of the existing field.
            wField.Delete
            
            ' Enter our table information including headers.
            ' Ideally, I would get this data from an ADO recordset
            ' using GetString().
            With wRange
        
                .Text = lstrText
                .FormattedText.Font.Name = "Arial"
                .FormattedText.Font.Size = "8"
            
                ' Once the data is there, we can convert it to a table
                ' structure and format it to look pretty!
                .ConvertToTable "~", , , 55, wdTableFormatColorful2
                .Columns(1).Width = 110
                .Columns(2).Width = 220
            End With
            
            ' Send back blank string as field does not exist anymore
            lstrWordValue = ""
    
        Case Else
            
            ' Get the value of the field from the user
            lstrWordValue = lstrWord
    
    End Select
    
    ParseDocument = lstrWordValue
    Exit Function

Error_Handler:
    llngErrNo = IIf(llngErrNo = 0, Err.Number, llngErrNo)
    lstrErrDescr = IIf(Len(lstrErrDescr) = 0, Err.Description, Err.Description)
    lstrErrSource = IIf(Len(Trim(Err.Source)) = 0, "", Err.Source & vbCrLf) & cstrErrorSource
    Call Err.Raise(llngErrNo, lstrErrSource, lstrErrDescr)
End Function

Public Function CheckForOustandingItems() As Boolean

    On Error GoTo Error_Handler
    Const cstrErrorSource As String = cstrApplicationName & "." & cstrModuleName & ".CheckForOustandingItems()"
    Dim llngErrNo As Long
    Dim lstrErrDescr As String
    Dim lstrErrSource As String
    llngErrNo = 0
    lstrErrDescr = ""
    lstrErrSource = ""
    Dim lstrSQL As String
    Dim lobjPORS As ADODB.Recordset
    Dim llngAffected As Long
    
    lstrSQL = " select POPtStatus from PODetails where POID=" & mlngPOID
    Set lobjPORS = CnnBTMS.Execute(lstrSQL, llngAffected, adCmdText)
    If Not lobjPORS Is Nothing Then
        lobjPORS.ActiveConnection = Nothing
        If Not (lobjPORS.BOF And lobjPORS.EOF) Then
            lobjPORS.MoveFirst
            lobjPORS.Find "POPtStatus='Outstanding'"
            If lobjPORS.BOF = False And lobjPORS.EOF = False Then
                CheckForOustandingItems = True
            End If
        End If
    End If
    Exit Function

Error_Handler:
    llngErrNo = IIf(llngErrNo = 0, Err.Number, llngErrNo)
    lstrErrDescr = IIf(Len(lstrErrDescr) = 0, Err.Description, Err.Description)
    lstrErrSource = IIf(Len(Trim(Err.Source)) = 0, "", Err.Source & vbCrLf) & cstrErrorSource
    Call Err.Raise(llngErrNo, lstrErrSource, lstrErrDescr)

End Function

        lobjPODetRS.Close
    End If
    Set lobjPODetRS = Nothing
    
    Dim llngNewPOID As Long
    llngNewPOID = GetID("PO", "POID", "PONum", lstrPONum)

    
    ' ==== Copy the PO Details across ====
    lstrTableName = "PODetails"
    
    ' Get empty recordset
    Set lobjADORS = New ADODB.Recordset
    lobjADORS.CursorType = adOpenForwardOnly
    lobjADORS.CursorLocation = adUseClient
    
    lstrSQLQuery = "select * from " & lstrTableName & " where POID=" & lngPOID
    Set lobjPODetRS = CnnBTMS.Execute(lstrSQLQuery, llngAffected, adCmdText)
    If Not lobjPODetRS Is Nothing Then
        lobjPODetRS.ActiveConnection = Nothing
        If Not (lobjPODetRS.BOF And lobjPODetRS.EOF) Then
            lstrSQLQuery = "Select * from " & lstrTableName & " where 1=0"
            lobjADORS.Open lstrSQLQuery, CnnBTMS, adOpenForwardOnly, adLockOptimistic
            Set lobjADORS.ActiveConnection = Nothing
            
            lobjPODetRS.MoveFirst
            Do While lobjPODetRS.EOF = False
                lobjADORS.AddNew
                For Each lobjADOField In lobjPODetRS.Fields
                    If Not UCase(Trim(lobjADOField.Name)) = "POID" Then
                        If IsNull(lobjADOField.Value) = False Then
                            If UCase(Trim(lobjADOField.Name)) = "POPTSTATUS" Then
                                lobjADORS.Fields("POPtStatus").Value = "Active"
                            ElseIf UCase(Trim(lobjADOField.Name)) = "POPTQTYRCVD" Then
                                lobjADORS.Fields(lobjADOField.Name).Value = 0
                            Else
                                lobjADORS.Fields(lobjADOField.Name).Value = lobjADOField.Value
                            End If
                        End If
                        If UCase(Trim(lobjADOField.Name)) = "PONUM" Then
                            lobjADORS.Fields("PONum").Value = lstrPONum
                        End If
                    Else
                        lobjADORS.Fields("POID").Value = llngNewPOID
                    End If
                Next
                
                lobjPODetRS.MoveNext
            Loop
            
            ' Insert the PO or PO child record
            If Not lobjADORS Is Nothing Then
                If Not (lobjADORS.BOF And lobjADORS.EOF) Then
                    lobjADORS.MoveFirst
                    
                    Do While lobjADORS.EOF = False
                        lstrErrDescr = "Error building insert statement for " & lstrTableName
                        lstrSQLQuery = Build_Insert(lobjADORS, lstrTableName, "")
                        lstrErrDescr = "Error inserting data into " & lstrTableName
                        Call CnnBTMS.Execute(lstrSQLQuery, 0, adCmdText And adExecuteNoRecords)
                        lstrErrDescr = ""
                        lobjADORS.MoveNext
                    Loop
                    lobjADORS.Close
                End If
                
                lstrSQLQuery = "update PODetails Set PoPtQtyBO=POPtQtyOrd where POID=" & llngNewPOID
                Call CnnBTMS.Execute(lstrSQLQuery, llngAffected, adCmdText And adExecuteNoRecords)
            End If
            Set lobjADORS = Nothing
            
        End If
        lobjPODetRS.Close
    End If
    Set lobjPODetRS = Nothing
    
    mlngPOID = llngNewPOID
    
    Call UpdateOnPOsForPO(mlngPOID, False)
    CnnBTMS.CommitTrans
    lblnInTrans = False
    Exit Function

Error_Handler:
    llngErrNo = IIf(llngErrNo = 0, Err.Number, llngErrNo)
    lstrErrDescr = IIf(Len(lstrErrDescr) = 0, Err.Description, lstrErrDescr & vbCrLf & Err.Description)
    lstrErrSource = IIf(Len(Trim(Err.Source)) = 0, "", Err.Source & vbCrLf) & cstrErrorSource
    If lblnInTrans = True Then CnnBTMS.RollbackTrans
    Call Err.Raise(llngErrNo, lstrErrSource, lstrErrDescr)
End Function

Private Function CountDetails() As Long
    On Error GoTo Error_Handler
    Const cstrErrorSource As String = cstrApplicationName & "." & cstrModuleName & ".CountDetails()"
    Dim llngErrNo As Long
    Dim lstrErrDescr As String
    Dim lstrErrSource As String
    llngErrNo = 0
    lstrErrDescr = ""
    lstrErrSource = ""

    Dim lstrSQL As String
    Dim llngAffected As Long
    Dim lobjADORS As ADODB.Recordset
    ' Find the number of QuoteDetails
    Dim llngCount As Long

    lstrSQL = "select Count(*) from PODetails where POID=" & mlngPOID
    Set lobjADORS = CnnBTMS.Execute(lstrSQL, llngAffected, adCmdText)
    If Not lobjADORS Is Nothing Then
        lobjADORS.ActiveConnection = Nothing
        If Not (lobjADORS.BOF And lobjADORS.EOF) Then
            lobjADORS.MoveFirst
            llngCount = IIf(IsNull(lobjADORS.Fields(0).Value), 0, DBLValue(lobjADORS.Fields(0).Value))
        End If
        lobjADORS.Close
    End If
    Set lobjADORS = Nothing

    CountDetails = llngCount
    
    Exit Function

Error_Handler:
    llngErrNo = IIf(llngErrNo = 0, Err.Number, llngErrNo)
    lstrErrDescr = IIf(Len(lstrErrDescr) = 0, Err.Description, lstrErrDescr & vbCrLf & Err.Description)
    lstrErrSource = IIf(Len(Trim(Err.Source)) = 0, "", Err.Source & vbCrLf) & cstrErrorSource
    Call Err.Raise(llngErrNo, lstrErrSource, lstrErrDescr)

End Function

Private Function CanUndo(ByVal plngPOID As Long) As Boolean
    On Error GoTo Error_Handler
    Const cstrErrorSource As String = cstrApplicationName & "." & cstrModuleName & ".CanUndo()"
    Dim llngErrNo As Long
    Dim lstrErrDescr As String
    Dim lstrErrSource As String
    llngErrNo = 0
    lstrErrDescr = ""
    lstrErrSource = ""
    
    Dim lstrSQL As String
    Dim lobjRS As ADODB.Recordset
    Dim llngAffected As Long
    
    CanUndo = True
    
    lstrSQL = "select POID, BomAlloc from BoM where POID=" & plngPOID
    lstrSQL = lstrSQL & " and BomAlloc=true union "
    lstrSQL = lstrSQL & " select POID, QuoteAlloc from Quote where POID=" & plngPOID
    lstrSQL = lstrSQL & " and QuoteAlloc=true"
    Set lobjRS = CnnBTMS.Execute(lstrSQL, llngAffected, adCmdText)
    
    If Not lobjRS Is Nothing Then
        If Not (lobjRS.EOF And lobjRS.BOF) Then
            CanUndo = False
        End If
    End If
    Set lobjRS = Nothing
    Exit Function

Error_Handler:
    llngErrNo = IIf(llngErrNo = 0, Err.Number, llngErrNo)
    lstrErrDescr = IIf(Len(lstrErrDescr) = 0, Err.Description, lstrErrDescr & vbCrLf & Err.Description)
    lstrErrSource = IIf(Len(Trim(Err.Source)) = 0, "", Err.Source & vbCrLf) & cstrErrorSource
    Call Err.Raise(llngErrNo, lstrErrSource, lstrErrDescr)

End Function
