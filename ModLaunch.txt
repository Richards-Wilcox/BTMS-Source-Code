Public Function GetDbVersion() As String 'returns version of RWMaster.mdb (blank database) for the 2.0.10 to 2.0.11 upgrade.
    On Error GoTo Error_Handler
    Const cstrErrorSource As String = cstrApplicationName & "." & cstrModuleName & ".GetDbVersion()"
    Dim llngErrNo As Long
    Dim lstrErrDescr As String
    Dim lstrErrSource As String
    llngErrNo = 0
    lstrErrDescr = ""
    lstrErrSource = ""

    Dim llngAffected As Long
    Dim lobjRS As ADODB.Recordset
    Dim cnnTemp As ADODB.Connection
    Set cnnTemp = New ADODB.Connection
    cnnTemp.Provider = "Microsoft.Jet.OLEDB.4.0"
    
    cnnTemp.CursorLocation = adUseClient
    cnnTemp.IsolationLevel = adXactReadCommitted
    cnnTemp.Open App.Path & "\RWMaster.mdb"
    Set lobjRS = cnnTemp.Execute("select version from version", llngAffected, adCmdText)
    If Not lobjRS Is Nothing Then
        lobjRS.MoveFirst
        GetDbVersion = lobjRS.Fields(0).Value
    End If
    Set lobjRS = Nothing
    cnnTemp.Close
    Set cnnTemp = Nothing
    Exit Function

Public Function GetUserLocaleInfo(ByVal dwLocaleID As Long, ByVal dwLCType As Long) As String
    
    On Error GoTo Error_Handler
    Const cstrErrorSource As String = cstrApplicationName & "." & cstrModuleName & ".GetUserLocaleInfo()"
    Dim llngErrNo As Long
    Dim lstrErrDescr As String
    Dim lstrErrSource As String
    llngErrNo = 0
    lstrErrDescr = ""
    lstrErrSource = ""

     Dim sReturn As String
     Dim R As Long
    
    'call the function passing the Locale type
    'variable to retrieve the required size of
    'the string buffer needed
     R = GetLocaleInfo(dwLocaleID, dwLCType, sReturn, Len(sReturn))
      
    'if successful..
     If R Then
      
       'pad the buffer with spaces
        sReturn = Space$(R)
         
       'and call again passing the buffer
        R = GetLocaleInfo(dwLocaleID, dwLCType, sReturn, Len(sReturn))
       
       'if successful (r > 0)
        If R Then
        
          'r holds the size of the string
          'including the terminating null
           GetUserLocaleInfo = Left$(sReturn, R - 1)
        
        End If
       
    End If
     
    Exit Function

Error_Handler:
    llngErrNo = IIf(llngErrNo = 0, Err.Number, llngErrNo)
    lstrErrDescr = IIf(Len(lstrErrDescr) = 0, Err.Description, Err.Description)
    lstrErrSource = IIf(Len(Trim(Err.Source)) = 0, "", Err.Source & vbCrLf) & cstrErrorSource
    Call Err.Raise(llngErrNo, lstrErrSource, lstrErrDescr)
   
End Function

'----------------------------------------------------------
'  FUNCTION Buid_Insert
'----------------------------------------------------------
Public Function Build_Insert(ByRef objADORS As ADODB.Recordset, _
                                ByVal strTableName As String, _
                                Optional ByVal strInsertFields As String = "##") As String
    On Error GoTo Build_Insert_Error
    Const cstrErrorSource As String = cstrApplicationName & "." & cstrModuleName & ".Build_Insert()"
    Dim llngErrNo As Long
    Dim lstrErrDescr As String, lstrErrSource As String
    llngErrNo = 0
    lstrErrDescr = ""
    
    Dim lstrFields As String
    Dim lstrValues As String
    Dim lobjADOField As ADODB.Field
    Dim llngFldCtr As Long
    Dim lstrDataType As String
    
    llngFldCtr = 0
    lstrFields = ""
    lstrValues = ""
    Build_Insert = "Insert into " & strTableName & " ("
    lstrValues = "Values ("
    
    ' Loop through the fields in the recordset and set two strings.
    ' One equal to the fieldnames and the other equal the the values for
    ' each field.
    For Each lobjADOField In objADORS.Fields
        If Not IsNull(lobjADOField.Value) Then
            If InStr(UCase(strInsertFields), UCase(lobjADOField.Name)) = 0 Then
                If llngFldCtr > 0 Then lstrFields = lstrFields & ","
                lstrFields = lstrFields & "[" & lobjADOField.Name & "]"
                
                Select Case lobjADOField.Type
                    Case adBSTR, adChar, adVarChar, adVarWChar, adWChar, adLongVarChar, adLongVarWChar
                                If llngFldCtr > 0 Then lstrValues = lstrValues & ","
                                lstrValues = lstrValues & SQLString(lobjADOField.Value)
                    Case adDate, adDBDate, adDBTime, adDBTimeStamp
                                If llngFldCtr > 0 Then lstrValues = lstrValues & ","
                                lstrValues = lstrValues & SQLString(lobjADOField.Value)
                    Case adBoolean
                                If llngFldCtr > 0 Then lstrValues = lstrValues & ","
                                lstrValues = lstrValues & IIf(lobjADOField.Value = True, "1", "0")
                    Case Else
                                If llngFldCtr > 0 Then lstrValues = lstrValues & ","
                                lstrValues = lstrValues & Replace(lobjADOField.Value, ",", ".")
                End Select
                
                llngFldCtr = llngFldCtr + 1
            End If
        End If
    Next
    
    lstrFields = lstrFields & ") "
    lstrValues = lstrValues & ")"
    Build_Insert = Build_Insert & lstrFields & lstrValues
    
    Set lobjADOField = Nothing
    Exit Function
    
Build_Insert_Error:
    llngErrNo = IIf(llngErrNo = 0, Err.Number, llngErrNo)
    lstrErrDescr = IIf(Len(lstrErrDescr) = 0, Err.Description, Err.Description & vbCrLf & lstrErrDescr)
    Call Err.Raise(llngErrNo, Err.Source & vbCrLf & cstrErrorSource, lstrErrDescr)
End Function

'----------------------------------------------------------
'  FUNCTION Buid_Update
'----------------------------------------------------------
Public Function Build_Update(ByRef objADORS As ADODB.Recordset, _
                                ByVal strTableName As String, _
                                Optional ByVal strWhereClause As String = "", _
                                Optional ByVal strUpdateFields As String = "##") As String
    On Error GoTo Build_Update_Error
    Const cstrErrorSource As String = cstrApplicationName & "." & cstrModuleName & ".Build_Update()"
    Dim llngErrNo As Long
    Dim lstrErrDescr As String, lstrErrSource As String
    llngErrNo = 0
    lstrErrDescr = ""
    
    Dim lstrValues As String
    Dim lobjADOField As ADODB.Field
    Dim llngFldCtr As Long
    Dim lstrDataType As String
    
    llngFldCtr = 0
    lstrValues = "Update " & strTableName & " Set "
    
    ' Loop through the fields in the recordset and build the update string
    For Each lobjADOField In objADORS.Fields
        If Not IsNull(lobjADOField.Value) Then
            If InStr(UCase(strUpdateFields), UCase(lobjADOField.Name)) = 0 Then
                If llngFldCtr > 0 Then lstrValues = lstrValues & ","
                Select Case lobjADOField.Type
                    Case adBSTR, adChar, adVarChar, adVarWChar, adWChar, adLongVarChar, adLongVarWChar
                                lstrValues = lstrValues & "[" & lobjADOField.Name & "]=" & SQLString(IIf(IsNull(lobjADOField.Value), "", Trim(lobjADOField.Value)))
                    Case adDate, adDBDate, adDBTime, adDBTimeStamp
                                lstrValues = lstrValues & "[" & lobjADOField.Name & "]=" & SQLString(lobjADOField.Value)
                    Case adBoolean
                                lstrValues = lstrValues & "[" & lobjADOField.Name & "]=" & IIf(lobjADOField.Value = True, "1", "0")
                    Case Else
                                lstrValues = lstrValues & "[" & lobjADOField.Name & "]=" & Replace(lobjADOField.Value, ",", ".")
                End Select
                llngFldCtr = llngFldCtr + 1
            End If
        Else
            If InStr(UCase(strUpdateFields), UCase(lobjADOField.Name)) = 0 Then
                If lobjADOField.Type = adDate Then
                    If llngFldCtr > 0 Then lstrValues = lstrValues & ","
                    lstrValues = lstrValues & "[" & lobjADOField.Name & "]=" & "Null"
                    llngFldCtr = llngFldCtr + 1
                End If
            End If
        End If
    Next
    
    If llngFldCtr > 0 And Len(Trim(strWhereClause)) > 0 Then
        lstrValues = lstrValues & " Where " & strWhereClause
    End If
    Build_Update = lstrValues
    
    Set lobjADOField = Nothing
    Exit Function
    
Build_Update_Error:
    llngErrNo = IIf(llngErrNo = 0, Err.Number, llngErrNo)
    lstrErrDescr = IIf(Len(lstrErrDescr) = 0, Err.Description, Err.Description & vbCrLf & lstrErrDescr)
    Call Err.Raise(llngErrNo, Err.Source & vbCrLf & cstrErrorSource, lstrErrDescr)
End Function

'----------------------------------------------------------
'  FUNCTION SQLString
'----------------------------------------------------------

Public Function SQLString(ByVal strSQLString As Variant) As String
    On Error GoTo SQLString_Error
    Const cstrErrorSource As String = cstrApplicationName & "." & cstrModuleName & ".SQLString()"
    Dim llngErrNo As Long
    Dim lstrErrDescr As String, lstrErrSource As String
    llngErrNo = 0
    lstrErrDescr = ""
    
    Dim lintCommaOffset As Integer
    Dim lintLastCommaOffset As Integer
                
    ' replace Apostrophes 's with double Apostrophes's so make the
    ' strings SQL friendly
    lintCommaOffset = 1
    lintLastCommaOffset = 1
    SQLString = ""
    If IsNull(strSQLString) Then
        SQLString = "NULL"
    Else
        Do
            lintCommaOffset = InStr(lintLastCommaOffset, strSQLString, "'")
            If lintCommaOffset > 0 Then
                SQLString = SQLString & Mid$(strSQLString, lintLastCommaOffset, lintCommaOffset - lintLastCommaOffset + 1) & "'"
            Else
                SQLString = SQLString & Mid$(strSQLString, lintLastCommaOffset, Len(strSQLString) - lintLastCommaOffset + 1)
            End If
            lintLastCommaOffset = lintCommaOffset + 1
        Loop While lintCommaOffset <> 0
        SQLString = "'" & Trim(SQLString) & "'"
    End If
    Exit Function
    
SQLString_Error:
    llngErrNo = IIf(llngErrNo = 0, Err.Number, llngErrNo)
    lstrErrDescr = IIf(Len(lstrErrDescr) = 0, Err.Description, Err.Description & vbCrLf & lstrErrDescr)
    Call Err.Raise(llngErrNo, Err.Source & vbCrLf & cstrErrorSource, lstrErrDescr)
End Function

Public Function GetNumberOfPartsRequired(ByVal lstrPN As String, _
                                        ByVal lsngTotLen As Single, _
                                        pArrBOM() As Long, _
                                        pArrQuote() As Long, _
                                        Optional ByVal psngPartLen As Single) As Integer
    
    On Error GoTo Error_Handler
    Const cstrErrorSource As String = cstrApplicationName & "." & cstrModuleName & ".GetNumberOfPartsRequired()"
    Dim llngErrNo As Long
    Dim lstrErrDescr As String
    Dim lstrErrSource As String
    llngErrNo = 0
    lstrErrDescr = ""
    lstrErrSource = ""
    
    Dim lintNSprings As Integer, lintNPieces As Integer, lintCounter As Integer
    Dim i As Integer, lArr() As Double, lobjLenRS As ADODB.Recordset
    Dim NBOMs As Integer, NQuotes As Integer, lstrSQL As String
    Dim llngAffected As Long
    
    'get number of documents and save in variables
    NBOMs = IIf(pArrBOM(0) = 0, 0, UBound(pArrBOM()) + 1)
    NQuotes = IIf(pArrQuote(0) = 0, 0, UBound(pArrQuote()) + 1)
    lintCounter = 0
    If NBOMs + NQuotes > 0 Then
        
        'get number of pieces from the Length tables
        For i = 0 To NBOMs - 1
            lstrSQL = "select sum(Qty) from BomPartLenDetails where PartNum=" & SQLString(lstrPN)
            lstrSQL = lstrSQL & " and Docid=" & pArrBOM(i)
            Set lobjLenRS = CnnBTMS.Execute(lstrSQL, llngAffected, adCmdText)
            If IsNull(lobjLenRS.Fields(0).Value) Then
                lintNPieces = lintNPieces + 0
            Else
                lintNPieces = lintNPieces + DBLValue(lobjLenRS.Fields(0).Value)
            End If
        Next
        
        For i = 0 To NQuotes - 1
            lstrSQL = "select sum(Qty) from QuotePartLenDetails where PartNum=" & SQLString(lstrPN)
            lstrSQL = lstrSQL & " and Docid=" & pArrQuote(0)
            Set lobjLenRS = CnnBTMS.Execute(lstrSQL, llngAffected, adCmdText)
            If IsNull(lobjLenRS.Fields(0).Value) Then
                lintNPieces = lintNPieces + 0
            Else
                lintNPieces = lintNPieces + DBLValue(lobjLenRS.Fields(0).Value)
            End If
        Next
        
        lstrSQL = ""
        'select all records from the Bom Length Table that match that Part Number, later add records from Quote Length table.
        If NBOMs > 0 Then
        
            lstrSQL = "select Length,Qty,DocID from BomPartLenDetails where PartNum=" & SQLString(lstrPN)
            lstrSQL = lstrSQL & " and Docid=" & pArrBOM(0)
            
            If NBOMs > 1 Then
                For i = 1 To NBOMs - 1
                    lstrSQL = lstrSQL & " union select Length,Qty,DocID from BomPartLenDetails where PartNum=" & SQLString(lstrPN)
                    lstrSQL = lstrSQL & " and Docid=" & pArrBOM(i)
                Next
            End If
        End If
        
        If NQuotes > 0 Then
            If NBOMs > 0 Then
                lstrSQL = lstrSQL & " union "
            End If
            lstrSQL = lstrSQL & " select Length,Qty,DocID "
            lstrSQL = lstrSQL & " from QuotePartLenDetails where PartNum=" & SQLString(lstrPN)
            lstrSQL = lstrSQL & " and Docid=" & pArrQuote(0)
            If NQuotes > 1 Then
                For i = 1 To NQuotes - 1
                    lstrSQL = lstrSQL & " union select Length,Qty,DocID from QuotePartLenDetails where PartNum=" & SQLString(lstrPN)
                    lstrSQL = lstrSQL & " and Docid=" & pArrQuote(i)
                Next
            End If
        End If
        
        lstrSQL = lstrSQL & " order by length DESC"
        
        Set lobjLenRS = CnnBTMS.Execute(lstrSQL, llngAffected, adCmdText)
    End If
    'start with assigning each little piece to a whole spring
    lintNSprings = lintNPieces '+ IIf(Not IsMissing(psngPartLen) And Not (psngPartLen = 0), 1, 0)
    If lintNSprings = 0 Then Exit Function
    ReDim lArr(lintNSprings - 1)
    
    If lintNSprings > 1 Then
        lobjLenRS.MoveFirst
        
        'transfer the recordset lengths to an array
        Do While lobjLenRS.EOF = False
            For i = 0 To DBLValue(lobjLenRS.Fields("Qty").Value) - 1
                lArr(lintCounter) = lobjLenRS.Fields("Length").Value
                lintCounter = lintCounter + 1
            Next
                
            lobjLenRS.MoveNext
        Loop
        
        If Not IsMissing(psngPartLen) And Not (psngPartLen = 0) Then
            If psngPartLen < lsngTotLen Then
                
                ReDim Preserve lArr(UBound(lArr) + 1)
                lArr(UBound(lArr)) = psngPartLen
                lintNSprings = lintNSprings + 1
            Else
                Do Until psngPartLen <= 0
                    ReDim Preserve lArr(UBound(lArr) + 1)
                    lArr(UBound(lArr)) = IIf(psngPartLen < lsngTotLen, psngPartLen, lsngTotLen)
                    psngPartLen = psngPartLen - lsngTotLen
                    lintNSprings = lintNSprings + 1
                Loop
            End If
        End If
        
        i = 0
        lintCounter = 0
        
        Call SortArray(lArr) 'sort lengths in ascending order
        
        Do While i = 0
            If lintNSprings - lintCounter = 1 Then Exit Do
            If lArr(lintCounter) + lArr(lintNSprings - 1) <= lsngTotLen Then
                lArr(lintCounter) = lArr(lintCounter) + lArr(lintNSprings - 1)
                lintNSprings = lintNSprings - 1
                ReDim Preserve lArr(lintNSprings)
                If lintNSprings - lintCounter = 1 Then i = 1
            Else
                lintCounter = lintCounter + 1
            End If
        Loop
        
    End If
    
    Set lobjLenRS = Nothing
    GetNumberOfPartsRequired = lintNSprings
    
    Exit Function

Error_Handler:
    llngErrNo = IIf(llngErrNo = 0, Err.Number, llngErrNo)
    lstrErrDescr = IIf(Len(lstrErrDescr) = 0, Err.Description, lstrErrDescr & vbCrLf & Err.Description)
    lstrErrSource = IIf(Len(Trim(Err.Source)) = 0, "", Err.Source & vbCrLf) & cstrErrorSource
    Call Err.Raise(llngErrNo, lstrErrSource, lstrErrDescr)

End Function

Private Function SortArray(ByRef ldblArray() As Double)

    On Error GoTo Error_Handler
    Const cstrErrorSource As String = cstrApplicationName & "." & cstrModuleName & ".SortArray()"
    Dim llngErrNo As Long
    Dim lstrErrDescr As String
    Dim lstrErrSource As String
    llngErrNo = 0
    lstrErrDescr = ""
    lstrErrSource = ""

    Dim X1 As Integer
    Dim Y1 As Integer
    Dim ldblTemp As Double
     
    For X1 = 0 To UBound(ldblArray())
        For Y1 = X1 + 1 To UBound(ldblArray())
            If Not (ldblArray(X1) > ldblArray(Y1)) Then
                ldblTemp = ldblArray(X1)
                ldblArray(X1) = ldblArray(Y1)
                ldblArray(Y1) = ldblTemp
             End If
        Next Y1
    Next X1
    
    Exit Function

Error_Handler:
    llngErrNo = IIf(llngErrNo = 0, Err.Number, llngErrNo)
    lstrErrDescr = IIf(Len(lstrErrDescr) = 0, Err.Description, lstrErrDescr & vbCrLf & Err.Description)
    lstrErrSource = IIf(Len(Trim(Err.Source)) = 0, "", Err.Source & vbCrLf) & cstrErrorSource
    Call Err.Raise(llngErrNo, lstrErrSource, lstrErrDescr)

End Function

Public Sub SpellCheck(ByRef varTextBox As Variant)
    On Error GoTo Error_Handler
    Const cstrErrorSource As String = cstrApplicationName & "." & cstrModuleName & ".cmdSpell_Click()"
    Dim llngErrNo As Long
    Dim lstrErrDescr As String
    Dim lstrErrSource As String
    llngErrNo = 0
    lstrErrDescr = ""
    lstrErrSource = ""

    If gblnWordFlag = False Then Exit Sub
    
    Dim lobjSpell As Object
    Dim lstrText As String
    Dim lstrNewText As String
    Dim lstrOldText As String
    Dim lintPos As Integer
            
    Set lobjSpell = CreateObject("Word.Basic")
    
    lobjSpell.FileNew
    lstrOldText = varTextBox.Text
    lobjSpell.Insert lstrOldText
    lobjSpell.ToolsSpelling
    lobjSpell.EditSelectAll
    lstrText = lobjSpell.Selection()
    lobjSpell.FileExit 2

    If Right$(lstrText, 1) = Chr(13) Then _
        lstrText = Left$(lstrText, Len(lstrText) - 1)
    lstrNewText = ""
    lintPos = InStr(lstrText, Chr(13))
    Do While lintPos > 0
        'When a character is Carriage Return (13), replace is with vbCrLf (line feed)
        lstrNewText = lstrNewText & Left$(lstrText, lintPos - 1) & vbCrLf
        lstrText = Right$(lstrText, Len(lstrText) - lintPos)
        lintPos = InStr(lstrText, Chr(13))
    Loop
    lstrNewText = lstrNewText & lstrText
    
    varTextBox.Text = lstrNewText
    If lstrNewText = lstrOldText Then
        MsgBox "Spell Check completed.", vbInformation + vbApplicationModal, "Spell Check"
    End If
    Exit Sub

Error_Handler:
    llngErrNo = IIf(llngErrNo = 0, Err.Number, llngErrNo)
    lstrErrDescr = IIf(Len(lstrErrDescr) = 0, Err.Description, lstrErrDescr & vbCrLf & Err.Description)
    Call DisplayError(llngErrNo, IIf(Len(Trim(Err.Source)) = 0, "", Err.Source & vbCrLf) & cstrErrorSource, lstrErrDescr)
End Sub

Public Function GetImpString(ByVal pintInches As Single) As String

    On Error GoTo Error_Handler
    Const cstrErrorSource As String = cstrApplicationName & "." & cstrModuleName & ".GetImpString()"
    Dim llngErrNo As Long
    Dim lstrErrDescr As String
    Dim lstrErrSource As String
    llngErrNo = 0
    lstrErrDescr = ""
    lstrErrSource = ""

    Dim lsngFeet As Single, lsngFract As Single, lintInch As Integer
    Dim lstrFract As String
    Dim lstrImpString As String
    
    ClearConObj
    aCon.Inch = pintInches
    aCon.Feetcalc
    lsngFeet = aCon.Feet
    lintInch = Fix(pintInches) Mod 12
    lsngFract = pintInches - Fix(pintInches) 'problem, the result is not formatted to fit into getfraction function.
    
    Dim Min As Single
    Dim Max As Single
    Dim Dec As Single
    'to solve the problem:
    'need to get the lower and higher fraction, compare which is closer to the decimal and round up or down accordingly.
    Max = RoundFractionUp(lsngFract - Fix(lsngFract))
    Min = RoundFractionDown(lsngFract - Fix(lsngFract))
    Dec = IIf(Max - (lsngFract - Fix(lsngFract)) > (lsngFract - Fix(lsngFract)) - Min, Min, Max)

    If Dec = 1 Then 'if dec is 1 then add it to inches
        lintInch = lintInch + 1
        lstrFract = ""
    Else
        lstrFract = GetFraction(CStr(Dec))
    End If
    
    lsngFeet = Fix(lsngFeet)
    
    If lintInch > 0 And lintInch Mod 12 = 0 Then 'if inches = 12 then add 1 to feet and set inches to 0
        lsngFeet = lsngFeet + 1
        lintInch = 0
    End If
    
    'that's how it worked before:
    'GetImpString = (IIf(lsngFeet > 0, lsngFeet & Chr(39) & Chr(32), "")) & IIf(lintInch > 0, lintInch & IIf(lstrFract <> "", Chr(32) & lstrFract, "") & Chr(34), IIf(lstrFract <> "", Chr(32) & lstrFract & Chr(34), ""))
    
    'this is new solution
    If lsngFeet > 0 Then
        If lstrFract <> "" Then
            If lintInch > 0 Then 'feet, inches and fraction
                lstrImpString = CStr(lsngFeet) & Chr(39) & Chr(45) & Chr(32) & CStr(lintInch) & Chr(32) & lstrFract & Chr(34)
            Else 'feet, fraction
                lstrImpString = CStr(lsngFeet) & Chr(39) & Chr(45) & Chr(32) & lstrFract & Chr(34)
            End If
        Else 'feet, inches
            lstrImpString = CStr(lsngFeet) & Chr(39) & Chr(45) & CStr(lintInch) & Chr(34)
        End If
    Else
        If lstrFract <> "" And lintInch > 0 Then 'inches and fraction
            lstrImpString = CStr(lintInch) & Chr(32) & lstrFract & Chr(34)
        Else
            If lstrFract = "" Then 'inches
                lstrImpString = CStr(lintInch) & Chr(34)
            Else 'fraction
                lstrImpString = lstrFract & Chr(34)
            End If
        End If
    End If
    
    GetImpString = lstrImpString
    
    Exit Function

Error_Handler:
    llngErrNo = IIf(llngErrNo = 0, Err.Number, llngErrNo)
    lstrErrDescr = IIf(Len(lstrErrDescr) = 0, Err.Description, Err.Description)
    lstrErrSource = IIf(Len(Trim(Err.Source)) = 0, "", Err.Source & vbCrLf) & cstrErrorSource
    Call Err.Raise(llngErrNo, lstrErrSource, lstrErrDescr)

End Function

Public Function ValNumericText(ByVal aTextBox As TextBox, ByVal pintKey As Integer, ByVal pintLen As Integer) As Boolean

    On Error GoTo Error_Handler
    Const cstrErrorSource As String = cstrApplicationName & "." & cstrModuleName & ".ValNumericText()"
    Dim llngErrNo As Long
    Dim lstrErrDescr As String
    Dim lstrErrSource As String
    llngErrNo = 0
    lstrErrDescr = ""
    lstrErrSource = ""
    
    ValNumericText = True
    
    If DecimU = 1 Then
        If Not IsNumeric(Chr(pintKey)) And pintKey <> 46 And pintKey <> 8 Then
            Beep
            ValNumericText = False
            Exit Function
        End If
        If pintKey = 46 Then
            If Not aTextBox.SelLength = Len(aTextBox.Text) Then
                If InStr(1, aTextBox.Text, Chr(46)) <> 0 Then
                    Beep
                    ValNumericText = False
                    Exit Function
                End If
            End If
        End If
    Else
        If Not IsNumeric(Chr(pintKey)) And pintKey <> 44 And pintKey <> 8 Then
            Beep
            ValNumericText = False
            Exit Function
        End If
        If pintKey = 44 Then
            If Not aTextBox.SelLength = Len(aTextBox.Text) Then
                If InStr(1, aTextBox.Text, Chr(44)) <> 0 Then
                    Beep
                    ValNumericText = False
                    Exit Function
                End If
            End If
        End If

    End If
    
    Exit Function

    llngErrNo = IIf(llngErrNo = 0, Err.Number, llngErrNo)

Public Function Dec2Frac(ByVal f As Double) As String

    Dim df As Double
    Dim lUpperPart As Long
    Dim lLowerPart As Long
    
    lUpperPart = 1
    lLowerPart = 1
    
    df = lUpperPart / lLowerPart
    While (df <> f)
       If (df < f) Then
          lUpperPart = lUpperPart + 1
       Else
          lLowerPart = lLowerPart + 1
          lUpperPart = f * lLowerPart
       End If
       df = lUpperPart / lLowerPart
    Wend
    Dec2Frac = CStr(lUpperPart) & "/" & CStr(lLowerPart)

End Function

Public Function ReturnBTMSVersion() As String

    On Error GoTo Error_Handler
    Const cstrErrorSource As String = cstrApplicationName & "." & cstrModuleName & ".ReturnBTMSVersion()"
    Dim llngErrNo As Long
    Dim lstrErrDescr As String
    Dim lstrErrSource As String
    llngErrNo = 0
    lstrErrDescr = ""
    lstrErrSource = ""
    
    Dim vffi As VS_FIXEDFILEINFO  ' version info structure
    Dim buffer() As Byte          ' buffer for version info resource
    Dim pData As Long             ' pointer to version info data
    Dim nDataLen As Long          ' length of info pointed at by pData
    Dim dispstr As String         ' string used to display version information
    Dim retval As Long            ' generic return value
    
    ' First, get the size of the version info resource.
    nDataLen = GetFileVersionInfoSize(App.Path & "\BTMS.exe", pData)
    
    ' Make the buffer large enough to hold the version info resource.
    ReDim buffer(0 To nDataLen - 1) As Byte
    
    ' Get the version information resource.
    retval = GetFileVersionInfo(App.Path & "\BTMS.exe", 0, nDataLen, buffer(0))
    
    ' Get a pointer to a structure that holds a bunch of data.
    retval = VerQueryValue(buffer(0), "\", pData, nDataLen)
    
    ' Copy that structure into the one we can access.
    CopyMemory vffi, ByVal pData, nDataLen
    
    ' holds the full version number of the file.
    dispstr = Trim(Str(HIWORD(vffi.dwFileVersionMS))) & "." & _
        Trim(Str(LOWORD(vffi.dwFileVersionMS))) & "." & _
        Trim(Str(HIWORD(vffi.dwFileVersionLS))) & "." & _
        Trim(Str(LOWORD(vffi.dwFileVersionLS)))
    
    'return the full version number of the file.
    ReturnBTMSVersion = dispstr

    Exit Function

Error_Handler:
    llngErrNo = IIf(llngErrNo = 0, Err.Number, llngErrNo)
    lstrErrDescr = IIf(Len(lstrErrDescr) = 0, Err.Description, lstrErrDescr & vbCrLf & Err.Description)
    lstrErrSource = IIf(Len(Trim(Err.Source)) = 0, "", Err.Source & vbCrLf) & cstrErrorSource
    Call Err.Raise(llngErrNo, lstrErrSource, lstrErrDescr)
 
End Function
