Public Sub SaveFullView()

    On Error GoTo Error_Handler
    Const cstrErrorSource As String = cstrApplicationName & "." & cstrModuleName & ".SaveFullView()"
    Dim llngErrNo As Long
    Dim lstrErrDescr As String
    Dim lstrErrSource As String
    llngErrNo = 0
    lstrErrDescr = ""
    lstrErrSource = ""
    
    Dim lstrLiteType As String
    Dim i As Integer
    
    For i = 0 To IIf(chkSealedUnit.Value = 1, 1, 0)
        
        '1st digit - colour
        '7-anodized
        '8-white
        '9-by others
        If chkByOthers.Value = 1 Then
            lstrLiteType = lstrLiteType & "9"
        Else
            If optWhite.Value = True Then
                lstrLiteType = lstrLiteType & "8"
            ElseIf optAnodized.Value = True Then
                lstrLiteType = lstrLiteType & "7"
            End If
        End If
        
        '2nd digit - size
        '0-1/8
        '1-1/4
        If cboThickness.ItemData(cboThickness.ListIndex) = 1 Or cboThickness.ItemData(cboThickness.ListIndex) = 3 Then
            lstrLiteType = lstrLiteType & "0"
        ElseIf cboThickness.ItemData(cboThickness.ListIndex) = 2 Then
            lstrLiteType = lstrLiteType & "1"
        End If
        
        '3rd digit - material
        '0-glass
        '1-acrylic
        '2-lexan
        '3-other
        lstrLiteType = lstrLiteType & CStr(cboMaterial(i).ListIndex)
        
        '4th digit - finish
        '1-clear
        '2-tinted
        '3-tempered
        '4-other
        '0-n/a (for other than glass)
        If cboMaterial(i).ListIndex = 0 Then
            lstrLiteType = lstrLiteType & CStr(cboFinish(i).ListIndex + 1)
        Else
            lstrLiteType = lstrLiteType & "0"
        End If
    
    Next
    varDoorData.LiteType = CLng(lstrLiteType)
    
    Exit Sub

Error_Handler:
    llngErrNo = IIf(llngErrNo = 0, Err.Number, llngErrNo)
    lstrErrDescr = IIf(Len(lstrErrDescr) = 0, Err.Description, lstrErrDescr & vbCrLf & Err.Description)
    lstrErrSource = IIf(Len(Trim(Err.Source)) = 0, "", Err.Source & vbCrLf) & cstrErrorSource
    Call Err.Raise(llngErrNo, lstrErrSource, lstrErrDescr)

End Sub
