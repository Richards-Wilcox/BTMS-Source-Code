Private Sub Form_Load()
    On Error GoTo Error_Handler
    Const cstrErrorSource As String = cstrApplicationName & "." & cstrModuleName & ".Form_Load()"
    Dim llngErrNo As Long
    Dim lstrErrDescr As String
    Dim lstrErrSource As String
    llngErrNo = 0
    lstrErrDescr = ""
    lstrErrSource = ""
    
    Dim LCID As Long
    Dim currBPP As Long
    
    currBPP = GetDeviceCaps(hdc, BITSPIXEL)
    If currBPP <= 8 Then Call GoGrey(Me)
    LCID = GetUserDefaultLCID()
    
    Label5.Caption = GetUserLocaleInfo(LCID, LOCALE_SSHORTDATE)
    
    If objParent Is Nothing Then Unload Me
    
    Me.Move Screen.Width / 2 - Me.Width / 2, Screen.Height / 2 - Me.Height / 2
    
    Dim i As Integer, Ind As Integer
    For i = 1 To UBound(PriceArr)
        cboPrice.AddItem PriceArr(i)
        Select Case cboPrice.List(i - 1)
            Case "Canadian": cboPrice.ItemData(cboPrice.NewIndex) = 1
            Case "US21": cboPrice.ItemData(cboPrice.NewIndex) = 2
            Case "US22": cboPrice.ItemData(cboPrice.NewIndex) = 3
            Case "US23": cboPrice.ItemData(cboPrice.NewIndex) = 4
            'Mod Hanover
            
            Case "Canadian RVSD": cboPrice.ItemData(cboPrice.NewIndex) = 5
            Case "US21 RVSD": cboPrice.ItemData(cboPrice.NewIndex) = 6
            Case "US22 RVSD": cboPrice.ItemData(cboPrice.NewIndex) = 7
            Case "US23 RVSD": cboPrice.ItemData(cboPrice.NewIndex) = 8
 
            
            
        End Select
        If cboPrice.ItemData(cboPrice.NewIndex) = intCheck Then Ind = i - 1
    Next i
    
    If cboPrice.ListCount > 0 Then cboPrice.ListIndex = Ind
    If cboPrice.ListCount < 2 Then
        lblPrice.Visible = False
        cboPrice.Visible = False
    Else
        lblPrice.Visible = True
        cboPrice.Visible = True
    End If
    
    txtDate.Text = Format(Now() + 14, cstrDateMask)
    Call LoadDist
    Call LoadShip
    LoadHelpText Me
    MousePos1.Init Me
    Exit Sub

Error_Handler:
    llngErrNo = IIf(llngErrNo = 0, Err.Number, llngErrNo)
    lstrErrDescr = IIf(Len(lstrErrDescr) = 0, Err.Description, lstrErrDescr & vbCrLf & Err.Description)
    Call DisplayError(llngErrNo, IIf(Len(Trim(Err.Source)) = 0, "", Err.Source & vbCrLf) & cstrErrorSource, lstrErrDescr)
End Sub

Private Sub LoadDist()
    On Error GoTo Error_Handler
    Const cstrErrorSource As String = cstrApplicationName & "." & cstrModuleName & ".LoadDist()"
    Dim llngErrNo As Long
    Dim lstrErrDescr As String
    Dim lstrErrSource As String
    llngErrNo = 0
    lstrErrDescr = ""
    lstrErrSource = ""

    Dim llngAffected As Long
    Dim lstrSQLQuery As String
    Dim lobjDistRS As ADODB.Recordset
    
    lstrSQLQuery = "select DistNum,DistName from Distributor order by DistName"
    Set lobjDistRS = CnnBTMS.Execute(lstrSQLQuery, llngAffected, adCmdText)
    If Not lobjDistRS Is Nothing Then
        lobjDistRS.ActiveConnection = Nothing
        If Not (lobjDistRS.BOF And lobjDistRS.EOF) Then
            lobjDistRS.MoveFirst
            txtDist.Text = (lobjDistRS.Fields("DistName").Value)

''''''''''Change in .NET''''''''''''''''
            txtDist.Enabled = False
''''''''''END Change in .NET
            
            mlngDistID = lobjDistRS.Fields("DistNum").Value
            lobjDistRS.Close
        End If
    End If
    Set lobjDistRS = Nothing
    
    Exit Sub

Error_Handler:
    llngErrNo = IIf(llngErrNo = 0, Err.Number, llngErrNo)
    lstrErrDescr = IIf(Len(lstrErrDescr) = 0, Err.Description, lstrErrDescr & vbCrLf & Err.Description)
    lstrErrSource = IIf(Len(Trim(Err.Source)) = 0, "", Err.Source & vbCrLf) & cstrErrorSource
    Call Err.Raise(llngErrNo, lstrErrSource, lstrErrDescr)
End Sub

Private Sub LoadShip()
    On Error GoTo Error_Handler
    Const cstrErrorSource As String = cstrApplicationName & "." & cstrModuleName & ".LoadDist()"
    Dim llngErrNo As Long
    Dim lstrErrDescr As String
    Dim lstrErrSource As String
    llngErrNo = 0
    lstrErrDescr = ""
    lstrErrSource = ""

'    If cmbDist.ListIndex < 0 Then Exit Sub
'    If cmbDist.ItemData(cmbDist.ListIndex) < 0 Then Exit Sub
    
    Dim llngAffected As Long
    Dim lstrSQLQuery As String
    Dim lobjShipRS As ADODB.Recordset
    
'    lstrSQLQuery = "select ShipToNum, ShipToName from ShipTo where ShipToDistrNum=" & mlngDistID
    
    lstrSQLQuery = "select ShipToNum, ShipToStrAddress1 from ShipTo where ShipToDistrNum=" & mlngDistID
    lstrSQLQuery = lstrSQLQuery & " Order By ShipToStrAddress1"
    Set lobjShipRS = CnnBTMS.Execute(lstrSQLQuery, llngAffected, adCmdText)
    If Not lobjShipRS Is Nothing Then
        lobjShipRS.ActiveConnection = Nothing
        If Not (lobjShipRS.BOF And lobjShipRS.EOF) Then
            lobjShipRS.MoveFirst
            Do While lobjShipRS.EOF = False
'                Call cmbShip.AddItem(lobjShipRS.Fields("ShipToName").Value)
                Call cmbShip.AddItem(IIf(IsNull(lobjShipRS.Fields("ShipToStrAddress1").Value), "", lobjShipRS.Fields("ShipToStrAddress1").Value))
                cmbShip.ItemData(cmbShip.NewIndex) = lobjShipRS.Fields("ShipToNum").Value
                lobjShipRS.MoveNext
            Loop
            lobjShipRS.Close
        End If
    End If
    Set lobjShipRS = Nothing

    If cmbShip.ListCount = 0 Then
        cmbShip.AddItem ("")
        cmbShip.ItemData(cmbShip.NewIndex) = -1
        cmbShip.ListIndex = cmbShip.NewIndex
    Else
        cmbShip.ListIndex = 0
    End If
    Exit Sub

Error_Handler:
    llngErrNo = IIf(llngErrNo = 0, Err.Number, llngErrNo)
    lstrErrDescr = IIf(Len(lstrErrDescr) = 0, Err.Description, lstrErrDescr & vbCrLf & Err.Description)
    lstrErrSource = IIf(Len(Trim(Err.Source)) = 0, "", Err.Source & vbCrLf) & cstrErrorSource
    Call Err.Raise(llngErrNo, lstrErrSource, lstrErrDescr)
End Sub
