Private Sub Form_Load()

    On Error GoTo Error_Handler
    Const cstrErrorSource As String = cstrApplicationName & "." & cstrModuleName & ".Form_Load()"
    Dim llngErrNo As Long
    Dim lstrErrDescr As String
    Dim lstrErrSource As String
    llngErrNo = 0
    lstrErrDescr = ""
    lstrErrSource = ""

    Dim currBPP As Long
    currBPP = GetDeviceCaps(hdc, BITSPIXEL)
    If currBPP <= 8 Then Call GoGrey(Me)
    
    Me.Move Screen.Width / 2 - Me.Width / 2, Screen.Height / 2 - Me.Height / 2
    
    Call LoadGrid
    LoadHelpText Me
    MousePos1.Init Me
    
    Exit Sub

Error_Handler:
    llngErrNo = IIf(llngErrNo = 0, Err.Number, llngErrNo)
    lstrErrDescr = IIf(Len(lstrErrDescr) = 0, Err.Description, lstrErrDescr & vbCrLf & Err.Description)

Public Sub LoadGrid()

    On Error GoTo Error_Handler
    Const cstrErrorSource As String = cstrApplicationName & "." & cstrModuleName & ".LoadGrid()"
    Dim llngErrNo As Long
    Dim lstrErrDescr As String
    Dim lstrErrSource As String
    llngErrNo = 0
    lstrErrDescr = ""
    lstrErrSource = ""

    Dim llngAffected As Long
    Dim lstrSQLQuery As String
    Dim lobjSectRS As ADODB.Recordset
    DtGrid1.Clear
    
    lstrSQLQuery = "select * from BOMSection BoM where BOMID=" & mlngBOMID
    lstrSQLQuery = lstrSQLQuery & " order by BoMSectionNum"
    Set lobjSectRS = CnnBTMS.Execute(lstrSQLQuery, llngAffected, adCmdText)
    If Not lobjSectRS Is Nothing Then
        lobjSectRS.ActiveConnection = Nothing
        If Not (lobjSectRS.BOF And lobjSectRS.EOF) Then
            DtGrid1.VirtualData = False
            DtGrid1.DataMode = flexDMFree
            Set DtGrid1.DataSource = lobjSectRS
        End If
    End If
    Call SetGridTitles
    Exit Sub

Error_Handler:
    llngErrNo = IIf(llngErrNo = 0, Err.Number, llngErrNo)
    lstrErrDescr = IIf(Len(lstrErrDescr) = 0, Err.Description, lstrErrDescr & vbCrLf & Err.Description)
    lstrErrSource = IIf(Len(Trim(Err.Source)) = 0, "", Err.Source & vbCrLf) & cstrErrorSource
    Call Err.Raise(llngErrNo, lstrErrSource, lstrErrDescr)

End Sub
