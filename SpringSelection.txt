Private Sub cmdAccept_Click()

    On Error GoTo Error_Handler
    Const cstrErrorSource As String = cstrApplicationName & "." & cstrModuleName & ".cmdAccept_Click()"
    Dim llngErrNo As Long
    Dim lstrErrDescr As String
    Dim lstrErrSource As String
    llngErrNo = 0
    lstrErrDescr = ""
    lstrErrSource = ""
    Screen.MousePointer = vbHourglass
    
    Call SaveDrum
    
    Call AcceptSpringSolution
    
    Call ModTracks.checkExtraChainReq
    Screen.MousePointer = vbHourglass
    
    If varDoorData.arrSelectedSolution(SpringIndex).arrSpring(0).PVC > 0 Or varDoorData.arrSelectedSolution(SpringIndex).arrSpring(UBound(varDoorData.arrSelectedSolution(SpringIndex).arrSpring())).PVC > 0 Then Call SelectPVC
    Call Select_Plug_And_Winders
    Call SelectCenterPlatesAndSupports
    
''''''''''Change in .NET''''''''''''''''
'moved from AcceptSpringSolution
    If varDoorData.gblnSpringFailure = False Then Call SelectSpringFastenerPkg
''''''''''END Change in .NET
    
    If varDoorData.AuxillaryShaft = True Then
        Call ModTracks.calShaftLength_AuxillaryShaft
    Else
        Call ModTracks.calShaftLength
    End If
    
    
    
    If LiftType = "Vertical Lift" Or LiftType = "Low Headroom Vertical Lift" Then cValue = ModTracks.GetBracketMountC_VL_VLLHR    're-calculates the 'C' value for D1350-336 drum
    
    If varDoorData.Track.strLowerSplice = "w/ Upper Wall Angle" Then
    
        If Not varDoorData.Track.strMountType = "EU ADCA" Then
            
            Call ModTracks.selectUpperWallAngle_VL_LHRVL
            Call ModTracks.selectUWATrackCips_VL_LHRVL
         Else
         
            Call selectEU_UpperWallAngle_VL_LHRVL
            Call selectEU_UWATrackClips_VL_VLLHR
         End If
               
    ElseIf varDoorData.Track.strLowerSplice = "w/ Upper Bracket Mount" Then
        Call selectUpperBrackets_VL_VLHRVL
    End If
        
    If Not varDoorData.aDrum.strStamping = "D1350-336" And (LiftType = "Vertical Lift" Or LiftType = "Low Headroom Vertical Lift") Then 'execute this routine except D1350-336 drum case
        ModTracks.VLTrackSupport
    End If
     
     
   
    Screen.MousePointer = vbHourglass
    If varDoorData.Track.strJambType = "Wood" Then Call ModTracks.SelectLagBolts
    
    If LiftType = "Vertical Lift" Or LiftType = "Low Headroom Vertical Lift" Then Call ModTracks.SelectTrackPkgs

    Call SelectHeadPlates
     'only select bearing if drum type is not D6375-164
    If varDoorData.aDrum.strStamping <> "D6375-164" Then
        If frmTrackEntry.chkFlangeBearing.Value = 1 Or DrWgt >= 1000 Then varDoorData.FlangeBearing = True
        Call ModTracks.SelectBearings
    End If

    Call SelectHeadPlateFastenerPkgs
    
    If varDoorData.OperatorType = 6 Then
        mblnInFocus = False
        frmDynaHoist.Show
        mblnInFocus = True
    Else
        frmShipInstructions.Show
    End If
    
    Me.Hide
    Screen.MousePointer = vbDefault
    Exit Sub

Error_Handler:
    Screen.MousePointer = vbDefault
    llngErrNo = IIf(llngErrNo = 0, Err.Number, llngErrNo)
    lstrErrDescr = IIf(Len(lstrErrDescr) = 0, Err.Description, Err.Description)
    Call DisplayError(llngErrNo, IIf(Len(Trim(Err.Source)) = 0, "", Err.Source & vbCrLf) & cstrErrorSource, lstrErrDescr)

End Sub

Private Sub cmdBack_Click()

    On Error GoTo Error_Handler
    Const cstrErrorSource As String = cstrApplicationName & "." & cstrModuleName & ".cmdBack_Click()"
    Dim llngErrNo As Long
    Dim lstrErrDescr As String
    Dim lstrErrSource As String
    llngErrNo = 0
    lstrErrDescr = ""
    lstrErrSource = ""
    Dim lblnRoll As Boolean
    
    If mblnSelectionCommit = True Then 'they have a selection picked.
        
        Dim lstrMsg As String, lintBut As Integer, lintRep As Integer
        
        lstrMsg = "Are you sure you want to return to the previous screen?" & vbCrLf & _
            "Your spring selection will be lost."
        lintBut = 4 '(yes,no)
        
        If frmGrid.txtSprNotes.Text <> "" Then
            lstrMsg = "By returning to the previous screen you will loose the spring selection." & vbCrLf & _
                "Do you wish to clear the Springing Notes?"
            lintBut = 3 '(yes,no,cancel)
        End If
        
        lintRep = MsgBox(lstrMsg, vbApplicationModal + vbInformation + lintBut, "Return to the Previous Screen.")
        
        If lintRep = vbCancel Or (lintBut = 4 And lintRep = vbNo) Then Exit Sub
            
        If lintRep = vbYes Then frmGrid.txtSprNotes.Text = ""
    
    Else
        If frmGrid.txtSprNotes.Text <> "" Then
            lstrMsg = "Do you wish to clear the Springing Notes?"
            lintBut = 4 '(yes,no)
        
            lintRep = MsgBox(lstrMsg, vbApplicationModal + vbInformation + lintBut, "Return to the Previous Screen.")
            
            If lintRep = vbYes Then frmGrid.txtSprNotes.Text = ""
                
        End If
    End If
    
    Screen.MousePointer = vbHourglass
    lblnRoll = RollBack("Track", True)

''''''''''Change in .NET''''''''''''''''
    If varDoorData.Track.strWeatherStrip = "Complete Aluminum black vinyl" Then
        Call frmTrackEntry.addItem_WeatherStrip(2)
    ElseIf varDoorData.Track.strWeatherStrip = "Vertical Aluminum black vinyl" Then
        Call frmTrackEntry.addItem_WeatherStrip(3)
    ElseIf varDoorData.Track.strWeatherStrip = "Complete steel white vinyl" Then
        Call frmTrackEntry.addItem_WeatherStrip(4)
    ElseIf varDoorData.Track.strWeatherStrip = "Vertical steel white vinyl" Then
        Call frmTrackEntry.addItem_WeatherStrip(5)
    End If
''''''''''END Change in .NET
    
    lblnRoll = RollBack("Drums", False, "411-227")
    lblnRoll = RollBack("Drums", False, "411-189")
    lblnRoll = RollBack("Drums", False, "411-226")
    mblnSelectionCommit = False

    Unload Me
    frmSprSolutions.Show
    
    Screen.MousePointer = vbDefault
    Exit Sub

Error_Handler:
    Screen.MousePointer = vbDefault
    llngErrNo = IIf(llngErrNo = 0, Err.Number, llngErrNo)
    lstrErrDescr = IIf(Len(lstrErrDescr) = 0, Err.Description, Err.Description)
    Call DisplayError(llngErrNo, IIf(Len(Trim(Err.Source)) = 0, "", Err.Source & vbCrLf) & cstrErrorSource, lstrErrDescr)

End Sub

Public Sub AcceptSpringSolution()

    On Error GoTo Error_Handler
    Const cstrErrorSource As String = cstrApplicationName & "." & cstrModuleName & ".AcceptSpringSolution()"
    Dim llngErrNo As Long
    Dim lstrErrDescr As String
    Dim lstrErrSource As String
    llngErrNo = 0
    lstrErrDescr = ""
    lstrErrSource = ""
    Dim Q As Integer, Qb As Integer, i As Integer
    Dim MetSprLen1 As Single
    Dim MetSprLen2 As Single
    aCon.Inch = varDoorData.arrSelectedSolution(mintInd).arrSpring(0).Length
    aCon.mmcalc
    MetSprLen1 = Round(aCon.mm, 0)
    aCon.Inch = varDoorData.arrSelectedSolution(mintInd).arrSpring(UBound(varDoorData.arrSelectedSolution(mintInd).arrSpring())).Length
    aCon.mmcalc
    MetSprLen2 = Round(aCon.mm, 0)
    
    Call FillSummaryGrid(varDoorData.aDrum.strPN, 1, "Torsion", GetSubCategory(varDoorData.aDrum.strPN), 0)
    Call FillSummaryGrid(Swap(varDoorData.aDrum.strPN, True), 1, "Torsion", GetSubCategory(Swap(varDoorData.aDrum.strPN, True)), 0)
    
    If varDoorData.arrSelectedSolution(mintInd).blnMismatched = False Then
        Q = GetSprQtyPerSide(mintInd, True) 'left wound springs
        
        'auxillary shaft
        If varDoorData.AuxillaryShaft = True Then Q = Q * 2
        
        If Q > 0 Then
            If varDoorData.arrSelectedSolution(mintInd).strSpringWireType = "Square" Then
                Call FillSummaryGrid(varDoorData.arrSelectedSolution(mintInd).arrSpring(0).strPN, Q * varDoorData.arrSelectedSolution(mintInd).arrSpring(0).Length, "Torsion", GetSubCategory(varDoorData.arrSelectedSolution(mintInd).arrSpring(0).strPN), Q * varDoorData.arrSelectedSolution(mintInd).arrSpring(0).Length, "Cut (" & Q & ") to " & CStr(Format(varDoorData.arrSelectedSolution(mintInd).arrSpring(0).Length, "##0.0")) & " coils")

        Grid3.TextMatrix(4, 5) = Format(varDoorData.arrSelectedSolution(mintInd).arrSpring(0).WoundSpringLength, "Fixed") & Chr(34) & Chr(13) & Format(varDoorData.arrSelectedSolution(mintInd).arrSpring(UBound(varDoorData.arrSelectedSolution(mintInd).arrSpring())).WoundSpringLength, "Fixed") & Chr(34)
    Else
        Call ClearConObj
        aCon.Inch = varDoorData.arrSelectedSolution(mintInd).arrSpring(0).WoundSpringLength
        aCon.mmcalc
        lsngWL1 = aCon.mm
        
        Call ClearConObj
        aCon.Inch = varDoorData.arrSelectedSolution(mintInd).arrSpring(UBound(varDoorData.arrSelectedSolution(mintInd).arrSpring())).WoundSpringLength
        aCon.mmcalc
        lsngWL2 = aCon.mm
        
        Grid3.TextMatrix(4, 5) = Round(lsngWL1, 0) & " mm" & Chr(13) & Round(lsngWL2, 0) & " mm"
    End If
    
    'price
    Grid3.TextMatrix(4, 6) = Format(varDoorData.arrSelectedSolution(mintInd).arrSpring(0).Price, "Fixed") & Chr(13) & Format(varDoorData.arrSelectedSolution(mintInd).arrSpring(UBound(varDoorData.arrSelectedSolution(mintInd).arrSpring())).Price, "Fixed")
    
    'line price
    Grid3.TextMatrix(4, 7) = Format((varDoorData.arrSelectedSolution(mintInd).arrSpring(0).Price * (qtyLs + qtyRs)) + (varDoorData.arrSelectedSolution(mintInd).arrSpring(UBound(varDoorData.arrSelectedSolution(mintInd).arrSpring())).Price * (qtyLb + qtyRb)), "Fixed")

    Exit Sub

Error_Handler:
    llngErrNo = IIf(llngErrNo = 0, Err.Number, llngErrNo)
    lstrErrDescr = IIf(Len(lstrErrDescr) = 0, Err.Description, Err.Description)
    lstrErrSource = IIf(Len(Trim(Err.Source)) = 0, "", Err.Source & vbCrLf) & cstrErrorSource
    Call Err.Raise(llngErrNo, lstrErrSource, lstrErrDescr)

End Sub
