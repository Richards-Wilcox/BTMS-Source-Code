Public Sub ShowPic()

    Dim lstrErrSource As String
    llngErrNo = 0
    lstrErrDescr = ""
    lstrErrSource = ""

    Dim a As Single, b As Single

Public Sub stamping()

    On Error GoTo Error_Handler
    Const cstrErrorSource As String = cstrApplicationName & "." & cstrModuleName & ".stamping()"
    Dim llngErrNo As Long
    Dim lstrErrDescr As String
    Dim lstrErrSource As String
    llngErrNo = 0
    lstrErrDescr = ""
    lstrErrSource = ""

    Dim tempTyp As String
    Dim HL As Single
    Dim Q As Integer
    Dim lobjDrumsRS As ADODB.Recordset
    Dim Price As Double, Num As Integer
    Dim MultiU As Single
    Dim lobjMainRS As ADODB.Recordset
    Dim lstrSQL As String
    Dim llngAffected As Long
    
    Set lobjDrumsRS = New ADODB.Recordset
    
    lobjDrumsRS.Open "Drums", CnnBTMS, adOpenKeyset, adLockOptimistic
    
    Q = 0
    
    If LiftType = "Low Headroom Front Mount" Or LiftType = "Low Headroom Rear Mount" Or LiftType = "Standard Lift" Then
        tempTyp = "S"
    End If
    If LiftType = "Low Headroom Vertical Lift" Or LiftType = "Vertical Lift" Then
        tempTyp = "V"
    End If
    If LiftType = "High Lift" Then
        tempTyp = "H"
    End If
    
    lobjDrumsRS.MoveFirst
    Do Until lobjDrumsRS.EOF = True
        If lobjDrumsRS.Fields("Type").Value = tempTyp Then
            If DrWgt < DBLValue(lobjDrumsRS.Fields("MaxDrWgt").Value) Then
                If DrHgt <= DBLValue(lobjDrumsRS.Fields("MaxDrHgt").Value) Then
                    If varDoorData.ShaftDia = DBLValue(lobjDrumsRS.Fields("Shafts").Value) Then
                        If tempTyp = "H" And varDoorData.sngHeadroom > 0 Then
                            Select Case lobjDrumsRS.Fields("Stamping").Value
                                Case "D400-54"
                                    HL = IIf(HWType < 3, varDoorData.sngHeadroom - 7.49, varDoorData.sngHeadroom - 8.553)
                                Case "D525-54"
                                    HL = IIf(HWType < 3, varDoorData.sngHeadroom - 8.844, varDoorData.sngHeadroom - 10.334)
                                Case "D575-120"
                                    HL = IIf(HWType < 3, varDoorData.sngHeadroom - 10.219, varDoorData.sngHeadroom - 11.969)
                                Case "D6375-164"
                                    HL = varDoorData.sngHeadroom - 17.75
                                Case "D800-120"
                                    HL = varDoorData.sngHeadroom - 18.2
                                Case "D850-132"
                                    HL = IIf(LiftType = "Vertical Lift", DrHgt * 2 + 22, DrHgt * 2 + 10)
                                Case "D1100-216"
                                    HL = IIf(LiftType = "Vertical Lift", DrHgt * 2 + 10, DrHgt * 2 + 22)
                                Case Else
                                    HL = 0
                            End Select
                            If HL > DrHgt - 9 Or HL > 164 Then 'set the Hl to the maximum allowed.
                                HL = IIf(DrHgt - 9 < 164, DrHgt - 9, 164)
                            End If
                        Else
                            HL = HILIFT
                        End If
                        If (Not tempTyp = "H") Or (tempTyp = "H" And (DrHgt - HL <= lobjDrumsRS.Fields("MaxDrHgt").Value - lobjDrumsRS.Fields("MaxHilift").Value And HL <= lobjDrumsRS.Fields("MaxHilift").Value And HL >= 17 And HWType >= GetMinHWSizeForStamping(lobjDrumsRS.Fields("Stamping").Value))) Then
                            
                            
                        '-- Mod Hanover  MultId changed to strMultId
                            
                            
                           ' lstrSQL = "select PartNum, G,I,N,T,MultID,SpecialMult from Main where PartNum=" & SQLString(lobjDrumsRS.Fields("Part N").Value)
                            lstrSQL = "select PartNum, G,I,N,T,J,M,S,V," & strMultId & ",SpecialMult from Main where PartNum=" & SQLString(lobjDrumsRS.Fields("Part N").Value)

                            
                            Set lobjMainRS = CnnBTMS.Execute(lstrSQL, llngAffected, adCmdText)
                            If Not lobjMainRS Is Nothing Then
                                If Not (lobjMainRS.EOF And lobjMainRS.BOF) Then
                            
                                    Select Case intCheck
                                    Case 1
                                        Price = Decrypt(GetSym(lobjMainRS.Fields("PartNum").Value), lobjMainRS.Fields("G").Value)
                                    Case 2
                                        Price = Decrypt(GetSym(lobjMainRS.Fields("PartNum").Value), lobjMainRS.Fields("I").Value)
                                    Case 3
                                        Price = Decrypt(GetSym(lobjMainRS.Fields("PartNum").Value), lobjMainRS.Fields("N").Value)
                                    Case 4
                                        Price = Decrypt(GetSym(lobjMainRS.Fields("PartNum").Value), lobjMainRS.Fields("T").Value)
                                    
                                    Case 5
                                        Price = Decrypt(GetSym(lobjMainRS.Fields("PartNum").Value), lobjMainRS.Fields("J").Value)
                                    Case 6
                                        Price = Decrypt(GetSym(lobjMainRS.Fields("PartNum").Value), lobjMainRS.Fields("M").Value)
                                    Case 7
                                        Price = Decrypt(GetSym(lobjMainRS.Fields("PartNum").Value), lobjMainRS.Fields("S").Value)
                                    Case 8
                                        Price = Decrypt(GetSym(lobjMainRS.Fields("PartNum").Value), lobjMainRS.Fields("V").Value)
    
                                    End Select

                                    If lobjMainRS.Fields("SpecialMult").Value <> "0" Then
                                        MultiU = lobjMainRS.Fields("SpecialMult").Value
                                    Else
                                    
                     '-- Mod Hanover  MultId changed to strMultId

                                        Num = lobjMainRS.Fields(strMultId).Value
                                        MultiU = MultipliersCase(Num)
                                    End If
                                    
                                    If MultiU < 1 Then
                                        Price = Price * MultiU
                                    End If
                                    
                                    Set lobjMainRS = Nothing
                                    'load stamping and price into invisible grid for sorting by price
                                    frmSprSolutions.VSFlexGrid1.Rows = Q + 1
                                    frmSprSolutions.VSFlexGrid1.Row = Q
                                    frmSprSolutions.VSFlexGrid1.Col = 0
                                    frmSprSolutions.VSFlexGrid1.Text = lobjDrumsRS.Fields("Stamping")
                                    frmSprSolutions.VSFlexGrid1.Col = 1
                                    frmSprSolutions.VSFlexGrid1.Text = Format(Price, "Fixed")
                                    Q = Q + 1
                                End If
                            End If
                        End If
                    End If
                End If
            End If
        End If
        lobjDrumsRS.MoveNext
        lobjDrumsRS.MoveNext
    Loop
    
    lobjDrumsRS.Close
    Set lobjDrumsRS = Nothing
    
    If frmSprSolutions.VSFlexGrid1.TextMatrix(0, 0) = "" Then
''''''''''Change in .NET''''''''''''''''
'        frmSprSolutions.cboStamping.AddItem ""
'        frmSprSolutions.cboStamping.ListIndex = 0
''''''''''END Change in .NET
        frmSprSolutions.cboStamping.Enabled = False
    Else

        frmSprSolutions.VSFlexGrid1.Select 0, 1, frmSprSolutions.VSFlexGrid1.Rows - 1, 1
        frmSprSolutions.VSFlexGrid1.Sort = flexSortNumericAscending
    
        If frmSprSolutions.VSFlexGrid1.Rows < 1 Then
            Exit Sub
        End If
        
        For Q = 0 To frmSprSolutions.VSFlexGrid1.Rows - 1
            frmSprSolutions.cboStamping.List(Q) = frmSprSolutions.VSFlexGrid1.TextMatrix(Q, 0)
        Next Q
    End If
    
    Exit Sub

Error_Handler:
    llngErrNo = IIf(llngErrNo = 0, Err.Number, llngErrNo)
    lstrErrDescr = IIf(Len(lstrErrDescr) = 0, Err.Description, Err.Description)
    lstrErrSource = IIf(Len(Trim(Err.Source)) = 0, "", Err.Source & vbCrLf) & cstrErrorSource
    Call Err.Raise(llngErrNo, lstrErrSource, lstrErrDescr)

End Sub
Public Function GetDrumPartN(ByVal pstrLeft As String) As String

    On Error GoTo Error_Handler
    Const cstrErrorSource As String = cstrApplicationName & "." & cstrModuleName & ".GetDrumPartN()"
    Dim llngErrNo As Long
    Dim lstrErrDescr As String
    Dim lstrErrSource As String
    Dim lobjDrumsRS As ADODB.Recordset
    Set lobjDrumsRS = New ADODB.Recordset
    lobjDrumsRS.Open "Drums", CnnBTMS, adOpenKeyset, adLockOptimistic
        
    llngErrNo = 0
    lstrErrDescr = ""
    lstrErrSource = ""

    Dim e As Byte
    e = 0
    lobjDrumsRS.MoveFirst
    Do Until e = 1
        If (lobjDrumsRS.Fields("Stamping").Value = varDoorData.aDrum.strStamping) And (pstrLeft = lobjDrumsRS.Fields("Position").Value) Then
            If varDoorData.ShaftDia = lobjDrumsRS.Fields("Shafts").Value Then
                GetDrumPartN = lobjDrumsRS.Fields("Part N")
                e = e + 1
            End If
        End If
        lobjDrumsRS.MoveNext
    Loop
    lobjDrumsRS.Close
    Set lobjDrumsRS = Nothing
    Exit Function

Error_Handler:
    llngErrNo = IIf(llngErrNo = 0, Err.Number, llngErrNo)
    lstrErrDescr = IIf(Len(lstrErrDescr) = 0, Err.Description, Err.Description)
    lstrErrSource = IIf(Len(Trim(Err.Source)) = 0, "", Err.Source & vbCrLf) & cstrErrorSource
    Call Err.Raise(llngErrNo, lstrErrSource, lstrErrDescr)

End Function

    Dim llngErrNo As Long

    Select Case PartN
    Case "950-044", "950-045" ' 400-144 drums
        If LiftType = "Low Headroom Rear Mount" Then
            If HWType = 2 Then
                sngCblLen = Sqr(121 + (DrHgt + 5.781) ^ 2) + DrHgt + 17 + 8

                sngCblLen = Sqr(154 + (DrHgt + 3) ^ 2) + DrHgt + 20.5 + 8
            End If

            If HWType = 2 Then
                sngCblLen = DrHgt + 16.4375
            Else
                sngCblLen = DrHgt + 17.625
            End If

            If HRadius = 12 Then
                sngCblLen = DrHgt + 18.4375
            Else
                If HWType = 2 Then
                    sngCblLen = DrHgt + 21.4375
                Else
                    sngCblLen = DrHgt + 22.5
                End If
            End If
        End If
        PN = "133-021"
        
    Case "773-127", "773-128", "773-137", "773-138" ' 5250-216 drums
        If LiftType = "Low Headroom Rear Mount" Then
            If HWType = 2 Then
                sngCblLen = Sqr(121 + (DrHgt + 5.781) ^ 2) + DrHgt + 17 + 10
            Else
                sngCblLen = Sqr(154 + (DrHgt + 3) ^ 2) + DrHgt + 20.5 + 10
            End If
        ElseIf LiftType = "Low Headroom Front Mount" Then
            If HWType = 2 Then
                sngCblLen = DrHgt + 19
            Else
                sngCblLen = DrHgt + 19.625
            End If
        Else
            If HRadius = 12 Then
                sngCblLen = DrHgt + 21
            Else
                If HWType = 2 Then
                    sngCblLen = DrHgt + 24
                Else
                    sngCblLen = DrHgt + 25.5
                End If
            End If
        End If
        PN = "133-061"
        
    Case "950-048", "950-049", "950-050", "950-051" ' 800-384 drums
        If LiftType = "Low Headroom Rear Mount" Then
            If HWType = 2 Then
                sngCblLen = Sqr(121 + (DrHgt + 5.781) ^ 2) + DrHgt + 17 + 14
            Else
                sngCblLen = Sqr(154 + (DrHgt + 3) ^ 2) + DrHgt + 20.5 + 14
            End If
        ElseIf LiftType = "Low Headroom Front Mount" Then
            If HWType = 2 Then

                sngCblLen = DrHgt + 24.5625

                If HRadius = 12 Then
                    sngCblLen = DrHgt + 25.375
                Else
                    sngCblLen = DrHgt + 28.375
                End If
            Else
                sngCblLen = DrHgt + 30.125

        End If
        PN = "133-081"
        
    Case "950-052", "950-053" ' 400-54 drums

'            sngCblLen = DrHgt + 67.4375
'        Else
'            sngCblLen = DrHgt + 68.5

   Case "950-058", "950-059", "950-060", "950-061" ' 850-132 drums

   Case "773-139", "773-140", "773-129", "773-130"

'

Public Function GetMaxSpringLen(ByVal CID As Single) As Integer
    
    On Error GoTo Error_Handler
    Const cstrErrorSource As String = cstrApplicationName & "." & cstrModuleName & ".GetMaxSpringLen()"
    Dim llngErrNo As Long
    Dim lstrErrDescr As String
    Dim lstrErrSource As String
    llngErrNo = 0
    lstrErrDescr = ""
    lstrErrSource = ""
    
    Dim lstrSQL As String
    Dim lobjMaxLenRS As ADODB.Recordset
    Dim llngAffected As Long
    lstrSQL = "SELECT * FROM MaxLen WHERE CoilID=" & Replace(CID, ",", ".")
    Set lobjMaxLenRS = CnnBTMS.Execute(lstrSQL, llngAffected, adCmdText)
    If Not lobjMaxLenRS Is Nothing Then
        If Not (lobjMaxLenRS.EOF And lobjMaxLenRS.BOF) Then
            If varDoorData.ShaftTube = False Then
                GetMaxSpringLen = Int(lobjMaxLenRS.Fields!Solid)
            Else
                GetMaxSpringLen = Int(lobjMaxLenRS.Fields!tube)
            End If
        End If
        lobjMaxLenRS.Close
    End If
    Set lobjMaxLenRS = Nothing
    Exit Function

Error_Handler:
    llngErrNo = IIf(llngErrNo = 0, Err.Number, llngErrNo)
    lstrErrDescr = IIf(Len(lstrErrDescr) = 0, Err.Description, Err.Description)
    lstrErrSource = IIf(Len(Trim(Err.Source)) = 0, "", Err.Source & vbCrLf) & cstrErrorSource
    Call Err.Raise(llngErrNo, lstrErrSource, lstrErrDescr)

End Function

Public Sub DrumInfoFromTable()

    On Error GoTo Error_Handler
    Const cstrErrorSource As String = cstrApplicationName & "." & cstrModuleName & ".DrumInfoFromTable()"
    Dim llngErrNo As Long
    Dim lstrErrDescr As String
    Dim lstrErrSource As String
    Dim lobjDrumsRS As ADODB.Recordset
    Set lobjDrumsRS = New ADODB.Recordset
    lobjDrumsRS.Open "Drums", CnnBTMS, adOpenKeyset, adLockOptimistic

    llngErrNo = 0
    lstrErrDescr = ""
    lstrErrSource = ""

    Dim Pos1 As Integer, Pos2 As Integer, TAngle As Single, CableDia As Single
    Dim lstrDesc As String, DWUP As Single
    Dim sym As String ' will hold second symbol in Part N
    Dim lstrSQL As String
    Dim llngAffected As Long
    Dim lobjMainRS As ADODB.Recordset
    
    'If Val(frmSectionsOptions.cboSwitches.Text) > 0 Then MRSBWgt = GetMRSBWeight(Val(frmSectionsOptions.cboSwitches.Text))
    TAngle = Sin(SlopHAng * (3.14159265358979 / 180))
    
    lobjDrumsRS.MoveFirst
    Do Until lobjDrumsRS.EOF = True
        If lobjDrumsRS.Fields("Part N").Value = frmSprSolutions.lblPnRight.Caption Then  'drum part number
            sym = GetSym(lobjDrumsRS.Fields("Part N").Value)
            varDoorData.aDrum.sngMaxWeight = lobjDrumsRS.Fields("MaxDrWgt").Value
            If MeasU = 0 Then ' metric
                aCon.KG = 0
                aCon.LB = varDoorData.aDrum.sngMaxWeight
                aCon.KGcalc
                frmSprSolutions.lblDrWght.Caption = Format(aCon.KG, "###0.00") & " kg"
            Else
                frmSprSolutions.lblDrWght.Caption = varDoorData.aDrum.sngMaxWeight & " lb"
            End If
            varDoorData.aDrum.sngMaxHgt = DBLValue(lobjDrumsRS.Fields("MaxDrHgt").Value)
            If MeasU = 0 Then ' metric
                aCon.mm = 0
                aCon.Inch = varDoorData.aDrum.sngMaxHgt
                aCon.mmcalc
                frmSprSolutions.lblDrHgt = Round(aCon.mm, 0)
            Else
                frmSprSolutions.lblDrHgt = varDoorData.aDrum.sngMaxHgt & Chr$(34)
            End If
            varDoorData.sngMaxHiLift = DBLValue(lobjDrumsRS.Fields("maxhilift").Value)
            If HILIFT > 0 Then
                If MeasU = 0 Then ' metric
                    aCon.mm = 0
                    aCon.Inch = varDoorData.sngMaxHiLift
                    aCon.mmcalc
                    frmSprSolutions.lblMaxHiLift.Caption = Round(aCon.mm, 0)
                Else
                    frmSprSolutions.lblMaxHiLift.Caption = varDoorData.sngMaxHiLift & Chr$(34)
                End If
            Else
                frmSprSolutions.lblMaxHiLift.Caption = "N/A"
            End If
            
            varDoorData.aDrum.DrumDia = DBLValue(lobjDrumsRS.Fields("Diam").Value)
            varDoorData.aDrum.RoR = DBLValue(lobjDrumsRS.Fields("RoRise").Value)
            If MeasU = 0 Then ' metric
                aCon.mm = 0
                aCon.Inch = varDoorData.aDrum.DrumDia
                aCon.mmcalc
                frmSprSolutions.lblDia.Caption = Round(aCon.mm, 0)
            Else
                frmSprSolutions.lblDia.Caption = Format(varDoorData.aDrum.DrumDia, "Fixed") & Chr$(34)
            End If
            lstrDesc = lobjDrumsRS.Fields("Description").Value
            Pos1 = InStr(lstrDesc, "LH")
            Pos2 = InStr(lstrDesc, "RH")
            
            If Pos1 <> 0 Then
                lstrDesc = Mid(lstrDesc, 1, Pos1 - 1)
            Else
                If Pos2 <> 0 Then
                    lstrDesc = Mid(lstrDesc, 1, Pos2 - 1)
                End If
            End If
            
            frmSprSolutions.lblDescr.Caption = lstrDesc
            varDoorData.aDrum.strDescr = lstrDesc
            varDoorData.aDrum.Circ = lobjDrumsRS.Fields("Circ")
            CableDia = GetCableDia(lobjDrumsRS.Fields("Stamping"))
            varDoorData.aDrum.FMA = DBLValue(lobjDrumsRS.Fields("FMArm").Value) + CableDia / 2
            
            Select Case LiftType
            Case "Standard Lift"
                Select Case HRadius
                Case 16 'Can be 2" or 3" or Super 3"
                    If HWType = 2 Then    '2"
                        If SlopHAng > 0 Then
                            DWUP = ((18 / DrHgt) + ((DrHgt - 18) / DrHgt) * TAngle) * (DrWgt - MRSBWgt) + MRSBWgt
                        Else
                            DWUP = (18 / DrHgt) * (DrWgt - MRSBWgt) + MRSBWgt
                        End If
                    Else  '3" or Super 3"
                        If SlopHAng > 0 Then
                            DWUP = ((18.625 / DrHgt) + ((DrHgt - 18.625) / DrHgt) * TAngle) * (DrWgt - MRSBWgt) + MRSBWgt
                        Else
                            DWUP = (18.625 / DrHgt) * (DrWgt - MRSBWgt) + MRSBWgt
                        End If
                    End If
                Case 12 'Only 2" track can have a 12" radius
                    If HWType = 2 Then
                        If SlopHAng > 0 Then
                            DWUP = ((15 / DrHgt) + ((DrHgt - 15) / DrHgt) * TAngle) * (DrWgt - MRSBWgt) + MRSBWgt
                        Else
                            DWUP = (15 / DrHgt) * (DrWgt - MRSBWgt) + MRSBWgt
                        End If
                    End If
                End Select
                varDoorData.aDrum.TTDU = DWUP * varDoorData.aDrum.FMA
                varDoorData.aDrum.HMA = DBLValue(lobjDrumsRS.Fields("HMArm").Value) + CableDia / 2
                varDoorData.aDrum.TTDD = DrWgt * varDoorData.aDrum.HMA
                varDoorData.aDrum.REVS = DrHgt / (varDoorData.aDrum.Circ + CableDia * 3.14159265358979)
                varDoorData.aDrum.IPPT = (varDoorData.aDrum.TTDD - varDoorData.aDrum.TTDU) / varDoorData.aDrum.REVS
                
            Case "Low Headroom Front Mount", "Low Headroom Rear Mount"
                Select Case HWType
                Case 2    '2" NOTE: Special Radius - different track required than other lift types
                      'The radius is reduced here to compensate for more of the door being in the radius when in the open position
                    
                    If SlopHAng > 0 Then
                            DWUP = ((9 / DrHgt) + ((DrHgt - 9) / DrHgt) * TAngle) * (DrWgt - MRSBWgt) + MRSBWgt
                    Else
                        DWUP = (9 / DrHgt) * (DrWgt - MRSBWgt) + MRSBWgt
                    End If
                        
'                    DWUP = (11 / DrHgt) * (DrWgt - MRSBWgt - varPassDoor.sngWgt) + MRSBWgt + varPassDoor.sngWgt
                    'DWUP = (9 / DrHgt) * (DrWgt - MRSBWgt) + MRSBWgt   commented in version 2.0.18  - added slop dimensions above
                    
                Case 3, 3.25    '3" or Super 3" NOTE: Special Radius - different track required than other lift types
                      'The radius is reduced here to compensate for more of the door being in the radius when in the open position
                    
                    'DWUP = (9 / DrHgt) * (DrWgt - MRSBWgt) + MRSBWgt  commented in version 2.0.18  - added slop dimensions above
                
                    If SlopHAng > 0 Then
                            DWUP = ((9 / DrHgt) + ((DrHgt - 9) / DrHgt) * TAngle) * (DrWgt - MRSBWgt) + MRSBWgt
                    Else
                        DWUP = (9 / DrHgt) * (DrWgt - MRSBWgt) + MRSBWgt
                    End If
                    
                End Select
                
                varDoorData.aDrum.TTDU = DWUP * varDoorData.aDrum.FMA
                varDoorData.aDrum.HMA = DBLValue(lobjDrumsRS.Fields("HMArm").Value) + CableDia / 2
                varDoorData.aDrum.TTDD = DrWgt * varDoorData.aDrum.HMA
                varDoorData.aDrum.REVS = DrHgt / (varDoorData.aDrum.Circ + CableDia * 3.14159265358979)
                varDoorData.aDrum.IPPT = (varDoorData.aDrum.TTDD - varDoorData.aDrum.TTDU) / varDoorData.aDrum.REVS
            
            Case "High Lift"
                If SlopHAng > 0 Then
                    DWUP = (((HILIFT + 9) / DrHgt) + ((DrHgt - HILIFT + 9) / DrHgt) * TAngle) * (DrWgt - MRSBWgt) + MRSBWgt
                Else
                    If (HILIFT + 9) <= DrHgt Then
                        DWUP = ((HILIFT + 9) / DrHgt) * (DrWgt - MRSBWgt) + MRSBWgt
                    Else
                        DWUP = DrWgt
                    End If
                End If
                varDoorData.aDrum.TTDU = DWUP * varDoorData.aDrum.FMA
                varDoorData.aDrum.HMA = ((varDoorData.aDrum.FMA ^ 2) + (HILIFT * lobjDrumsRS.Fields("RoRise").Value) / 3.14159265358979) ^ 0.5 '+ CableDia / 2
                varDoorData.aDrum.TTDD = DrWgt * varDoorData.aDrum.HMA
                varDoorData.aDrum.REVS = (varDoorData.aDrum.HMA - varDoorData.aDrum.FMA) / lobjDrumsRS.Fields("RoRise").Value + (DrHgt - HILIFT) / (varDoorData.aDrum.Circ + CableDia * 3.14159265358979)
                varDoorData.aDrum.IPPT = (varDoorData.aDrum.TTDD - varDoorData.aDrum.TTDU) / varDoorData.aDrum.REVS
            Case "Vertical Lift", "Low Headroom Vertical Lift"
                If SlopHAng > 0 Then 'if you arrived here through the exclusion offered when using either Standard or High lift style then one of the following formulas will apply.
                    If HILIFT = 0 Then
                        DWUP = (DrWgt - MRSBWgt) * TAngle + MRSBWgt
                    Else
                        DWUP = (((HILIFT + 9) / DrHgt) + ((DrHgt - HILIFT + 9) / DrHgt) * TAngle) * (DrWgt - MRSBWgt) + MRSBWgt
                    End If
                Else
                    DWUP = DrWgt
                End If
             
                varDoorData.aDrum.TTDU = DWUP * varDoorData.aDrum.FMA
                varDoorData.aDrum.HMA = ((varDoorData.aDrum.FMA ^ 2) + ((lobjDrumsRS.Fields("RoRise").Value) * DrHgt / 3.14159265358979)) ^ 0.5 ' + CableDia / 2
                varDoorData.aDrum.RoR = DBLValue(lobjDrumsRS.Fields("RoRise").Value)
                varDoorData.aDrum.TTDD = DrWgt * varDoorData.aDrum.HMA
                varDoorData.aDrum.REVS = (varDoorData.aDrum.HMA - varDoorData.aDrum.FMA) / varDoorData.aDrum.RoR
                varDoorData.aDrum.IPPT = (varDoorData.aDrum.TTDD - varDoorData.aDrum.TTDU) / varDoorData.aDrum.REVS
               
            End Select
            
            '-- Mod Hanover  MultId changed to strMultId

'               lstrSQL = "select PartNum, G,I,N,T,MultID,SpecialMult from Main where PartNum=" & SQLString(lobjDrumsRS.Fields("Part N").Value)
                lstrSQL = "select PartNum, G,I,N,T, J,M,S,V, " & strMultId & ",SpecialMult from Main where PartNum=" & SQLString(lobjDrumsRS.Fields("Part N").Value)
   
            
            Set lobjMainRS = CnnBTMS.Execute(lstrSQL, llngAffected, adCmdText)
            If Not lobjMainRS Is Nothing Then
                If Not (lobjMainRS.EOF And lobjMainRS.BOF) Then
                    Dim ldblPrice As Double, MultiU As Single
                    Select Case intCheck
                        Case 1
                        ldblPrice = Decrypt(sym, DBLValue(lobjMainRS.Fields("G").Value))
                        Case 2
                        ldblPrice = Decrypt(sym, DBLValue(lobjMainRS.Fields("I").Value))
                        Case 3
                        ldblPrice = Decrypt(sym, DBLValue(lobjMainRS.Fields("N").Value))
                        Case 4
                        ldblPrice = Decrypt(sym, DBLValue(lobjMainRS.Fields("T").Value))
                                  
                        Case 5
                        ldblPrice = Decrypt(sym, DBLValue(lobjMainRS.Fields("J").Value))
                        Case 6
                        ldblPrice = Decrypt(sym, DBLValue(lobjMainRS.Fields("M").Value))
                        Case 7
                        ldblPrice = Decrypt(sym, DBLValue(lobjMainRS.Fields("S").Value))
                        Case 8
                        ldblPrice = Decrypt(sym, DBLValue(lobjMainRS.Fields("V").Value))
     
                        
                    End Select

                    Dim Num As Integer
                    If lobjMainRS.Fields("SpecialMult").Value <> "0" Then
                        MultiU = lobjMainRS.Fields("SpecialMult").Value
                    Else
                    
                '-- Mod Hanover  MultId changed to strMultId
                     
                        Num = lobjMainRS.Fields(strMultId).Value
                        MultiU = MultipliersCase(Num)
                    End If
                    If MultiU = 1 Then
                        varDoorData.aDrum.Price = Format(ldblPrice, cstrDollarMask)
                    Else
                        varDoorData.aDrum.Price = Format(ldblPrice * MultiU, cstrDollarMask)
                    End If
                    frmSprSolutions.lblPrice = varDoorData.aDrum.Price
                End If
            End If

''''''''''Change in .NET''''''''''''''''
'            frmSprSolutions.lblRevs = Format(varDoorData.aDrum.REVS, "Fixed")
            
            'Floor to shaft centre
            frmSprSolutions.lblRevs = Format(GetFloorToShaftCenter, "Fixed")

''''''''''END Change in .NET
            
            'IPPT
            frmSprSolutions.lblIPPT = Format(varDoorData.aDrum.IPPT, "Fixed")
            
            'winds
            Winds = varDoorData.aDrum.TTDD / varDoorData.aDrum.IPPT
            
            'springing weight
             If MeasU = 0 Then ' metric
                aCon.KG = 0
                aCon.LB = DrWgt
                aCon.KGcalc
                frmSprSolutions.lblSpringingWeight.Caption = Format(aCon.KG, "Fixed") & " kg"
            Else
                frmSprSolutions.lblSpringingWeight.Caption = Format(DrWgt, "Fixed") & " lb"
            End If
            
            'Min ceiling height
            If MeasU = 0 Then ' metric
                aCon.mm = 0
                aCon.Inch = GetMinCeilingHeight
                aCon.mmcalc
                frmSprSolutions.lblMinCeilingHgt.Caption = Round(aCon.mm, 0) & " mm"
            Else
                frmSprSolutions.lblMinCeilingHgt.Caption = Format(GetMinCeilingHeight, "Fixed") & Chr(34)
            End If
            
            Exit Do
        End If
        lobjDrumsRS.MoveNext
    Loop
    lobjDrumsRS.Close
    Set lobjDrumsRS = Nothing
    Set lobjMainRS = Nothing
    Exit Sub

Error_Handler:
    llngErrNo = IIf(llngErrNo = 0, Err.Number, llngErrNo)
    lstrErrDescr = IIf(Len(lstrErrDescr) = 0, Err.Description, Err.Description)
    lstrErrSource = IIf(Len(Trim(Err.Source)) = 0, "", Err.Source & vbCrLf) & cstrErrorSource
    Call Err.Raise(llngErrNo, lstrErrSource, lstrErrDescr)

End Sub

Public Function Springs(ByVal WireType As String, ByVal clr As Boolean) As Boolean

    On Error GoTo Error_Handler
    Const cstrErrorSource As String = cstrApplicationName & "." & cstrModuleName & ".Springs()"
    Dim llngErrNo As Long
    Dim lstrErrDescr As String
    Dim lstrErrSource As String
    llngErrNo = 0
    lstrErrDescr = ""
    lstrErrSource = ""

    Screen.MousePointer = vbHourglass
    
    Dim lintCLife As Integer, lintCoilIDCnt As Integer
    Dim lsngTempLen As Single
    Dim lintSprQty As Integer
    Dim lsngCLFact As Single
    Dim lsngSMTen As Single
    Dim lsngTorten As Single
    Dim lblTight As Boolean
    Dim i As Integer
    Dim lsngSprSpace As Single, lsngsqfact As Single
    Dim AvailableShaft As Single
    Dim lsngCycleLife As Single
    Dim T As Double
    Dim lsngCoilID As Single
    Dim lsngDeadcoils As Single
    Dim lsngActiveCoils As Single
    Dim lsngWSizeConverted As Single, lsngFAC As Single, lsngMD As Single
    Dim ldblPrice As Double
    Dim lsngMultiU As Single
    Dim lsngTempPrice As Double
    Dim lsngWireSize As Single
    Dim lsngIPPTspring As Single
    Dim lobjWireSizeRS As ADODB.Recordset
    Dim lobjSpringPriceListRS As ADODB.Recordset
    Dim lstrSQL As String
    Dim llngAffected As Long
    Dim DrumW As Single
    
    DrumW = GetDrumWidth(varDoorData.aDrum.strStamping)
    AvailableShaft = GetAvailableShaft(DrumW)
    If clr = True Then SpringIndex = 0
    
    If varDoorData.aDrum.TTDD = 0 Or MaxCycLife = 0 Then
        Screen.MousePointer = vbDefault
        Exit Function
    End If
    
    If StandardVariance = 1 Then
        If frmSprSolutions.cboMin.Text = "10" Then
            MinCycLife = 9.5
        End If
    Else
        If frmSprSolutions.cboMin.Text = "10" Then
            MinCycLife = 10
        End If
    End If
    
    Set lobjWireSizeRS = New ADODB.Recordset
    
    If WireType = "Round" Then
        lstrSQL = "SELECT * FROM WireSize WHERE SqWire=0"
    Else
        lstrSQL = "SELECT * FROM WireSize WHERE SqWire=1"
    End If
    lobjWireSizeRS.CursorType = adOpenForwardOnly
    Set lobjWireSizeRS = CnnBTMS.Execute(lstrSQL, llngAffected, adCmdText)
    
    If Not lobjWireSizeRS Is Nothing Then
        lobjWireSizeRS.ActiveConnection = Nothing
        If Not (lobjWireSizeRS.EOF And lobjWireSizeRS.BOF) Then
            For lintSprQty = 1 To Fix((AvailableShaft - (CouplNum * 6)) \ 15)
                lobjWireSizeRS.MoveFirst
                Do Until lobjWireSizeRS.EOF = True
                    'DoEvents
                    For lintCLife = Int(MinCycLife) To MaxCycLife
                        If lintCLife = 9 Then
                            lsngCLFact = 1 / (1.38498 - (Log(9.5) / 5.9809))
                        Else
                            lsngCLFact = 1 / (1.38498 - (Log(lintCLife) / 5.9809))
                        End If
                        
'                        lsngSMTen = (varDoorData.aDrum.TTDD / lintSprQty) * lsngCLFact  'MIP
                        lsngSMTen = (varDoorData.aDrum.TTDD / lintSprQty)   'MIP per spring - added for 2.0.14 release
                        
                        If varDoorData.gblnSpringFailure = True And lsngSMTen > 1858.5 Then Exit For  'smten will increase when increasing cycle life, but for the next torten, smten drops.
                        If T = DBLValue(lobjWireSizeRS.Fields("TorTen").Value) Then Exit For
                        If DBLValue(lobjWireSizeRS.Fields("TorTen").Value) > lsngSMTen Then
                            
                            Dim startLoopIndex As Integer
                            If varDoorData.ShaftDia = 1.25 Then
                                startLoopIndex = 5
                            Else
                                startLoopIndex = 2
                            End If
                            
                            For lintCoilIDCnt = startLoopIndex To 8
                                If lobjWireSizeRS.Fields(lintCoilIDCnt).Value <> "N/A" Then
                                    T = DBLValue(lobjWireSizeRS.Fields("TorTen").Value)
                                    lsngWireSize = DBLValue(lobjWireSizeRS.Fields("WSize").Value)
                                        
                                    Select Case lintCoilIDCnt
                                    
                                    Case 2
                                        lsngCoilID = 1.75
                                        lsngDeadcoils = 7
                                    Case 3
                                        lsngCoilID = 2
                                        lsngDeadcoils = 7
                                    Case 4
                                        lsngCoilID = 2.625
                                        lsngDeadcoils = 7
                                    Case 5
                                        lsngCoilID = 3.625
                                        lsngDeadcoils = 1
                                    Case 6
                                        lsngCoilID = 3.75
                                        lsngDeadcoils = 5
                                    Case 7
                                        If lobjWireSizeRS.Fields("SqWire").Value = 1 And lobjWireSizeRS.Fields("WSize").Value > 0.375 Then
                                            lsngDeadcoils = 1
                                        Else
                                            lsngDeadcoils = 5
                                        End If
                                        lsngCoilID = 5.25
                                    Case 8
                                        lsngCoilID = 6
                                        lsngDeadcoils = 5
                                    End Select
                                    lsngTorten = DBLValue(lobjWireSizeRS.Fields("TorTen").Value)
                                    
                                    lsngCycleLife = Exp(8.2834 - 5.9809 * (varDoorData.aDrum.TTDD / (lsngTorten * lintSprQty)))
                                    If MaxCycLife >= lsngCycleLife Then
                                        If MinCycLife <= lsngCycleLife Then
                                            
                                            lsngWSizeConverted = lsngWireSize
                                            If WireType = "Round" Then
                                                lsngFAC = 3.14159265358979 / 32
                                            End If
                                            If WireType = "Square" Then
                                                lsngFAC = 1 / 6
                                                If lsngWireSize = 0.3125 Then
                                                    lsngWSizeConverted = 0.3050536 '0.30442 '0.3031
                                                ElseIf lsngWireSize = 0.34375 Then
                                                    lsngWSizeConverted = 0.3345314 '0.33906 '0.3379
                                                ElseIf lsngWireSize = 0.375 Then 'added after 2.0.13 + torten field for sq.wire needs also be updated
                                                    lsngWSizeConverted = 0.361006 '0.3707  '0.360442
                                                ElseIf lsngWireSize = 0.40625 Then
                                                    lsngWSizeConverted = 0.3960648 '0.40228
                                                ElseIf lsngWireSize = 0.4375 Then
                                                    lsngWSizeConverted = 0.4255254 '0.43382
                                                End If
                                            End If
                                            lsngMD = lsngCoilID + lsngWireSize
                                            
                                            lsngActiveCoils = (30000000 * (lsngWSizeConverted ^ 4) * lsngFAC) / ((varDoorData.aDrum.IPPT / lintSprQty) * lsngMD)
                                            lsngActiveCoils = Round(lsngActiveCoils, 1)
                                            
                                            If WireType = "Round" Then
                                                lsngTempLen = (lsngActiveCoils + lsngDeadcoils) * lsngWireSize
                                            Else
                                                lsngsqfact = GetSquareFactor(lsngWireSize, lsngCoilID)
                                                lsngTempLen = (lsngActiveCoils + lsngDeadcoils) * lsngWireSize * lsngsqfact
                                            End If
                                            
                                            If frmSprSolutions.chkMaxLen.Value = 0 Then
                                                MaxSpringLength = AvailableShaft / lintSprQty
                                            Else
                                                MaxSpringLength = GetMaxSpringLen(lsngCoilID)
                                            End If
                                            
                                            If lsngTempLen <= MaxSpringLength And lsngTempLen > 15 Then
                                                'actual space taken up by the springs
                                                
                                                If WireType = "Round" Then
                                                    lsngSprSpace = lintSprQty * ((lsngActiveCoils + lsngDeadcoils + Winds) * lsngWireSize + 6) '(3 for plugs and centerplate, 3 for key)
                                                Else
                                                    
                                                    lsngSprSpace = lintSprQty * ((lsngActiveCoils + lsngDeadcoils + Winds) * lsngWireSize * lsngsqfact + 6) '(3 for plugs and centerplate, 3 for key)
                                                End If
                                                If lsngSprSpace <= AvailableShaft - (CouplNum * 6) Then
                                                    lblTight = True
                                                    If lsngSprSpace <= AvailableShaft - (CouplNum * 8) Then lblTight = False
                                                    Dim X As Integer, lblnSel As Boolean
                                                    lblnSel = False
                                                    If Not isEmptySelectedSpringSolutionArray Then 'do not add already selected solutions
                                                        For X = 0 To UBound(varDoorData.arrSelectedSolution)
                                                            If varDoorData.arrSelectedSolution(X).blnMismatched = False Then
                                                                If varDoorData.arrSelectedSolution(X).strDrumStamping = varDoorData.aDrum.strStamping Then
                                                                    'For y = 0 To UBound(varDoorData.arrSelectedSolution(x).arrSpring)
                                                                        If varDoorData.arrSelectedSolution(X).arrSpring(0).strPN = lobjWireSizeRS.Fields(lintCoilIDCnt).Value Then
                                                                            If UBound(varDoorData.arrSelectedSolution(X).arrSpring) + 1 = lintSprQty Then
                                                                                lblnSel = True
                                                                                'Exit For
                                                                            End If
                                                                        End If
                                                                    'Next
                                                                End If
                                                            End If
                                                        Next
                                                    End If
                                                    If lblnSel = False Then
                                                    
                                                        Springs = True
                                                        Dim sym As String
    
                                                        sym = GetSym(lobjWireSizeRS.Fields(lintCoilIDCnt).Value)
                                                        
                                                        'lstrSQL = "SELECT * FROM SpringPriceList WHERE [Part N]='" & lobjWireSizeRS.Fields(lintCoilIDCnt).Value & "'"
                                                        
                                         '             lstrSQL = "SELECT * FROM Main WHERE PartNum=" & SQLString(lobjWireSizeRS.Fields(lintCoilIDCnt).Value)
                                                        lstrSQL = "SELECT * FROM Main WHERE PartNum=" & SQLString(lobjWireSizeRS.Fields(lintCoilIDCnt).Value)
                                                        Set lobjSpringPriceListRS = CnnBTMS.Execute(lstrSQL, llngAffected, adCmdText)
                                                        
                                                        lobjSpringPriceListRS.MoveFirst
                                                        lobjSpringPriceListRS.Find "PartNum=" & SQLString(lobjWireSizeRS.Fields(lintCoilIDCnt).Value)

Public Function Swap(ByVal strSwap As String, ByVal chk As Boolean) As String

    On Error GoTo Error_Handler
    Const cstrErrorSource As String = cstrApplicationName & "." & cstrModuleName & ".Swap()"
    Dim llngErrNo As Long
    Dim lstrErrDescr As String
    Dim lstrErrSource As String
    llngErrNo = 0
    lstrErrDescr = ""
    lstrErrSource = ""

    Dim aa As Integer
    Dim a As String
    Dim b As String
    If chk = True Then
        aa = InStr(strSwap, Chr$(45))
        If aa <> 0 Then
            a = Mid(strSwap, aa + 1)
            b = Mid(strSwap, 1, aa)
            a = Val(a) + 1
            If Len(a) = 2 Then a = "0" & a
            Swap = b & a
        End If
    End If
    If chk = False Then
        aa = InStr(strSwap, Chr$(45))
        If aa <> 0 Then
            a = Mid(strSwap, aa + 1)
            b = Mid(strSwap, 1, aa)
            a = Val(a) - 1
            If Len(a) = 2 Then a = "0" & a
            Swap = b & a
        End If
    End If

    Exit Function

Error_Handler:
    llngErrNo = IIf(llngErrNo = 0, Err.Number, llngErrNo)
    lstrErrDescr = IIf(Len(lstrErrDescr) = 0, Err.Description, Err.Description)
    lstrErrSource = IIf(Len(Trim(Err.Source)) = 0, "", Err.Source & vbCrLf) & cstrErrorSource
    Call Err.Raise(llngErrNo, lstrErrSource, lstrErrDescr)

End Function

                                                                varDoorData.arrSpringSolution(SpringIndex).arrSpring(i).Turns = Winds 'added for 2.0.14 - could have been added into varDoorData.arrSpringSolution() but we used varDoorData.arrSpringSolution().arrSpring() to save (just like in standard solution)

    lstrErrSource = ""

Public Sub SelectPVC()

    On Error GoTo Error_Handler
    Const cstrErrorSource As String = cstrApplicationName & "." & cstrModuleName & ".SelectPVC()"
    Dim llngErrNo As Long
    Dim lstrErrDescr As String
    Dim lstrErrSource As String
    llngErrNo = 0
    lstrErrDescr = ""
    lstrErrSource = ""
    
    Dim lsngLen As Single, lsngMetLen As Single
    Dim lsngLenM As Single, lsngMetLenM As Single
    Dim QtySm As Integer, QtyBig As Integer
    
    'small length and big length
    lsngLen = varDoorData.arrSelectedSolution(SpringIndex).arrSpring(0).PVC
    lsngLenM = varDoorData.arrSelectedSolution(SpringIndex).arrSpring(UBound(varDoorData.arrSelectedSolution(SpringIndex).arrSpring())).PVC
    'round to 1/16"
    lsngLen = RoundFraction(lsngLen - Fix(lsngLen)) + Fix(lsngLen)
    lsngLenM = RoundFraction(lsngLenM - Fix(lsngLenM)) + Fix(lsngLenM)
    
    
    QtySm = GetSelectedSprQtyPerGauge(SpringIndex, varDoorData.arrSelectedSolution(SpringIndex).arrSpring(0).WireSize)
    QtyBig = GetSelectedSprQtyPerGauge(SpringIndex, varDoorData.arrSelectedSolution(SpringIndex).arrSpring(UBound(varDoorData.arrSelectedSolution(SpringIndex).arrSpring())).WireSize)
    
    
     'auxillary shaft - double the QTY (in-house feature only that's why imperial measurements are considered)
    If varDoorData.AuxillaryShaft = True Then
        QtySm = QtySm * 2
        QtyBig = QtyBig * 2
    End If
    
    If MeasU = 0 Then
        aCon.Inch = lsngLen
        aCon.mmcalc
        lsngMetLen = aCon.mm
        aCon.Inch = lsngLenM
        aCon.mmcalc
        lsngMetLenM = aCon.mm
    End If
    
    If varDoorData.arrSelectedSolution(SpringIndex).blnMismatched Then
        If MeasU = 1 Then
            Call FillSummaryGrid("328-227", _
                QtySm * lsngLen + QtyBig * lsngLenM, "Torsion", _
                GetSubCategory("328-227"), QtySm * lsngLen + QtyBig * lsngLenM, _
                "Cut (" & QtySm & ") to " & _
                GetImpString(lsngLen) & " (" & QtyBig & _
                ") to " & GetImpString(lsngLenM))
        Else
            Call FillSummaryGrid("328-227", _
                QtySm * lsngMetLen + QtyBig * lsngMetLenM, "Torsion", _
                GetSubCategory("328-227"), QtySm * lsngLen + QtyBig * lsngLenM, _
                "Cut (" & QtySm & ") to " & _
                Format(lsngMetLen, "Fixed") & " (" & QtyBig & _
                ") to " & Round(lsngMetLenM, 0) & " mm")
        End If
    Else
        If MeasU = 1 Then
            Call FillSummaryGrid("328-227", QtySm * lsngLen, "Torsion", _
                GetSubCategory("328-227"), QtySm * lsngLen, _
                "Cut (" & QtySm & ") to " & GetImpString(lsngLen))
        Else
            Call FillSummaryGrid("328-227", QtySm * lsngMetLen, "Torsion", _
                GetSubCategory("328-227"), QtySm * lsngLen, _
                "Cut (" & QtySm & ") to " & Round(lsngMetLen, 0) & " mm")
        End If
    End If
    
    Exit Sub

Error_Handler:
    llngErrNo = IIf(llngErrNo = 0, Err.Number, llngErrNo)
    lstrErrDescr = IIf(Len(lstrErrDescr) = 0, Err.Description, lstrErrDescr & vbCrLf & Err.Description)
    lstrErrSource = IIf(Len(Trim(Err.Source)) = 0, "", Err.Source & vbCrLf) & cstrErrorSource
    Call Err.Raise(llngErrNo, lstrErrSource, lstrErrDescr)

End Sub

    Dim lstrErrDescr As String
    Dim lstrErrSource As String
    llngErrNo = 0
    lstrErrDescr = ""
    lstrErrSource = ""

    Dim lstrSQL As String
    Dim lobjSpringRS As ADODB.Recordset
    
    lstrSQL = "select Square, Round from Main where PartNum= " & SQLString(pstrPN)
    Set lobjSpringRS = CnnBTMS.Execute(lstrSQL, 0, adCmdText)
    If Not lobjSpringRS Is Nothing Then
        If Not (lobjSpringRS.EOF And lobjSpringRS.BOF) Then
            If DBLValue(lobjSpringRS.Fields("Square").Value) > 0 Then
                IsSquare = True

            End If
        End If
    End If

    On Error GoTo Error_Handler
    Const cstrErrorSource As String = cstrApplicationName & "." & cstrModuleName & ".SortSpringsArray()"
    Dim llngErrNo As Long
    Dim lstrErrDescr As String
    Dim lstrErrSource As String
    llngErrNo = 0
    lstrErrDescr = ""
    lstrErrSource = ""
    
    Dim X1 As Integer 'existing array index before sort
    Dim Y1 As Integer 'new array index, array being created by sort
    Dim ldblTemp As Double
    
    Dim lArrSolution() As SpringSolutionData
    Dim lArrSpring() As SpringData
    ReDim lArrSolution(UBound(varDoorData.arrSpringSolution))
    
    'sort by price. (all other criteria are fullfilled by the Springs routine, the only sort left
    ' is by price)
    
    For X1 = 0 To UBound(varDoorData.arrSpringSolution()) 'evaluate one element vs. element + 1
        For Y1 = X1 + 1 To UBound(varDoorData.arrSpringSolution())
            If (varDoorData.arrSpringSolution(X1).arrSpring(0).WireSize = varDoorData.arrSpringSolution(Y1).arrSpring(0).WireSize) Then
                If Not (varDoorData.arrSpringSolution(X1).arrSpring(0).Price <= varDoorData.arrSpringSolution(Y1).arrSpring(0).Price) Then
                    'using temporary array swap elements.
                    lArrSolution(X1) = varDoorData.arrSpringSolution(X1)
                    varDoorData.arrSpringSolution(X1) = varDoorData.arrSpringSolution(Y1)
                    varDoorData.arrSpringSolution(Y1) = lArrSolution(X1)
                End If
             End If
        Next Y1
    Next X1
    
    Exit Sub

Error_Handler:
