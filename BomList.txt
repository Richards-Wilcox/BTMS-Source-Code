Private Sub cmdDelete_Click() 'delete BoM
    On Error GoTo Error_Handler
    Const cstrErrorSource As String = cstrApplicationName & "." & cstrModuleName & ".cmdDelete_Click()"
    Dim llngErrNo As Long
    Dim lstrErrDescr As String
    Dim lstrErrSource As String
    llngErrNo = 0
    lstrErrDescr = ""
    lstrErrSource = ""

    Dim llngRow As Long
    Dim llngBOMID As Long, llngPOID As Long, llngQuoteID As Long
    Dim llngAffected As Long
    Dim lblnAlloc As Boolean
    Dim lstrSQLQuery As String

    llngRow = grdData.RowSel
    llngBOMID = DBLValue(grdData.TextMatrix(llngRow, 10))
    llngPOID = DBLValue(grdData.TextMatrix(llngRow, 11))
    llngQuoteID = DBLValue(grdData.TextMatrix(llngRow, 12))
    If grdData.TextMatrix(llngRow, 15) <> "" Then lblnAlloc = CBool(grdData.TextMatrix(llngRow, 15))
    
    If llngBOMID > 0 Then
        If llngPOID + llngQuoteID > 0 Then
            Call MsgBox("A Quote or PO exists for this BoM", vbApplicationModal + vbOKOnly + vbExclamation, "Cannot delete BoM: " & LTrim(grdData.TextMatrix(llngRow, 1)))
            Exit Sub
        Else
            If lblnAlloc = True Then
                If MsgBox("This BoM has inventory allocated." & vbCrLf & "Are you sure you wish to delete this BoM?", vbApplicationModal + vbYesNo + vbExclamation, "Delete BoM: " & LTrim(grdData.TextMatrix(llngRow, 1))) = vbNo Then
                    Exit Sub
                Else
                    lstrSQLQuery = "delete * from " & cstrTableName & " where BOMID=" & llngBOMID
                    Call CnnBTMS.Execute(lstrSQLQuery, llngAffected, adCmdText And adExecuteNoRecords)
                    Call LoadData
                End If
            Else
                If grdData.TextMatrix(llngRow, 14) <> "" Then
                    If MsgBox("The items on this BoM are reserved in the inventory. Deleting this BoM will un-reserve those parts." & vbCrLf & _
                        "Do you wish to delete this BoM?", vbQuestion + vbApplicationModal + vbYesNo, "Delete BoM") = vbYes Then
                            Call UnAllocateBOM(DBLValue(grdData.TextMatrix(llngRow, 10)))
                            
                            lstrSQLQuery = "delete * from " & cstrTableName & " where BOMAlloc=0 and BOMID=" & llngBOMID
                            Call CnnBTMS.Execute(lstrSQLQuery, llngAffected, adCmdText And adExecuteNoRecords)
                            Call LoadData
                    End If
                Else
                    If MsgBox("Are you sure you wish to delete BoM " & LTrim(grdData.TextMatrix(llngRow, 1)), vbApplicationModal + vbYesNo + vbExclamation, "Delete BoM") = vbYes Then
                        lstrSQLQuery = "delete * from " & cstrTableName & " where BOMAlloc=0 and BOMID=" & llngBOMID
                        Call CnnBTMS.Execute(lstrSQLQuery, llngAffected, adCmdText And adExecuteNoRecords)
                        Call LoadData
                    End If
                End If
                
            End If
        End If
    End If
    Exit Sub

Error_Handler:
    llngErrNo = IIf(llngErrNo = 0, Err.Number, llngErrNo)
    lstrErrDescr = IIf(Len(lstrErrDescr) = 0, Err.Description, Err.Description)
    Call DisplayError(llngErrNo, IIf(Len(Trim(Err.Source)) = 0, "", Err.Source & vbCrLf) & cstrErrorSource, lstrErrDescr)
End Sub

Private Sub LoadData()
    On Error GoTo Error_Handler
    Const cstrErrorSource As String = cstrApplicationName & "." & cstrModuleName & ".LoadData()"
    Dim llngErrNo As Long
    Dim lstrErrDescr As String
    Dim lstrErrSource As String
    llngErrNo = 0
    lstrErrDescr = ""
    lstrErrSource = ""

    Dim llngAffected As Long
    Dim llngRecords As Long
    Dim lstrSQLQuery As String
    
    grdData.Clear
    llngRecords = 0
    lstrSQLQuery = "Select " & cstrSelectList & " from " & cstrTableName
    lstrSQLQuery = lstrSQLQuery & " order by BomNum "
    Set mobjADORS = CnnBTMS.Execute(lstrSQLQuery, llngAffected, adCmdText)
    If Not mobjADORS Is Nothing Then

    fraGrid.Caption = " " & llngRecords & " Records "
    
    Call SetGridTitles
    Exit Sub

    llngErrNo = IIf(llngErrNo = 0, Err.Number, llngErrNo)

    Call Err.Raise(llngErrNo, lstrErrSource, lstrErrDescr)

Private Sub grdData_BeforeSort(ByVal Col As Long, Order As Integer)
    On Error GoTo Error_Handler
    Const cstrErrorSource As String = cstrApplicationName & "." & cstrModuleName & ".grdData_BeforeSort()"
    Dim llngErrNo As Long
    Dim lstrErrDescr As String
    Dim lstrErrSource As String
    llngErrNo = 0
    lstrErrDescr = ""
    lstrErrSource = ""
   
    ' no flags? apply custom sort
    If grdData.ExplorerBar > &H1000& Then Exit Sub

    ' save selection
    Dim R&, c&, rs&, cs&
    grdData.GetSelection R, c, rs, cs
    grdData.Redraw = flexRDNone

    ' apply sort to non-empty range
    Dim Row%
    For Row = grdData.Rows - 1 To grdData.FixedRows Step -1
        If Len(grdData.TextMatrix(Row, Col)) Then Exit For
    Next
    If Row > grdData.FixedRows Then
        Dim llngCol As Long
        llngCol = Col
        If Col = 14 Then llngCol = 1
        grdData.Select grdData.FixedRows, llngCol, Row, llngCol
        grdData.Sort = Order
    End If

    ' restore selection
    grdData.Select R, c, rs, cs
    grdData.Redraw = flexRDDirect

    ' cancel default sort
    Order = 0
    Exit Sub

Error_Handler:
    llngErrNo = IIf(llngErrNo = 0, Err.Number, llngErrNo)
    lstrErrDescr = IIf(Len(lstrErrDescr) = 0, Err.Description, Err.Description)
    lstrErrSource = IIf(Len(Trim(Err.Source)) = 0, "", Err.Source & vbCrLf) & cstrErrorSource
    Call Err.Raise(llngErrNo, lstrErrSource, lstrErrDescr)
End Sub

Public Function ParseDocument(wField As Object, WordDoc As Object) As String
    On Error GoTo Error_Handler
    Const cstrErrorSource As String = cstrApplicationName & "." & cstrModuleName & ".GetNewResult()"
    Dim llngErrNo As Long
    Dim lstrErrDescr As String
    Dim lstrErrSource As String
    llngErrNo = 0
    lstrErrDescr = ""
    lstrErrSource = ""
    
    Dim lstrStopPos As Long
    Dim lstrWord As String
    Dim lstrText As String
    Dim lstrWordValue As String
    Dim wRange As Object
    
    ' These three lines strip down the field code to find
    ' out it's name
    lstrStopPos = InStrRev(wField.Code, "\*")
    lstrWord = Left(wField.Code, lstrStopPos - 3)
    lstrWord = Right(lstrWord, Len(lstrWord) - 14)
    Set wRange = WordDoc.Range
    
    ' Check this field hasn't already appeared in this
    ' document.
    Select Case UCase(lstrWord)
        Case "RPTTITLE":         lstrWordValue = "BoM Listing"
            
        ' I don't simply want to insert a string -
        ' I wish to insert a table at the Product Field.
        Case "RPTLIST", "RPTHEADER"
                            
                lstrText = "BoM No" & "~" & "Model" & "~" & "Type" & "~" & "Width" & "~" & "Height" & "~" & "Qty" & vbCrLf
                If UCase(lstrWord) = "RPTLIST" Then
                    Dim lstrSQLQuery As String
                    Dim llngAffected As String
                    lstrSQLQuery = "select BOMNum, BOMDrModel, BOMDrType, BOMDrWidth, BOMDrHeight, BOMDrQty"
                    lstrSQLQuery = lstrSQLQuery & " from BoM"
                    lstrSQLQuery = lstrSQLQuery & " order by BOMNum"
                    Dim lobjBOMRS As ADODB.Recordset
                    Set lobjBOMRS = CnnBTMS.Execute(lstrSQLQuery, llngAffected, adCmdText)
                    
                    lstrText = " ~ ~ ~ ~ ~ ~ ~ " & vbCrLf
                    If Not lobjBOMRS Is Nothing Then
                        lobjBOMRS.ActiveConnection = Nothing
                        If Not (lobjBOMRS.BOF And lobjBOMRS.EOF) Then
                            lstrText = ""
                            lobjBOMRS.MoveFirst
                            lstrText = lobjBOMRS.GetString(adClipString, -1, "~", vbCrLf)
                            lobjBOMRS.Close
                        End If
                    End If
                    Set lobjBOMRS = Nothing
                End If
                
                ' Get the range (location) of the product field
                Set wRange = wField.Code
                ' Delete the field, as any text will be inserted into the
                ' {} of the existing field.
                wField.Delete
                
                ' Enter our table information including headers.
                ' Ideally, I would get this data from an ADO recordset
                ' using GetString().
                With wRange
            
                    .Text = lstrText
                    .FormattedText.Font.Name = "Arial"
                    .FormattedText.Font.Size = "8"
                
                    ' Once the data is there, we can convert it to a table
                    ' structure and format it to look pretty!
                    If UCase(lstrWord) = "RPTLIST" Then
                        .ConvertToTable "~", , , 70, wdTableFormatColorful2, , , , , 0
                    Else
                        .ConvertToTable "~", , , 70, wdTableFormatColorful2
                    End If
                                                   
                    .Columns(1).Width = 110
                    .Columns(2).Width = 100
                    .Columns(3).Width = 100
                End With
                
                ' Send back blank string as field does not exist anymore
                lstrWordValue = ""
        
        Case Else
            
            ' Get the value of the field from the user
            lstrWordValue = lstrWord
    
    End Select
    
    ParseDocument = lstrWordValue
    Exit Function

Error_Handler:
    llngErrNo = IIf(llngErrNo = 0, Err.Number, llngErrNo)
    lstrErrDescr = IIf(Len(lstrErrDescr) = 0, Err.Description, Err.Description)
    lstrErrSource = IIf(Len(Trim(Err.Source)) = 0, "", Err.Source & vbCrLf) & cstrErrorSource
    Call Err.Raise(llngErrNo, lstrErrSource, lstrErrDescr)
End Function
