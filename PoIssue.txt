Private Sub cmdPrint_Click()

    On Error GoTo Error_Handler
    Const cstrErrorSource As String = cstrApplicationName & "." & cstrModuleName & ".cmdPrint_Click()"
    Dim llngErrNo As Long
    Dim lstrErrDescr As String
    Dim lstrErrSource As String
    llngErrNo = 0
    lstrErrDescr = ""
    lstrErrSource = ""
    Dim lblnChkIssue As Boolean
    
''''''''''Change in .NET''''''''''''''''
    If Not IsDate(Trim(txtDate.Text)) Then
     
        MsgBox "Invalid Date.", vbApplicationModal + vbCritical, "Invalid Date"
        If txtDate.Enabled = True Then txtDate.SetFocus
        Exit Sub
    Else
        If DateValue(Now()) > DateValue(Me.txtDate.Text) Then
            MsgBox "Invalid Date.", vbApplicationModal + vbCritical, "Invalid Date"
            If txtDate.Enabled = True Then txtDate.SetFocus
            Exit Sub
        End If
    End If
''''''''''END Change in .NET
    
    MsgBox "This action will issue this Purchase Order." & vbCrLf & _
        "Please print the document and send to Richards-Wilcox", vbApplicationModal, "Issue PO"
    CnnBTMS.BeginTrans
    Call ChangePOStatus(mlngPOID, "Issued", Me.txtDate.Text)
    If IsInvMaint = 1 And CheckPORunInventory(mlngPOID) = True Then
        Call UpdateOnOrder(mlngPOID)
    End If
    Call UpdateOnPOsForPO(mlngPOID, True)
    If IsInvMaint = 1 And CheckPORunInventory(mlngPOID) = True Then
        If chkStock.Value = 1 Then
            If MsgBox("This will permanently change your minimum quantities and restocking levels in your inventory for all items included in this PO." & vbCrLf & _
                "Are sure you want to set restocking levels for these items?", vbExclamation + vbYesNo + vbApplicationModal, "Set Restocking Levels.") = vbNo Then
                    Me.chkStock.Value = 0
                    MsgBox "The PO was issued without setting restocking levels.", vbInformation + vbApplicationModal, "PO Issue"
            Else
                Call SetRestockingLevels(mlngPOID)
            End If
        End If
    End If
    CnnBTMS.CommitTrans
    lblnChkIssue = True
    frmPrint.mstrReportName = "PO.rpt"
    frmPrint.mstrTitle = "Purchase Order"
    frmPrint.mlngIDValue = mlngPOID

    frmPrint.mstrComputerID = mstrComputerID

    frmPrint.Show vbModal
    Unload Me
    
    Exit Sub
Error_Handler:
    
    If lblnChkIssue = True Then
        MsgBox "The PO was Issued but a printer error ocurred." & vbCrLf & _
        "Make sure Richards Wilcox receives a copy of this Issued PO.", vbExclamation + vbApplicationModal, "Printer Error"
    Else
        CnnBTMS.RollbackTrans
        MsgBox "The PO was NOT Issued." & vbCrLf & _
        "Unexpected error ocurred.", vbExclamation + vbApplicationModal, "Error"
    End If
    llngErrNo = IIf(llngErrNo = 0, Err.Number, llngErrNo)
    lstrErrDescr = IIf(Len(lstrErrDescr) = 0, Err.Description, Err.Description)
    Call DisplayError(llngErrNo, IIf(Len(Trim(Err.Source)) = 0, "", Err.Source & vbCrLf) & cstrErrorSource, lstrErrDescr)

End Sub
