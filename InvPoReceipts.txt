Private Sub LoadData()
    On Error GoTo Error_Handler
    Const cstrErrorSource As String = cstrApplicationName & "." & cstrModuleName & ".LoadData()"
    Dim llngErrNo As Long
    Dim lstrErrDescr As String
    Dim lstrErrSource As String
    llngErrNo = 0
    lstrErrDescr = ""
    lstrErrSource = ""

    Dim llngAffected As Long
    Dim llngRecords As Long
    Dim lstrSQLQuery As String
    
    Screen.MousePointer = vbDefault
    grdData.Clear
    mintMode = MODE_VIEW
    
    ' Get the PO Header Info
    Dim lobjPORS As ADODB.Recordset
    If Not mlngPOID > 0 Then Exit Sub
    lstrSQLQuery = " select * from PO where POID=" & mlngPOID
    Set lobjPORS = CnnBTMS.Execute(lstrSQLQuery, llngAffected, adCmdText)
    If Not lobjPORS Is Nothing Then
        lobjPORS.ActiveConnection = Nothing
        If Not (lobjPORS.BOF And lobjPORS.EOF) Then
            lobjPORS.MoveFirst
            Dim lstrFieldName As String
            lstrFieldName = "POID"
            mlngPOID = IIf(IsNull(lobjPORS.Fields(lstrFieldName).Value), 0, lobjPORS.Fields(lstrFieldName).Value)
            lstrFieldName = "PONum"
            lblPONum.Caption = IIf(IsNull(lobjPORS.Fields(lstrFieldName).Value), "", Trim(lobjPORS.Fields(lstrFieldName).Value))
            lstrFieldName = "POIssueDate"
            lblPOIssuedDate.Caption = IIf(IsNull(lobjPORS.Fields(lstrFieldName).Value), Format(Now(), "Short Date"), lobjPORS.Fields(lstrFieldName).Value)
            lstrFieldName = "POStatus"
            lblStatus.Caption = IIf(IsNull(lobjPORS.Fields("POStatus").Value), "", (lobjPORS.Fields("POStatus").Value))
        End If
        lobjPORS.Close
    End If
    Set lobjPORS = Nothing
    
    'Get the PODetail Info
    llngRecords = 0
    lstrSQLQuery = "Select " & cstrSelectList & " from " & cstrTableName & " where POID=" & mlngPOID & " order by POPtNum"
    Set mobjADORS = CnnBTMS.Execute(lstrSQLQuery, llngAffected, adCmdText)
    If Not mobjADORS Is Nothing Then
        mobjADORS.ActiveConnection = Nothing
        If Not (mobjADORS.BOF And mobjADORS.EOF) Then
            llngRecords = mobjADORS.RecordCount
            grdData.VirtualData = False
            grdData.DataMode = flexDMFree
            Set grdData.DataSource = mobjADORS
        End If
    End If
    
    Call SetGridTitles
    fraGrid.Caption = " " & llngRecords & " Records "
    Call SetControls
    
    Exit Sub

Error_Handler:
    llngErrNo = IIf(llngErrNo = 0, Err.Number, llngErrNo)
    lstrErrDescr = IIf(Len(lstrErrDescr) = 0, Err.Description, Err.Description)
    lstrErrSource = IIf(Len(Trim(Err.Source)) = 0, "", Err.Source & vbCrLf) & cstrErrorSource
    Call Err.Raise(llngErrNo, lstrErrSource, lstrErrDescr)
End Sub

Private Sub grdData_AfterEdit(ByVal Row As Long, ByVal Col As Long)
    On Error GoTo Error_Handler
    Const cstrErrorSource As String = cstrApplicationName & "." & cstrModuleName & ".grdData_AfterEdit()"
    Dim llngErrNo As Long
    Dim lstrErrDescr As String
    Dim lstrErrSource As String
    llngErrNo = 0
    lstrErrDescr = ""
    lstrErrSource = ""
    
    Dim llngRow As Long
    Dim ldblRecvdNow As Double, ldblToBeRecvd As Double
    
    If Col = 5 And Row = grdData.RowSel Then
        grdData.TextMatrix(Row, Col) = DBLValue(grdData.TextMatrix(Row, Col))
        If grdData.TextMatrix(Row, Col) <> "" Then
            ldblRecvdNow = DBLValue(grdData.TextMatrix(Row, Col)) + DBLValue(grdData.TextMatrix(Row, 6))
            ldblToBeRecvd = DBLValue(grdData.TextMatrix(Row, 3)) '- (DBLValue(GrdData.TextMatrix(Row, 6)) + DBLValue(GrdData.TextMatrix(Row, 5)))
            
            If DBLValue(Format(ldblRecvdNow, "fixed")) = DBLValue(Format(ldblToBeRecvd, "fixed")) Then
'                Dim ldblOutst As Double
'                ldblOutst = GrdData.TextMatrix(Row, 4)
                grdData.TextMatrix(Row, 4) = DBLValue(grdData.TextMatrix(Row, 3)) - (DBLValue(grdData.TextMatrix(Row, 6)) + DBLValue(grdData.TextMatrix(Row, 5)))
                'GrdData.TextMatrix(Row, 5) = ldblOutst
                If DBLValue(grdData.TextMatrix(Row, 4)) < 0 Then grdData.TextMatrix(Row, 4) = 0
                If DBLValue(grdData.TextMatrix(Row, 4)) = 0 Then grdData.TextMatrix(Row, 8) = "Complete"

            ElseIf DBLValue(Format(ldblRecvdNow, "fixed")) < DBLValue(Format(ldblToBeRecvd, "fixed")) Then
                
                grdData.TextMatrix(Row, 4) = DBLValue(grdData.TextMatrix(Row, 3)) - DBLValue((grdData.TextMatrix(Row, 5)) + DBLValue(grdData.TextMatrix(Row, 6)))
                If DBLValue(grdData.TextMatrix(Row, 4)) < 0 Then grdData.TextMatrix(Row, 4) = 0
                If DBLValue(grdData.TextMatrix(Row, 4)) = 0 Then grdData.TextMatrix(Row, 8) = "Complete"
            Else 'If DBLValue(GrdData.TextMatrix(Row, Col)) > DBLValue(GrdData.TextMatrix(Row, 3)) Then
                MsgBox "You cannot receive more than you ordered.", vbExclamation + vbApplicationModal, "Invalid Entry"
                grdData.TextMatrix(Row, Col) = 0
                grdData.Col = 5
                grdData.Row = Row
                Exit Sub
            End If
        End If
    End If
    
    Exit Sub

Error_Handler:
    llngErrNo = IIf(llngErrNo = 0, Err.Number, llngErrNo)
    lstrErrDescr = IIf(Len(lstrErrDescr) = 0, Err.Description, Err.Description)
    Call DisplayError(llngErrNo, IIf(Len(Trim(Err.Source)) = 0, "", Err.Source & vbCrLf) & cstrErrorSource, lstrErrDescr)
End Sub

Private Sub grdData_BeforeSort(ByVal Col As Long, Order As Integer)
    On Error GoTo Error_Handler
    Const cstrErrorSource As String = cstrApplicationName & "." & cstrModuleName & ".grdData_BeforeSort()"
    Dim llngErrNo As Long
    Dim lstrErrDescr As String
    Dim lstrErrSource As String
    llngErrNo = 0
    lstrErrDescr = ""
    lstrErrSource = ""
    
    ' no flags? apply custom sort
    If grdData.ExplorerBar > &H1000& Then Exit Sub

    ' save selection
    Dim R&, c&, rs&, cs&
    grdData.GetSelection R, c, rs, cs
    grdData.Redraw = flexRDNone

    ' apply sort to non-empty range
    Dim Row As Integer
    For Row = grdData.Rows - 1 To grdData.FixedRows Step -1
        If Len(Trim(grdData.TextMatrix(Row, Col))) Then Exit For
    Next
    
    If Row > grdData.FixedRows Then
        Dim llngCol As Long
        llngCol = Col
        If Col = 3 Then llngCol = 1
        grdData.Select grdData.FixedRows, llngCol, Row, llngCol
        grdData.Sort = Order
    End If

    ' restore selection
    grdData.Select R, c, rs, cs
    grdData.Redraw = flexRDDirect

    ' cancel default sort
    Order = 0
    Exit Sub

Error_Handler:
    llngErrNo = IIf(llngErrNo = 0, Err.Number, llngErrNo)
    lstrErrDescr = IIf(Len(lstrErrDescr) = 0, Err.Description, Err.Description)
    lstrErrSource = IIf(Len(Trim(Err.Source)) = 0, "", Err.Source & vbCrLf) & cstrErrorSource
    Call Err.Raise(llngErrNo, lstrErrSource, lstrErrDescr)
End Sub

Private Sub grdData_Click()
    On Error GoTo Error_Handler
    Const cstrErrorSource As String = cstrApplicationName & "." & cstrModuleName & ".grdData_click()"
    Dim llngErrNo As Long
    Dim lstrErrDescr As String
    Dim lstrErrSource As String
    llngErrNo = 0
    lstrErrDescr = ""
    lstrErrSource = ""
    
    Dim llngRow As Long
    Dim llngCol As Long
    
    If mintMode = MODE_EDIT Then
        If mlngRow > 0 And mlngCol = 8 Then
            llngRow = grdData.RowSel
            If grdData.TextMatrix(llngRow, 8) = "Cancelled" Then
                grdData.TextMatrix(llngRow, 8) = "Outstanding"
            ElseIf grdData.Text = "Outstanding" Then
                If DBLValue(grdData.TextMatrix(llngRow, 6)) + DBLValue(grdData.TextMatrix(llngRow, 5)) < DBLValue(grdData.TextMatrix(llngRow, 3)) Then
                    grdData.TextMatrix(llngRow, 8) = "Cancelled"
                End If
            ElseIf grdData.TextMatrix(llngRow, 8) = "Complete" Then
                'GrdData.TextMatrix(llngRow, 6) = 0
                grdData.TextMatrix(llngRow, 4) = DBLValue(grdData.TextMatrix(llngRow, 3)) - DBLValue(grdData.TextMatrix(llngRow, 6))
                grdData.TextMatrix(llngRow, 5) = 0
                grdData.TextMatrix(llngRow, 8) = "Outstanding"
            End If
        End If
    End If
    Exit Sub

Error_Handler:

Private Sub grdData_DblClick()
    On Error GoTo Error_Handler
    Const cstrErrorSource As String = cstrApplicationName & "." & cstrModuleName & ".grdData_DblClick()"
    Dim llngErrNo As Long
    Dim lstrErrDescr As String
    Dim lstrErrSource As String
    llngErrNo = 0
    lstrErrDescr = ""
    lstrErrSource = ""
    
    If mintMode = MODE_EDIT And mlngRow = 0 Then
        Dim llngRow As Long
        Dim ldblNewValue As Double
        
        Select Case mlngCol
            Case 5  ' Received Qty
                    If MsgBox("Do you wish to do this?", vbApplicationModal + vbExclamation + vbYesNo, "This will receive all outstanding stock.") = vbYes Then
                        For llngRow = 1 To grdData.Rows - 1
                            If Trim(grdData.TextMatrix(llngRow, 1)) = "" Then Exit For
                            If UCase(Trim(grdData.TextMatrix(llngRow, 8))) = "OUTSTANDING" Then
                                If grdData.TextMatrix(llngRow, 5) <> grdData.TextMatrix(llngRow, 3) Then
                                    ldblNewValue = DBLValue(grdData.TextMatrix(llngRow, 3)) - (DBLValue(grdData.TextMatrix(llngRow, 5)) + DBLValue(grdData.TextMatrix(llngRow, 6)))
                                    If ldblNewValue < 0 Then ldblNewValue = 0
                                    
                                    grdData.TextMatrix(llngRow, 5) = ldblNewValue
                                    grdData.TextMatrix(llngRow, 4) = 0
                                End If
                            End If
                        Next llngRow
                    End If
                    
            Case 7  ' Status
                    If MsgBox("Do you wish to do this?", vbApplicationModal + vbExclamation + vbYesNo, "This will cancel all outstanding stock.") = vbYes Then
                        For llngRow = 1 To grdData.Rows - 1
                            If Trim(grdData.TextMatrix(llngRow, 1)) = "" Then Exit For
                            If UCase(Trim(grdData.TextMatrix(llngRow, 8))) = "OUTSTANDING" Then
                                grdData.TextMatrix(llngRow, 8) = "Cancelled"
                                ldblNewValue = DBLValue(grdData.TextMatrix(llngRow, 3)) - DBLValue(grdData.TextMatrix(llngRow, 6))
                                If ldblNewValue < 0 Then ldblNewValue = 0
                                grdData.TextMatrix(llngRow, 5) = 0
                                grdData.TextMatrix(llngRow, 4) = ldblNewValue
                            End If
                        Next llngRow
                    End If
        End Select
    End If
    
    Exit Sub

Error_Handler:
    llngErrNo = IIf(llngErrNo = 0, Err.Number, llngErrNo)
    lstrErrDescr = IIf(Len(lstrErrDescr) = 0, Err.Description, Err.Description)
    Screen.MousePointer = vbDefault
    CnnBTMS.RollbackTrans
    Call DisplayError(llngErrNo, IIf(Len(Trim(Err.Source)) = 0, "", Err.Source & vbCrLf) & cstrErrorSource, lstrErrDescr)

End Sub

Private Sub SaveData()
    On Error GoTo Error_Handler
    Const cstrErrorSource As String = cstrApplicationName & "." & cstrModuleName & ".SaveData()"
    Dim llngErrNo As Long
    Dim lstrErrDescr As String
    Dim lstrErrSource As String
    llngErrNo = 0
    lstrErrDescr = ""
    lstrErrSource = ""

    Dim lblnInTrans As Boolean
    Dim lstrSQL As String
    Dim lstrPOStatus As String
    Dim lstrWhereClause As String
    Dim llngAffected As Long
    Dim lngTag As Long
    Dim lobjADORS As ADODB.Recordset
    Dim lobjPORS As ADODB.Recordset

    lblnInTrans = False
    Screen.MousePointer = vbHourglass
    ' Copy the fields from the screen to an ADO Recordset
    CnnBTMS.BeginTrans
    lblnInTrans = True
    Set lobjADORS = BuildRS()
    
    Set lobjPORS = New ADODB.Recordset
    
    lstrSQL = "select POID, POFirstReceive from PO where POID=" & mlngPOID
    lobjPORS.Open lstrSQL, CnnBTMS, adOpenForwardOnly, adLockOptimistic
    If Not lobjPORS Is Nothing Then
        If Not (lobjPORS.EOF = True And lobjPORS.BOF = True) Then
            If lobjPORS.Fields("POFirstReceive").Value = False Then lobjPORS.Fields("POFirstReceive").Value = True
            lobjPORS.Update
        End If
        Set lobjPORS = Nothing
    End If
    
    'Updating any changes made to PODetails table
    
    If Not lobjADORS Is Nothing Then
        If Not (lobjADORS.BOF And lobjADORS.EOF) Then
            lobjADORS.MoveFirst
            Do While lobjADORS.EOF = False
                lstrErrDescr = "Error building update statement for PODetails"
                lstrWhereClause = "POID=" & mlngPOID
                lstrWhereClause = lstrWhereClause & " and POPtNum=" & SQLString(lobjADORS.Fields("POPtNum").Value)
                lstrSQL = Build_Update(lobjADORS, cstrTableName, lstrWhereClause, "POID, PONum")
                lstrErrDescr = "Error updating PODetails"

                Call CnnBTMS.Execute(lstrSQL, 0, adCmdText And adExecuteNoRecords)
                lstrErrDescr = ""
                lobjADORS.MoveNext
            Loop
            lobjADORS.Close
        End If
    End If
    Set lobjADORS = Nothing

    ' Get the minimum status of the PODetails
    lstrPOStatus = "Issued"
    lstrSQL = "SELECT MIN(IIf(PODetails.POPtStatus='Outstanding',1,IIf(PODetails.POPtStatus='Cancelled',2,3))) from PODetails"
    lstrSQL = lstrSQL & " where PODetails.POID=" & mlngPOID
    Set lobjADORS = CnnBTMS.Execute(lstrSQL, llngAffected, adCmdText)
    If Not lobjADORS Is Nothing Then
        lobjADORS.ActiveConnection = Nothing
        If Not (lobjADORS.BOF And lobjADORS.EOF) Then
            lobjADORS.MoveFirst
            Select Case lobjADORS.Fields(0).Value
                Case 1: lstrPOStatus = "Issued"
                Case 2, 3
                    lstrPOStatus = "Complete"
                    frmPO.mlngTag = 1
                    frmPO.mlBytCompletePO = 0
            End Select
        End If
        lobjADORS.Close
    End If
    Set lobjADORS = Nothing
    
    'Update the status on the PO Header
    lstrSQL = "Update PO Set POStatus=" & SQLString(lstrPOStatus)
    lstrSQL = lstrSQL & IIf(lstrPOStatus = "Complete", ", POCompletedate=" & SQLString(Date), "") & " where POID=" & mlngPOID
    Call CnnBTMS.Execute(lstrSQL, llngAffected, adCmdText And adExecuteNoRecords)
    
    CnnBTMS.CommitTrans
    lblnInTrans = False
    Exit Sub

Error_Handler:
    llngErrNo = IIf(llngErrNo = 0, Err.Number, llngErrNo)
    lstrErrDescr = IIf(Len(lstrErrDescr) = 0, Err.Description, Err.Description)
    lstrErrSource = IIf(Len(Trim(Err.Source)) = 0, "", Err.Source & vbCrLf) & cstrErrorSource
    Screen.MousePointer = vbDefault
    If lblnInTrans = True Then CnnBTMS.RollbackTrans
    Call Err.Raise(llngErrNo, lstrErrSource, lstrErrDescr)
    
End Sub

Private Function BuildRS() As ADODB.Recordset
    On Error GoTo Error_Handler
    Const cstrErrorSource As String = cstrApplicationName & "." & cstrModuleName & ".BuildRS()"
    Dim llngErrNo As Long
    Dim lstrErrDescr As String
    Dim lstrErrSource As String
    llngErrNo = 0
    lstrErrDescr = ""
    lstrErrSource = ""

    Dim lstrSQL As String
    Dim lstrSchema As String
    Dim llngAffected As Long
    Dim lobjADORS As ADODB.Recordset
    Dim lobjINVRS As ADODB.Recordset
    Dim ldblReceived As Double
    
    'Get empty recordset
    lstrErrDescr = "Error getting empty ADO recordset."
    lstrSQL = "Select * from PODetails where 1=0"
    Set lobjADORS = New ADODB.Recordset
    lobjADORS.CursorType = adOpenForwardOnly
    lobjADORS.CursorLocation = adUseClient
    lobjADORS.Open lstrSQL, CnnBTMS, adOpenForwardOnly, adLockOptimistic
    Set lobjADORS.ActiveConnection = Nothing
    lstrErrDescr = ""
        
    Dim llngRow As Long
    For llngRow = 1 To grdData.Rows - 1
        If Trim(grdData.TextMatrix(llngRow, 1)) = "" Then Exit For
        lobjADORS.AddNew
        
        lobjADORS.Fields("POPtNum").Value = Trim(grdData.TextMatrix(llngRow, 1))
        lobjADORS.Fields("POPtQtyBO").Value = DBLValue(grdData.TextMatrix(llngRow, 4))
        lobjADORS.Fields("POPtQtyOrd").Value = DBLValue(grdData.TextMatrix(llngRow, 3))
        lobjADORS.Fields("POPtQtyRcvd").Value = DBLValue(grdData.TextMatrix(llngRow, 5)) + DBLValue(grdData.TextMatrix(llngRow, 6))
        lobjADORS.Fields("POPtStatus").Value = grdData.TextMatrix(llngRow, 8)
        
        If UCase(lobjADORS.Fields("POPtStatus").Value) = "OUTSTANDING" And _
             lobjADORS.Fields("POPtQtyRcvd").Value >= DBLValue(grdData.TextMatrix(llngRow, 3)) Then
             lobjADORS.Fields("POPtStatus").Value = "Complete"
        End If
         
        If grdData.TextMatrix(llngRow, 9) = False And grdData.TextMatrix(llngRow, 8) = "Cancelled" Then
            lobjADORS.Fields("POPtInvCancelled").Value = True
        End If
         
        lstrSQL = "select PartNum,InvUM,QtyInStock,QtyBkOrder,QtyOnOrder from Inventory"
        Set lobjINVRS = New ADODB.Recordset
        lobjINVRS.CursorType = adOpenForwardOnly
        lobjINVRS.CursorLocation = adUseClient
        lobjINVRS.Open lstrSQL, CnnBTMS, adOpenForwardOnly, adLockOptimistic
        
        ldblReceived = DBLValue(lobjADORS.Fields("POPtQtyOrd").Value)
        If IsInvMaint = 1 And CheckPORunInventory(mlngPOID) = True Then
            lobjINVRS.MoveFirst
            lobjINVRS.Find "PartNum=" & SQLString(lobjADORS.Fields("POPtNum").Value)
            If lobjINVRS.EOF = False And lobjINVRS.BOF = False Then
               If (grdData.TextMatrix(llngRow, 7) = "EACH") And lobjINVRS.Fields("InvUM").Value = "INCHES" Then
                    If grdData.TextMatrix(llngRow, 9) = True And grdData.TextMatrix(llngRow, 8) = "Outstanding" Then 'when item cancelled previously and set to outstanding.
                        lobjINVRS.Fields("QtyInStock").Value = DBLValue(lobjINVRS.Fields("QtyInStock").Value) - DBLValue(lobjADORS.Fields("POPtQtyRcvd").Value) * GetLengthFromMain(lobjADORS.Fields("POPtNum").Value) + DBLValue(grdData.TextMatrix(llngRow, 5)) * GetLengthFromMain(lobjADORS.Fields("POPtNum").Value)
                        lobjINVRS.Fields("QtyOnOrder").Value = DBLValue(lobjINVRS.Fields("QtyOnOrder").Value) + DBLValue(grdData.TextMatrix(llngRow, 4)) * GetLengthFromMain(lobjADORS.Fields("POPtNum").Value)
                        lobjINVRS.Fields("QtyBkOrder").Value = DBLValue(lobjINVRS.Fields("QtyBkOrder").Value) + DBLValue(grdData.TextMatrix(llngRow, 4)) * GetLengthFromMain(lobjADORS.Fields("POPtNum").Value)
                    ElseIf grdData.TextMatrix(llngRow, 9) = True And grdData.TextMatrix(llngRow, 8) = "Complete" Then 'when item cancelled previously and set to complete.
                        lobjINVRS.Fields("QtyInStock").Value = DBLValue(lobjINVRS.Fields("QtyInStock").Value) - DBLValue(grdData.TextMatrix(llngRow, 6)) * GetLengthFromMain(lobjADORS.Fields("POPtNum").Value) + DBLValue(grdData.TextMatrix(llngRow, 3)) * GetLengthFromMain(lobjADORS.Fields("POPtNum").Value)
                    Else
                        lobjINVRS.Fields("QtyInStock").Value = DBLValue(lobjINVRS.Fields("QtyInStock").Value) + DBLValue(grdData.TextMatrix(llngRow, 5)) * GetLengthFromMain(lobjADORS.Fields("POPtNum").Value)
                        If grdData.TextMatrix(llngRow, 8) = "Cancelled" Then
                            If grdData.TextMatrix(llngRow, 9) = False Then
                                lobjINVRS.Fields("QtyOnOrder").Value = DBLValue(lobjINVRS.Fields("QtyOnOrder").Value) - (DBLValue(grdData.TextMatrix(llngRow, 4)) * GetLengthFromMain(lobjADORS.Fields("POPtNum").Value) + DBLValue(grdData.TextMatrix(llngRow, 5)) * GetLengthFromMain(lobjADORS.Fields("POPtNum").Value))
                                If WasPOPreviouslyReceived(mlngPOID) = True Then
                                    lobjINVRS.Fields("QtyBkOrder").Value = DBLValue(lobjINVRS.Fields("QtyBkOrder").Value) - ((DBLValue(grdData.TextMatrix(llngRow, 4)) * GetLengthFromMain(lobjADORS.Fields("POPtNum").Value)) + DBLValue(grdData.TextMatrix(llngRow, 5)) * GetLengthFromMain(lobjADORS.Fields("POPtNum").Value))
                                End If
                            End If
                        Else
                            lobjINVRS.Fields("QtyOnOrder").Value = DBLValue(lobjINVRS.Fields("QtyOnOrder").Value) - DBLValue(grdData.TextMatrix(llngRow, 5)) * GetLengthFromMain(lobjADORS.Fields("POPtNum").Value)
                            If WasPOPreviouslyReceived(mlngPOID) = True Then
                                lobjINVRS.Fields("QtyBkOrder").Value = DBLValue(lobjINVRS.Fields("QtyBkOrder").Value) - DBLValue(grdData.TextMatrix(llngRow, 5)) * GetLengthFromMain(lobjADORS.Fields("POPtNum").Value)
                            Else
                                lobjINVRS.Fields("QtyBkOrder").Value = DBLValue(lobjINVRS.Fields("QtyBkOrder").Value) + DBLValue(grdData.TextMatrix(llngRow, 4)) * GetLengthFromMain(lobjADORS.Fields("POPtNum").Value)
                            End If
                        End If 'if WasPOPreviouslyReceived = true then this is not the first time it is received.
                    End If
                ElseIf grdData.TextMatrix(llngRow, 7) = "LFT" Then
                    If grdData.TextMatrix(llngRow, 9) = True And grdData.TextMatrix(llngRow, 8) = "Outstanding" Then 'when item cancelled previously and set to outstanding.
                        lobjINVRS.Fields("QtyInStock").Value = DBLValue(lobjINVRS.Fields("QtyInStock").Value) - DBLValue(lobjADORS.Fields("POPtQtyRcvd").Value) * 12 + DBLValue(grdData.TextMatrix(llngRow, 5)) * 12
                        lobjINVRS.Fields("QtyOnOrder").Value = DBLValue(lobjINVRS.Fields("QtyOnOrder").Value) + DBLValue(grdData.TextMatrix(llngRow, 4)) * 12
                        lobjINVRS.Fields("QtyBkOrder").Value = DBLValue(lobjINVRS.Fields("QtyBkOrder").Value) + DBLValue(grdData.TextMatrix(llngRow, 4)) * 12
                    ElseIf grdData.TextMatrix(llngRow, 9) = True And grdData.TextMatrix(llngRow, 8) = "Complete" Then 'when item cancelled previously and set to complete.
                        lobjINVRS.Fields("QtyInStock").Value = DBLValue(lobjINVRS.Fields("QtyInStock").Value) - DBLValue(grdData.TextMatrix(llngRow, 6)) * 12 + DBLValue(grdData.TextMatrix(llngRow, 3)) * 12
                    Else
                        lobjINVRS.Fields("QtyInStock").Value = DBLValue(lobjINVRS.Fields("QtyInStock").Value) + DBLValue(grdData.TextMatrix(llngRow, 5)) * 12
                        If grdData.TextMatrix(llngRow, 8) = "Cancelled" Then
                            If grdData.TextMatrix(llngRow, 9) = False Then
                                lobjINVRS.Fields("QtyOnOrder").Value = DBLValue(lobjINVRS.Fields("QtyOnOrder").Value) - (DBLValue(grdData.TextMatrix(llngRow, 4)) * 12 + DBLValue(grdData.TextMatrix(llngRow, 5)) * 12)
                                If WasPOPreviouslyReceived(mlngPOID) = True Then
                                    lobjINVRS.Fields("QtyBkOrder").Value = DBLValue(lobjINVRS.Fields("QtyBkOrder").Value) - (DBLValue(grdData.TextMatrix(llngRow, 4)) * 12 + DBLValue(grdData.TextMatrix(llngRow, 5)) * 12)
                                End If
                            End If
                        Else
                            'if format(DBLValue(GrdData.TextMatrix(llngRow, 5)) * 12,"Fixed") =
                            lobjINVRS.Fields("QtyOnOrder").Value = DBLValue(lobjINVRS.Fields("QtyOnOrder").Value) - DBLValue(grdData.TextMatrix(llngRow, 5)) * 12
                            
                            If lobjINVRS.Fields("QtyOnOrder").Value < 0.01 Then lobjINVRS.Fields("QtyOnOrder").Value = 0 'due to VB rounding, have to prevent the value to go negative decimal (4 spaces after decimal)
                            
                            If WasPOPreviouslyReceived(mlngPOID) = True Then
                                lobjINVRS.Fields("QtyBkOrder").Value = DBLValue(lobjINVRS.Fields("QtyBkOrder").Value) - DBLValue(grdData.TextMatrix(llngRow, 5)) * 12
                            Else
                                lobjINVRS.Fields("QtyBkOrder").Value = DBLValue(lobjINVRS.Fields("QtyBkOrder").Value) + DBLValue(grdData.TextMatrix(llngRow, 4)) * 12
                            End If
                            If lobjINVRS.Fields("QtyBkOrder").Value < 0.01 Then lobjINVRS.Fields("QtyBkOrder").Value = 0 'rounding issue arises when a number divided by 12 is an infinite number. Then multiply it back by 12, and you get a variance.
                                                                                                                                    'in metric the same issue occurs because we round mm to the nearest mm. Then when they see value in mm on the screen (especially on the BomAlloc screen)
                                                                                                                                    'that's what we order, and receive. But when they try to allocate, we take the value from the db, and variance is left in inventory.------AW.
                        End If
                    End If
                Else
                    If grdData.TextMatrix(llngRow, 9) = True And grdData.TextMatrix(llngRow, 8) = "Outstanding" Then 'when item cancelled previously and set to outstanding.
                        lobjINVRS.Fields("QtyInStock").Value = DBLValue(lobjINVRS.Fields("QtyInStock").Value) - DBLValue(lobjADORS.Fields("POPtQtyRcvd").Value) + DBLValue(grdData.TextMatrix(llngRow, 5))
                        lobjINVRS.Fields("QtyOnOrder").Value = DBLValue(lobjINVRS.Fields("QtyOnOrder").Value) + DBLValue(grdData.TextMatrix(llngRow, 4))
                        lobjINVRS.Fields("QtyBkOrder").Value = DBLValue(lobjINVRS.Fields("QtyBkOrder").Value) + DBLValue(grdData.TextMatrix(llngRow, 4))
                    ElseIf grdData.TextMatrix(llngRow, 9) = True And grdData.TextMatrix(llngRow, 8) = "Complete" Then 'when item cancelled previously and set to complete.
                        lobjINVRS.Fields("QtyInStock").Value = DBLValue(lobjINVRS.Fields("QtyInStock").Value) - DBLValue(grdData.TextMatrix(llngRow, 6)) + DBLValue(grdData.TextMatrix(llngRow, 3))
                    Else
                        lobjINVRS.Fields("QtyInStock").Value = DBLValue(lobjINVRS.Fields("QtyInStock").Value) + DBLValue(grdData.TextMatrix(llngRow, 5))
                        If grdData.TextMatrix(llngRow, 8) = "Cancelled" Then
                            If grdData.TextMatrix(llngRow, 9) = False Then
                                lobjINVRS.Fields("QtyOnOrder").Value = DBLValue(lobjINVRS.Fields("QtyOnOrder").Value) - (DBLValue(grdData.TextMatrix(llngRow, 4)) + DBLValue(grdData.TextMatrix(llngRow, 5)))
                                If WasPOPreviouslyReceived(mlngPOID) = True Then
                                    lobjINVRS.Fields("QtyBkOrder").Value = DBLValue(lobjINVRS.Fields("QtyBkOrder").Value) - (DBLValue(grdData.TextMatrix(llngRow, 4)) + DBLValue(grdData.TextMatrix(llngRow, 5)))
                                End If
                            End If
                        Else
                            lobjINVRS.Fields("QtyOnOrder").Value = DBLValue(lobjINVRS.Fields("QtyOnOrder").Value) - DBLValue(grdData.TextMatrix(llngRow, 5))
                            If WasPOPreviouslyReceived(mlngPOID) = True Then
                                lobjINVRS.Fields("QtyBkOrder").Value = DBLValue(lobjINVRS.Fields("QtyBkOrder").Value) - DBLValue(grdData.TextMatrix(llngRow, 5))
                            Else
                                lobjINVRS.Fields("QtyBkOrder").Value = DBLValue(lobjINVRS.Fields("QtyBkOrder").Value) + DBLValue(grdData.TextMatrix(llngRow, 4))
                            End If
                        End If
                    End If
                End If
                
                lobjINVRS.Update
            End If
        End If
        If grdData.TextMatrix(llngRow, 9) = True And (grdData.TextMatrix(llngRow, 8) = "Outstanding" Or grdData.TextMatrix(llngRow, 8) = "Complete") Then 'when item cancelled previously and set to outstanding.
            lobjADORS.Fields("POPtInvCancelled").Value = False
        End If
    Next llngRow
    
    Set BuildRS = lobjADORS
    Set lobjINVRS = Nothing
    Set lobjADORS = Nothing
    Exit Function

Error_Handler:
    llngErrNo = IIf(llngErrNo = 0, Err.Number, llngErrNo)
    lstrErrDescr = IIf(Len(lstrErrDescr) = 0, Err.Description, Err.Description)
    lstrErrSource = IIf(Len(Trim(Err.Source)) = 0, "", Err.Source & vbCrLf) & cstrErrorSource
    Call Err.Raise(llngErrNo, lstrErrSource, lstrErrDescr)

End Function

Private Sub SetControls()
    On Error GoTo Error_Handler
    Const cstrErrorSource As String = cstrApplicationName & "." & cstrModuleName & ".SetControls()"
    Dim llngErrNo As Long
    Dim lstrErrDescr As String
    Dim lstrErrSource As String
    llngErrNo = 0
    lstrErrDescr = ""
    lstrErrSource = ""

    Select Case mintMode
        Case MODE_EDIT
                        cmdEdit.Caption = "Cancel &Edit"
                        cmdEdit.Enabled = True
                        cmdSave.Enabled = True
                        cmdBack.Enabled = False
                        cmdPrint.Enabled = False
                        Call SetColColor(5, &HE0E0E0)
                        grdData.Row = 1
                        grdData.Col = 5
        Case Else
                        cmdEdit.Caption = "&Edit Receipt"
                        cmdEdit.Enabled = True
                        If UCase(lblStatus.Caption) = "COMPLETE" Then cmdEdit.Enabled = False
                        cmdSave.Enabled = False
                        cmdBack.Enabled = True
                        cmdPrint.Enabled = True
                        Call SetColColor(5, &HFFFFFF)
    End Select
    Exit Sub

Error_Handler:
    llngErrNo = IIf(llngErrNo = 0, Err.Number, llngErrNo)
    lstrErrDescr = IIf(Len(lstrErrDescr) = 0, Err.Description, Err.Description)
    lstrErrSource = IIf(Len(Trim(Err.Source)) = 0, "", Err.Source & vbCrLf) & cstrErrorSource
    Call Err.Raise(llngErrNo, lstrErrSource, lstrErrDescr)
End Sub

Private Sub SetColColor(ByVal intCol As Integer, ByVal lngColor As Long, Optional blnFlg As Boolean = False)
    On Error GoTo Error_Handler
    Const cstrErrorSource As String = cstrApplicationName & "." & cstrModuleName & ".SetGridColor()"
    Dim llngErrNo As Long
    Dim lstrErrDescr As String
    Dim lstrErrSource As String
    llngErrNo = 0
    lstrErrDescr = ""
    lstrErrSource = ""

    Dim llngRow As Long
    grdData.Redraw = flexRDNone
    grdData.Col = intCol
    For llngRow = 1 To grdData.Rows - 1
            grdData.Row = llngRow
            If blnFlg = True Or Not (Trim(grdData.TextMatrix(llngRow, 1)) = "") Then grdData.CellBackColor = lngColor
    Next llngRow
    grdData.Row = 1
    grdData.Col = 1
    grdData.Redraw = flexRDDirect
    Exit Sub

Error_Handler:
    llngErrNo = IIf(llngErrNo = 0, Err.Number, llngErrNo)
    lstrErrDescr = IIf(Len(lstrErrDescr) = 0, Err.Description, lstrErrDescr & vbCrLf & Err.Description)
    lstrErrSource = IIf(Len(Trim(Err.Source)) = 0, "", Err.Source & vbCrLf) & cstrErrorSource
    Call Err.Raise(llngErrNo, lstrErrSource, lstrErrDescr)
End Sub
