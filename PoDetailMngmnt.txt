Private Sub cmdPN_Click()
    On Error GoTo Error_Handler
    Const cstrErrorSource As String = cstrApplicationName & "." & cstrModuleName & ".SubName()"
    Dim llngErrNo As Long
    Dim lstrErrDescr As String
    Dim lstrErrSource As String
    llngErrNo = 0
    lstrErrDescr = ""
    lstrErrSource = ""
    
    Dim larrParms() As String
    Dim lstrPart As String, lstrNewPart As String
    Dim lobjRS As ADODB.Recordset
    
    mdblSectionQTY = 0
    
    If chkPhantom.Value = 1 Then
        If mintMode <> MODE_VIEW Then
            Dim frmW As frmWidth
            Set frmW = New frmWidth

            frmW.mstrPart = mstrPtNum
            Set frmW.frmParent = Me
'            frmW.mstrDesc = Trim(txtPtDesc.Text)
            mblnInFocus = False
            frmW.Show vbModal
            mblnInFocus = True
            Set frmW = Nothing
            If mstrParms <> "" Then
                larrParms = Split(mstrParms, "~")

                'Update the Part Number
                lstrNewPart = larrParms(0)
                txtPartNum.Text = (lstrNewPart)

                'Update the description
                If Left(lstrNewPart, 1) <> "9" Then
                    txtPtDesc.Text = loadDefaultBasePNDescription(txtPartNum.Text) & larrParms(1)
                Else
                    txtPtDesc.Text = txtPtDesc.Text & larrParms(1)
                End If
                
                lblPOUM.Caption = "EACH"

                ''''''''''Change in .NET''''''''''''''''
'                Me.lblBookCost.Caption = (DBLValue(lblBookCost.Caption)) * (DBLValue(larrParms(2)) / 12) * IIf(Left(lstrNewPart, 1) = "9", 1, (IIf(Mid(lstrNewPart, 5, 1) = "1", 1.75, 2)))
                txtPtQty.Text = mdblSectionQTY
                Me.lblBookCost.Caption = Format((DBLValue(mdblCost)) * (DBLValue(larrParms(2)) / 12) * IIf(Left(lstrNewPart, 1) = "9", 1, (IIf(Mid(lstrNewPart, 6, 1) = "1", 1.75, 2))), cstrDollarMask)
                Me.lblListPrice.Caption = Format((DBLValue(mdblListPrice)) * (DBLValue(larrParms(2)) / 12) * IIf(Left(lstrNewPart, 1) = "9", 1, (IIf(Mid(lstrNewPart, 6, 1) = "1", 1.75, 2))), cstrDollarMask)
                ''''END Change in .NET
                txtPtCost.Text = Format(DBLValue(lblBookCost.Caption), "Fixed")
                Me.lblWgt.Caption = (DBLValue(mdblShipWeight)) * (DBLValue(larrParms(2)) / 12) * (IIf(Mid(lstrNewPart, 6, 1) = "1", 1.75, 2))
                Me.lblTotalWgt.Caption = 0
                Me.lblTotalCost.Caption = DBLValue(txtPtCost.Text) * DBLValue(txtPtQty.Text)
            Else
                Call cmdBack_Click
                Exit Sub
            End If
        End If
    End If
'    lblStatus.Caption = mstrStatus

    If txtPtQty.Enabled = True Then txtPtQty.SetFocus
    txtPtQty.SelStart = 0
    txtPtQty.SelLength = Len(txtPtQty.Text)
    mblnInFocus = True
    Exit Sub

Error_Handler:
    llngErrNo = IIf(llngErrNo = 0, Err.Number, llngErrNo)
    lstrErrDescr = IIf(Len(lstrErrDescr) = 0, Err.Description, lstrErrDescr & vbCrLf & Err.Description)
    Call DisplayError(llngErrNo, IIf(Len(Trim(Err.Source)) = 0, "", Err.Source & vbCrLf) & cstrErrorSource, lstrErrDescr)

End Sub

Private Sub cmdPrint_Click()
    On Error GoTo Error_Handler
    Const cstrErrorSource As String = cstrApplicationName & "." & cstrModuleName & ".cmdPrint_Click()"
    Dim llngErrNo As Long
    Dim lstrErrDescr As String
    Dim lstrErrSource As String
    llngErrNo = 0
    lstrErrDescr = ""
    lstrErrSource = ""

'    CommonDialog1.CancelError = True
'    CommonDialog1.ShowPrinter
    Call ShowTaskbar(False) 'will hide the widows taskbar for the time of printing
    
    Set Picture1.Picture = CaptureScreen()
    PrintPictureToFitPage Printer, Picture1.Picture
    
    Printer.EndDoc
    
    Call ShowTaskbar(True)
    Exit Sub

Error_Handler:
    If Err.Number = 32755 Then
        Call ShowTaskbar(True)
        Exit Sub
    End If
    
    If Err.Number = 482 Or Err.Number = 484 Then 'Make sure all connections between computer and printer are solid. Most printers provide a display for error information such as "Out of paper," "Offline," and so on.
        MsgBox Err.Number & ":" & " Printer error!", vbCritical + vbApplicationModal, "Printer Error."
        Exit Sub
    End If
    
    If Err.Number = 9 Then  'no printer installed
        MsgBox Err.Number & ":" & " Printer error!", vbCritical + vbApplicationModal, "Printer Error."
        Exit Sub
    End If
    
    Call ShowTaskbar(True)
    llngErrNo = IIf(llngErrNo = 0, Err.Number, llngErrNo)
    lstrErrDescr = IIf(Len(lstrErrDescr) = 0, Err.Description, Err.Description)

End Sub

Private Sub cmdSave_Click()
    On Error GoTo Error_Handler
    Const cstrErrorSource As String = cstrApplicationName & "." & cstrModuleName & ".cmdSave_Click()"
    Dim llngErrNo As Long
    Dim lstrErrDescr As String
    Dim lstrErrSource As String
    llngErrNo = 0

    lstrErrSource = ""
    Dim ldblOrdQty As Double
    Dim ldblPkgQtyNew As Double
    txtPtQty.Text = Format(DBLValue(txtPtQty.Text), cstrQtyMask)
    lblTotalCost.Caption = Format(DBLValue(txtPtQty.Text) * DBLValue(txtPtCost.Text), cstrDollarMask)
    lblTotalWgt.Caption = Format(DBLValue(lblWgt.Caption) * DBLValue(txtPtQty.Text), "Fixed")
    
    If mlngPkgQty > 0 Then
        If mlngPkgQty > 0 And (CLng(DBLValue(txtPtQty.Text)) Mod mlngPkgQty) > 0 Then

''''''''''Change in .NET''''''''''''''''
'            If MsgBox("Quantity is not a multiple of Package Quantity.  Do you wish to continue?", vbQuestion + vbApplicationModal + vbYesNo) = vbNo Then
'                If txtPtQty.Enabled = True Then txtPtQty.SetFocus
'            Else
            If MsgBox("Quantity is not a multiple of Package Quantity. Do you wish to increase the order quantity to a multiple of Package Quantity?", vbQuestion + vbApplicationModal + vbYesNo) = vbYes Then
                
                ldblOrdQty = DBLValue(txtPtQty.Text)
                ldblPkgQtyNew = mlngPkgQty
                Do While ldblPkgQtyNew < ldblOrdQty
                    ldblPkgQtyNew = ldblPkgQtyNew + mlngPkgQty
                Loop
                txtPtQty.Text = ldblPkgQtyNew
                If txtPtQty.Enabled = True Then txtPtQty.SetFocus
            End If
            Screen.MousePointer = vbHourglass
            Call SaveData
'            End If
''''''''''END Change in .NET
            Screen.MousePointer = vbDefault
            Exit Sub
        End If
    End If

    If MsgBox("Save Data?", vbApplicationModal + vbYesNo + vbQuestion, "Save Data.") = vbYes Then
        Screen.MousePointer = vbHourglass
        Call SaveData
    End If
    Screen.MousePointer = vbDefault
    Exit Sub

    Screen.MousePointer = vbDefault

End Sub

Private Sub Form_Load()
    On Error GoTo Error_Handler
    Const cstrErrorSource As String = cstrApplicationName & "." & cstrModuleName & ".Form_Load()"
    Dim llngErrNo As Long
    Dim lstrErrDescr As String
    Dim lstrErrSource As String
    llngErrNo = 0
    lstrErrDescr = ""
    lstrErrSource = ""

    Dim currBPP As Long
    currBPP = GetDeviceCaps(hdc, BITSPIXEL)
    If currBPP <= 8 Then Call GoGrey(Me)
    
    Me.Move Screen.Width / 2 - Me.Width / 2, Screen.Height / 2 - Me.Height / 2
    
    Call LoadPartInfo(Me.mstrPtNum)
    If mintMode = MODE_ADD Then
        If mblnSpecial = False Then
            mintMode = MODE_VIEW
            Call AddPart
        Else
            Call ClearScreen(True)
            Call SetControls(mintMode)
            
            txtPartNum.Text = GetSpecialPart("NSTK")
            chkCall.Value = 1
            chkCall.Enabled = False
            txtPtDesc.Enabled = True
            txtPtDesc.BackColor = vbWhite
            txtSpecialWeight.Visible = True

Private Sub SaveData()
    On Error GoTo Error_Handler
    Const cstrErrorSource As String = cstrApplicationName & "." & cstrModuleName & ".SaveData()"
    Dim llngErrNo As Long
    Dim lstrErrDescr As String
    Dim lstrErrSource As String
    llngErrNo = 0
    lstrErrDescr = ""
    lstrErrSource = ""

    Dim lstrSQL As String
    Dim lstrWhereClause As String
    Dim lstrExcludeFieldList As String
    Dim llngAffected As Long
    Dim lobjADORS As ADODB.Recordset
    Dim lobjParentRS As ADODB.Recordset
    Dim lintTempMODE As Integer
    Dim lblnInTrans As Boolean
    Dim lblnPartExist As Boolean
    
    lintTempMODE = mintMode
    CnnBTMS.BeginTrans
    lblnInTrans = True
    If txtSpecialWeight.Visible = True Then Me.lblWgt.Caption = DBLValue(txtSpecialWeight.Text)
    
    ' Copy the fields from the screen to an ADO Recordset
    Set lobjADORS = BuildRS()
    
    If Not ValidateData(lobjADORS) Then
        CnnBTMS.RollbackTrans
        Exit Sub
    End If
    
    ' Process through the recordset and insert or update rows
    If Not lobjADORS Is Nothing Then
        If Not (lobjADORS.BOF And lobjADORS.EOF) Then
            lobjADORS.MoveFirst
            lstrExcludeFieldList = ""
            
            Do While lobjADORS.EOF = False
                ' Check to see if the record exists in the
                ' BOMDetails table already
                If mintMode = MODE_ADD Then
                    lstrSQL = "select POPtNum, POPtQtyOrd, POptMtCost,POPtCallPrice from POdetails "
                    lstrSQL = lstrSQL & "where POnum=" & SQLString(Trim(mstrPONum))
                    lstrSQL = lstrSQL & "and POptnum=" & SQLString(lobjADORS.Fields("POPtNum").Value)
                    Set lobjParentRS = CnnBTMS.Execute(lstrSQL, llngAffected, adCmdText)
                    If Not lobjParentRS Is Nothing Then
                        If Not (lobjParentRS.BOF And lobjParentRS.EOF) Then
                            lobjParentRS.MoveFirst
                            ' Reset the Mode so that we do an update of the existing part
                            lblnPartExist = True
                            
                            ' Average the cost and sum the qty's
                            If DBLValue(lobjADORS.Fields("POPtMtCost").Value) <> DBLValue(lobjParentRS.Fields("POPtMtCost").Value) Then lobjADORS.Fields("POPtCallPrice").Value = True

                            ''''''''''Change in .NET''''''''''''''''
'                            lobjADORS.Fields("POPtMtCost").Value = (DBLValue(lobjADORS.Fields("POPtQtyOrd").Value) * DBLValue(lobjADORS.Fields("POPtMtCost").Value) + DBLValue(lobjParentRS.Fields("POPtQtyOrd").Value) * DBLValue(lobjParentRS.Fields("POPtMtCost").Value)) / (DBLValue(lobjParentRS.Fields("POPtQtyOrd").Value) + DBLValue(lobjADORS.Fields("POPtQtyOrd").Value))
                            If lobjADORS.Fields("POPtCallPrice").Value = True Then
                                If lobjADORS.Fields("POPtMtCost").Value <> lobjParentRS.Fields("POPtMtCost").Value Then
                                    MsgBox "Part number " & lobjParentRS.Fields("POPtNum").Value & " has been converted back to List Pricing on the Purchase Order. ", vbApplicationModal + vbInformation, "Price Changed"
                                    lobjADORS.Fields("POPtMtCost").Value = DBLValue(ConvertToListPrice(lobjADORS))
                                    lobjADORS.Fields("POPtCallPrice").Value = 0
                                    lobjADORS.Fields("POCallPriceExpDate").Value = Null
                                End If
                            End If

                            ''''''''''END Change in .NET
                            
                            lobjADORS.Fields("POPtQtyOrd").Value = DBLValue(lobjADORS.Fields("POPtQtyOrd").Value) + DBLValue(lobjParentRS.Fields("POPtQtyOrd").Value)
                            lobjADORS.Fields("POPtQtyBO").Value = DBLValue(lobjADORS.Fields("POPtQtyOrd").Value)
                        End If
                        lobjParentRS.Close
                    End If
                    Set lobjParentRS = Nothing
                End If

                'Build the SQL Insert or Update statement
                If mintMode = MODE_ADD And lblnPartExist = False Then
                    lstrErrDescr = "Error building insert statement for " & cstrTableName
                    lstrSQL = Build_Insert(lobjADORS, cstrTableName, lstrExcludeFieldList)
                    lstrErrDescr = "Error inserting data into " & cstrTableName
                Else
                    lstrErrDescr = "Error building update statement for " & cstrTableName
                    lstrWhereClause = "PONum=" & SQLString(Trim(lobjADORS.Fields("PONum").Value))
                    lstrWhereClause = lstrWhereClause & " and POPtNum=" & SQLString(mstrPtNum)
                    lstrSQL = Build_Update(lobjADORS, cstrTableName, lstrWhereClause, lstrExcludeFieldList)
                    lstrErrDescr = "Error updating " & cstrTableName
                End If
                Call CnnBTMS.Execute(lstrSQL, 0, adCmdText And adExecuteNoRecords)
                lstrErrDescr = ""
                
                mstrPtNum = Trim(lobjADORS.Fields("POPtNum").Value)
                    ''''''''''Change in .NET''''''''''''''''
                mstrAddedPN = mstrPtNum
                    ''''''''''END Change in .NET
                lobjADORS.MoveNext
            Loop
            lobjADORS.Close
        End If
    End If
    
    Call UpdateModDate
    CnnBTMS.CommitTrans
    lblnInTrans = False
    Set lobjADORS = Nothing
    
    If lintTempMODE = MODE_ADD Then
        If mblnSpecial = True Then
            aReg.OpenKey LocalMachine, "\Software\Richards-Wilcox Canada\BTMS", True 'opens or creates key
            aReg.WriteValue "LastNumber", CStr(DBLValue(Mid(Me.txtPartNum.Text, 11)))
            aReg.CloseKey
        End If
        Call cmdBack_Click
        Exit Sub
    End If
    
    Call LoadData
    
    Exit Sub

Error_Handler:
    llngErrNo = IIf(llngErrNo = 0, Err.Number, llngErrNo)
    lstrErrDescr = IIf(Len(lstrErrDescr) = 0, Err.Description, Err.Description)
    lstrErrSource = IIf(Len(Trim(Err.Source)) = 0, "", Err.Source & vbCrLf) & cstrErrorSource
    If lblnInTrans Then CnnBTMS.RollbackTrans
    Call Err.Raise(llngErrNo, lstrErrSource, lstrErrDescr)
End Sub

Private Function BuildRS() As ADODB.Recordset
    On Error GoTo Error_Handler
    Const cstrErrorSource As String = cstrApplicationName & "." & cstrModuleName & ".BuildRS()"
    Dim llngErrNo As Long
    Dim lstrErrDescr As String
    Dim lstrErrSource As String
    llngErrNo = 0
    lstrErrDescr = ""
    lstrErrSource = ""

    Dim lstrSQL As String
    Dim lstrSchema As String
    Dim llngAffected As Long
    Dim lobjADORS As ADODB.Recordset
    
    ' Get empty recordset
    lstrErrDescr = "Error getting empty ADO recordset."
    lstrSQL = "Select * from " & cstrTableName & " where 1=0"
    Set lobjADORS = New ADODB.Recordset
    lobjADORS.CursorType = adOpenForwardOnly
    lobjADORS.CursorLocation = adUseClient
    lobjADORS.Open lstrSQL, CnnBTMS, adOpenForwardOnly, adLockOptimistic
    lobjADORS.AddNew
    Set lobjADORS.ActiveConnection = Nothing
    lstrErrDescr = ""
    
    lobjADORS.Fields("POPtStatus").Value = IIf(lblStatus.Caption = "Active", "Active", "Outstanding")
    lobjADORS.Fields("POID").Value = mlngPOID
    lobjADORS.Fields("PONum").Value = Trim(mstrPONum)
    lobjADORS.Fields("POPtNum").Value = Trim(txtPartNum.Text)
    lobjADORS.Fields("POPtDesc").Value = Trim(txtPtDesc.Text)
    lobjADORS.Fields("POPtUM").Value = IIf(Trim(lblPOUM.Caption) = "", "EACH", Trim(lblPOUM.Caption))
    lobjADORS.Fields("POPtQtyOrd").Value = DBLValue(txtPtQty.Text)
    lobjADORS.Fields("POPtQtyBO").Value = DBLValue(txtPtQty.Text)
    lobjADORS.Fields("POCat").Value = GetPOCat(lobjADORS.Fields("POPtNum").Value)
    lobjADORS.Fields("POSubCategory").Value = GetSubCategory(lobjADORS.Fields("POPtNum").Value)
    lobjADORS.Fields("POPtWgt").Value = DBLValue(lblWgt.Caption)
    lobjADORS.Fields("POPtMtCost").Value = DBLValue(txtPtCost.Text)
    lobjADORS.Fields("POPtMultID").Value = DBLValue(lblMultID.Caption)
    lobjADORS.Fields("PODiscount").Value = DBLValue(lblSpMult.Caption)
    lobjADORS.Fields("POptCallPrice").Value = IIf(chkCall.Value = 1, True, False)
    lobjADORS.Fields("POPtPhantomPN").Value = IIf(chkPhantom.Value = 1, True, False)
    lobjADORS.Fields("POPtNotes").Value = Trim(txtPtNotes.Text)
    If mdblOldQty <> DBLValue(txtPtQty.Text) Then
        Dim ldblTempQtyOld As Double, ldblTempQtyNew As Double
        If GetUMfromInventory(Trim(txtPartNum.Text)) = "INCHES" And Trim(lblPOUM.Caption) = "LFT" Then
            ldblTempQtyOld = mdblOldQty * 12
            ldblTempQtyNew = DBLValue(txtPtQty.Text) * 12
        ElseIf GetUMfromInventory(Trim(txtPartNum.Text)) = "INCHES" And Trim(lblPOUM.Caption) = "EACH" Then
            ldblTempQtyOld = mdblOldQty * GetLengthFromMain(Trim(txtPartNum.Text))
            ldblTempQtyNew = DBLValue(txtPtQty.Text) * GetLengthFromMain(Trim(txtPartNum.Text))
        Else
            ldblTempQtyOld = mdblOldQty
            ldblTempQtyNew = DBLValue(txtPtQty.Text)
        End If
        If ldblTempQtyOld > 0 Then Call UpdateOnPOsForPN(lobjADORS.Fields("POPtNum").Value, 0 - ldblTempQtyOld)
        Call UpdateOnPOsForPN(lobjADORS.Fields("POPtNum").Value, ldblTempQtyNew)
    
    End If
    Set BuildRS = lobjADORS
    Set lobjADORS = Nothing
    Exit Function

Error_Handler:
    llngErrNo = IIf(llngErrNo = 0, Err.Number, llngErrNo)
    lstrErrDescr = IIf(Len(lstrErrDescr) = 0, Err.Description, Err.Description)
    lstrErrSource = IIf(Len(Trim(Err.Source)) = 0, "", Err.Source & vbCrLf) & cstrErrorSource
    Call Err.Raise(llngErrNo, lstrErrSource, lstrErrDescr)
End Function

            If UCase(Trim(mstrPOUM)) = "METRIC" Then lstrFieldName = "DescMet"
            txtPtDesc.Text = IIf(IsNull(lobjMainRS.Fields(lstrFieldName).Value), "", Trim(lobjMainRS.Fields(lstrFieldName).Value))
             '-- Mod Hanobver  MultId changed to strMultId
            lstrFieldName = strMultId
            lblMultID.Caption = IIf(IsNull(lobjMainRS.Fields(lstrFieldName).Value), "", Trim(lobjMainRS.Fields(lstrFieldName).Value))
            lstrFieldName = "UM"
            lblPOUM.Caption = IIf(IsNull(lobjMainRS.Fields(lstrFieldName).Value), "", Trim(lobjMainRS.Fields(lstrFieldName).Value))
            lstrFieldName = "PkgQty"
            mlngPkgQty = IIf(lobjMainRS.Fields(lstrFieldName).Value > 0, lobjMainRS.Fields(lstrFieldName).Value, 0)
            lblPkgQty.Caption = "Package Qty: " & IIf(mlngPkgQty > 0, Format(mlngPkgQty, cstrQtyMask), "N/A")
            'If DBLValue(txtPtQty.Text) = 0 Then txtPtQty.Text = Format(mlngPkgQty, cstrQtyMask)
            'txtOrderQty.Text = txtPtQty.Text
            lstrFieldName = "SpecialMult"
            lblSpMult.Caption = Format(IIf(IsNull(lobjMainRS.Fields(lstrFieldName).Value), "", Trim(lobjMainRS.Fields(lstrFieldName).Value)), cstrQtyMask)
            
            'this label will hold a value of the total discount.
'            If DBLValue(lblSpMult.Caption) = 0 Then
'                lblSpMult.Caption = MultipliersCase(DBLValue(Trim(lblMultID.Caption)))
'            End If
            
            lstrFieldName = "ShipWeight"
            lblWgt.Caption = IIf(IsNull(lobjMainRS.Fields(lstrFieldName).Value), "", Trim(lobjMainRS.Fields(lstrFieldName).Value))
            mdblShipWeight = lblWgt.Caption 'use only for section's ship weight
            If Left(strPartNum, 4) = "433-" Then lblWgt.Caption = DBLValue(lblWgt.Caption) * GetLengthFromMain(strPartNum)
            lblWgt.Caption = Format(lblWgt.Caption, "Fixed")
            lblTotalWgt.Caption = Format(DBLValue(lblWgt.Caption) * DBLValue(txtPtQty.Text), "Fixed")
            
            lstrFieldName = "PhantomPN"
            chkPhantom.Value = IIf(lobjMainRS.Fields(lstrFieldName).Value = True, 1, 0)
            lstrFieldName = "CallPrice"
            chkCall.Value = IIf(lobjMainRS.Fields(lstrFieldName).Value = True, 1, 0)
            
            ' Calculate Cost of part
            Dim ldblCost As Double
            Dim dblListPrice As Double
            Select Case intCheck
                Case 1: ldblCost = DBLValue(lobjMainRS.Fields("G").Value)
                Case 2: ldblCost = DBLValue(lobjMainRS.Fields("I").Value)
                Case 3: ldblCost = DBLValue(lobjMainRS.Fields("N").Value)
                Case 4: ldblCost = DBLValue(lobjMainRS.Fields("T").Value)
                Case 5: ldblCost = DBLValue(lobjMainRS.Fields("J").Value)
                Case 6: ldblCost = DBLValue(lobjMainRS.Fields("M").Value)
                Case 7: ldblCost = DBLValue(lobjMainRS.Fields("S").Value)
                Case 8: ldblCost = DBLValue(lobjMainRS.Fields("V").Value)

                
                
            End Select
            ldblCost = Decrypt(GetSym(strPartNum), ldblCost)
            dblListPrice = ldblCost

            Dim ldblSpMult As Double
            ldblSpMult = GetSpMultFromDetails(strPartNum)
            If ldblSpMult > 0 Then 'there is Sp Mult for that PN on the PO.
                ldblCost = ldblCost * ldblSpMult
            ElseIf ldblSpMult = -1 Then 'the PN does not exist on the PO, in this case will take the Sp Mult from Main and Mult from Registry
                If lobjMainRS.Fields("SpecialMult").Value > 0 Then
                    ldblCost = ldblCost * DBLValue(lobjMainRS.Fields("SpecialMult").Value)
                Else
                    ldblCost = ldblCost * MultipliersCase(DBLValue(Trim(lblMultID.Caption)))
                End If
            Else 'PN is on the PO, but has no Sp. Mult. in this case take the Mult values from PO table.
                ldblCost = ldblCost * (GetMultiplierFromSavedDocument(DBLValue(Trim(lblMultID.Caption)), mlngPOID, "PO"))
            End If
            
            mdblCost = ldblCost 'use only for section's cost
            mdblListPrice = dblListPrice 'use for section's cost
            
            If Left(strPartNum, 4) = "433-" Then
                ldblCost = ldblCost * GetLengthFromMain(strPartNum)
                dblListPrice = dblListPrice * GetLengthFromMain(strPartNum)
            End If
                
            lblBookCost.Caption = Format(ldblCost, cstrDollarMask)
            lblListPrice.Caption = Format(dblListPrice, cstrDollarMask)
            If chkCall.Value = 0 Then txtPtCost.Text = Format(ldblCost, cstrDollarMask)
            lblTotalCost.Caption = Format(DBLValue(txtPtQty.Text) * DBLValue(txtPtCost.Text), cstrDollarMask)
            lobjMainRS.Close
        Else
            
        End If
    End If
    Set lobjMainRS = Nothing
    
    Exit Sub

Error_Handler:
    llngErrNo = IIf(llngErrNo = 0, Err.Number, llngErrNo)
    lstrErrDescr = IIf(Len(lstrErrDescr) = 0, Err.Description, Err.Description)
    lstrErrSource = IIf(Len(Trim(Err.Source)) = 0, "", Err.Source & vbCrLf) & cstrErrorSource
    Call Err.Raise(llngErrNo, lstrErrSource, lstrErrDescr)
End Sub
Private Function GetSpMultFromDetails(ByVal lstrPN As String) As Single
    On Error GoTo Error_Handler
    Const cstrErrorSource As String = cstrApplicationName & "." & cstrModuleName & ".GetSpMultFromDetails()"
    Dim llngErrNo As Long
    Dim lstrErrDescr As String
    Dim lstrErrSource As String
    llngErrNo = 0
    lstrErrDescr = ""
    lstrErrSource = ""
    
    'the function will return -1 if the PN is not found on the PO, or Special Multiplier value from the PODetails table that was saved
    'at the time of creating the PO. In all other cases, the function will return 0.
    
    GetSpMultFromDetails = 0
    
    Dim lstrSQL As String
    Dim lobjRS As ADODB.Recordset
    Dim llngAffected As Long
    lstrSQL = "select PODiscount from PODetails where POID=" & mlngPOID & " and POPtNum=" & SQLString(lstrPN)

        If lobjRS.EOF = False And lobjRS.BOF = False Then
            GetSpMultFromDetails = IIf(IsNull(lobjRS.Fields("PODiscount").Value), 0, lobjRS.Fields("PODiscount").Value)
        Else

        End If
    End If
    Set lobjRS = Nothing

    Dim llngErrNo As Long
    Dim lstrErrDescr As String
    Dim lstrErrSource As String
    llngErrNo = 0

    lstrErrSource = ""

    ValidateData = True

    If mintMode = MODE_ADD Then
    End If

    End If

Private Function ConvertToListPrice(ByRef lobjADORS As ADODB.Recordset) As Double
                                
    On Error GoTo Error_Handler
    Const cstrErrorSource As String = cstrApplicationName & "." & cstrModuleName & ".ConvertToListPrice()"
    Dim llngErrNo As Long
    Dim lstrErrDescr As String
    Dim lstrErrSource As String
    llngErrNo = 0
    lstrErrDescr = ""
    lstrErrSource = ""
    
    Dim lobjMainRS As ADODB.Recordset
    Dim ldblCost As Double
    Dim lstrPriceField As String
    Dim lstrSQL As String
    Dim llngAffected As Long
    Dim ldblSpMult As Double
    
    Select Case intCheck
        Case 1:  lstrPriceField = "G"
        Case 2:  lstrPriceField = "I"
        Case 3:  lstrPriceField = "N"
        Case 4:  lstrPriceField = "T"
        
        Case 5:  lstrPriceField = "J"
        Case 6:  lstrPriceField = "M"
        Case 7:  lstrPriceField = "S"
        Case 8:  lstrPriceField = "V"

    End Select
     '-- Mod Hanobver  MultId changed to strMultId
'    lstrSQL = "Select PartNum, descimp, Length, shipweight, UM, SpecialMult, MultID, " & lstrPriceField & " from Main where PartNum=" & SQLString(lobjADORS.Fields("POPtNum").Value)
    lstrSQL = "Select PartNum, descimp, Length, shipweight, UM, SpecialMult," & strMultId & ", " & lstrPriceField & " from Main where PartNum=" & SQLString(lobjADORS.Fields("POPtNum").Value)

    
    Set lobjMainRS = CnnBTMS.Execute(lstrSQL, llngAffected, adCmdText)
    If Not lobjMainRS Is Nothing Then
        lobjMainRS.ActiveConnection = Nothing
        If Not (lobjMainRS.BOF And lobjMainRS.EOF) Then
        
            ldblCost = Decrypt(GetSym(lobjADORS.Fields("POPtNum").Value), lobjMainRS.Fields(lstrPriceField).Value)
            
            ldblSpMult = GetSpMultFromDetails(lobjADORS.Fields("POPtNum").Value)
            If ldblSpMult > 0 Then 'there is Sp Mult for that PN on the PO.
                ldblCost = ldblCost * ldblSpMult
            ElseIf ldblSpMult = -1 Then 'the PN does not exist on the PO, in this case will take the Sp Mult from Main and Mult from Registry
                If DBLValue(lobjMainRS.Fields("SpecialMult").Value) > 0 Then
                    ldblCost = ldblCost * DBLValue(lobjMainRS.Fields("SpecialMult").Value)
                Else
                    ldblCost = ldblCost * MultipliersCase(DBLValue(lobjADORS.Fields("POPttMultID").Value))
                End If
            Else 'PN is on the PO, but has no Sp. Mult. in this case take the Mult values from PO table.
                ldblCost = ldblCost * (GetMultiplierFromSavedDocument(DBLValue(lobjADORS.Fields("POPtMultID").Value), mlngPOID, "PO"))
            End If
        End If
    End If
    
    If Left(lobjADORS.Fields("POPtNum").Value, 4) = "433-" Then ldblCost = ldblCost * GetLengthFromMain(lobjADORS.Fields("POPtNum").Value)
    
    ConvertToListPrice = ldblCost
    Set lobjMainRS = Nothing
    
    Exit Function

Error_Handler:
    llngErrNo = IIf(llngErrNo = 0, Err.Number, llngErrNo)
    lstrErrDescr = IIf(Len(lstrErrDescr) = 0, Err.Description, lstrErrDescr & vbCrLf & Err.Description)
    lstrErrSource = IIf(Len(Trim(Err.Source)) = 0, "", Err.Source & vbCrLf) & cstrErrorSource
    Call Err.Raise(llngErrNo, lstrErrSource, lstrErrDescr)

End Function
