Private Sub Form_Activate()
    On Error GoTo Error_Handler
    Const cstrErrorSource As String = cstrApplicationName & "." & cstrModuleName & ".Form_Activate()"
    Dim llngErrNo As Long
    Dim lstrErrDescr As String
    Dim lstrErrSource As String
    llngErrNo = 0
    lstrErrDescr = ""
    lstrErrSource = ""

    mblnInFocus = True
    Dim i As Integer, IsLite As Boolean
    If Not isEmptyArraySpItems Then
        For i = 0 To UBound(varSpItem)
            If varSpItem(i).intType = 1 Then IsLite = True 'Lite
        Next
    End If
    If varDoorData.LiteType = 8 And IsLite = False Then
        Me.cboLiteType.ListIndex = 0
        Me.txtNofLiteSec.Text = "0"
        Me.txtLitesPerSection.Text = "0"
        varDoorData.LiteType = 0
    End If
    
    If AbleToChangeMult Then
    
        aReg.OpenKey LocalMachine, "\Software\Richards-Wilcox Canada", True
        lblMult.Caption = "BTS: " & IIf(aReg.ReadValue("BulkThermatiteSections") = "1", "0", aReg.ReadValue("BulkThermatiteSections"))
        lblMult.Caption = lblMult.Caption & "/ OA: " & IIf(aReg.ReadValue("OptionsandAccessories") = "1", "0", aReg.ReadValue("OptionsandAccessories"))
        lblMult.Caption = lblMult.Caption & "/ BH: " & IIf(aReg.ReadValue("BulkHardware") = "1", "0", aReg.ReadValue("BulkHardware"))
        aReg.CloseKey
        lblMult.Visible = True
        cmdChangeMult.Visible = True
    Else
        lblMult.Visible = False
        cmdChangeMult.Visible = False
    End If
    
    Exit Sub

Error_Handler:
    llngErrNo = IIf(llngErrNo = 0, Err.Number, llngErrNo)
    lstrErrDescr = IIf(Len(lstrErrDescr) = 0, Err.Description, lstrErrDescr & vbCrLf & Err.Description)

Private Sub HRoomInch_Validate(Cancel As Boolean)
    On Error GoTo Error_Handler
    Const cstrErrorSource As String = cstrApplicationName & "." & cstrModuleName & ".HRoomInch_Validate()"
    Dim llngErrNo As Long
    Dim lstrErrDescr As String
    Dim lstrErrSource As String
    llngErrNo = 0
    lstrErrDescr = ""
    lstrErrSource = ""
    
    If DBLValue(HRoomInch.Text) > 0 Then
        If IsUnrestricted = False Then 'restricted
            If Val(HRoomInch.Text) < 24 Then
                MsgBox "Minimum door Headroom is " & Format(24.5, "Fixed") & Chr(34), vbApplicationModal + vbExclamation, "Invalid Headroom."
                Cancel = True
            End If
    
            If Val(HRoomInch.Text) = 164 Then
                HRoomFr.Enabled = False
            ElseIf Val(HRoomInch.Text) > 164 Then
                MsgBox "Maximum door Headroom is 164" & Chr(34), vbApplicationModal + vbExclamation, "Invalid Headroom."
                Cancel = True
            End If
        End If
    End If
    
    Exit Sub

Error_Handler:
    llngErrNo = IIf(llngErrNo = 0, Err.Number, llngErrNo)
    lstrErrDescr = IIf(Len(lstrErrDescr) = 0, Err.Description, lstrErrDescr & vbCrLf & Err.Description)
    Call DisplayError(llngErrNo, IIf(Len(Trim(Err.Source)) = 0, "", Err.Source & vbCrLf) & cstrErrorSource, lstrErrDescr)

End Sub

    On Error GoTo Error_Handler
    Const cstrErrorSource As String = cstrApplicationName & "." & cstrModuleName & ".txtHeight_Change()"

    Dim lstrErrDescr As String
    Dim lstrErrSource As String

    lstrErrSource = ""
     
    CECompliant = IIf(chkCE.Value = 1, 1, 0)

    Dim lsngHeight As Single
    Dim lsngWidth As Single
    
    Call ClearConObj
    aCon.Inch = DBLValue(txtHeight.Text)
    aCon.Feetcalc
    lsngHeight = Round(aCon.Feet, 2)
    
    Call ClearConObj
    aCon.Inch = DBLValue(txtWidth.Text)
    aCon.Feetcalc
    lsngWidth = Round(aCon.Feet, 2)

        If cboOperatorType.ListIndex = 1 Then
            cboChainHType.Enabled = True

        End If
    Else
        If cboOperatorType.ListIndex = 1 Then
            cboChainHType.Enabled = True
            cboChainHType.ListIndex = 2
        End If
    End If
    
    DrType = cboDrModel.Text
    ProdType = IIf(CboDrType.Text = "", "Replacement", CboDrType.Text)
    Grd.Clear
    Call SetGrdTitles
    If ProdType = "Alumatite" Then
        Grd.RowHeight(2) = 0
        Call CalcAlumatiteSections
        Call PopulateSectionArrayDefault
        Dim i As Integer
        If N21 + N24 - 1 > 0 Then Me.txtNofLiteSec.Text = N21 + N24 - 1
        
        Me.txtLitesPerSection.Text = GetNumInsideHingesPerWidth + 1
        Me.txtLitesPerSection.Enabled = False
        For i = 0 To cboLiteType.ListCount - 1
            If cboLiteType.ItemData(i) = 7 Then
                cboLiteType.ListIndex = i
                Exit For
            End If
        Next
    Else
        Grd.RowHeight(2) = Grd.RowHeight(1)
'        If IsUnrestricted = True And ((LiftType = "High Lift" And DrHgt > 264) Or _
'        (LiftType = "Vertical Lift" And DrHgt > 336) Or _
'        ((LiftType = "Standard Lift" Or LiftType = "Low Headroom Rear Mount" Or LiftType = "Low Headroom Front Mount") And DrHgt > 384)) Then
        If IsUnrestricted = True And DrHgt > 264 Then
            Call SectionsNumCombinationOversized
            If N21 + N24 >= 1 Then
                Call PopulateSectionArrayDefault
                Call FillGrd
            End If
        Else
            Call SectionsNumCombination   'get the # of 24 and 21 inch sections
            If N21 + N24 - 1 >= 1 Then
                Call PopulateSectionArrayDefault
                Call FillGrd
            End If
        End If
    End If
    Exit Sub

Private Sub PopulateSectionArrayDefault()

    On Error GoTo Error_Handler
    Const cstrErrorSource As String = cstrApplicationName & "." & cstrModuleName & ".PopulateSectionArrayDefault()"
    Dim llngErrNo As Long
    Dim lstrErrDescr As String
    Dim lstrErrSource As String
    llngErrNo = 0
    lstrErrDescr = ""
    lstrErrSource = ""

    Dim i As Integer, j As Integer
    Dim Temp24 As Integer
    Dim Temp21 As Integer
    Dim CutTop As Single
    Dim CutBtm As Single
    Dim lintUpperBound As Integer
    
    Temp24 = N24
    Temp21 = N21
    
    If N21 + N24 < 1 Then Exit Sub
    
    Dim larrTemp() As Integer
    
    
    'transfer existing lites to temp array, because on redim of sections array they will be cleared
    If Not isEmptySectionsArray Then
        lintUpperBound = UBound(varDoorData.arrSections)
        ReDim larrTemp(lintUpperBound)
        For i = 0 To lintUpperBound
            larrTemp(i) = varDoorData.arrSections(i).intNLites
        Next
    End If
    
    ReDim varDoorData.arrSections(N21 + N24 - 1) As SectionData

    For i = N21 + N24 - 1 To 0 Step -1
        If Temp24 > 0 Then
            varDoorData.arrSections(i).intHeight = 24
            Temp24 = Temp24 - 1
        Else
            varDoorData.arrSections(i).intHeight = 21
            Temp21 = Temp21 - 1
        End If
        ''''''''''''''''''''''''''
'        If i > 0 Then
'            If cmdShowAllLites.Enabled = False Then
'                varDoorData.arrSections(i).intNLites = Val(txtLitesPerSection.Text)
'            Else
'                Exit For
'            End If
'        End If
        ''''''''''''''''''''''''''''
    Next
    
    If lintUpperBound < UBound(varDoorData.arrSections) Then
        
        'reload lites from temp array.
        If Not isEmptyArray2(larrTemp) Then
            For i = 0 To UBound(larrTemp)
                varDoorData.arrSections(i).intNLites = larrTemp(i)
            Next
        End If
    ElseIf lintUpperBound > UBound(varDoorData.arrSections) Then
        Call ClearLiteInfo
    End If
    
    If ProdType = "Thermatite" Or ProdType = "Thermalume" Or ProdType = "Replacement" Or DrType = "A175T" Then
        CutTop = GetCut(False)
        CutBtm = GetCut(True)
        
        'there is a difference between the drhgt and hight in sections, have to add that difference to the cut.
        'when calculating cut, only height in sections is taken into consideration.
        If RoundUp(DrHgt) > DrHgt Then
            If IIf(varDoorData.arrSections(0).intHeight = 21, 21, 24) - CutTop + (RoundUp(DrHgt) - DrHgt) <= 15 Then
                CutTop = CutTop + (RoundUp(DrHgt) - DrHgt)

''''''''''Change in .NET''''''''''''''''
'                If CutTop < 1 Then
'                    If CutBtm > 0 Then CutBtm = CutBtm + CutTop
'                End If
'                CutTop = 0 'ignore cuts smaller than an inch.
                If CutTop < 1 Then
                    If CutBtm > 0 Then
                        CutBtm = CutBtm + CutTop
                        CutTop = 0 'ignore cuts smaller than an inch.
                    End If
                End If

Public Function CheckHLift() As Boolean

    On Error GoTo Error_Handler
    Const cstrErrorSource As String = cstrApplicationName & "." & cstrModuleName & ".CheckHLift()"
    Dim llngErrNo As Long
    Dim lstrErrDescr As String
    Dim lstrErrSource As String
    llngErrNo = 0
    lstrErrDescr = ""
    lstrErrSource = ""
    
    Dim lsngHL As Single
    Dim lsngHR As Single
    Dim lstrSQL As String
    Dim lobjDrumRS As ADODB.Recordset
    Dim llngAffected As Long
    lstrSQL = "select Stamping, MaxHiLift, Type from Drums"
    Set lobjDrumRS = CnnBTMS.Execute(lstrSQL, llngAffected, adCmdText)
    If Not lobjDrumRS Is Nothing Then
        lobjDrumRS.ActiveConnection = Nothing
        If Not (lobjDrumRS.EOF = True And lobjDrumRS.BOF = True) Then
            lobjDrumRS.MoveFirst
            Do While lobjDrumRS.EOF = False
                If lobjDrumRS.Fields("Type").Value = "H" Then
                    If OptHL.Value = True Then
                        If MeasU = 1 Then
                            lsngHL = DBLValue(txtHiLift.Text)
                        Else
                            aCon.mm = DBLValue(txtHiLift.Text)
                            aCon.Inchcalc
                            lsngHL = aCon.Inch
                        End If
                    Else
                        If MeasU = 1 Then
                            lsngHR = DBLValue(txtHRoom.Text)
                        Else
                            aCon.mm = DBLValue(txtHRoom.Text)
                            aCon.Inchcalc
                            lsngHR = aCon.Inch
                        End If
                        varDoorData.sngHeadroom = lsngHR
                        Select Case lobjDrumRS.Fields("Stamping").Value
                            Case "D400-54"
                                lsngHL = IIf(cboHWSize.ItemData(cboHWSize.ListIndex) < 3, lsngHR - 7.49, lsngHR - 8.553)
                            Case "D525-54"
                                lsngHL = IIf(cboHWSize.ItemData(cboHWSize.ListIndex) < 3, lsngHR - 8.844, lsngHR - 10.334)
                            Case "D575-120"
                                lsngHL = IIf(cboHWSize.ItemData(cboHWSize.ListIndex) < 3, lsngHR - 10.219, lsngHR - 11.969)
                            Case "D6375-164"
                                lsngHL = lsngHR - 17.75
                            Case "D800-120"
                                lsngHL = lsngHR - 18.2
                            Case "D850-132"
                                lsngHL = IIf(LiftType = "Vertical Lift", DrHgt * 2 + 22, DrHgt * 2 + 10)
                            Case "D1100-216"
                                lsngHL = IIf(LiftType = "Vertical Lift", DrHgt * 2 + 10, DrHgt * 2 + 22)
                            Case Else
                                lsngHL = 0
                        End Select
                    End If
                    If lsngHL <= lobjDrumRS.Fields("MaxHiLift").Value Then
                        CheckHLift = True
                        lobjDrumRS.Close
                        Set lobjDrumRS = Nothing
                        Exit Function
                    End If
                End If
                lobjDrumRS.MoveNext
            Loop
        End If
        lobjDrumRS.Close
    End If
    Set lobjDrumRS = Nothing
    Exit Function

Error_Handler:
    llngErrNo = IIf(llngErrNo = 0, Err.Number, llngErrNo)
    lstrErrDescr = IIf(Len(lstrErrDescr) = 0, Err.Description, lstrErrDescr & vbCrLf & Err.Description)
    lstrErrSource = IIf(Len(Trim(Err.Source)) = 0, "", Err.Source & vbCrLf) & cstrErrorSource
    Call Err.Raise(llngErrNo, lstrErrSource, lstrErrDescr)

End Function

    lstrErrDescr = IIf(Len(lstrErrDescr) = 0, Err.Description, lstrErrDescr & vbCrLf & Err.Description)
    lstrErrSource = IIf(Len(Trim(Err.Source)) = 0, "", Err.Source & vbCrLf) & cstrErrorSource
    Call Err.Raise(llngErrNo, lstrErrSource, lstrErrDescr)

End Sub

Private Sub ClearLiteInfo()
    On Error GoTo Error_Handler
    Const cstrErrorSource As String = cstrApplicationName & "." & cstrModuleName & ".ClearLiteInfo()"
    Dim llngErrNo As Long
    Dim lstrErrDescr As String
    Dim lstrErrSource As String
    llngErrNo = 0
    lstrErrDescr = ""
    lstrErrSource = ""
    
    Me.txtNofLiteSec.Text = ""
    Me.txtLitesPerSection.Text = ""
    Me.cboLiteType.ListIndex = 0
    
    Exit Sub

Error_Handler:
    llngErrNo = IIf(llngErrNo = 0, Err.Number, llngErrNo)
    lstrErrDescr = IIf(Len(lstrErrDescr) = 0, Err.Description, lstrErrDescr & vbCrLf & Err.Description)
    lstrErrSource = IIf(Len(Trim(Err.Source)) = 0, "", Err.Source & vbCrLf) & cstrErrorSource
    Call Err.Raise(llngErrNo, lstrErrSource, lstrErrDescr)

End Sub
