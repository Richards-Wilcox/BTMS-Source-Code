Private Sub cmdDelete_Click()
    On Error GoTo Error_Handler
    Const cstrErrorSource As String = cstrApplicationName & "." & cstrModuleName & ".cmdDelete_Click()"
    Dim llngErrNo As Long
    Dim lstrErrDescr As String
    Dim lstrErrSource As String
    llngErrNo = 0
    lstrErrDescr = ""
    lstrErrSource = ""

    Dim llngRow As Long
    Dim llngQuoteID As Long
    Dim llngPOID As Long
    Dim llngAffected As Long
    Dim lblnInTrans As Boolean
    Dim lstrSQLQuery As String
    Dim lblnAlloc As Boolean
    lblnInTrans = False
    llngRow = grdData.RowSel
    llngQuoteID = DBLValue(grdData.TextMatrix(llngRow, 9))
    llngPOID = DBLValue(grdData.TextMatrix(llngRow, 10))
    If grdData.TextMatrix(llngRow, 11) <> "" Then lblnAlloc = CBool(grdData.TextMatrix(llngRow, 11))
    
    If llngQuoteID > 0 Then
        If llngPOID > 0 Then
            MsgBox "This Quote is attached to a PO.", vbApplicationModal + vbOKOnly + vbExclamation, "Cannot Delete Quote"
            Exit Sub
        Else
            If lblnAlloc = True Then
                If MsgBox("This Quote has inventory allocated." & vbCrLf & "Are you sure you wish to delete this Quote?", vbApplicationModal + vbYesNo + vbExclamation, "Delete Quote") = vbNo Then
                    Exit Sub
                Else
                    CnnBTMS.BeginTrans
                    lblnInTrans = True
                    
                    ' Remove the quote from the BoM
                    lstrSQLQuery = "Update BoM Set QuoteID=0, QuoteNum='' where quoteid=" & llngQuoteID
                    Call CnnBTMS.Execute(lstrSQLQuery, llngAffected, adCmdText And adExecuteNoRecords)
                    
                    ' Delete the quote
                    lstrSQLQuery = "delete * from " & cstrTableName & " where quoteid=" & llngQuoteID
                    Call CnnBTMS.Execute(lstrSQLQuery, llngAffected, adCmdText And adExecuteNoRecords)
                    CnnBTMS.CommitTrans
                    lblnInTrans = False
                    Call LoadData
                End If
            Else
                If grdData.TextMatrix(llngRow, 6) <> "" Then
                    If MsgBox("The items on this Quote are reserved in the inventory. Deleting this Quote will un-reserve those parts." & vbCrLf & _
                        "Do you wish to delete this Quote?", vbQuestion + vbApplicationModal + vbYesNo, "Delete Quote") = vbYes Then
                            Call UnAllocateQuote(llngQuoteID)
                            
                            CnnBTMS.BeginTrans
                            lblnInTrans = True
                            
                            ' Remove the quote from the BoM
                            lstrSQLQuery = "Update BoM Set QuoteID=0, QuoteNum='' where quoteid=" & llngQuoteID
                            Call CnnBTMS.Execute(lstrSQLQuery, llngAffected, adCmdText And adExecuteNoRecords)
                            
                            ' Delete the quote
                            lstrSQLQuery = "delete * from " & cstrTableName & " where quoteid=" & llngQuoteID
                            Call CnnBTMS.Execute(lstrSQLQuery, llngAffected, adCmdText And adExecuteNoRecords)
                            CnnBTMS.CommitTrans
                            lblnInTrans = False
                            Call LoadData
                    End If
                Else
                    If MsgBox("Are you sure you wish to delete quote " & Trim(grdData.TextMatrix(llngRow, 2)), vbApplicationModal + vbYesNo + vbExclamation, "Delete Quote") = vbYes Then
                        
                        CnnBTMS.BeginTrans
                        lblnInTrans = True
                        
                        ' Remove the quote from the BoM
                        lstrSQLQuery = "Update BoM Set QuoteID=0, QuoteNum='' where quoteid=" & llngQuoteID
                        Call CnnBTMS.Execute(lstrSQLQuery, llngAffected, adCmdText And adExecuteNoRecords)
                        
                        ' Delete the quote
                        lstrSQLQuery = "delete * from " & cstrTableName & " where quoteid=" & llngQuoteID
                        Call CnnBTMS.Execute(lstrSQLQuery, llngAffected, adCmdText And adExecuteNoRecords)
                        CnnBTMS.CommitTrans
                        lblnInTrans = False
                        Call LoadData
                    End If
                End If
            End If
        End If
    End If
    Exit Sub

Error_Handler:
    llngErrNo = IIf(llngErrNo = 0, Err.Number, llngErrNo)
    lstrErrDescr = IIf(Len(lstrErrDescr) = 0, Err.Description, Err.Description)
    If lblnInTrans = True Then CnnBTMS.RollbackTrans
    Call DisplayError(llngErrNo, IIf(Len(Trim(Err.Source)) = 0, "", Err.Source & vbCrLf) & cstrErrorSource, lstrErrDescr)
End Sub

Private Sub LoadData()
    On Error GoTo Error_Handler
    Const cstrErrorSource As String = cstrApplicationName & "." & cstrModuleName & ".LoadData()"
    Dim llngErrNo As Long
    Dim lstrErrDescr As String
    Dim lstrErrSource As String
    llngErrNo = 0
    lstrErrDescr = ""
    lstrErrSource = ""

    Dim llngAffected As Long
    Dim llngRecords As Long
    Dim lstrSQL As String
    Dim lobjRS As ADODB.Recordset
    Dim i As Integer
    
    grdData.Clear
    llngRecords = 0
    lstrSQL = "Select " & cstrSelectList & " from " & cstrTableName
    Set mobjADORS = CnnBTMS.Execute(lstrSQL, llngAffected, adCmdText)
    If Not mobjADORS Is Nothing Then
        mobjADORS.ActiveConnection = Nothing
        If Not (mobjADORS.BOF And mobjADORS.EOF) Then
            llngRecords = mobjADORS.RecordCount
            grdData.VirtualData = False
            grdData.DataMode = flexDMFree
            Set grdData.DataSource = mobjADORS
        End If
    End If
    fraGrid.Caption = " " & llngRecords & " Records "
    
    For i = 1 To grdData.Rows - 1
        If grdData.TextMatrix(i, 1) <> "" Then
            lstrSQL = "select Bomid from bom where quoteid=" & grdData.TextMatrix(i, 9)
            Set lobjRS = CnnBTMS.Execute(lstrSQL, llngAffected, adCmdText)
            If Not lobjRS Is Nothing Then
                If Not (lobjRS.EOF And lobjRS.BOF) Then
                    grdData.TextMatrix(i, 3) = "click to view"
                Else
                    grdData.TextMatrix(i, 3) = "none"
                End If
            Else
                grdData.TextMatrix(i, 3) = "none"
            End If
        End If
    Next
    
    Call SetGridTitles
    
    Exit Sub

Error_Handler:
    llngErrNo = IIf(llngErrNo = 0, Err.Number, llngErrNo)
    lstrErrDescr = IIf(Len(lstrErrDescr) = 0, Err.Description, Err.Description)
    lstrErrSource = IIf(Len(Trim(Err.Source)) = 0, "", Err.Source & vbCrLf) & cstrErrorSource
    Call Err.Raise(llngErrNo, lstrErrSource, lstrErrDescr)
End Sub

Private Sub grdData_BeforeSort(ByVal Col As Long, Order As Integer)
    On Error GoTo Error_Handler
    Const cstrErrorSource As String = cstrApplicationName & "." & cstrModuleName & ".grdData_BeforeSort()"
    Dim llngErrNo As Long
    Dim lstrErrDescr As String
    Dim lstrErrSource As String
    llngErrNo = 0
    lstrErrDescr = ""
    lstrErrSource = ""

    
    ' no flags? apply custom sort
    If grdData.ExplorerBar > &H1000& Then Exit Sub
    
    ' save selection
    Dim R&, c&, rs&, cs&
    grdData.GetSelection R, c, rs, cs
    grdData.Redraw = flexRDNone
    
    ' apply sort to non-empty range
    Dim Row%
    For Row = grdData.Rows - 1 To grdData.FixedRows Step -1
        If Len(grdData.TextMatrix(Row, Col)) Then Exit For
    Next
    If Row > grdData.FixedRows Then
        Dim llngCol As Long
        llngCol = Col
        If Col = 4 Then llngCol = 1
        grdData.Select grdData.FixedRows, llngCol, Row, llngCol
        grdData.Sort = Order
    End If
    
    ' restore selection
    grdData.Select R, c, rs, cs
    grdData.Redraw = flexRDDirect
    
    ' cancel default sort
    Order = 0
    Exit Sub

Error_Handler:
    llngErrNo = IIf(llngErrNo = 0, Err.Number, llngErrNo)
    lstrErrDescr = IIf(Len(lstrErrDescr) = 0, Err.Description, Err.Description)
    lstrErrSource = IIf(Len(Trim(Err.Source)) = 0, "", Err.Source & vbCrLf) & cstrErrorSource
    Call Err.Raise(llngErrNo, lstrErrSource, lstrErrDescr)
End Sub

Public Function ParseDocument(wField As Object, WordDoc As Object) As String
    On Error GoTo Error_Handler
    Const cstrErrorSource As String = cstrApplicationName & "." & cstrModuleName & ".GetNewResult()"
    Dim llngErrNo As Long
    Dim lstrErrDescr As String
    Dim lstrErrSource As String
    llngErrNo = 0
    lstrErrDescr = ""
    lstrErrSource = ""
    
    Dim lstrStopPos As Long
    Dim lstrWord As String
    Dim lstrText As String
    Dim lstrWordValue As String
    Dim wRange As Object
    
    ' These three lines strip down the field code to find
    ' out it's name
    lstrStopPos = InStrRev(wField.Code, "\*")
    lstrWord = Left(wField.Code, lstrStopPos - 3)
    lstrWord = Right(lstrWord, Len(lstrWord) - 14)
    Set wRange = WordDoc.Range
    
    ' Check this field hasn't already appeared in this
    ' document.
    Select Case UCase(lstrWord)
        Case "RPTTITLE":         lstrWordValue = "Quote Listing"
            
        ' I don't simply want to insert a string -
        ' I wish to insert a table at the Product Field.
        Case "RPTLIST", "RPTHEADER"
                            
                lstrText = "Quote No" & "~" & "Customer Name" & "~" & "Contact" & "~" & "Phone" & "~" & "Price" & vbCrLf
                If UCase(lstrWord) = "RPTLIST" Then
                    Dim lstrSQLQuery As String
                    Dim llngAffected As String
                    lstrSQLQuery = "select QuoteNum, CustName, CustContact, CustPhone, QuoteTotPrice"
                    lstrSQLQuery = lstrSQLQuery & " from Quote"
                    lstrSQLQuery = lstrSQLQuery & " order by QuoteNum"
                    Dim lobjQuoteRS As ADODB.Recordset
                    Set lobjQuoteRS = CnnBTMS.Execute(lstrSQLQuery, llngAffected, adCmdText)
                    lstrText = ""
                    
                    lstrText = " ~ ~ ~ ~ ~ ~ ~ " & vbCrLf
                    If Not lobjQuoteRS Is Nothing Then
                        lobjQuoteRS.ActiveConnection = Nothing
                        If Not (lobjQuoteRS.BOF And lobjQuoteRS.EOF) Then
                            lobjQuoteRS.MoveFirst
                            lstrText = lobjQuoteRS.GetString(adClipString, -1, "~", vbCrLf)
                            lobjQuoteRS.Close
                        End If
                    End If
                    Set lobjQuoteRS = Nothing
                End If
                
                ' Get the range (location) of the product field
                Set wRange = wField.Code
                ' Delete the field, as any text will be inserted into the
                ' {} of the existing field.
                wField.Delete
                
                ' Enter our table information including headers.
                ' Ideally, I would get this data from an ADO recordset
                ' using GetString().
                With wRange
            
                    .Text = lstrText
                    .FormattedText.Font.Name = "Arial"
                    .FormattedText.Font.Size = "8"
                
                    ' Once the data is there, we can convert it to a table
                    ' structure and format it to look pretty!
                    If UCase(lstrWord) = "RPTLIST" Then
                        .ConvertToTable "~", , , 60, wdTableFormatColorful2, , , , , 0
                    Else
                        .ConvertToTable "~", , , 60, wdTableFormatColorful2
                    End If
                                                   
                .Columns(1).Width = 140
                .Columns(2).Width = 130
                .Columns(3).Width = 110
                .Columns(4).Width = 100
                End With
                
                ' Send back blank string as field does not exist anymore
                lstrWordValue = ""
    
        Case Else
            
            ' Get the value of the field from the user
            lstrWordValue = lstrWord
    
    End Select
    
    ParseDocument = lstrWordValue
    Exit Function

Error_Handler:
    llngErrNo = IIf(llngErrNo = 0, Err.Number, llngErrNo)
    lstrErrDescr = IIf(Len(lstrErrDescr) = 0, Err.Description, Err.Description)
    lstrErrSource = IIf(Len(Trim(Err.Source)) = 0, "", Err.Source & vbCrLf) & cstrErrorSource
    Call Err.Raise(llngErrNo, lstrErrSource, lstrErrDescr)
End Function


Private Sub LoadDocuments(ByVal Row As Long, ByVal Col As Long)
    On Error GoTo Error_Handler
    Const cstrErrorSource As String = cstrApplicationName & "." & cstrModuleName & ".LoadDocuments()"
    Dim llngErrNo As Long
    Dim lstrErrDescr As String
    Dim lstrErrSource As String
    llngErrNo = 0
    lstrErrDescr = ""
    lstrErrSource = ""
        
    Dim lstrSQL As String
    Dim lobjRS As ADODB.Recordset
    Dim llngAffected As Long
    Dim i As Integer
    Dim lstrList As String
    
    If grdData.TextMatrix(Row, Col) <> "" Then
        lstrSQL = "select BOMNum,BOMID from BoM where QuoteID=" & Me.grdData.TextMatrix(Row, 9)
        Set lobjRS = CnnBTMS.Execute(lstrSQL, llngAffected, adCmdText)
        If Not lobjRS Is Nothing Then
            If Not (lobjRS.EOF And lobjRS.BOF) Then
                lstrList = ""
                ReDim llngBOMID(lobjRS.RecordCount)
                Do While lobjRS.EOF = False
                    lstrList = lstrList & lobjRS.Fields("BOMNum").Value & "|"
                    llngBOMID(i) = lobjRS.Fields("BOMID").Value
                    i = i + 1
                    lobjRS.MoveNext
                Loop
            Else
                grdData.TextMatrix(Row, Col) = "none"
            End If
        End If
    End If
    grdData.ColComboList(Col) = lstrList
    Exit Sub

Error_Handler:
    llngErrNo = IIf(llngErrNo = 0, Err.Number, llngErrNo)
    lstrErrDescr = IIf(Len(lstrErrDescr) = 0, Err.Description, lstrErrDescr & vbCrLf & Err.Description)
    lstrErrSource = IIf(Len(Trim(Err.Source)) = 0, "", Err.Source & vbCrLf) & cstrErrorSource
    Call Err.Raise(llngErrNo, lstrErrSource, lstrErrDescr)

End Sub
